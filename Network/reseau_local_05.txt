        // Si oui, on se connecte à SA PROPRE adresse : ça marche en local

        // (ngrok / LocalTunnel) sans rien saisir. Seule une page ouverte en
        // double-cliquant sur le fichier (file://) utilise les champs ci-dessous.
        const SERVED = /^https?:$/.test(window.location.protocol);

        // Certains tunnels affichent une page d'avertissement tant qu'un en-tête
        // spécial n'est pas envoyé : on l'ajoute à toutes les requêtes.
        const _originalFetch = window.fetch;
        window.fetch = function (input, init = {}) {
            const headers = new Headers(init.headers || {});
            headers.set('Bypass-Tunnel-Reminder', 'true');
            headers.set('ngrok-skip-browser-warning', 'true');
            return _originalFetch(input, { ...init, headers });
        };

        // Configuration

            AUTO_REFRESH: true,

        };

        // Adresses possibles du serveur à partir des champs (dans l'ordre d'essai)
        function serverUrlCandidates(ip, port) {
            ip = String(ip || '').trim();
            port = String(port || '').trim();
            // URL complète collée (ex: https://xxxx.ngrok-free.app) : utilisée telle quelle
            if (/^https?:\/\//i.test(ip)) return [ip.replace(/\/+$/, '')];
            // Même adresse que la page : on reprend exactement son origine
            if (SERVED && ip === window.location.hostname && port === window.location.port) {
                return [window.location.origin];
            }
            const host = ip.replace(/\/+$/, '');
            const suffix = port ? `:${port}` : '';
            if (SERVED) return [`${window.location.protocol}//${host}${suffix}`];
            // Fichier ouvert directement : le serveur peut être en HTTPS ou en HTTP
            return [`https://${host}${suffix}`, `http://${host}${suffix}`];
        }

        // Tester la connexion
        async function testConnection() {
            for (const url of serverUrlCandidates(CONFIG.SERVER_IP, CONFIG.SERVER_PORT)) {
                try {
                    const response = await fetch(`${url}/health`, {
                        method: 'GET',
                        mode: 'cors'
                    }).catch(() => null);

                    if (response && response.ok) {
                        updateStatus(true, 'Serveur accessible');
                        return true;
                    }
                } catch (error) {
                    // Ignorer les erreurs de test
                }

            return false;

        // Connecter au serveur
        async function connectServer() {
            const ip = DOM.serverIP.value.trim();
            const port = DOM.serverPort.value.trim();
            
            if (!ip) {
                showToast('Veuillez entrer une adresse IP ou l\'URL du tunnel', 'error');
                return;
            }
            
            if (port && (port < 1 || port > 65535)) {
                showToast('Port invalide', 'error');
                return;
            }
            
            // Sauvegarder la configuration
            localStorage.setItem('admin_server_ip', ip);
            localStorage.setItem('admin_server_port', port);
            
            CONFIG.SERVER_IP = ip;
            CONFIG.SERVER_PORT = port;
            
            // Afficher le chargement
            DOM.btnStart.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Connexion...';
            DOM.btnStart.disabled = true;
            
            try {
                let connected = false;
                let lastError = null;

                for (const url of serverUrlCandidates(ip, port)) {
                    try {
                        const response = await fetch(`${url}/users`, {
                            method: 'GET',
                            mode: 'cors',
                            headers: {
                                'Accept': 'application/json'
                            }
                        });

                        if (response.ok) {
                            STATE.serverUrl = url;
                            connected = true;
                            break;
                        }
                        lastError = new Error('Réponse du serveur : HTTP ' + response.status);
                    } catch (e) {
                        lastError = e;
                    }
                }

                if (!connected) {
                    throw lastError || new Error('Serveur inaccessible');
                }

                updateStatus(true);
                
                // Charger la configuration du serveur
                await loadServerConfig();
                
                // Charger les données
                await Promise.all([
                    loadUsers(),
                    loadStats()
                ]);
                
                // Activer le rafraîchissement automatique
                startAutoRefresh();
                
                showToast('Connecté au serveur', 'success');
                
            } catch (error) {
                console.error('Erreur connexion:', error);
                showToast(
                    'Impossible de se connecter au serveur. Vérifiez:\n1. L\'adresse (ou l\'URL du tunnel) et le port\n2. Le serveur est en cours d\'exécution\n3. En HTTPS avec certificat auto-signé : ouvrez d\'abord l\'adresse dans le navigateur et acceptez l\'avertissement',
                    'error'
                );
                updateStatus(false);
            } finally {
                DOM.btnStart.innerHTML = '<i class="fas fa-plug"></i> Se connecter';
                DOM.btnStart.disabled = false;
            }
        }

        // Charger la configuration du serveur
        async function loadServerConfig() {
            if (!STATE.isConnected) return;
            
            try {
                const response = await fetch(`${STATE.serverUrl}/config`);
                if (response.ok) {
                    const config = await response.json();
                    STATE.serverConfig = config;
                    
                    // Mettre à jour l'interface
                    DOM.minUsernameLength.value = config.min_username_length || 3;
                    DOM.maxUsernameLength.value = config.max_username_length || 20;
                    DOM.minPasswordLength.value = config.min_password_length || 6;
                    DOM.maxPasswordLength.value = config.max_password_length || 50;
                }
            } catch (error) {
                console.error('Erreur chargement config:', error);
            }
        }

        // Sauvegarder la configuration système
        async function saveSystemConfig() {
            if (!STATE.isConnected) {
                showToast('Veuillez d\'abord vous connecter au serveur', 'error');
                return;
            }
            
            const config = {
                min_username_length: parseInt(DOM.minUsernameLength.value),
                max_username_length: parseInt(DOM.maxUsernameLength.value),
                min_password_length: parseInt(DOM.minPasswordLength.value),
                max_password_length: parseInt(DOM.maxPasswordLength.value)
            };
            
            // Validation
            if (config.min_username_length < 1 || config.min_username_length > 50) {
                showToast('Longueur min nom d\'utilisateur invalide (1-50)', 'error');
                return;
            }
            
            if (config.max_username_length < 1 || config.max_username_length > 100) {
                showToast('Longueur max nom d\'utilisateur invalide (1-100)', 'error');
                return;
            }
            
            if (config.min_username_length > config.max_username_length) {
                showToast('Le min ne peut pas être supérieur au max', 'error');
                return;
            }
            
            if (config.min_password_length < 1 || config.min_password_length > 100) {
                showToast('Longueur min mot de passe invalide (1-100)', 'error');
                return;
            }
            
            if (config.max_password_length < 1 || config.max_password_length > 200) {
                showToast('Longueur max mot de passe invalide (1-200)', 'error');
                return;
            }
            
            if (config.min_password_length > config.max_password_length) {
                showToast('Le min ne peut pas être supérieur au max', 'error');
                return;
            }
            
            try {
                const response = await fetch(`${STATE.serverUrl}/update_config`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        username: 'admin',
                        ...config
                    })
                });
                
                if (response.ok) {
                    STATE.serverConfig = config;
                    showToast('Configuration sauvegardée avec succès', 'success');
                } else {
                    const error = await response.json();
                    throw new Error(error.detail || 'Erreur de sauvegarde');
                }
            } catch (error) {
                showToast(error.message, 'error');
            }
        }

        // Formater la date
        function formatDate(dateString) {
            try {
                const date = new Date(dateString);
                const now = new Date();
                const diffMs = now - date;
                const diffMins = Math.floor(diffMs / 60000);
                const diffHours = Math.floor(diffMs / 3600000);
                const diffDays = Math.floor(diffMs / 86400000);
                
                if (diffMins < 1) return 'à l\'instant';
                if (diffMins < 60) return `il y a ${diffMins} min`;
                if (diffHours < 24) return `il y a ${diffHours} h`;
                if (diffDays < 7) return `il y a ${diffDays} j`;
                
                return date.toLocaleDateString('fr-FR', {
                    day: 'numeric',
                    month: 'short',
                    year: 'numeric'
                });
            } catch {
                return dateString;
            }
        }

# --- Lecture / écriture JSON sécurisées -------------------------------------
# Écriture "atomique" (fichier temporaire puis remplacement) : une coupure ou un
# plantage pendant l'écriture ne peut plus laisser un fichier tronqué. Un
# fichier illisible n'est JAMAIS écrasé en silence : on essaie la copie .bak,
# sinon on le met de côté (.corrompu-DATE) avant de repartir de zéro.
def _read_json_safe(path, default, label):
    for candidate in (path, path + ".bak"):
        if not os.path.exists(candidate):
            continue
        try:
            with open(candidate, "r", encoding="utf-8") as f:
                data = json.load(f)
            if candidate != path:
                logger.warning(f"⚠️ {label} : fichier principal illisible, restauré depuis {os.path.basename(candidate)}")
            return data
        except Exception as e:
            logger.error(f"❌ {label} : lecture de {os.path.basename(candidate)} impossible ({e})")
            if candidate == path:
                try:
                    aside = f"{path}.corrompu-{datetime.now().strftime('%Y%m%d_%H%M%S')}"
                    os.replace(path, aside)
                    logger.error(f"   Fichier mis de côté : {aside}")
                except Exception:
                    pass
    return default

def _write_json_atomic(path, data):
    text = json.dumps(data, indent=2)
    try:
        if os.path.exists(path):
            with open(path, "r", encoding="utf-8") as f:
                if f.read() == text:
                    return  # rien n'a changé : on évite d'écrire pour rien
    except Exception:
        pass
    tmp = path + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        f.write(text)
        f.flush()
        os.fsync(f.fileno())
    if os.path.exists(path):
        try:
            shutil.copy2(path, path + ".bak")
        except Exception:
            pass
    for attempt in range(4):
        try:
            os.replace(tmp, path)
            return
        except OSError:
            time.sleep(0.1)  # Windows : fichier momentanément verrouillé (antivirus...)
    with open(path, "w", encoding="utf-8") as f:  # dernier recours
        f.write(text)

# Charger les données
def load_data():
    users_data = _read_json_safe(USERS_FILE, {}, "Utilisateurs")
    if not isinstance(users_data, dict):
        users_data = {}
    messages_data = _read_json_safe(MESSAGES_FILE, {"messages": [], "last_id": 0}, "Messages")
    if not isinstance(messages_data, dict) or "messages" not in messages_data:
        messages_data = {"messages": [], "last_id": 0}
    return users_data, messages_data

def save_data(users, messages):
    try:
        _write_json_atomic(USERS_FILE, users)
        _write_json_atomic(MESSAGES_FILE, messages)
    except Exception as e:
        logger.error(f"Erreur sauvegarde: {e}")

def snapshot_users(keep: int = 20):
    """Copie datée de users.json dans data/backups (au démarrage), pour ne plus jamais perdre les comptes."""
    try:
        if not os.path.exists(USERS_FILE):
            return
        os.makedirs(BACKUP_DIR, exist_ok=True)
        with open(USERS_FILE, "rb") as f:
            raw = f.read()
        existing = sorted(glob.glob(os.path.join(BACKUP_DIR, "users_*.json")))
        if existing:
            with open(existing[-1], "rb") as f:
                if f.read() == raw:
                    return
        dest = os.path.join(BACKUP_DIR, f"users_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json")
        with open(dest, "wb") as f:
            f.write(raw)
        for old in sorted(glob.glob(os.path.join(BACKUP_DIR, "users_*.json")))[:-keep]:
            try:
                os.remove(old)
            except Exception:
                pass
    except Exception as e:
        logger.error(f"Sauvegarde datée des utilisateurs impossible: {e}")

def create_session_token(username: str) -> str:
    token = secrets.token_urlsafe(32)
    now = datetime.now()
    expires = now + timedelta(days=CONFIG.get("session_duration_days", 30))
    sessions[token] = {
        "username": username,
        "created_at": now.isoformat(),
        "expires_at": expires.isoformat()
    }
    save_sessions(sessions)
    return token

def revoke_session_token(token: str):
    if token and token in sessions:
        del sessions[token]
        save_sessions(sessions)

def revoke_all_sessions_for_user(username: str):
    to_remove = [t for t, s in sessions.items() if s.get("username") == username]
    for t in to_remove:
        del sessions[t]
    if to_remove:
        save_sessions(sessions)

def get_valid_session(token: str):
    """Retourne le nom d'utilisateur associé à un jeton valide, ou None."""
    if not token or token not in sessions:
        return None
    session = sessions[token]
    try:
        expires_at = datetime.fromisoformat(session["expires_at"])
    except Exception:
        return None
    if datetime.now() > expires_at:
        # Jeton expiré : nettoyage
        del sessions[token]
        save_sessions(sessions)
        return None
    username = session.get("username")
    if username not in users:
        del sessions[token]
        save_sessions(sessions)
        return None
    return username

def new_node_id() -> str:
    return uuid.uuid4().hex[:10]

def make_node(name: str, node_id: Optional[str] = None):
    return {
        "id": node_id or new_node_id(),
        "name": name,
        "mastered": False,
        "children": []
    }

def reassign_node_ids(node: dict):
    """Régénère récursivement les id d'un arbre (utilisé à l'import pour éviter les collisions)."""
    node["id"] = new_node_id()
    node.setdefault("name", "Sans titre")
    node.setdefault("mastered", False)
    node.setdefault("children", [])
    for child in node["children"]:
        reassign_node_ids(child)
    return node

def count_nodes(node: dict) -> int:
    total = 1
    for child in node.get("children", []):
        total += count_nodes(child)
    return total

def find_node(node: dict, node_id: str):
    if node.get("id") == node_id:
        return node
    for child in node.get("children", []):
        found = find_node(child, node_id)
        if found:
            return found
    return None

# Gestion WebSocket
class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[str, WebSocket] = {}
        self.user_colors = {}
    
    async def connect(self, websocket: WebSocket, username: str):
        await websocket.accept()
        self.active_connections[username] = websocket
        users[username]["online"] = True
        save_data(users, messages)
        await self.broadcast_status()
    
    def disconnect(self, username: str):
        if username in self.active_connections:
            del self.active_connections[username]
        if username in users:
            users[username]["online"] = False
            save_data(users, messages)
    
    async def send_message(self, message: dict, username: str):
        if username in self.active_connections:
            try:
                await self.active_connections[username].send_json(message)
            except:
                self.disconnect(username)
    
    async def broadcast(self, message: dict):
        for username, websocket in list(self.active_connections.items()):
            try:
                await websocket.send_json(message)
            except:
                self.disconnect(username)
    
    async def broadcast_status(self):
        online_users = list(self.active_connections.keys())
        status_message = {
            "type": "user_status",
            "online_users": online_users,
            "count": len(online_users)
        }
        await self.broadcast(status_message)
    
    def get_user_color(self, username: str):
        if username in users and "color" in users[username]:
            return users[username]["color"]
        # Générer une couleur si elle n'existe pas
        color = generate_random_color()
        if username in users:
            users[username]["color"] = color
            save_data(users, messages)
        return color
    
    def set_user_color(self, username: str, color: str):
        if username in users:
            users[username]["color"] = color
            save_data(users, messages)

# ==================== APPELS VOCAUX (signalisation WebRTC) ====================
class CallManager:
    """Appels vocaux, y compris de groupe.

    Le serveur ne transporte PAS l'audio : il sert uniquement de "standard
    téléphonique" (qui appelle qui, qui a rejoint, échange des offres WebRTC).
    L'audio circule directement de navigateur à navigateur (maillage).
    """

    RING_SECONDS = 45        # durée pendant laquelle l'appel sonne
    MAX_PARTICIPANTS = 8     # participants + invités par appel

    def __init__(self):
        self.calls: Dict[str, dict] = {}
        self.user_call: Dict[str, str] = {}   # username -> call_id (1 appel à la fois)

    # -- utilitaires --
    def _public(self, call: dict) -> dict:
        return {
            "call_id": call["id"],
            "host": call["host"],
            "participants": list(call["participants"]),
            "invited": sorted(call["invited"]),
        }

    async def _send(self, username: str, payload: dict):
        await manager.send_message(payload, username)

    async def _to_participants(self, call: dict, payload: dict, exclude: Optional[str] = None):
        for u in list(call["participants"]):
            if u != exclude:
                await self._send(u, payload)

    async def _error(self, username: str, text: str, call_id: str = ""):
        await self._send(username, {"type": "call_error", "text": text, "call_id": call_id})

    # -- point d'entrée --
    async def handle(self, username: str, msg: dict):
        t = msg.get("type")
        try:
            if t == "call_start":
                invitees = msg.get("invitees")
                await self.start(username, invitees if isinstance(invitees, list) else [])
            elif t == "call_join":
                await self.join(username, str(msg.get("call_id", "")))
            elif t == "call_decline":
                await self.decline(username, str(msg.get("call_id", "")), str(msg.get("reason", "declined")))
            elif t == "call_leave":
                await self.leave(username)
            elif t == "call_invite":
                invitees = msg.get("invitees")
                await self.invite(username, str(msg.get("call_id", "")), invitees if isinstance(invitees, list) else [])
            elif t == "call_signal":
                await self.signal(username, str(msg.get("call_id", "")), str(msg.get("to", "")), msg.get("payload"))
        except Exception:
            logger.exception(f"Erreur appel vocal ({t}) pour {username}")

    # -- actions --
    async def start(self, username: str, invitees: list):
        if username in self.user_call:
            return await self._error(username, "Vous êtes déjà dans un appel.")
        call_id = uuid.uuid4().hex[:12]
        call = {
            "id": call_id,
            "host": username,
            "participants": [username],
            "invited": set(),
            "created_at": datetime.now().isoformat(),
        }
        self.calls[call_id] = call
        self.user_call[username] = call_id
        logger.info(f"📞 Appel {call_id} démarré par {username}")
        await self._send(username, {"type": "call_started", "call": self._public(call), "peers": []})
        notice = await self._invite(call, username, invitees)
        if notice:
            await self._send(username, {"type": "call_notice", "text": notice})
        await self._check_end(call)

    async def _invite(self, call: dict, inviter: str, invitees: list) -> str:
        offline, busy, invited_now = [], [], []
        for u in dict.fromkeys(invitees):
            if (not isinstance(u, str) or u not in users or u == inviter
                    or u in call["participants"] or u in call["invited"]):
                continue
            if len(call["participants"]) + len(call["invited"]) >= self.MAX_PARTICIPANTS:
                break
            if u not in manager.active_connections:
                offline.append(u)
                continue
            if u in self.user_call:
                busy.append(u)
                continue
            call["invited"].add(u)
            invited_now.append(u)
            await self._send(u, {"type": "call_incoming", "call": self._public(call), "from": inviter})
        if invited_now:
            await self._to_participants(call, {
                "type": "call_invited", "call_id": call["id"], "invited": sorted(call["invited"])
            })
            asyncio.create_task(self._expire(call["id"], invited_now))
        parts = []
        if offline:
            parts.append("hors ligne : " + ", ".join(offline))
        if busy:
            parts.append("déjà en appel : " + ", ".join(busy))
        return ("Non joignable(s) - " + " ; ".join(parts)) if parts else ""

    async def _expire(self, call_id: str, usernames: list):
        await asyncio.sleep(self.RING_SECONDS)
        call = self.calls.get(call_id)
        if not call:
            return
        for u in usernames:
            if u in call["invited"]:
                call["invited"].discard(u)
                await self._send(u, {"type": "call_invite_cancelled", "call_id": call_id, "reason": "missed"})
                await self._to_participants(call, {
                    "type": "call_declined", "call_id": call_id, "username": u, "reason": "no_answer"
                })
        await self._check_end(call)

    async def join(self, username: str, call_id: str):
        call = self.calls.get(call_id)
        if not call:
            return await self._error(username, "Cet appel est déjà terminé.", call_id)
        if username in self.user_call:
            if self.user_call[username] == call_id:
                return
            return await self._error(username, "Vous êtes déjà dans un autre appel.", call_id)
        if username not in call["invited"]:
            return await self._error(username, "Vous n'êtes pas invité à cet appel.", call_id)
        peers = list(call["participants"])
        call["invited"].discard(username)
        call["participants"].append(username)
        self.user_call[username] = call_id
        logger.info(f"📞 {username} a rejoint l'appel {call_id} ({len(call['participants'])} participants)")
        # Le nouvel arrivant envoie une offre WebRTC à chaque personne déjà présente
        await self._send(username, {"type": "call_joined", "call": self._public(call), "peers": peers})
        await self._to_participants(call, {
            "type": "call_participant_joined", "call_id": call_id,
            "username": username, "call": self._public(call)
        }, exclude=username)

    async def decline(self, username: str, call_id: str, reason: str = "declined"):
        call = self.calls.get(call_id)
        if not call or username not in call["invited"]:
            return
        call["invited"].discard(username)
        if reason not in ("declined", "busy"):
            reason = "declined"
        await self._to_participants(call, {
            "type": "call_declined", "call_id": call_id, "username": username, "reason": reason
        })
        await self._check_end(call)

    async def leave(self, username: str):
        call_id = self.user_call.pop(username, None)
        call = self.calls.get(call_id) if call_id else None
        if not call:
            return
        if username in call["participants"]:
            call["participants"].remove(username)
        logger.info(f"📞 {username} a quitté l'appel {call_id}")
        await self._to_participants(call, {
            "type": "call_participant_left", "call_id": call_id, "username": username
        })
        await self._check_end(call)

    async def invite(self, username: str, call_id: str, invitees: list):
        call = self.calls.get(call_id)
        if not call or username not in call["participants"]:
            return await self._error(username, "Vous n'êtes pas dans cet appel.", call_id)
        notice = await self._invite(call, username, invitees)
        if notice:
            await self._send(username, {"type": "call_notice", "text": notice})

    async def signal(self, username: str, call_id: str, to: str, payload):
        call = self.calls.get(call_id)
        if (not call or not isinstance(payload, dict)
                or username not in call["participants"] or to not in call["participants"]):
            return
        await self._send(to, {"type": "call_signal", "call_id": call_id, "from": username, "payload": payload})

    async def leave_all(self, username: str):
        """Appelé quand un utilisateur se déconnecte (fermeture d'onglet, logout, etc.)."""
        await self.leave(username)
        for call in list(self.calls.values()):
            if username in call["invited"]:
                call["invited"].discard(username)
                await self._to_participants(call, {
                    "type": "call_declined", "call_id": call["id"], "username": username, "reason": "declined"
                })
                await self._check_end(call)

    async def _check_end(self, call: dict):
        """Un appel se termine quand plus personne n'est dedans, ou quand il ne reste
        qu'une personne et que plus personne n'est en train d'être appelé."""
        if call["id"] not in self.calls:
            return
        alone = len(call["participants"]) == 1 and not call["invited"]
        if call["participants"] and not alone:
            return
        for u in list(call["participants"]):
            self.user_call.pop(u, None)
            await self._send(u, {
                "type": "call_ended", "call_id": call["id"], "reason": "Plus personne d'autre dans l'appel."
            })
        for u in list(call["invited"]):
            await self._send(u, {"type": "call_invite_cancelled", "call_id": call["id"], "reason": "cancelled"})
        del self.calls[call["id"]]
        logger.info(f"📞 Appel {call['id']} terminé")

@app.websocket("/ws/{username}")
async def websocket_endpoint(websocket: WebSocket, username: str):
    if username not in users:
        await websocket.close(code=1008, reason="Utilisateur inconnu")
        return
    
    await manager.connect(websocket, username)
    
    try:
        while True:
            data = await websocket.receive_text()
            try:
                message = json.loads(data)
                if message.get("type") == "ping":
                    await websocket.send_json({"type": "pong"})
                elif message.get("type") == "typing":
                    await manager.broadcast({
                        "type": "user_typing",
                        "username": username,
                        "is_typing": message.get("is_typing", False)
                    })
                elif str(message.get("type", "")).startswith("call_"):
                    await call_manager.handle(username, message)
            except:
                pass
    except WebSocketDisconnect:
        await call_manager.leave_all(username)
        manager.disconnect(username)
        await manager.broadcast_status()

# Upload de fichiers - PAS DE LIMITE DE TAILLE
@app.post("/upload_file")
async def upload_file(file: UploadFile = File(...)):
    if not file:
        raise HTTPException(status_code=400, detail="Aucun fichier")
    
    try:
        # Liste étendue des extensions autorisées
        ALLOWED_EXTENSIONS = {
            '.jpg', '.jpeg', '.png', '.gif', '.webp', '.bmp',
            '.pdf', '.txt', '.doc', '.docx', '.xls', '.xlsx', '.ppt', '.pptx',
            '.zip', '.rar', '.7z', '.tar', '.gz',
            '.mp3', '.m4a', '.wav', '.ogg', '.flac', '.aac',
            '.mp4', '.avi', '.mkv', '.mov', '.wmv', '.flv', '.webm',
            '.csv', '.json', '.xml', '.html', '.css', '.js', '.py', '.java', '.cpp', '.c',
            '.rtf', '.odt', '.ods', '.odp'
        }
        
        file_ext = os.path.splitext(file.filename)[1].lower()
        
        # Si config permet tous les types ou si extension est dans la liste
        if not CONFIG.get("allow_all_file_types", True) and file_ext not in ALLOWED_EXTENSIONS:
            logger.warning(f"Tentative d'upload avec extension non autorisée: {file_ext}")
            # Accepter quand même avec avertissement
            # raise HTTPException(status_code=400, detail="Type de fichier non autorisé")
        
        # Lire le fichier en chunks
        CHUNK_SIZE = 1024 * 1024  # 1MB chunks
        file_content = b""
        total_size = 0
        
        while True:
            chunk = await file.read(CHUNK_SIZE)
            if not chunk:
                break
            file_content += chunk
            total_size += len(chunk)
            
            # Log toutes les 10MB
            if total_size % (10 * 1024 * 1024) == 0:
                logger.info(f"Upload en cours: {file.filename} - {total_size / (1024*1024):.1f} MB")
        
        file_size = total_size
        
        # PAS DE VÉRIFICATION DE TAILLE - ACCEPTE TOUS LES FICHIERS
        logger.info(f"✅ Fichier uploadé: {file.filename} - Taille: {file_size} bytes ({file_size/(1024*1024):.1f} MB)")
        
        unique_filename = f"{uuid.uuid4()}{file_ext}"
        file_path = os.path.join(UPLOADS_DIR, unique_filename)
        
        # Écrire le fichier
        with open(file_path, "wb") as f:
            f.write(file_content)
        
        file_url = f"/uploads/{unique_filename}"
        
        # Formater la taille
        def format_size(size):
            if size < 1024:
                return f"{size} B"
            elif size < 1024 * 1024:
                return f"{size/1024:.1f} KB"
            elif size < 1024 * 1024 * 1024:
                return f"{size/(1024*1024):.1f} MB"
            else:
                return f"{size/(1024*1024*1024):.1f} GB"
        
        return {
            "file_url": file_url,
            "file_name": file.filename,
            "file_size": format_size(file_size),
            "file_size_bytes": file_size,
            "detail": "Fichier uploadé avec succès"
        }
        
    except Exception as e:
        logger.error(f"❌ Erreur upload fichier: {e}")
        raise HTTPException(status_code=500, detail=f"Erreur lors de l'upload: {str(e)}")

# --- Génération de segments via IA locale (Ollama) ---
def _call_ollama_sync(base_url: str, model: str, prompt: str) -> str:
    payload = json.dumps({
        "model": model,
        "prompt": prompt,
        "stream": False
    }).encode("utf-8")
    
    req = urllib.request.Request(
        url=base_url.rstrip("/") + "/api/generate",
        data=payload,
        headers={"Content-Type": "application/json"},
        method="POST"
    )
    with urllib.request.urlopen(req, timeout=120) as resp:
        body = json.loads(resp.read().decode("utf-8"))
        return body.get("response", "")

def _parse_segments_from_ai_text(text: str) -> List[str]:
    # 1) Essayer d'extraire un tableau JSON de chaînes
    match = re.search(r"\[[\s\S]*\]", text)
    if match:
        try:
            parsed = json.loads(match.group(0))
            if isinstance(parsed, list):
                cleaned = [str(item).strip(" \"'.-") for item in parsed if str(item).strip()]
                if cleaned:
                    return cleaned[:12]
        except Exception:
            pass
    
    # 2) Repli: découper par lignes et nettoyer les puces/numéros
    lines = [l.strip() for l in text.splitlines() if l.strip()]
    cleaned = []
    for line in lines:
        line = re.sub(r"^[\-\*\d\.\)\s]+", "", line).strip(' "\'')
        if line and len(line) < 120:
            cleaned.append(line)
    return cleaned[:12]

# --- Certificat HTTPS auto-signé ---
def _local_ips():
    ips = {"127.0.0.1"}
    try:
        for info in socket.getaddrinfo(socket.gethostname(), None, socket.AF_INET):
            ips.add(info[4][0])
    except Exception:
        pass
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(("10.255.255.255", 1))  # rien n'est envoyé, sert juste à trouver l'IP locale
        ips.add(s.getsockname()[0])
        s.close()
    except Exception:
        pass
    return sorted(ips)

def _split_extra_hosts():
    ips, names = set(), set()
    for h in EXTRA_CERT_HOSTS:
        try:
            ips.add(str(ipaddress.ip_address(h)))
        except ValueError:
            names.add(h)
    return ips, names

def _read_cert_info():
    """(ips, noms, expiration) du certificat actuel, ou None s'il est absent / illisible."""
    if not (os.path.exists(CERT_FILE) and os.path.exists(KEY_FILE)):
        return None
    try:
        from cryptography import x509
        with open(CERT_FILE, "rb") as f:
            cert = x509.load_pem_x509_certificate(f.read())
        try:
            san = cert.extensions.get_extension_for_class(x509.SubjectAlternativeName).value
            ips = {str(i) for i in san.get_values_for_type(x509.IPAddress)}
            names = set(san.get_values_for_type(x509.DNSName))
        except x509.ExtensionNotFound:
            ips, names = set(), set()
        expires = getattr(cert, "not_valid_after_utc", None)
        if expires is None:
            expires = cert.not_valid_after.replace(tzinfo=timezone.utc)
        return ips, names, expires
    except Exception as e:
        logger.warning(f"Certificat existant illisible ({e}) -> il sera recréé")
        return None

def ensure_certificate(force: bool = False) -> bool:
    """Prépare data/cert.pem + data/key.pem (localhost + IP locales du PC).

    Le certificat existant est CONTRÔLÉ à chaque démarrage : s'il ne couvre plus
    une IP actuelle du PC (changement de Wi-Fi, nouvelle adresse...) ou s'il
    expire bientôt, il est recréé automatiquement, en gardant les anciennes
    adresses. 'python serveur.py --regen-cert' force une recréation complète.
    """
    try:
        from cryptography import x509
        from cryptography.x509.oid import NameOID
        from cryptography.hazmat.primitives import hashes, serialization
        from cryptography.hazmat.primitives.asymmetric import rsa
    except ImportError:
        logger.warning("Module 'cryptography' absent -> pas de HTTPS. Installez-le avec : pip install cryptography")
        return False

    extra_ips, extra_names = _split_extra_hosts()
    wanted_ips = set(_local_ips()) | extra_ips
    wanted_names = {"localhost"} | extra_names
    try:
        wanted_names.add(socket.gethostname())
    except Exception:
        pass

    if not force:
        info = _read_cert_info()
        if info:
            old_ips, old_names, expires = info
            missing = (wanted_ips - old_ips) | (wanted_names - old_names)
            soon = expires < datetime.now(timezone.utc) + timedelta(days=30)
            if not missing and not soon:
                return True
            if missing:
                logger.info(f"🔒 Le certificat ne couvre pas : {', '.join(sorted(missing))} -> recréation")
            if soon:
                logger.info("🔒 Le certificat expire bientôt -> recréation")
            wanted_ips |= old_ips      # on garde les anciennes adresses valides
            wanted_names |= old_names

    try:
        key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        name = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, "Reseau Local")])
        alt_names = []
        for n in sorted(wanted_names):
            try:
                alt_names.append(x509.DNSName(n))
            except Exception:
                pass
        for ip in sorted(wanted_ips):
            try:
                alt_names.append(x509.IPAddress(ipaddress.ip_address(ip)))
            except Exception:
                pass
        now = datetime.now(timezone.utc)
        cert = (
            x509.CertificateBuilder()
            .subject_name(name)
            .issuer_name(name)
            .public_key(key.public_key())
            .serial_number(x509.random_serial_number())
            .not_valid_before(now - timedelta(days=1))
            .not_valid_after(now + timedelta(days=825))
            .add_extension(x509.SubjectAlternativeName(alt_names), critical=False)
            .add_extension(x509.BasicConstraints(ca=False, path_length=None), critical=True)
            .sign(key, hashes.SHA256())
        )
        with open(KEY_FILE, "wb") as f:
            f.write(key.private_bytes(
                serialization.Encoding.PEM,
                serialization.PrivateFormat.TraditionalOpenSSL,
                serialization.NoEncryption()
            ))
        with open(CERT_FILE, "wb") as f:
            f.write(cert.public_bytes(serialization.Encoding.PEM))
        logger.info("🔒 Certificat HTTPS auto-signé créé dans le dossier data (valable pour : %s)"
                    % ", ".join(sorted(wanted_names) + sorted(wanted_ips)))
        return True
    except Exception as e:
        logger.error(f"Impossible de créer le certificat HTTPS: {e}")
        return False

def _port_free(port: int) -> bool:
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    try:
        s.bind(("0.0.0.0", port))
        return True
    except OSError:
        return False
    finally:
        s.close()

async def _serve_all(http_port, https_port):
    """Lance le serveur HTTP et le serveur HTTPS côte à côte, dans le même programme."""
    import uvicorn

    class _Server(uvicorn.Server):
        # Deux serveurs dans un même processus : on laisse Python gérer Ctrl+C
        # (sinon le second remplacerait la gestion des signaux du premier).
        def install_signal_handlers(self):
            pass

        @contextlib.contextmanager
        def capture_signals(self):
            yield

    servers = []
    if http_port:
        servers.append(_Server(uvicorn.Config(app, host="0.0.0.0", port=http_port, log_level="info")))
    if https_port:
        servers.append(_Server(uvicorn.Config(
            app, host="0.0.0.0", port=https_port, log_level="info",
            ssl_certfile=CERT_FILE, ssl_keyfile=KEY_FILE)))
    await asyncio.gather(*(srv.serve() for srv in servers))

    // alors qu'un accès direct en réseau local en a besoin (ex: :8001).
    function buildServerUrl(ip, port) {
        // Même adresse que la page : on reprend exactement son origine
        if (SERVED && ip === window.location.hostname && String(port || '') === window.location.port) {
            return window.location.origin;
        }
        const protocol = window.location.protocol === 'https:' ? 'https:' : 'http:';
        return port ? `${protocol}//${ip}:${port}` : `${protocol}//${ip}`;
    }

    function buildWsUrl(ip, port, path) {
        const wsProtocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
        if (SERVED && ip === window.location.hostname && String(port || '') === window.location.port) {
            return `${wsProtocol}//${window.location.host}${path}`;
        }
        const base = port ? `${wsProtocol}//${ip}:${port}` : `${wsProtocol}//${ip}`;
        return `${base}${path}`;

    function setupContextMenu() {

    // Configurer l'appui long pour mobile
    function setupLongPressForMobile() {
        if (!STATE.contextMenu.isTouchDevice) return;
        
        let pressTimer;
        
        document.addEventListener('touchstart', (e) => {
            if (!STATE.currentUser) return;
            
            const messageElement = e.target.closest('.message');
            if (messageElement && messageElement.dataset.messageId) {
                // Empêcher le menu contextuel par défaut
                e.preventDefault();
                
                // Démarrer le timer pour l'appui long
                pressTimer = setTimeout(() => {
                    // Calculer la position pour le menu
                    const touch = e.touches[0] || e.changedTouches[0];
                    showContextMenu(touch.pageX, touch.pageY, messageElement);
                }, CONFIG.LONG_PRESS_DURATION);
                
                // Ajouter l'effet visuel
                messageElement.classList.add('highlight');
            }
        });
        
        document.addEventListener('touchend', (e) => {
            clearTimeout(pressTimer);
            const messageElement = e.target.closest('.message');
            if (messageElement) {
                messageElement.classList.remove('highlight');
            }
        });
        
        document.addEventListener('touchmove', (e) => {
            clearTimeout(pressTimer);
            const messageElement = e.target.closest('.message');
            if (messageElement) {
                messageElement.classList.remove('highlight');
            }
        });
    }

        // Ajuster la position si le menu dépasse de l'écran
        let posX = x;
        let posY = y;
        
        if (x + menuWidth > windowWidth) {

        }
        
        if (y + menuHeight > windowHeight) {
            posY = windowHeight - menuHeight - 10;
        }
        
        DOM.contextMenu.style.left = posX + 'px';
        DOM.contextMenu.style.top = posY + 'px';
        STATE.contextMenu.active = true;
        
        // Sur mobile, ajouter un effet visuel

            messageElement.classList.add('highlight');

    }

    // Télécharger depuis le menu contextuel
    function handleDownloadFromContext() {
        if (!STATE.contextMenu.selectedMessage) return;
        
        const messageElement = STATE.contextMenu.selectedMessage;
        
        // Chercher différents types de fichiers
        const fileAttachment = messageElement.querySelector('.file-attachment');
        const imagePreview = messageElement.querySelector('.image-preview img');
        const audioPlayer = messageElement.querySelector('.audio-player audio source');
        
        if (fileAttachment && fileAttachment.href) {
            // Télécharger le fichier attaché
            const url = fileAttachment.href;
            const fileName = fileAttachment.querySelector('.file-name')?.textContent || `file_${Date.now()}`;
            forceDownload(url, fileName);
        } else if (imagePreview && imagePreview.src) {
            // Télécharger l'image
            const url = imagePreview.src;
            const fileName = `image_${Date.now()}.${url.split('.').pop()}`;
            forceDownload(url, fileName);
        } else if (audioPlayer && audioPlayer.src) {
            // Télécharger l'audio
            const url = audioPlayer.src;
            const fileName = `audio_${Date.now()}.${url.split('.').pop()}`;
            forceDownload(url, fileName);
        } else {
            // Si c'est un message texte, l'offrir en téléchargement
            const contentElement = messageElement.querySelector('.message-content');
            if (contentElement) {
                const text = contentElement.textContent;
                const blob = new Blob([text], { type: 'text/plain' });
                const url = URL.createObjectURL(blob);
                const sender = messageElement.dataset.sender || 'message';
                const time = messageElement.querySelector('.message-time')?.textContent || Date.now();
                const fileName = `message_${sender}_${time.replace(/:/g, '-')}.txt`;
                
                forceDownload(url, fileName);
                
                // Libérer l'URL après le téléchargement
                setTimeout(() => URL.revokeObjectURL(url), 100);
            }
        }
        
        hideContextMenu();
    }

    // Fonction pour forcer le téléchargement (pas l'ouverture)
    function forceDownload(url, filename) {
        fetch(url)
            .then(response => response.blob())
            .then(blob => {
                const blobUrl = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = blobUrl;
                a.download = filename;
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(blobUrl);
                showToast('Téléchargement démarré', 'success');
            })
            .catch(error => {
                console.error('Erreur de téléchargement:', error);
                showToast('Erreur lors du téléchargement', 'error');
            });
    }

    // Vérifier une session existante (connexion persistante réelle via /verify_token)
    async function checkExistingSession() {
        const savedUser = localStorage.getItem('chat_user');
        const savedToken = localStorage.getItem('chat_token');
        const savedColor = localStorage.getItem('chat_user_color');
        const savedRole = localStorage.getItem('chat_role');
        
        if (!savedUser || !savedToken) {
            return;
        }
        
        STATE.currentUser = savedUser;
        STATE.userColor = savedColor;
        STATE.userRole = savedRole || 'user';
        STATE.sessionToken = savedToken;
        STATE.serverUrl = buildServerUrl(CONFIG.SERVER_IP, CONFIG.SERVER_PORT);
        
        try {
            const response = await fetch(`${STATE.serverUrl}/verify_token`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ token: savedToken })
            });
            
            if (response.ok) {
                const data = await response.json();
                STATE.currentUser = data.username;
                STATE.userColor = data.color;
                STATE.userRole = data.role || 'user';
                localStorage.setItem('chat_user_color', STATE.userColor);
                localStorage.setItem('chat_role', STATE.userRole);
                await restoreSession(data.username);
                return;
            }
        } catch (error) {
            console.log('Serveur inaccessible pour la vérification de session:', error);
        }
        
        // Jeton invalide/expiré ou serveur injoignable au démarrage : nettoyer
        clearSession();
    }

    // Restaurer la session
    async function restoreSession(username) {
        showScreen('chat');
        
        // Mettre à jour l'interface
        DOM.currentUserName.textContent = username;
        DOM.userAvatar.textContent = username.charAt(0).toUpperCase();
        applyRoleUI();
        
        // Récupérer la couleur de l'utilisateur
        if (!STATE.userColor) {
            try {
                const response = await fetch(`${STATE.serverUrl}/user/${username}`);
                if (response.ok) {
                    const userData = await response.json();
                    STATE.userColor = userData.color;
                    localStorage.setItem('chat_user_color', STATE.userColor);
                }
            } catch (error) {
                console.error('Erreur récupération couleur:', error);
            }
        }
        
        // Connecter au WebSocket
        await connectWebSocket();
        
        // Charger les messages
        await loadMessages();
        
        showToast('Session restaurée', 'success');
    }

    // Gérer les fichiers déposés
    async function handleDroppedFiles(files) {
        if (!STATE.currentUser) {
            showToast('Veuillez vous connecter d\'abord', 'error');
            return;
        }
        
        const validFiles = Array.from(files).filter(file => {
            // PAS DE VÉRIFICATION DE TAILLE
            // Seulement vérifier les types très dangereux
            const dangerousTypes = [
                'application/x-msdownload', // .exe
                'application/x-dosexec',    // .exe
                'application/x-sh',         // shell scripts
                'application/x-bat',        // batch files
                'application/x-csh',        // csh scripts
                'text/x-python',            // python (potentiellement dangereux)
                'application/x-php'         // php
            ];
            
            if (dangerousTypes.includes(file.type)) {
                showToast(`${file.name} : Type de fichier potentiellement dangereux`, 'error');
                return false;
            }
            
            return true;
        });
        
        if (validFiles.length === 0) return;
        
        if (validFiles.length === 1) {
            const file = validFiles[0];
            // Si c'est une image, l'envoyer directement avec le texte actuel
            if (CONFIG.ALLOWED_IMAGE_TYPES.includes(file.type)) {
                const currentText = DOM.messageInput.value;
                await sendFileDirectly(file, currentText || `Image: ${file.name}`);
                DOM.messageInput.value = '';
            } else {
                // Sinon ouvrir le modal
                openFileModal();
                handleMultipleFilesSelect(validFiles);
            }
        } else {
            // Plusieurs fichiers : ouvrir le modal
            openFileModal();
            handleMultipleFilesSelect(validFiles);
        }
    }

    // Gérer le défilement

        STATE.shouldScrollToBottom = isAtBottom && CONFIG.AUTO_SCROLL_ENABLED;

        }

    // Configuration du drag & drop global
    function setupGlobalDragAndDrop() {
        // Empêcher le comportement par défaut
        ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
            document.addEventListener(eventName, preventDefaults, false);
        });

        // Afficher l'overlay de drag & drop
        ['dragenter', 'dragover'].forEach(eventName => {
            document.addEventListener(eventName, (e) => {
                if (STATE.currentUser && e.dataTransfer.types.includes('Files')) {
                    STATE.isDragging = true;
                    DOM.dragOverlay.classList.add('active');
                }
            }, false);
        });

        ['dragleave', 'drop'].forEach(eventName => {
            document.addEventListener(eventName, () => {
                STATE.isDragging = false;
                DOM.dragOverlay.classList.remove('active');
            }, false);
        });

        // Gérer le drop
        document.addEventListener('drop', handleGlobalDrop, false);

        function preventDefaults(e) {
            e.preventDefault();
            e.stopPropagation();
        }

        function handleGlobalDrop(e) {
            const dt = e.dataTransfer;
            const files = dt.files;
            
            if (files.length && STATE.currentUser) {
                e.preventDefault();
                handleDroppedFiles(files);
            }
        }
    }

    // Configuration du collage d'images
    function setupPasteHandler() {
        document.addEventListener('paste', async (e) => {
            if (!STATE.currentUser) return;
            
            const items = e.clipboardData.items;
            let imageFound = false;
            
            for (let item of items) {
                if (item.type.indexOf('image') !== -1) {
                    const blob = item.getAsFile();
                    if (blob) {
                        // PAS DE VÉRIFICATION DE TAILLE
                        const fileName = `paste_${Date.now()}.png`;
                        const file = new File([blob], fileName, { type: 'image/png' });
                        
                        imageFound = true;
                        const currentText = DOM.messageInput.value;
                        await sendFileDirectly(file, currentText || 'Image collée');
                        DOM.messageInput.value = '';
                        break;
                    }
                }
            }
            
            if (!imageFound) {
                // Vérifier si du texte a été collé
                const text = e.clipboardData.getData('text');
                if (text && text.trim()) {
                    // Ajouter le texte collé au champ de message
                    const cursorPos = DOM.messageInput.selectionStart;
                    const textBefore = DOM.messageInput.value.substring(0, cursorPos);
                    const textAfter = DOM.messageInput.value.substring(cursorPos);
                    
                    DOM.messageInput.value = textBefore + text + textAfter;
                    DOM.messageInput.focus();
                    DOM.messageInput.setSelectionRange(cursorPos + text.length, cursorPos + text.length);
                    
                    // Déclencher le redimensionnement
                    DOM.messageInput.dispatchEvent(new Event('input'));
                }
            }
        });
    }

    // Gérer la connexion
    async function handleLogin() {
        const username = DOM.username.value.trim();
        const password = DOM.password.value.trim();
        const serverIP = DOM.serverIP.value.trim();
        const serverPort = DOM.serverPort.value.trim();

        // Validation
        if (!username || !password) {
            showNotification('Veuillez remplir tous les champs', 'error');
            return;
        }

        if (!serverIP) {
            showNotification('Veuillez spécifier l\'adresse du serveur', 'error');
            return;
        }

        // Sauvegarder la configuration
        localStorage.setItem('chat_server_ip', serverIP);
        localStorage.setItem('chat_server_port', serverPort);
        
        CONFIG.SERVER_IP = serverIP;
        CONFIG.SERVER_PORT = serverPort;
        
        STATE.serverUrl = buildServerUrl(serverIP, serverPort);

        // Afficher le chargement
        DOM.loginBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Connexion...';
        DOM.loginBtn.disabled = true;

        try {
            const response = await fetch(`${STATE.serverUrl}/login`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Accept': 'application/json'
                },
                body: JSON.stringify({ username, password })
            });

            if (response.ok) {
                const data = await response.json();
                
                // Sauvegarder la session (connexion persistante : jeton réel vérifié côté serveur)
                STATE.currentUser = username;
                STATE.userColor = data.color;
                STATE.userRole = data.role || 'user';
                STATE.sessionToken = data.token;
                localStorage.setItem('chat_user', username);
                localStorage.setItem('chat_token', data.token);
                localStorage.setItem('chat_user_color', data.color);
                localStorage.setItem('chat_role', STATE.userRole);
                
                // Mettre à jour l'interface
                DOM.currentUserName.textContent = username;
                DOM.userAvatar.textContent = username.charAt(0).toUpperCase();
                applyRoleUI();
                
                // Basculer vers le chat
                showScreen('chat');
                
                // Connecter au WebSocket
                await connectWebSocket();
                
                // Charger les messages
                await loadMessages();
                
                showToast('✅ Connexion réussie', 'success');
                
                // Réinitialiser le formulaire
                DOM.username.value = '';
                DOM.password.value = '';
                
            } else {
                const error = await response.json();
                throw new Error(error.detail || 'Erreur de connexion');
            }
        } catch (error) {
            showNotification(error.message, 'error');
            console.error('Login error:', error);
        } finally {
            DOM.loginBtn.innerHTML = '<i class="fas fa-sign-in-alt"></i> Se connecter';
            DOM.loginBtn.disabled = false;
        }
    }

    // Gérer la déconnexion
    async function handleLogout() {
        if (!STATE.currentUser || !STATE.serverUrl) {
            clearSession();
            showScreen('login');
            return;
        }

        try {
            // Fermer le WebSocket
            if (STATE.websocket) {
                STATE.websocket.close();
                STATE.websocket = null;
            }

            // Appeler l'API de déconnexion (invalide le jeton côté serveur)
            await fetch(`${STATE.serverUrl}/logout`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ username: STATE.currentUser, token: STATE.sessionToken })
            }).catch(() => {}); // Ignorer les erreurs

        } catch (error) {
            console.error('Logout error:', error);
        } finally {
            clearSession();
            showScreen('login');
            showToast('Déconnexion réussie', 'success');
        }
    }

    // Connecter au WebSocket
    async function connectWebSocket() {
        if (!STATE.currentUser || !STATE.serverUrl) return;

        try {
            const wsUrl = buildWsUrl(CONFIG.SERVER_IP, CONFIG.SERVER_PORT, `/ws/${STATE.currentUser}`);
            
            STATE.websocket = new WebSocket(wsUrl);
            
            STATE.websocket.onopen = () => {
                console.log('WebSocket connecté');
                updateConnectionStatus('online', 'Connecté en temps réel');
                STATE.reconnectAttempts = 0;
                
                // Récupérer les utilisateurs en ligne
                fetchOnlineUsers();
            };
            
            STATE.websocket.onmessage = (event) => {
                try {
                    const data = JSON.parse(event.data);
                    handleWebSocketMessage(data);
                } catch (error) {
                    console.error('Erreur parsing WebSocket:', error);
                }
            };
            
            STATE.websocket.onerror = (error) => {
                console.error('Erreur WebSocket:', error);
                updateConnectionStatus('offline', 'Erreur de connexion');
            };
            
            STATE.websocket.onclose = () => {
                console.log('WebSocket déconnecté');
                updateConnectionStatus('offline', 'Déconnecté');
                
                // Tentative de reconnexion
                if (CONFIG.AUTO_RECONNECT && STATE.reconnectAttempts < CONFIG.MAX_RECONNECT_ATTEMPTS) {
                    STATE.reconnectAttempts++;
                    console.log(`Tentative de reconnexion ${STATE.reconnectAttempts}/${CONFIG.MAX_RECONNECT_ATTEMPTS}`);
                    
                    setTimeout(() => {
                        if (STATE.currentUser) {
                            connectWebSocket();
                        }
                    }, CONFIG.RECONNECT_DELAY * STATE.reconnectAttempts);
                }
            };
            
        } catch (error) {
            console.error('Erreur création WebSocket:', error);
            updateConnectionStatus('offline', 'Connexion impossible');
        }
    }

    // Gérer les messages WebSocket
    function handleWebSocketMessage(data) {
        switch (data.type) {
            case 'new_message':
                handleNewMessage(data.message);
                break;
                
            case 'delete_message':
                handleDeleteMessage(data.message_id);
                break;
                
            case 'clear_messages':
                handleClearMessages();
                break;
                
            case 'user_status':
                handleUserStatus(data);
                break;
                
            case 'user_typing':
                handleUserTyping(data);
                break;
                
            case 'user_color_updated':
                handleUserColorUpdated(data);
                break;
                
            case 'force_logout':
                handleForceLogout(data);
                break;
                
            case 'account_deleted':
                handleAccountDeleted(data);
                break;
                
            case 'pong':
                // Ping/pong pour maintenir la connexion
                break;
                
            default:
                console.log('Message WebSocket inconnu:', data);
        }
    }

    // Gérer un nouveau message
    function handleNewMessage(message) {
        // Vérifier si le message existe déjà
        const existingIndex = STATE.messages.findIndex(m => m.id === message.id);
        
        if (existingIndex === -1) {
            STATE.messages.push(message);
            
            // Stocker la couleur de l'utilisateur
            if (message.sender_color) {
                STATE.userColors.set(message.sender, message.sender_color);
            }
            
            renderMessage(message);
            
            // Toujours faire défiler vers le bas pour les nouveaux messages
            STATE.shouldScrollToBottom = true;
            scrollToBottomIfNeeded();
            
            // Notification sonore pour les nouveaux messages (sauf les siens)
            if (message.sender !== STATE.currentUser) {
                playNotificationSound();
            }
        }
    }

    // Gérer la suppression d'un message
    function handleDeleteMessage(messageId) {
        const messageElement = document.querySelector(`[data-message-id="${messageId}"]`);
        if (messageElement) {
            messageElement.remove();
        }
        
        // Mettre à jour le tableau des messages
        STATE.messages = STATE.messages.filter(m => m.id !== messageId);
    }

    // Afficher un message
    function renderMessage(message) {
        const isSent = message.sender === STATE.currentUser;
        const time = message.hour || (message.time ? message.time.split(' ')[1] : '');
        const fileIcon = getFileIcon(message.file_name);
        const isImage = message.file_name && /\.(jpg|jpeg|png|gif|webp|bmp|svg)$/i.test(message.file_name);
        const isAudio = message.file_name && (/\.(mp3|m4a|wav|ogg|flac|aac)$/i.test(message.file_name) || /^vocal_.+\.webm$/i.test(message.file_name));
        const senderColor = message.sender_color || STATE.userColors.get(message.sender) || '#6366f1';
        
        const messageElement = document.createElement('div');
        messageElement.className = `message ${isSent ? 'sent' : 'received'}`;
        messageElement.dataset.messageId = message.id;
        messageElement.dataset.sender = message.sender;
        
        let fileContent = '';
        if (message.file_url) {
            if (isImage) {
                fileContent = `
                    <div class="image-preview" onclick="openImage('${STATE.serverUrl}${message.file_url}', '${escapeHtml(message.file_name)}', '${message.sender}', '${time}')">
                        <img src="${STATE.serverUrl}${message.file_url}" alt="${escapeHtml(message.file_name)}" loading="lazy">
                        <div class="image-overlay">
                            <i class="fas fa-expand"></i>
                        </div>
                    </div>
                `;
            } else if (isAudio) {
                const audioType = getAudioType(message.file_name);
                fileContent = `

                        <audio controls>
                            <source src="${STATE.serverUrl}${message.file_url}" type="audio/${audioType}">

                        </audio>

                       class="file-attachment"
                       onclick="event.stopPropagation();">
                        <div class="file-icon">

    // Obtenir le type MIME pour l'audio
    function getAudioType(filename) {
        if (!filename) return 'mpeg';
        const ext = filename.toLowerCase().split('.').pop();
        
        const types = {
            mp3: 'mpeg',
            m4a: 'mp4',
            wav: 'wav',
            ogg: 'ogg',
            flac: 'flac',
            aac: 'aac',
            webm: 'webm'
        };
        
        return types[ext] || 'mpeg';
    }

    // Envoyer un message
    async function sendMessage() {
        const content = DOM.messageInput.value.trim();
        
        if (!content && STATE.selectedFiles.length === 0) {
            showToast('Le message ne peut pas être vide', 'error');
            return;
        }
        
        if (!STATE.currentUser || !STATE.serverUrl) {
            showToast('Vous n\'êtes pas connecté', 'error');
            return;
        }
        
        // Vérifier la longueur du message
        if (content.length > CONFIG.MAX_MESSAGE_LENGTH) {
            showToast(`Message trop long (max ${CONFIG.MAX_MESSAGE_LENGTH} caractères)`, 'error');
            return;
        }
        
        // Désactiver le bouton
        DOM.sendBtn.disabled = true;
        DOM.sendBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i>';
        
        try {
            let fileDataArray = [];
            
            // Si des fichiers sont sélectionnés, les uploader d'abord
            if (STATE.selectedFiles.length > 0) {
                for (const file of STATE.selectedFiles) {
                    try {
                        const fileData = await uploadFileToServer(file);
                        if (fileData) {
                            fileDataArray.push(fileData);
                        }
                    } catch (error) {
                        console.error('Erreur upload fichier:', error);
                        showToast(`Erreur pour ${file.name}: ${error.message}`, 'error');
                    }
                }
            }
            
            // S'il y a des fichiers uploadés, les envoyer
            if (fileDataArray.length > 0) {
                // Pour chaque fichier, envoyer un message séparé
                for (const fileData of fileDataArray) {
                    const messageData = {
                        sender: STATE.currentUser,
                        content: content || `Fichier: ${fileData.file_name}`,
                        file_url: fileData.file_url,
                        file_name: fileData.file_name,
                        file_size: fileData.file_size
                    };
                    
                    const response = await fetch(`${STATE.serverUrl}/send_message`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(messageData)
                    });
                    
                    if (!response.ok) {
                        const error = await response.json();
                        throw new Error(error.detail || 'Erreur d\'envoi du fichier');
                    }
                }
                
                // Si on a du texte en plus des fichiers, envoyer un message texte séparé
                if (content && fileDataArray.length > 0) {
                    const textMessageData = {
                        sender: STATE.currentUser,
                        content: content
                    };
                    
                    const response = await fetch(`${STATE.serverUrl}/send_message`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(textMessageData)
                    });
                    
                    if (!response.ok) {
                        const error = await response.json();
                        throw new Error(error.detail || 'Erreur d\'envoi du message texte');
                    }
                }
                
                // Confirmation améliorée
                showToast(`✅ ${fileDataArray.length} fichier(s) envoyé(s) avec succès`, 'success');
                
            } else {
                // Message texte seulement
                const messageData = {
                    sender: STATE.currentUser,
                    content: content
                };
                
                const response = await fetch(`${STATE.serverUrl}/send_message`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(messageData)
                });
                
                if (response.ok) {
                    // Confirmation améliorée
                    showToast('✅ Message envoyé avec succès', 'success');
                } else {
                    const error = await response.json();
                    throw new Error(error.detail || 'Erreur d\'envoi');
                }
            }
            
            // Réinitialiser l'interface
            DOM.messageInput.value = '';
            DOM.messageInput.style.height = 'auto';
            STATE.selectedFiles = [];
            
            // Animation de confirmation
            DOM.sendBtn.innerHTML = '<i class="fas fa-check"></i>';
            DOM.sendBtn.style.background = 'var(--secondary)';
            setTimeout(() => {
                DOM.sendBtn.innerHTML = '<i class="fas fa-paper-plane"></i>';
                DOM.sendBtn.style.background = '';
            }, 1500);
            
            // Focus sur le champ
            DOM.messageInput.focus();
            
            // Activer le défilement automatique après l'envoi
            STATE.shouldScrollToBottom = true;
            scrollToBottomIfNeeded();
            
        } catch (error) {
            console.error('Erreur envoi message:', error);
            showToast(error.message, 'error');
        } finally {
            // Réactiver le bouton
            DOM.sendBtn.disabled = false;
            DOM.sendBtn.innerHTML = '<i class="fas fa-paper-plane"></i>';
        }
    }

    // Uploader un fichier vers le serveur - VERSION CORRIGÉE
    async function uploadFileToServer(file) {
        if (!file) return null;
        
        console.log(`Début upload: ${file.name} (${file.size} bytes)`);
        
        return new Promise((resolve, reject) => {
            const xhr = new XMLHttpRequest();
            const formData = new FormData();
            formData.append('file', file);
            
            // Mettre à jour la progression
            xhr.upload.addEventListener('progress', (e) => {
                if (e.lengthComputable) {
                    const percent = Math.round((e.loaded / e.total) * 100);
                    updateUploadProgress(file.name, percent);
                }
            });
            
            xhr.onload = function() {
                hideUploadProgress();
                
                console.log(`Upload terminé - Status: ${this.status}`, this.responseText);
                
                if (this.status === 200) {
                    try {
                        const response = JSON.parse(this.responseText);
                        console.log('Upload réussi:', response);
                        resolve(response);
                    } catch (error) {
                        console.error('Erreur parsing réponse:', error);
                        reject(new Error('Format de réponse invalide du serveur'));
                    }
                } else {
                    let errorMessage = `Erreur serveur ${this.status}`;
                    try {
                        const errorResponse = JSON.parse(this.responseText);
                        errorMessage = errorResponse.detail || errorMessage;
                    } catch (e) {
                        // Ne rien faire, garder le message d'erreur par défaut
                    }
                    console.error('Erreur upload:', errorMessage);
                    reject(new Error(errorMessage));
                }
            };
            
            xhr.onerror = function() {
                hideUploadProgress();
                console.error('Erreur réseau lors de l\'upload');
                reject(new Error('Erreur réseau lors de l\'upload'));
            };
            
            xhr.ontimeout = function() {
                hideUploadProgress();
                console.error('Timeout de l\'upload');
                reject(new Error('Timeout de l\'upload - fichier trop grand ou connexion lente'));
            };
            
            // Configuration
            xhr.timeout = 300000; // 5 minutes timeout pour les gros fichiers
            xhr.open('POST', `${STATE.serverUrl}/upload_file`);
            xhr.setRequestHeader('Accept', 'application/json');
            
            // Afficher la progression
            showUploadProgress(file.name);
            
            // Envoyer la requête
            console.log('Envoi de la requête upload...');
            xhr.send(formData);
        });
    }

    // Uploader les fichiers via le modal - VERSION CORRIGÉE
    async function uploadFiles() {
        if (STATE.selectedFiles.length === 0) {
            showToast('Veuillez sélectionner des fichiers', 'error');
            return;

        // Désactiver le bouton
        DOM.uploadFileBtn.disabled = true;

        try {
            // Uploader chaque fichier et envoyer les messages
            let successfulUploads = 0;
            
            for (const file of STATE.selectedFiles) {
                try {
                    console.log(`Tentative d'upload: ${file.name}`);
                    const fileData = await uploadFileToServer(file);
                    
                    if (fileData) {
                        // Créer le message avec le fichier
                        const messageText = DOM.fileMessage.value.trim() || `Fichier: ${file.name}`;

                            sender: STATE.currentUser,
                            content: messageText,
                            file_url: fileData.file_url,

                            file_size: fileData.file_size
                        };
                        
                        // Envoyer le message via l'API
                        const response = await fetch(`${STATE.serverUrl}/send_message`, {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify(messageData)
                        });
                        
                        if (!response.ok) {
                            const error = await response.json();
                            throw new Error(error.detail || 'Erreur d\'envoi');
                        }
                        
                        successfulUploads++;
                        
                        // Ajouter le texte du fichier au champ de message principal
                        if (DOM.fileMessage.value.trim()) {
                            if (DOM.messageInput.value) {
                                DOM.messageInput.value += '\n' + `Fichier: ${file.name}`;
                            } else {
                                DOM.messageInput.value = `Fichier: ${file.name}`;
                            }
                        }
                    }
                } catch (error) {
                    console.error(`Erreur upload fichier ${file.name}:`, error);
                    showToast(`Erreur pour ${file.name}: ${error.message}`, 'error');
                }
            }
            
            // Fermer le modal
            closeFileModal();
            
            // Afficher un message de succès
            if (successfulUploads > 0) {
                showToast(`✅ ${successfulUploads} fichier(s) envoyé(s) avec succès`, 'success');
            } else {

            DOM.uploadFileBtn.disabled = false;

    // Gérer plusieurs fichiers sélectionnés
    function handleMultipleFilesSelect(files) {
        STATE.selectedFiles = [];
        
        // Liste étendue des extensions autorisées (PAS DE LIMITE DE TAILLE)
        const allowedExtensions = [
            // Images
            '.jpg', '.jpeg', '.png', '.gif', '.webp', '.bmp', '.tiff', '.svg', '.ico',
            // Documents
            '.pdf', '.doc', '.docx', '.xls', '.xlsx', '.txt', '.ppt', '.pptx', '.odt', '.ods', '.odp', '.rtf', '.csv',
            // Archives
            '.zip', '.rar', '.7z', '.tar', '.gz', '.bz2',
            // Audio
            '.mp3', '.m4a', '.wav', '.ogg', '.flac', '.aac', '.wma', '.mka', '.opus',
            // Video
            '.mp4', '.avi', '.mkv', '.mov', '.wmv', '.flv', '.webm', '.m4v', '.mpg', '.mpeg', '.3gp',
            // Autres
            '.json', '.xml', '.html', '.htm', '.css', '.js', '.py', '.java', '.cpp', '.c', '.php', '.sql',
            '.exe', '.msi', '.apk', '.dmg', '.iso', '.img'
        ];
        
        // Filtrer les fichiers valides
        const validFiles = files.filter(file => {
            const fileExt = '.' + file.name.split('.').pop().toLowerCase();
            
            // Vérifier les types très dangereux
            const dangerousExtensions = ['.bat', '.cmd', '.sh', '.bash', '.ps1', '.vbs', '.jar', '.js'];
            if (dangerousExtensions.includes(fileExt)) {
                showToast(`${file.name} : Type de fichier potentiellement dangereux`, 'warning');
                // Accepter quand même avec avertissement
            }
            
            if (!allowedExtensions.includes(fileExt)) {
                showToast(`${file.name} : Type de fichier non standard, tentative d'envoi...`, 'warning');
                // Essayer quand même
            }
            
            return true;
        });
        
        if (validFiles.length === 0) return;
        
        STATE.selectedFiles = validFiles;
        
        if (validFiles.length === 1) {
            // Afficher un seul fichier
            const file = validFiles[0];
            DOM.selectedFile.innerHTML = `
                <div class="selected-file-info">
                    <div class="selected-file-icon">
                        <i class="fas fa-file${getFileIcon(file.name)}"></i>
                    </div>
                    <div class="selected-file-details">
                        <h5>${escapeHtml(file.name)}</h5>
                        <span>${formatFileSize(file.size)}</span>
                    </div>
                    <button onclick="clearSelectedFiles()" style="background: none; border: none; color: var(--text-3); cursor: pointer;">
                        <i class="fas fa-times"></i>
                    </button>
                </div>
            `;
            DOM.selectedFile.style.display = 'block';
            DOM.multipleFilesInfo.style.display = 'none';
        } else {
            // Afficher plusieurs fichiers
            DOM.selectedFile.style.display = 'none';
            DOM.multipleFilesInfo.style.display = 'block';
            
            DOM.multipleFilesList.innerHTML = '';
            validFiles.forEach((file, index) => {
                const fileElement = document.createElement('div');
                fileElement.className = 'selected-file-info';
                fileElement.style.marginBottom = '0.5rem';

    // Formater la date
    function formatDate(dateString) {
        try {
            const date = new Date(dateString);
            const today = new Date();
            const yesterday = new Date(today);
            yesterday.setDate(yesterday.getDate() - 1);
            
            if (date.toDateString() === today.toDateString()) {
                return "Aujourd'hui";
            } else if (date.toDateString() === yesterday.toDateString()) {
                return "Hier";
            } else {
                return date.toLocaleDateString('fr-FR', { weekday: 'long', day: 'numeric', month: 'long' });
            }
        } catch {
            return dateString;
        }
    }

    // Formater la taille du fichier
    function formatFileSize(bytes) {
        if (bytes === 0) return '0 Bytes';
        const k = 1024;
        const sizes = ['Bytes', 'KB', 'MB', 'GB', 'TB'];
        const i = Math.floor(Math.log(bytes) / Math.log(k));
        return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
    }

        if (!filename) return '';

            jpg: '-image', jpeg: '-image', png: '-image', gif: '-image',

            // Spreadsheets
            xls: '-excel', xlsx: '-excel', csv: '-excel', ods: '-excel',
            // Archives
            zip: '-archive', rar: '-archive', '7z': '-archive', 
            tar: '-archive', gz: '-archive', bz2: '-archive',
            // Audio
            mp3: '-audio', m4a: '-audio', wav: '-audio', 
            ogg: '-audio', flac: '-audio', aac: '-audio', wma: '-audio', opus: '-audio',
            // Video
            mp4: '-video', avi: '-video', mkv: '-video',

    // Échapper le HTML
    function escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }

    // Formater le contenu du message
    function formatMessageContent(content) {
        if (!content) return '';
        
        // Échapper le HTML
        let safeContent = escapeHtml(content);
        
        // URLs
        safeContent = safeContent.replace(
            /(https?:\/\/[^\s]+)/g,
            '<a href="$1" target="_blank" style="color: var(--primary); text-decoration: underline;">$1</a>'
        );
        
        // Retours à la ligne
        safeContent = safeContent.replace(/\n/g, '<br>');
        
        return safeContent;
    }

    async function projectsApiCall(path, options = {}) {
        const url = `${STATE.serverUrl}${path}`;
        const response = await fetch(url, {
            headers: { 'Content-Type': 'application/json' },
            ...options
        });
        let data = null;
        try {
            data = await response.json();
        } catch (e) {

    function downloadJsonFile(data, filename) {

        a.href = url;

    function slugifyFilename(name) {
        return (name || 'projet')
            .toLowerCase()
            .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
            .replace(/[^a-z0-9]+/g, '_')
            .replace(/^_+|_+$/g, '') || 'projet';
    }

    function handleImportProjectFile(e) {
        const file = e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = async (event) => {
            try {
                const parsed = JSON.parse(event.target.result);
                const tree = parsed.tree || parsed; // tolère un export brut de l'arbre
                const name = parsed.name || tree.name || file.name.replace(/\.json$/i, '');

                if (!tree || typeof tree !== 'object' || !tree.name) {
                    throw new Error("Fichier de projet invalide");
                }

                const project = await projectsApiCall('/projects/import', {
                    method: 'POST',
                    body: JSON.stringify({ username: STATE.currentUser, name, tree })
                });
                showToast('Projet importé', 'success');
                await loadProjectsList();
            } catch (error) {
                showToast(`Import impossible : ${error.message}`, 'error');
            } finally {
                DOM.importProjectInput.value = '';
            }
        };
        reader.readAsText(file);
    }

    // Par défaut, on replie les branches à partir de la 2e génération pour éviter
    // que l'arbre entier ne s'affiche d'un coup et devienne illisible.
    function computeDefaultCollapsedNodes(tree) {
        const collapsed = new Set();
        const walk = (node, depth) => {
            if (depth >= 2 && node.children && node.children.length > 0) {
                collapsed.add(node.id);
            }
            (node.children || []).forEach(child => walk(child, depth + 1));
        };
        walk(tree, 0);
        return collapsed;
    }

    function countDescendants(node) {
        let count = 0;
        for (const child of (node.children || [])) {
            count += 1 + countDescendants(child);
        }
        return count;
    }

    function buildNodeElement(node, isRoot, depth) {
        const li = document.createElement('li');

        const nodeDiv = document.createElement('div');
        nodeDiv.className = 'mm-node' + (isRoot ? ' root' : '') + (node.mastered ? ' mastered' : '') + (node.id === PSTATE.selectedNodeId ? ' selected' : '');
        nodeDiv.dataset.nodeId = node.id;
        if (!isRoot) {
            nodeDiv.style.borderTopColor = getDepthColor(depth);
        }

        if (PSTATE.editingNodeId === node.id) {
            const input = document.createElement('input');
            input.type = 'text';
            input.className = 'mm-node-input';
            input.value = node.name;
            nodeDiv.appendChild(input);
            setTimeout(() => { input.focus(); input.select(); }, 0);

            const commit = () => {
                const newName = input.value.trim();
                if (newName) node.name = newName;
                PSTATE.editingNodeId = null;
                if (isRoot) {
                    DOM.projectTitleInput.value = node.name;
                    PSTATE.currentProject.name = node.name;
                }
                renderMindMap();
            };
            input.addEventListener('blur', commit);
            input.addEventListener('keydown', (e) => {
                if (e.key === 'Enter') { e.preventDefault(); input.blur(); }
                if (e.key === 'Escape') { PSTATE.editingNodeId = null; renderMindMap(); }
            });
        } else {
            nodeDiv.textContent = node.name;
            if (PSTATE.generatingNodeId === node.id) {
                const spinner = document.createElement('div');
                spinner.className = 'mm-generating';
                spinner.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Génération...';
                nodeDiv.appendChild(spinner);
            }
            nodeDiv.addEventListener('click', (e) => {
                e.stopPropagation();
                selectNode(node.id);
            });
            nodeDiv.addEventListener('dblclick', (e) => {
                e.stopPropagation();
                selectNode(node.id);
                PSTATE.editingNodeId = node.id;
                renderMindMap();
            });
        }

        const hasChildren = node.children && node.children.length > 0;
        if (hasChildren) {
            const collapsed = PSTATE.collapsedNodes.has(node.id);
            const toggle = document.createElement('button');
            toggle.type = 'button';
            toggle.className = 'mm-toggle';
            toggle.title = collapsed ? 'Déplier cette branche' : 'Replier cette branche';
            toggle.innerHTML = collapsed
                ? `<span>+${countDescendants(node)}</span>`
                : '<i class="fas fa-minus"></i>';
            toggle.addEventListener('click', (e) => {
                e.stopPropagation();
                if (PSTATE.collapsedNodes.has(node.id)) {
                    PSTATE.collapsedNodes.delete(node.id);
                } else {
                    PSTATE.collapsedNodes.add(node.id);
                }
                renderMindMap();
            });
            nodeDiv.appendChild(toggle);
        }

        li.appendChild(nodeDiv);

        if (hasChildren && !PSTATE.collapsedNodes.has(node.id)) {
            const ul = document.createElement('ul');
            node.children.forEach(child => {
                ul.appendChild(buildNodeElement(child, false, depth + 1));
            });
            li.appendChild(ul);
        }

        return li;
    }

    // Calcule la position d'un élément par rapport à un ancêtre donné, en additionnant
    // offsetLeft/offsetTop le long de la chaîne offsetParent — ces valeurs sont des
    // propriétés de mise en page, donc indépendantes du zoom/déplacement (transform CSS).
    function getOffsetRelativeTo(el, ancestor) {
        let x = 0, y = 0, node = el;
        while (node && node !== ancestor) {
            x += node.offsetLeft;
            y += node.offsetTop;
            node = node.offsetParent;
        }
        return { x, y };
    }

    // Dessine les branches (courbes organiques) entre chaque nœud visible et ses enfants visibles,
    // avec une couleur qui change selon la profondeur.
    function drawMindmapConnectors() {
        const svg = DOM.mmConnectors;
        const stage = DOM.mmTreeStage;
        if (!svg || !stage) return;
        if (!PSTATE.currentProject) { svg.innerHTML = ''; return; }

        const width = stage.offsetWidth;
        const height = stage.offsetHeight;
        svg.setAttribute('width', width);
        svg.setAttribute('height', height);
        svg.setAttribute('viewBox', `0 0 ${width} ${height}`);
        svg.innerHTML = '';

        const NS = 'http://www.w3.org/2000/svg';

        const walk = (node, depth) => {
            if (!node.children || node.children.length === 0) return;
            if (PSTATE.collapsedNodes.has(node.id)) return; // branche repliée : rien à tracer en dessous

            const parentEl = DOM.mmTree.querySelector(`[data-node-id="${node.id}"]`);
            if (!parentEl) return;
            const pPos = getOffsetRelativeTo(parentEl, stage);
            const startX = pPos.x + parentEl.offsetWidth / 2;
            const startY = pPos.y + parentEl.offsetHeight;
            const color = getDepthColor(depth + 1);
            const strokeWidth = Math.max(1.6, 3.4 - depth * 0.3);

            node.children.forEach(child => {
                const childEl = DOM.mmTree.querySelector(`[data-node-id="${child.id}"]`);
                if (!childEl) return;
                const cPos = getOffsetRelativeTo(childEl, stage);
                const endX = cPos.x + childEl.offsetWidth / 2;
                const endY = cPos.y;
                const midY = startY + (endY - startY) / 2;

                const path = document.createElementNS(NS, 'path');
                path.setAttribute('d', `M ${startX} ${startY} C ${startX} ${midY}, ${endX} ${midY}, ${endX} ${endY}`);
                path.setAttribute('class', 'mm-connector');
                path.setAttribute('fill', 'none');
                path.setAttribute('stroke', color);
                path.setAttribute('stroke-width', strokeWidth);
                path.setAttribute('stroke-linecap', 'round');
                path.setAttribute('opacity', '0.9');
                svg.appendChild(path);

                walk(child, depth + 1);
            });
        };

        walk(PSTATE.currentProject.tree, 0);
    }

    function findNodeInTree(node, id) {
        if (node.id === id) return node;
        for (const child of (node.children || [])) {
            const found = findNodeInTree(child, id);
            if (found) return found;
        }
        return null;
    }

    function findParentInTree(node, id, parent = null) {
        if (node.id === id) return parent;
        for (const child of (node.children || [])) {
            const found = findParentInTree(child, id, node);
            if (found) return found;
        }
        return null;
    }

    function generateNodeId() {
        return 'n_' + Math.random().toString(36).slice(2, 10) + Date.now().toString(36).slice(-4);
    }

    function getNodePath(node, id, path = []) {
        if (node.id === id) return [...path, node.name];
        for (const child of (node.children || [])) {
            const found = getNodePath(child, id, [...path, node.name]);
            if (found) return found;
        }
        return null;
    }

    /* ---- Zoom / déplacement (molette + clic-glissé) ---- */

    function applyMindmapTransform() {
        const { x, y, scale } = PSTATE.zoom;
        DOM.mmCanvasInner.style.transform = `translate(${x}px, ${y}px) scale(${scale})`;
        DOM.mmZoomLevel.textContent = `${Math.round(scale * 100)}%`;
    }

    function clampZoomScale(scale) {
        return Math.min(Math.max(scale, 0.25), 2.5);
    }

    function zoomBy(factor, centerX = null, centerY = null) {
        const wrap = DOM.mindmapCanvasWrap;
        const rect = wrap.getBoundingClientRect();
        const cx = centerX !== null ? centerX : rect.width / 2;
        const cy = centerY !== null ? centerY : rect.height / 2;
        const zoom = PSTATE.zoom;

        const contentX = (cx - zoom.x) / zoom.scale;
        const contentY = (cy - zoom.y) / zoom.scale;

        const newScale = clampZoomScale(zoom.scale * factor);
        zoom.x = cx - contentX * newScale;
        zoom.y = cy - contentY * newScale;
        zoom.scale = newScale;

        applyMindmapTransform();
    }

    function resetMindmapView() {
        PSTATE.zoom = { scale: 1, x: 0, y: 0 };
        PSTATE.panning = null;
        PSTATE.pinch = null;
        PSTATE.pointers.clear();
        DOM.mindmapCanvasWrap.classList.remove('panning');
        applyMindmapTransform();
    }

    // Recentre la vue sur une bulle donnée sans changer le niveau de zoom
    function panToNode(nodeId) {
        const el = DOM.mmTree.querySelector(`[data-node-id="${nodeId}"]`);
        if (!el) return;
        const wrap = DOM.mindmapCanvasWrap;
        const wrapRect = wrap.getBoundingClientRect();
        const elRect = el.getBoundingClientRect();

        const elCenterX = elRect.left + elRect.width / 2 - wrapRect.left;
        const elCenterY = elRect.top + elRect.height / 2 - wrapRect.top;
        const wrapCenterX = wrapRect.width / 2;
        const wrapCenterY = wrapRect.height / 2;

        PSTATE.zoom.x += (wrapCenterX - elCenterX);
        PSTATE.zoom.y += (wrapCenterY - elCenterY);
        applyMindmapTransform();
    }

    function handleMindmapWheel(e) {
        e.preventDefault();
        const wrap = DOM.mindmapCanvasWrap;
        const rect = wrap.getBoundingClientRect();
        const mouseX = e.clientX - rect.left;
        const mouseY = e.clientY - rect.top;
        const factor = e.deltaY < 0 ? 1.12 : 1 / 1.12;
        zoomBy(factor, mouseX, mouseY);
    }

    function pointerDistance(a, b) {
        return Math.hypot(a.x - b.x, a.y - b.y);
    }

    function pointerMidpoint(a, b) {
        return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
    }

    function handleMindmapPointerDown(e) {
        // Ne pas démarrer le déplacement au 1er doigt si on touche une bulle, un champ ou un bouton
        if (PSTATE.pointers.size === 0) {
            if (e.target.closest('.mm-node') || e.target.closest('button') || e.target.closest('input')) return;
            if (e.button !== undefined && e.button !== 0) return;
        }

        PSTATE.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
        try { DOM.mindmapCanvasWrap.setPointerCapture(e.pointerId); } catch (err) {}

        if (PSTATE.pointers.size === 1) {
            // Un seul doigt (ou souris) : déplacement classique
            PSTATE.panning = {
                active: true,
                startX: e.clientX,
                startY: e.clientY,
                originX: PSTATE.zoom.x,
                originY: PSTATE.zoom.y
            };
            DOM.mindmapCanvasWrap.classList.add('panning');
        } else if (PSTATE.pointers.size === 2) {
            // Deux doigts : démarrage du pincement (zoom tactile)
            PSTATE.panning = null;
            const pts = Array.from(PSTATE.pointers.values());
            const rect = DOM.mindmapCanvasWrap.getBoundingClientRect();
            const mid = pointerMidpoint(pts[0], pts[1]);
            PSTATE.pinch = {
                startDist: pointerDistance(pts[0], pts[1]),
                startScale: PSTATE.zoom.scale,
                startZoomX: PSTATE.zoom.x,
                startZoomY: PSTATE.zoom.y,
                startCenterX: mid.x - rect.left,
                startCenterY: mid.y - rect.top
            };
        }
    }

    function handleMindmapPointerMove(e) {
        if (!PSTATE.pointers.has(e.pointerId)) return;
        PSTATE.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });

        if (PSTATE.pointers.size >= 2 && PSTATE.pinch) {
            const pts = Array.from(PSTATE.pointers.values()).slice(0, 2);
            const rect = DOM.mindmapCanvasWrap.getBoundingClientRect();
            const mid = pointerMidpoint(pts[0], pts[1]);
            const centerX = mid.x - rect.left;
            const centerY = mid.y - rect.top;

            const dist = pointerDistance(pts[0], pts[1]);
            const scaleFactor = dist / PSTATE.pinch.startDist;
            const newScale = clampZoomScale(PSTATE.pinch.startScale * scaleFactor);

            // Point du contenu situé sous le centre du pincement au moment du démarrage
            const contentX = (PSTATE.pinch.startCenterX - PSTATE.pinch.startZoomX) / PSTATE.pinch.startScale;
            const contentY = (PSTATE.pinch.startCenterY - PSTATE.pinch.startZoomY) / PSTATE.pinch.startScale;

            PSTATE.zoom.scale = newScale;
            PSTATE.zoom.x = centerX - contentX * newScale;
            PSTATE.zoom.y = centerY - contentY * newScale;
            applyMindmapTransform();
            return;
        }

        if (!PSTATE.panning || !PSTATE.panning.active) return;
        const dx = e.clientX - PSTATE.panning.startX;
        const dy = e.clientY - PSTATE.panning.startY;
        PSTATE.zoom.x = PSTATE.panning.originX + dx;
        PSTATE.zoom.y = PSTATE.panning.originY + dy;
        applyMindmapTransform();
    }

    function handleMindmapPointerUp(e) {
        PSTATE.pointers.delete(e.pointerId);
        try { DOM.mindmapCanvasWrap.releasePointerCapture(e.pointerId); } catch (err) {}

        if (PSTATE.pointers.size < 2) {
            PSTATE.pinch = null;
        }

        if (PSTATE.pointers.size === 1) {
            // Il reste un doigt posé : reprendre le déplacement sans à-coup depuis sa position actuelle
            const [, remaining] = Array.from(PSTATE.pointers.entries())[0];
            PSTATE.panning = {
                active: true,
                startX: remaining.x,
                startY: remaining.y,
                originX: PSTATE.zoom.x,
                originY: PSTATE.zoom.y
            };
        } else if (PSTATE.pointers.size === 0) {
            PSTATE.panning = null;
            DOM.mindmapCanvasWrap.classList.remove('panning');
        }
    }

    function setupMindmapPanZoom() {
        const wrap = DOM.mindmapCanvasWrap;
        wrap.addEventListener('wheel', handleMindmapWheel, { passive: false });
        wrap.addEventListener('pointerdown', handleMindmapPointerDown);
        wrap.addEventListener('pointermove', handleMindmapPointerMove);
        wrap.addEventListener('pointerup', handleMindmapPointerUp);
        wrap.addEventListener('pointercancel', handleMindmapPointerUp);
        wrap.addEventListener('pointerleave', (e) => {
            // Ne relâcher que si le pointeur quitte vraiment la zone sans bouton enfoncé (évite de casser un pincement en cours)
            if (e.buttons === 0 && e.pointerType !== 'touch') handleMindmapPointerUp(e);
        });
        wrap.addEventListener('dblclick', (e) => {
            if (e.target.closest('.mm-node')) return; // le double-clic sur une bulle sert au renommage
            resetMindmapView();
        });
    }

    function handleDeleteSelectedNode() {
        const node = getSelectedNode();
        if (!node || !PSTATE.currentProject) return;
        if (node.id === PSTATE.currentProject.tree.id) return; // sécurité : racine non supprimable ici

        const childCount = countAllNodes(node) - 1;
        const warning = childCount > 0
            ? `Supprimer "${node.name}" et ses ${childCount} sous-élément(s) ?`
            : `Supprimer "${node.name}" ?`;
        if (!confirm(warning)) return;

        const parent = findParentInTree(PSTATE.currentProject.tree, node.id);
        if (parent) {
            parent.children = parent.children.filter(c => c.id !== node.id);
        }
        PSTATE.selectedNodeId = null;
        renderMindMap();
        updateToolbarState();
    }

    function countAllNodes(node) {
        let total = 1;
        for (const child of (node.children || [])) total += countAllNodes(child);
        return total;
    }

    async function handleGenerateSegments() {
        const node = getSelectedNode();
        if (!node || !PSTATE.currentProject) return;

        const path = getNodePath(PSTATE.currentProject.tree, node.id) || [];
        path.pop(); // retire le thème lui-même, ne garde que les ancêtres pour le contexte

        PSTATE.generatingNodeId = node.id;
        renderMindMap();
        DOM.mmGenerateBtn.disabled = true;

        try {
            const result = await projectsApiCall('/generate_segments', {
                method: 'POST',
                body: JSON.stringify({
                    username: STATE.currentUser,
                    theme: node.name,
                    path: path,
                    ollama_url: PSTATE.ai.url,
                    model: PSTATE.ai.model
                })
            });

            const existingNames = new Set((node.children || []).map(c => c.name.toLowerCase()));
            const newChildren = result.segments
                .filter(name => !existingNames.has(name.toLowerCase()))
                .map(name => ({ id: generateNodeId(), name, mastered: false, children: [] }));

            node.children = node.children || [];
            node.children.push(...newChildren);
            PSTATE.collapsedNodes.delete(node.id); // on déplie pour révéler les sous-thèmes générés

            showToast(`${newChildren.length} sous-thème(s) généré(s)`, 'success');
        } catch (error) {
            showToast(error.message, 'error');
        } finally {
            PSTATE.generatingNodeId = null;
            renderMindMap();
            updateToolbarState();
        }
    }
