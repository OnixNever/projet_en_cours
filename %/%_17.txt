    // --- DICTÉE & AUDIO ---
    function initAudioContext() {
        if (!audioCtx) {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            gainNodes = {};
            
            if (audioCtx.state === 'suspended') {
                const resumeAudio = async () => {
                    await audioCtx.resume();
                    console.log('AudioContext résumé');
                };
                resumeAudio();
            }
            
            document.addEventListener('click', async () => {
                if (audioCtx.state === 'suspended') {
                    await audioCtx.resume();
                }
            });
        }
        return audioCtx;
    }

    function boostAudio(id, multiplier) {
        try {
            const audioElement = document.getElementById(`audio-${id}`);
            if (!audioElement) return;
            
            const ctx = initAudioContext();
            
            if (!gainNodes[id]) {
                const source = ctx.createMediaElementSource(audioElement);
                const gainNode = ctx.createGain();
                source.connect(gainNode);
                gainNode.connect(ctx.destination);
                gainNodes[id] = gainNode;
            }
            
            const gainValue = parseFloat(multiplier);
            gainNodes[id].gain.value = gainValue;
            audioElement.volume = Math.min(gainValue, 5) / 5;
            
            console.log(`Boost audio ${id}: x${gainValue}`);
            showNotification(`Volume boosté x${gainValue}`, 'info', 1000);
        } catch (error) {
            console.error('Erreur boost audio:', error);
            const audioElement = document.getElementById(`audio-${id}`);
            if (audioElement) {
                audioElement.volume = Math.min(parseFloat(multiplier), 5) / 5;
            }
        }
    }

    // Initialisation de la reconnaissance vocale
    function initSpeechRecognition() {
        if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            recognition = new SpeechRecognition();
            recognition.lang = 'fr-FR';
            recognition.continuous = true;
            recognition.interimResults = true;
            recognition.maxAlternatives = 1;
            
            recognition.onresult = (e) => {
                let interim = '';
                for (let i = e.resultIndex; i < e.results.length; ++i) {
                    if (e.results[i].isFinal) {
                        const transcript = e.results[i][0].transcript;
                        finalTranscriptAccumulated += transcript + " ";
                        speakText(transcript);
                    } else {
                        interim += e.results[i][0].transcript;
                    }
                }
                document.getElementById('interimText').innerText = finalTranscriptAccumulated + interim;
            };
            
            recognition.onend = () => {
                if (isDictating) {
                    setTimeout(() => recognition.start(), 100);
                }
            };
            
            recognition.onerror = (e) => {
                console.error('Erreur reconnaissance vocale:', e.error);
                if (e.error === 'not-allowed') {
                    showNotification('Microphone non autorisé', 'error');
                }
            };
        }
    }

    // Lecture fluide des messages texte
    function speakText(text) {
        if ('speechSynthesis' in window && text.trim()) {
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'fr-FR';
            utterance.rate = 1.0;
            utterance.pitch = 1.0;
            utterance.volume = 1.0;
            
            const voices = speechSynthesis.getVoices();
            const frenchVoice = voices.find(voice => voice.lang === 'fr-FR' && voice.name.includes('female'));
            if (frenchVoice) {
                utterance.voice = frenchVoice;
            }
            
            speechSynthesis.speak(utterance);
        }
    }

    async function toggleAudio() {
        if (isRecording) {
            mediaRecorder.stop();
            isRecording = false;
            document.getElementById('btnRecord').classList.remove('btn-stop');
            document.getElementById('btnRecord').innerHTML = '<i class="fas fa-microphone"></i> Audio';
            showNotification('Enregistrement arrêté', 'info');
        } else {
            try {
                const stream = await navigator.mediaDevices.getUserMedia({ 
                    audio: {
                        echoCancellation: true,
                        noiseSuppression: true,
                        autoGainControl: true,
                        channelCount: 1,
                        sampleRate: 44100
                    } 
                });
                
                initAudioContext();
                
                mediaRecorder = new MediaRecorder(stream, {
                    mimeType: 'audio/webm;codecs=opus'
                });
                chunks = [];
                
                mediaRecorder.ondataavailable = e => {
                    if (e.data.size > 0) chunks.push(e.data);
                };
                
                mediaRecorder.onstop = () => {
                    const b = new Blob(chunks, { type: 'audio/webm' });
                    const r = new FileReader();
                    r.readAsDataURL(b);
                    r.onloadend = () => {
                        const newNote = {
                            id: Date.now(),
                            type: 'audio',
                            content: r.result,
                            date: new Date().toLocaleString(),
                            name: "Enregistrement " + new Date().toLocaleTimeString('fr-FR', {hour: '2-digit', minute:'2-digit'})
                        };
                        notes.unshift(newNote);
                        saveNotes();
                        
                        setTimeout(() => {
                            boostAudio(newNote.id, '1');
                            const audioElement = document.getElementById(`audio-${newNote.id}`);
                            if (audioElement) {
                                audioElement.play().catch(e => console.log('Lecture auto ignorée:', e));
                            }
                        }, 500);
                    };
                    
                    stream.getTracks().forEach(track => track.stop());
                };
                
                mediaRecorder.onerror = (e) => {
                    console.error('Erreur enregistrement:', e);
                    showNotification('Erreur d\'enregistrement', 'error');
                };
                
                mediaRecorder.start();
                isRecording = true;
                document.getElementById('btnRecord').classList.add('btn-stop');
                document.getElementById('btnRecord').innerHTML = '<i class="fas fa-stop"></i> Arrêter';
                switchTab('audio');
                showNotification('Enregistrement démarré', 'success');
                
            } catch (error) {
                console.error('Erreur accès microphone:', error);
                if (error.name === 'NotAllowedError') {
                    showNotification('Permission microphone refusée', 'error');
                } else if (error.name === 'NotFoundError') {
                    showNotification('Microphone non trouvé', 'error');
                } else {
                    showNotification('Erreur d\'accès au microphone', 'error');
                }
            }
        }
    }

        // Lignes de données
        table.rows.forEach((row, rowIndex) => {
            html += `<tr data-row-id="${row.id}" data-row-index="${rowIndex}"
                         draggable="true"
                         ondragstart="onRowDragStart(event, ${row.id}, ${rowIndex})"
                         ondragover="onRowDragOver(event)"
                         ondragend="onRowDragEnd(event)"
                         ondragenter="onRowDragEnter(event)"
                         ondragleave="onRowDragLeave(event)">`;
            
            // Première cellule
            html += `<td><div class="cell-container">`;
            
            if (numberingType === 'dates') {
                const dateValue = row.date || formatDate(new Date());
                if (!row.date) row.date = dateValue;
                html += `<div class="date-display" onclick="openDatepicker(${row.id})" data-row-id="${row.id}">
                            <i class="fas fa-calendar-alt"></i> ${dateValue}
                        </div>`;
            } else {
                html += `<span class="number-display">${rowIndex + 1}</span>`;
            }
            
            html += `<button class="cell-menu-btn" onclick="showRowMenu(${row.id}, event)">
                        <i class="fas fa-ellipsis-v"></i>
                    </button>
                </div></td>`;
            
            // Cellules de données
            table.columns.forEach(column => {
                const cell = row.cells.find(c => c.columnId === column.id);
                const cellValue = cell ? cell.value : '';
                const inputType = column.type === 'number' ? 'number' : 'text';
                
                html += `<td>
                    <input type="${inputType}" class="table-cell-input"
                           value="${cellValue}"
                           onchange="updateCellValue(${row.id}, ${column.id}, this.value)"
                           placeholder="${column.type === 'number' ? '0' : '...'}">
                </td>`;
            });
            
            html += `</tr>`;
        });
        
        // Ligne d'ajout
        html += `<tr>
            <td colspan="${table.columns.length + 1}" style="text-align: center; padding: 15px;">
                <button onclick="addTableRow()" class="secondary-btn w-full">
                    <i class="fas fa-plus"></i> Ajouter une ligne
                </button>
            </td>
        </tr></tbody></table>`;
        
        container.innerHTML = html;
    }

    function startResize(event, columnId) {
        event.preventDefault();
        event.stopPropagation();
        
        const th = event.target.closest('th');
        if (!th) return;
        
        currentResizeColumn = {
            id: columnId,
            element: th,
            startWidth: th.offsetWidth
        };
        
        currentResizeStartX = event.clientX;
        currentResizeStartWidth = th.offsetWidth;
        
        document.body.classList.add('resizing-cursor');
        th.classList.add('resizing');
        event.target.classList.add('resizing');
    }

    function onResize(event) {
        if (!currentResizeColumn) return;
        event.preventDefault();
        
        const dx = event.clientX - currentResizeStartX;
        let newWidth = Math.max(80, currentResizeStartWidth + dx);
        
        currentResizeColumn.element.style.width = newWidth + 'px';
        currentResizeColumn.element.style.minWidth = newWidth + 'px';
        currentResizeColumn.element.style.maxWidth = newWidth + 'px';
        
        const columnIndex = currentResizeColumn.element.cellIndex;
        const table = document.getElementById('main-data-table');
        if (table) {
            const rows = table.querySelectorAll('tbody tr');
            rows.forEach(row => {
                const cell = row.cells[columnIndex];
                if (cell) {
                    cell.style.width = newWidth + 'px';
                    cell.style.minWidth = newWidth + 'px';
                    cell.style.maxWidth = newWidth + 'px';
                }
            });
        }
    }

    function stopResize(event) {
        if (currentResizeColumn) {
            document.body.classList.remove('resizing-cursor');
            currentResizeColumn.element.classList.remove('resizing');
            
            const table = tables.find(t => t.id === currentTableId);
            if (table) {
                if (!table.columnWidths) table.columnWidths = {};
                table.columnWidths[currentResizeColumn.id] = currentResizeColumn.element.offsetWidth;
                saveTables();
            }
            currentResizeColumn = null;
        }
    }

    // DRAG & DROP
    function onColumnDragStart(event, columnId, index) {
        if (event.target.classList.contains('column-name-input') || 
            event.target.classList.contains('menu-btn') ||
            event.target.classList.contains('resize-handle')) {
            event.preventDefault();
            return false;
        }
        
        dragSourceType = 'column';
        dragSourceId = columnId;
        dragSourceIndex = index;
        
        event.dataTransfer.effectAllowed = 'move';
        event.dataTransfer.setData('text/plain', columnId);
        event.target.classList.add('dragging');
    }

    function onColumnDragOver(event) {
        event.preventDefault();
        event.dataTransfer.dropEffect = 'move';
    }

    function onColumnDragEnter(event) {
        const th = event.target.closest('th');
        if (!th || !th.dataset.columnId) return;
        
        th.classList.remove('drop-target-left', 'drop-target-right');
        
        const rect = th.getBoundingClientRect();
        const mouseX = event.clientX;
        const threshold = rect.left + rect.width / 2;
        
        if (mouseX < threshold) {
            th.classList.add('drop-target-left');
        } else {
            th.classList.add('drop-target-right');
        }
    }

    function onColumnDragLeave(event) {
        const th = event.target.closest('th');
        if (th) th.classList.remove('drop-target-left', 'drop-target-right');
    }

    function onColumnDragEnd(event) {
        document.querySelectorAll('th').forEach(th => {
            th.classList.remove('dragging', 'drop-target-left', 'drop-target-right');
        });
        dragSourceType = null;
        dragSourceId = null;
        dragSourceIndex = null;
    }

    function onRowDragStart(event, rowId, index) {
        if (event.target.classList.contains('cell-menu-btn') || 
            event.target.classList.contains('date-display')) {
            event.preventDefault();
            return false;
        }
        
        dragSourceType = 'row';
        dragSourceId = rowId;
        dragSourceIndex = index;
        
        event.dataTransfer.effectAllowed = 'move';
        event.dataTransfer.setData('text/plain', rowId);
        event.target.closest('tr').classList.add('dragging');
    }

    function onRowDragOver(event) {
        event.preventDefault();
        event.dataTransfer.dropEffect = 'move';
    }

    function onRowDragEnter(event) {
        const tr = event.target.closest('tr');
        if (!tr || !tr.dataset.rowId) return;
        
        tr.classList.remove('drop-target-above', 'drop-target-below');
        
        const rect = tr.getBoundingClientRect();
        const mouseY = event.clientY;
        const threshold = rect.top + rect.height / 2;
        
        if (mouseY < threshold) {
            tr.classList.add('drop-target-above');
        } else {
            tr.classList.add('drop-target-below');
        }
    }

    function onRowDragLeave(event) {
        const tr = event.target.closest('tr');
        if (tr) tr.classList.remove('drop-target-above', 'drop-target-below');
    }

    function onRowDragEnd(event) {
        document.querySelectorAll('tr').forEach(tr => {
            tr.classList.remove('dragging', 'drop-target-above', 'drop-target-below');
        });
        dragSourceType = null;
        dragSourceId = null;
        dragSourceIndex = null;
    }

    function handleDrop(event) {
        event.preventDefault();
        
        if (!currentTableId || !dragSourceType) return;
        
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        if (dragSourceType === 'column') {
            const targetTh = event.target.closest('th');
            if (!targetTh || !targetTh.dataset.columnId) return;
            
            const targetColumnId = parseInt(targetTh.dataset.columnId);
            const sourceColumnId = dragSourceId;
            
            if (sourceColumnId === targetColumnId) return;
            
            const sourceIndex = table.columns.findIndex(col => col.id === sourceColumnId);
            const targetIndex = table.columns.findIndex(col => col.id === targetColumnId);
            
            if (sourceIndex === -1 || targetIndex === -1) return;
            
            const rect = targetTh.getBoundingClientRect();
            const mouseX = event.clientX;
            let insertIndex = targetIndex;
            
            if (mouseX > rect.left + rect.width / 2) insertIndex = targetIndex + 1;
            
            const [movedColumn] = table.columns.splice(sourceIndex, 1);
            if (sourceIndex < insertIndex) insertIndex--;
            table.columns.splice(insertIndex, 0, movedColumn);
            
            table.rows.forEach(row => {
                const cells = [...row.cells];
                const movedCellIndex = cells.findIndex(cell => cell.columnId === sourceColumnId);
                const [movedCell] = cells.splice(movedCellIndex, 1);
                
                let targetCellIndex = cells.findIndex(cell => cell.columnId === targetColumnId);
                if (insertIndex > targetIndex) targetCellIndex++;
                
                cells.splice(targetCellIndex, 0, movedCell);
                row.cells = cells;
            });
            
            saveTables();
            renderCurrentTable();
            showNotification('Colonne déplacée', 'success');
            
        } else if (dragSourceType === 'row') {
            const targetTr = event.target.closest('tr');
            if (!targetTr || !targetTr.dataset.rowId) return;
            
            const targetRowId = parseInt(targetTr.dataset.rowId);
            const sourceRowId = dragSourceId;
            
            if (sourceRowId === targetRowId) return;
            
            const sourceIndex = table.rows.findIndex(row => row.id === sourceRowId);
            const targetIndex = table.rows.findIndex(row => row.id === targetRowId);
            
            if (sourceIndex === -1 || targetIndex === -1) return;
            
            const rect = targetTr.getBoundingClientRect();
            const mouseY = event.clientY;
            let insertIndex = targetIndex;
            
            if (mouseY > rect.top + rect.height / 2) insertIndex = targetIndex + 1;
            
            const [movedRow] = table.rows.splice(sourceIndex, 1);
            if (sourceIndex < insertIndex) insertIndex--;
            table.rows.splice(insertIndex, 0, movedRow);
            
            saveTables();
            renderCurrentTable();
            showNotification('Ligne déplacée', 'success');
        }
        
        document.querySelectorAll('th, tr').forEach(el => {
            el.classList.remove('drop-target-left', 'drop-target-right', 'drop-target-above', 'drop-target-below');
        });
        
        dragSourceType = null;
        dragSourceId = null;
        dragSourceIndex = null;
    }

    function openDatepicker(rowId) {
        const dateDisplay = document.querySelector(`.date-display[data-row-id="${rowId}"]`);
        if (!dateDisplay) return;
        
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const row = table.rows.find(r => r.id === rowId);
        if (!row) return;
        
        let currentDate = new Date();
        if (row.date) {
            const parts = row.date.split('/');
            if (parts.length === 3) {
                currentDate = new Date(parseInt(parts[2]), parseInt(parts[1]) - 1, parseInt(parts[0]));
            }
        }
        
        const input = document.createElement('input');
        input.type = 'text';
        input.style.position = 'absolute';
        input.style.opacity = '0';
        input.style.pointerEvents = 'none';
        document.body.appendChild(input);
        
        const datepicker = new Datepicker(input, {
            language: 'fr',
            format: 'dd/mm/yyyy',
            autohide: true,
            todayHighlight: true
        });
        
        datepicker.setDate(currentDate);
        
        const rect = dateDisplay.getBoundingClientRect();
        input.style.left = rect.left + 'px';
        input.style.top = rect.top + 'px';
        
        datepicker.show();
        
        input.addEventListener('changeDate', function(e) {
            const date = e.date;
            if (date) {
                const formattedDate = formatDate(date);
                row.date = formattedDate;
                saveTables();
                dateDisplay.innerHTML = `<i class="fas fa-calendar-alt"></i> ${formattedDate}`;
                showNotification(`Date mise à jour: ${formattedDate}`, 'success');
            }
            setTimeout(() => {
                datepicker.destroy();
                document.body.removeChild(input);
            }, 100);
        });
        
        setTimeout(() => {
            if (document.body.contains(input)) {
                datepicker.destroy();
                document.body.removeChild(input);
            }
        }, 30000);
    }

    function showContextMenu(x, y) {
        const contextMenu = document.getElementById('contextMenu');
        contextMenu.style.left = x + 'px';
        contextMenu.style.top = y + 'px';
        contextMenu.classList.add('show');
    }

    function handleMenuAction(action) {
        document.getElementById('contextMenu').classList.remove('show');
        
        if (!currentTableId || !currentMenuContext.id) return;
        
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        if (currentMenuContext.type === 'column') {
            if (currentMenuContext.id === 0) {
                if (action === 'delete') showNotification('Impossible de supprimer la colonne #', 'warning');
                return;
            }
            if (action === 'duplicate') duplicateColumn(currentMenuContext.id);
            else if (action === 'clear') clearColumn(currentMenuContext.id);
            else if (action === 'delete') deleteColumn(currentMenuContext.id);
        } else if (currentMenuContext.type === 'row') {
            if (action === 'duplicate') duplicateRow(currentMenuContext.id);
            else if (action === 'clear') clearRow(currentMenuContext.id);
            else if (action === 'delete') deleteRow(currentMenuContext.id);
        }
    }

    function getNextColumnNumber() {
        if (!currentTableId) return 1;
        const table = tables.find(t => t.id === currentTableId);
        return table ? table.columns.length + 1 : 1;
    }

    function addTableColumn(columnName, type) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const newColumn = { id: Date.now(), name: columnName, type: type };
        table.columns.push(newColumn);
        
        table.rows.forEach(row => {
            row.cells.push({ columnId: newColumn.id, value: type === 'number' ? '0' : '' });
        });
        
        saveTables();
        renderCurrentTable();
        showNotification(`Catégorie "${columnName}" ajoutée`, 'success');
    }

    function duplicateColumn(columnId) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const column = table.columns.find(col => col.id === columnId);
        if (!column) return;
        
        const newColumn = { id: Date.now(), name: `${column.name} (copie)`, type: column.type };
        table.columns.push(newColumn);
        
        table.rows.forEach(row => {
            const originalCell = row.cells.find(c => c.columnId === column.id);
            row.cells.push({
                columnId: newColumn.id,
                value: originalCell ? originalCell.value : (column.type === 'number' ? '0' : '')
            });
        });
        
        saveTables();
        renderCurrentTable();
        showNotification('Colonne dupliquée', 'success');
    }

    function clearColumn(columnId) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const column = table.columns.find(col => col.id === columnId);
        if (!column) return;
        
        table.rows.forEach(row => {
            const cell = row.cells.find(c => c.columnId === column.id);
            if (cell) cell.value = column.type === 'number' ? '0' : '';
        });
        
        saveTables();
        renderCurrentTable();
        showNotification('Contenu de la colonne effacé', 'success');
    }

    function deleteColumn(columnId) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table || table.columns.length <= 1) {
            showNotification('Un tableau doit avoir au moins une colonne', 'warning');
            return;
        }
        
        table.columns = table.columns.filter(col => col.id !== columnId);
        table.rows.forEach(row => {
            row.cells = row.cells.filter(cell => cell.columnId !== columnId);
        });
        
        if (table.columnWidths) delete table.columnWidths[columnId];
        
        saveTables();
        renderCurrentTable();
        showNotification('Colonne supprimée', 'success');
    }

    function updateColumnName(columnId, newName) {
        if (!currentTableId || !newName) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const column = table.columns.find(col => col.id === columnId);
        if (column) {
            column.name = newName;
            saveTables();
            showNotification('Nom de colonne mis à jour', 'success');
        }
    }

    function addTableRow() {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const todayFormatted = formatDate(new Date());
        const newRow = {
            id: Date.now(),
            date: todayFormatted,
            cells: table.columns.map(column => ({
                columnId: column.id,
                value: column.type === 'number' ? '0' : ''
            }))
        };
        
        table.rows.push(newRow);
        saveTables();
        renderCurrentTable();
        showNotification('Ligne ajoutée', 'success');
    }

    function duplicateRow(rowId) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const row = table.rows.find(r => r.id === rowId);
        if (!row) return;
        
        const newRow = {
            id: Date.now(),
            date: formatDate(new Date()),
            cells: JSON.parse(JSON.stringify(row.cells))

        table.rows.push(newRow);
        saveTables();
        renderCurrentTable();

    function clearRow(rowId) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;
        
        const row = table.rows.find(r => r.id === rowId);
        if (!row) return;

            if (column) cell.value = column.type === 'number' ? '0' : '';
        });
        
        saveTables();
        renderCurrentTable();
        showNotification('Contenu de la ligne effacé', 'success');
    }

    function deleteRow(rowId) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table || table.rows.length <= 1) {
            showNotification('Un tableau doit avoir au moins une ligne', 'warning');
            return;
        }
        
        table.rows = table.rows.filter(r => r.id !== rowId);
        saveTables();
        renderCurrentTable();
        showNotification('Ligne supprimée', 'success');
    }

    function duplicateCurrentRow() {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table || table.rows.length === 0) return;
        
        const lastRow = table.rows[table.rows.length - 1];
        const newRow = {
            id: Date.now(),
            date: formatDate(new Date()),
            cells: JSON.parse(JSON.stringify(lastRow.cells))
        };
        
        table.rows.push(newRow);
        saveTables();
        renderCurrentTable();
        showNotification('Ligne dupliquée', 'success');
    }

    function updateCellValue(rowId, columnId, newValue) {
        if (!currentTableId) return;
        const table = tables.find(t => t.id === currentTableId);
        if (!table) return;

        if (row) {
            const cell = row.cells.find(c => c.columnId === columnId);
            if (cell) {
                cell.value = newValue;
                saveTables();
            }
        }
    }

    function deleteTable(tableId, event = null) {
        if (event) event.stopPropagation();
        
        if (tables.length <= 1) {
            showNotification('Vous devez garder au moins un tableau', 'warning');
            return;
        }
        
        const table = tables.find(t => t.id === tableId);
        if (!table) return;
        
        if (confirm(`Supprimer le tableau "${table.name}" ?`)) {
            tables = tables.filter(t => t.id !== tableId);
            if (currentTableId === tableId) {
                currentTableId = tables.length > 0 ? tables[0].id : null;
            }
            
            saveTables();
            renderTableTabs();
            renderCurrentTable();
            showNotification('Tableau supprimé', 'success');
        }
    }

                { id: Date.now() + 2, name: 'Valeur', type: 'number' }

            rows: [
                { id: Date.now() + 3, date: formatDate(new Date()), cells: [] }

        saveTables();
        renderTableTabs();
        renderCurrentTable();
        showNotification(`Nouveau tableau créé: ${newTable.name}`, 'success');
    }

    function exportTableauData() {
        if (tables.length === 0) {
            showNotification('Aucun tableau à exporter', 'warning');
            return;
        }
        
        const exportData = {
            tables: tables,
            currentTableId: currentTableId,

        };

        const fileName = `tableaux_${new Date().toISOString().split('T')[0]}.json`;
        
        const link = document.createElement('a');
        link.href = dataUri;
        link.download = fileName;
        link.click();

    }

    function importTableauData() {
        document.getElementById('tableauFileInput').onchange = function(event) {
            const file = event.target.files[0];
            if (!file) return;
            
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const importedData = JSON.parse(e.target.result);
                    if (!importedData.tables || !Array.isArray(importedData.tables)) {
                        throw new Error('Format invalide');
                    }
                    
                    if (confirm(`Importer ${importedData.tables.length} tableau(x) ?\nLes données actuelles seront remplacées.`)) {
                        tables = importedData.tables;
                        currentTableId = importedData.currentTableId || (tables.length > 0 ? tables[0].id : null);
                        tableCounter = importedData.tableCounter || 1;
                        numberingType = importedData.numberingType || 'numbers';
                        
                        saveTables();
                        renderTableTabs();
                        renderCurrentTable();
                        showNotification(`${tables.length} tableau(x) importé(s)`, 'success');
                    }
                } catch (error) {
                    showNotification('Erreur lors de l\'importation', 'error');
                }
            };
            reader.readAsText(file);
            event.target.value = '';
        };
        document.getElementById('tableauFileInput').click();
    }

    // CALCULS RAPIDES
    // ============================================
    function updateCalculations() {
        const baseValue = parseFloat(document.getElementById('baseValue').value) || 0;
        const period = document.getElementById('timePeriod').value;
        const workingDays = parseInt(document.getElementById('workingDays').value) || 5;
        const hoursPerDay = parseFloat(document.getElementById('hoursPerDay').value) || 8;
        const holidays = parseInt(document.getElementById('holidaysPerYear').value) || 25;
        
        const workingDaysPerYear = (52 * workingDays) - holidays;
        const workingDaysPerMonth = (52 / 12) * workingDays;
        const workingDaysPerQuarter = workingDaysPerMonth * 3;
        const workingDaysPer5Years = workingDaysPerYear * 5;
        
        let conversions = {};
        
        switch(period) {
            case 'hour':
                conversions.hour = baseValue;
                conversions.day = baseValue * hoursPerDay;
                conversions.week = conversions.day * workingDays;
                conversions.month = conversions.week * (52 / 12);
                conversions.quarter = conversions.month * 3;
                conversions.year = conversions.day * workingDaysPerYear;
                conversions.fiveYears = conversions.year * 5;
                break;
            case 'day':
                conversions.hour = baseValue / hoursPerDay;
                conversions.day = baseValue;
                conversions.week = baseValue * workingDays;
                conversions.month = baseValue * workingDaysPerMonth;
                conversions.quarter = baseValue * workingDaysPerQuarter;
                conversions.year = baseValue * workingDaysPerYear;
                conversions.fiveYears = baseValue * workingDaysPer5Years;
                break;
            case 'week':
                conversions.hour = baseValue / (workingDays * hoursPerDay);
                conversions.day = baseValue / workingDays;
                conversions.week = baseValue;
                conversions.month = baseValue * (52 / 12);
                conversions.quarter = baseValue * 13;
                conversions.year = baseValue * 52;
                conversions.fiveYears = baseValue * 260;
                break;
            case 'month':
                conversions.hour = baseValue / (workingDaysPerMonth * hoursPerDay);
                conversions.day = baseValue / workingDaysPerMonth;
                conversions.week = baseValue / (workingDaysPerMonth / workingDays);
                conversions.month = baseValue;
                conversions.quarter = baseValue * 3;
                conversions.year = baseValue * 12;
                conversions.fiveYears = baseValue * 60;
                break;
            case 'quarter':
                conversions.hour = baseValue / (workingDaysPerQuarter * hoursPerDay);
                conversions.day = baseValue / workingDaysPerQuarter;
                conversions.week = baseValue / (workingDaysPerQuarter / workingDays);
                conversions.month = baseValue / 3;
                conversions.quarter = baseValue;
                conversions.year = baseValue * 4;
                conversions.fiveYears = baseValue * 20;
                break;
            case 'year':
                conversions.hour = baseValue / (workingDaysPerYear * hoursPerDay);
                conversions.day = baseValue / workingDaysPerYear;
                conversions.week = baseValue / (workingDaysPerYear / workingDays);
                conversions.month = baseValue / 12;
                conversions.quarter = baseValue / 4;
                conversions.year = baseValue;
                conversions.fiveYears = baseValue * 5;
                break;
            case 'fiveYears':
                conversions.hour = baseValue / (workingDaysPer5Years * hoursPerDay);
                conversions.day = baseValue / workingDaysPer5Years;
                conversions.week = baseValue / (workingDaysPer5Years / workingDays);
                conversions.month = baseValue / 60;
                conversions.quarter = baseValue / 20;
                conversions.year = baseValue / 5;
                conversions.fiveYears = baseValue;
                break;
        }
        
        const periodLabels = { 
            hour: 'par heure', 
            day: 'par jour', 
            week: 'par semaine', 
            month: 'par mois', 
            quarter: 'par trimestre',
            year: 'par an', 
            fiveYears: 'par 5 ans' 
        };
        document.getElementById('calcResultValue').textContent = `${formatNumber(baseValue)} € ${periodLabels[period]}`;
        
        const detailsContainer = document.getElementById('calcResultDetails');
        detailsContainer.innerHTML = '';
        
        const periods = [
            { key: 'hour', label: 'Par heure' },
            { key: 'day', label: 'Par jour' },
            { key: 'week', label: 'Par semaine' },
            { key: 'month', label: 'Par mois' },
            { key: 'quarter', label: 'Par trimestre' },
            { key: 'year', label: 'Par an' },
            { key: 'fiveYears', label: 'Par 5 ans' }
        ];
        
        periods.forEach(p => {
            if (conversions[p.key] !== undefined) {
                const item = document.createElement('div');
                item.className = 'detail-item';
                item.innerHTML = `<span class="detail-label">${p.label}</span>
                                 <span class="detail-value">${formatNumber(conversions[p.key])} €</span>`;
                detailsContainer.appendChild(item);
            }
        });
    }

    function updateRecentCalculations() {

            ...calculatorHistory.slice(0, 3).map(item => ({

            })),
            ...conversionHistory.slice(0, 2).map(item => ({
                type: 'devise',
                title: `Conversion: ${item.from.amount} ${item.from.currency} → ${item.to.currency}`,
                value: item.to.amount,
                timestamp: item.date,
                action: () => showMode('modeH')
            }))
        ].sort((a, b) => new Date(b.timestamp || b.date) - new Date(a.timestamp || a.date)).slice(0, 5);
        
        if (allRecent.length === 0) {
            container.innerHTML = `
                <div style="text-align: center; padding: 40px 20px; color: var(--text-muted);">
                    <i class="fas fa-history fa-2x" style="margin-bottom: 15px; opacity: 0.5;"></i>
                    <p>Aucun calcul récent</p>
                </div>
            `;
            return;
        }
        
        let html = '';
        allRecent.forEach(item => {
            const date = item.timestamp || item.date || 'Récemment';
            const value = typeof item.value === 'number' ? item.value.toFixed(2) : item.value;
            
            html += `
                <div class="recent-item" onclick="showMode('${item.action ? item.action().replace('showMode(', '').replace(')', '') : 'modeCalculator'}')">
                    <div>
                        <div style="font-weight: bold;">${item.title}</div>
                        <div style="font-size: 12px; color: var(--text-muted);">${date}</div>
                    </div>
                    <div>
                        <span class="recent-type">${item.type}</span>
                        <div style="font-weight: bold; color: var(--accent-color); text-align: right; margin-top: 5px;">${value}</div>
                    </div>
                </div>
            `;
        });
        
        container.innerHTML = html;
    }

    function quickCalculation(type) {
        switch(type) {
            case 'percentage':
                const value = prompt('Entrez la valeur :', '100');
                const percentage = prompt('Entrez le pourcentage :', '20');
                if (value && percentage) {
                    const result = (parseFloat(value) * parseFloat(percentage)) / 100;
                    showMode('modeCalculator');
                    setTimeout(() => {
                        addToExpression(`${value}*${percentage}/100`);
                        calculateResult();
                        addRecentCalculation('pourcentage', 
                            `Calcul de ${percentage}% de ${value}`, 
                            result
                        );
                    }, 100);
                }
                break;
                
            case 'vat':
                const ht = prompt('Montant HT :', '100');
                const tva = prompt('Taux TVA (%) :', '20');
                if (ht && tva) {
                    const ttc = parseFloat(ht) * (1 + parseFloat(tva)/100);

                        addRecentCalculation('tva', 
                            `TVA: ${ht} HT → ${ttc.toFixed(2)} TTC (${tva}%)`,

                const amount = prompt('Montant à convertir :', '100');

    function initKeyboardShortcuts() {
        document.addEventListener('keydown', function(event) {
            const key = getKeyCombination(event);
            
            const shortcut = keyboardShortcuts.find(s => s.key === key);
            if (shortcut) {
                event.preventDefault();
                showMode(shortcut.mode);
                showNotification(`Raccourci activé : ${shortcut.name}`, 'info');
            }
        });
        
        updateShortcutsDisplay();
    }

    function getKeyCombination(event) {
        const keys = [];
        if (event.ctrlKey || event.metaKey) keys.push('Ctrl');
        if (event.altKey) keys.push('Alt');
        if (event.shiftKey) keys.push('Shift');
        
        if (!['Control', 'Alt', 'Shift', 'Meta'].includes(event.key)) {
            const key = event.key.length === 1 ? event.key.toUpperCase() : event.key;
            keys.push(key);
        }
        
        return keys.join('+');
    }

    function addToExpression(value) {
        const expressionDiv = document.getElementById('calcExpression');
        const resultDiv = document.getElementById('calcResult');
        
        if (value === 'pi') {
            calcExpression += Math.PI;
        } else {
            calcExpression += value;
        }
        
        expressionDiv.textContent = calcExpression;
        resultDiv.textContent = calcExpression || '0';
    }

    function clearCalculator() {
        calcExpression = '';
        calcResult = '0';
        document.getElementById('calcExpression').textContent = '';
        document.getElementById('calcResult').textContent = '0';
    }

    function deleteLast() {
        calcExpression = calcExpression.slice(0, -1);
        document.getElementById('calcExpression').textContent = calcExpression;
        document.getElementById('calcResult').textContent = calcExpression || '0';
    }

    function calculateResult() {
        try {
            let expression = calcExpression;
            
            expression = expression.replace(/sqrt\(/g, 'Math.sqrt(');
            expression = expression.replace(/\^/g, '**');
            expression = expression.replace(/sin\(/g, 'Math.sin(');
            expression = expression.replace(/cos\(/g, 'Math.cos(');
            expression = expression.replace(/tan\(/g, 'Math.tan(');
            expression = expression.replace(/log\(/g, 'Math.log10(');
            expression = expression.replace(/ln\(/g, 'Math.log(');
            
            const result = eval(expression);
            
            if (isNaN(result) || !isFinite(result)) {
                throw new Error('Résultat non valide');
            }
            
            calcResult = result.toString();
            document.getElementById('calcResult').textContent = formatNumberWithSettings(result);
            
            saveToCalcHistory(calcExpression, result);
            addRecentCalculation('calculatrice', `Calcul: ${calcExpression}`, result);
            
            calcExpression = '';
            document.getElementById('calcExpression').textContent = '';
            
            saveAllData();
            
        } catch (error) {
            showNotification('Erreur de calcul : ' + error.message, 'error');
            document.getElementById('calcResult').textContent = 'Erreur';
        }
    }

    function autoCalculA() {
        const cont = document.getElementById("multiAContainer");
        const resDiv = document.getElementById("resultA");
        resDiv.innerHTML = "";
        
        Array.from(cont.getElementsByClassName("ref-block")).forEach((block, index) => {
            const ref = parseNumber(block.querySelector(".refA").value);
            const raw = (block.querySelector(".valsA").value || "").trim();
            if (raw === "") return;
            const vals = raw.split(/\s+/).map(parseNumber).filter(v => v !== null);
            
            let html = `<div style="margin:15px 0;"><h4>Bloc ${index + 1} - Référence : ${ref.toFixed(2)}€</h4>`;
            html += `<table style="width:100%;"><tr><th>Valeur</th><th>Différence</th><th>%</th></tr>`;
            
            vals.forEach(v => {
                const diff = v - ref;
                const pct = (ref === 0) ? null : (diff / ref * 100);
                const col = diff > 0 ? "var(--success-color)" : diff < 0 ? "var(--error-color)" : "var(--text-muted)";
                html += `<tr style="background:${col}; color:#fff;">
                    <td>${v.toFixed(2)} €</td>
                    <td>${diff>=0?'+':''}${diff.toFixed(2)} €</td>
                    <td>${pct===null?'—':pct.toFixed(2)+'%'}</td>
                </tr>`;
            });
            html += "</table></div>";
            resDiv.innerHTML += html;
        });
        saveAllData();
    }

        const tfoot = document.getElementById('cResultsTotal');
        let html = '';
        
        let totalInitial = 0;
        let totalFinal = 0;
        let totalVariation = 0;
        
        cBlocks.forEach(block => {
            block.variables.forEach(variable => {
                const initial = variable.value;
                const final = initial * (1 + variable.percent / 100);
                const variation = final - initial;
                
                totalInitial += initial;
                totalFinal += final;
                totalVariation += variation;
                
                html += `<tr>
                    <td>${block.name}</td>
                    <td>${variable.name}</td>
                    <td>${initial.toFixed(2)}</td>
                    <td>${variable.percent >= 0 ? '+' : ''}${variable.percent}%</td>
                    <td>${final.toFixed(2)}</td>
                    <td class="${variation >= 0 ? 'positive' : 'negative'}">${variation >= 0 ? '+' : ''}${variation.toFixed(2)}</td>
                </tr>`;
            });
        });
        
        tbody.innerHTML = html;
        
        tfoot.innerHTML = `
            <tr class="total-row">
                <td colspan="2"><strong>TOTAL GLOBAL</strong></td>
                <td><strong>${totalInitial.toFixed(2)}</strong></td>
                <td></td>
                <td><strong>${totalFinal.toFixed(2)}</strong></td>
                <td class="${totalVariation >= 0 ? 'positive' : 'negative'}">
                    <strong>${totalVariation >= 0 ? '+' : ''}${totalVariation.toFixed(2)}</strong>
                </td>
            </tr>
        `;
        saveAllData();
    }

    // Mode B : Budget
    function addBudgetCategory() {
        const category = {
            id: Date.now(),
            name: `Catégorie ${budgetCatCount++}`,
            percent: 0,
            subs: []
        };
        
        budgetCategories.push(category);
        renderBudgetCategories();
        updateBudgetRecap();
        saveAllData();
    }

    function removeBudgetCategory(index) {
        if (budgetCategories.length <= 1) {
            showNotification('Vous devez garder au moins une catégorie', 'warning');
            return;
        }
        
        if (confirm('Supprimer cette catégorie ?')) {
            budgetCategories.splice(index, 1);
            renderBudgetCategories();
            updateBudgetRecap();
            saveAllData();
        }
    }

    function addBudgetSubcategory(catIndex) {
        const category = budgetCategories[catIndex];
        if (category) {
            category.subs.push({
                id: Date.now(),
                name: `Sous-catégorie ${category.subs.length + 1}`,
                percent: 0
            });
            renderBudgetCategories();
            updateBudgetRecap();
            saveAllData();
        }
    }

    function removeBudgetSubcategory(catIndex, subIndex) {
        const category = budgetCategories[catIndex];
        if (category && category.subs.length > 0) {
            category.subs.splice(subIndex, 1);
            renderBudgetCategories();
            updateBudgetRecap();
            saveAllData();
        }
    }

    function renameBudgetCategory(index) {
        const newName = prompt("Renommer la catégorie", budgetCategories[index].name);
        if (newName && newName.trim() !== '') {
            budgetCategories[index].name = newName.trim();
            renderBudgetCategories();
            updateBudgetRecap();
            saveAllData();
        }
    }

    function renameBudgetSubcategory(catIndex, subIndex) {
        const sub = budgetCategories[catIndex].subs[subIndex];
        if (sub) {
            const newName = prompt("Renommer la sous-catégorie", sub.name);
            if (newName && newName.trim() !== '') {
                sub.name = newName.trim();
                renderBudgetCategories();
                updateBudgetRecap();
                saveAllData();
            }
        }
    }

    function adjustCategoryPercent(index, value) {
        const category = budgetCategories[index];
        if (category) {
            value = parseFloat(value) || 0;
            
            const otherPercent = budgetCategories.reduce((sum, cat, i) =>
                i !== index ? sum + cat.percent : sum, 0);
            
            const maxAllowed = 100 - otherPercent;
            category.percent = Math.min(Math.max(value, 0), maxAllowed);
            
            renderBudgetCategories();
            updateBudgetRecap();
            saveAllData();
        }
    }

    function adjustSubcategoryPercent(catIndex, subIndex, value) {
        const category = budgetCategories[catIndex];
        const subcategory = category.subs[subIndex];
        
        if (subcategory) {
            value = parseFloat(value) || 0;
            
            const otherPercent = category.subs.reduce((sum, sub, i) =>
                i !== subIndex ? sum + sub.percent : sum, 0);
            
            const maxAllowed = 100 - otherPercent;
            subcategory.percent = Math.min(Math.max(value, 0), maxAllowed);
            
            renderBudgetCategories();
            updateBudgetRecap();
            saveAllData();
        }
    }

    function renderBudgetCategories() {
        const container = document.getElementById('budgetCategories');
        if (!container) return;
        
        if (budgetCategories.length === 0) {
            container.innerHTML = `
                <div style="text-align: center; padding: 40px 20px; color: var(--text-muted);">
                    <i class="fas fa-wallet fa-2x" style="margin-bottom: 15px;"></i>
                    <p>Ajoutez une première catégorie pour commencer</p>
                </div>
            `;
            return;
        }
        
        let html = '';
        
        budgetCategories.forEach((cat, catIndex) => {
            html += `
                <div class="category">
                    <div class="category-content">
                        <span class="editable" ondblclick="renameBudgetCategory(${catIndex})">
                            <strong>${cat.name}</strong>
                        </span>
                        <input type="number" class="budget-input" value="${cat.percent}" min="0" max="100" step="0.01" 
                            onchange="adjustCategoryPercent(${catIndex}, this.value)">
                        <span class="percent-label">%</span>
                        <button class="delete-btn" onclick="removeBudgetCategory(${catIndex})">
                            <i class="fas fa-trash"></i>
                        </button>
                    </div>
            `;
            
            cat.subs.forEach((sub, subIndex) => {
                html += `
                    <div class="subcategory">
                        <div class="subcategory-content">
                            <span class="editable" ondblclick="renameBudgetSubcategory(${catIndex}, ${subIndex})">
                                ${sub.name}
                            </span>
                            <input type="number" class="budget-input" value="${sub.percent}" min="0" max="100" step="0.01"
                                onchange="adjustSubcategoryPercent(${catIndex}, ${subIndex}, this.value)">
                            <span class="percent-label">%</span>
                            <button class="delete-btn" onclick="removeBudgetSubcategory(${catIndex}, ${subIndex})">
                                <i class="fas fa-trash"></i>
                            </button>
                        </div>
                    </div>
                `;
            });
            
            html += `
                    <div style="text-align: right; margin-top: 10px;">
                        <button onclick="addBudgetSubcategory(${catIndex})" class="secondary">
                            <i class="fas fa-plus"></i> Sous-catégorie
                        </button>
                    </div>
                </div>
            `;
        });
        
        container.innerHTML = html;
    }

    function updateBudgetRecap() {
        const budget = parseFloat(document.getElementById('budget').value) || 0;
        const tbody = document.getElementById('budgetRecap');
        if (!tbody) return;
        
        let html = '';
        let totalPercent = 0;
        
        budgetCategories.forEach(cat => {
            const catAmount = budget * cat.percent / 100;
            totalPercent += cat.percent;
            
            if (cat.subs.length === 0) {
                html += `
                    <tr class="category-row">
                        <td>${cat.name}</td>
                        <td>-</td>
                        <td>${cat.percent}%</td>
                        <td>${catAmount.toFixed(2)} €</td>
                    </tr>
                `;
            } else {
                html += `
                    <tr class="category-row">
                        <td>${cat.name}</td>
                        <td>-</td>
                        <td>${cat.percent}%</td>
                        <td>${catAmount.toFixed(2)} €</td>
                    </tr>
                `;
                
                cat.subs.forEach(sub => {
                    const globalPercent = cat.percent * sub.percent / 100;
                    const amount = budget * globalPercent / 100;
                    
                    html += `
                        <tr class="sub-row">
                            <td></td>
                            <td>${sub.name}</td>
                            <td>${globalPercent.toFixed(2)}%</td>
                            <td>${amount.toFixed(2)} €</td>
                        </tr>
                    `;
                });
            }
        });
        
        html += `
            <tr class="total-row">
                <td>TOTAL</td>
                <td>-</td>
                <td>${totalPercent}%</td>
                <td>${budget.toFixed(2)} €</td>
            </tr>
        `;
        
        tbody.innerHTML = html;
        saveAllData();
    }

    function addDDate() {
        const useDates = document.getElementById('useDates')?.checked !== false;
        const id = Date.now();
        
        if (useDates) {
            const date = new Date().toISOString().split('T')[0];
            dDates.push({ id, date });
        } else {
            dDates.push({ id, label: `Point ${dDates.length + 1}` });
        }
        
        dSeries.forEach(s => {
            s.values[id] = 0;
        });
        
        updateDTable();
        saveAllData();
    }

    function removeDDate(id) {
        dDates = dDates.filter(d => d.id !== id);
        dSeries.forEach(s => {
            delete s.values[id];
        });
        updateDTable();
        saveAllData();
    }

    function addDSeries() {
        const newSeries = {
            id: Date.now(),
            name: `Série ${dSeries.length + 1}`,
            values: {},
            color: '#' + Math.floor(Math.random()*16777215).toString(16)
        };
        dDates.forEach(d => {
            newSeries.values[d.id] = 0;
        });
        dSeries.push(newSeries);
        updateDTable();
        saveAllData();
    }

    function removeDSeries(id) {
        dSeries = dSeries.filter(s => s.id !== id);
        updateDTable();
        saveAllData();
    }

    function updateDDate(id, newDate) {
        const dateObj = dDates.find(d => d.id === id);
        if (dateObj) dateObj.date = newDate;
        saveAllData();
    }

    function updateDPointLabel(id, newLabel) {
        const pointObj = dDates.find(d => d.id === id);
        if (pointObj) pointObj.label = newLabel;
        saveAllData();
    }

    function updateDSeriesName(id, newName) {
        const series = dSeries.find(s => s.id === id);
        if (series) series.name = newName;
        saveAllData();
    }

    function updateDSeriesColor(id, newColor) {
        const series = dSeries.find(s => s.id === id);
        if (series) series.color = newColor;
        saveAllData();
    }

    function updateDSeriesValue(seriesId, dateId, newValue) {
        const series = dSeries.find(s => s.id === seriesId);
        if (series) series.values[dateId] = parseFloat(newValue) || 0;
        saveAllData();
    }

    function updateDTable() {
        const dateHeader = document.getElementById('dateHeaderRow');
        const seriesBody = document.getElementById('seriesBody');
        const useDates = document.getElementById('useDates')?.checked !== false;
        
        let headerHTML = '<th style="min-width: 160px;">Série / Date</th>';
        dDates.forEach(d => {
            if (useDates) {
                headerHTML += `
                    <th>
                        <input type="date" value="${d.date || new Date().toISOString().split('T')[0]}" onchange="updateDDate(${d.id}, this.value)" style="width:120px;">
                        <button onclick="removeDDate(${d.id})" style="padding:2px 6px;font-size:11px; background: var(--error-color); color: white; border: none; border-radius: 3px; cursor: pointer;">✖</button>
                    </th>
                `;
            } else {
                headerHTML += `
                    <th>
                        <input type="text" value="${d.label || `Point ${dDates.indexOf(d) + 1}`}" onchange="updateDPointLabel(${d.id}, this.value)" style="width:100px;">
                        <button onclick="removeDDate(${d.id})" style="padding:2px 6px;font-size:11px; background: var(--error-color); color: white; border: none; border-radius: 3px; cursor: pointer;">✖</button>
                    </th>
                `;
            }
        });
        dateHeader.innerHTML = headerHTML;
        
        seriesBody.innerHTML = '';
        dSeries.forEach(s => {
            let row = document.createElement('tr');
            let rowHTML = `
                <td>
                    <input type="text" value="${s.name}" onchange="updateDSeriesName(${s.id}, this.value)" style="width:120px;">
                    <input type="color" value="${s.color}" onchange="updateDSeriesColor(${s.id}, this.value)" title="Couleur" style="width: 30px; height: 30px;">
                    <button onclick="removeDSeries(${s.id})" style="padding:2px 6px;font-size:11px; background: var(--error-color); color: white; border: none; border-radius: 3px; cursor: pointer;">✖</button>
                </td>
            `;
            
            dDates.forEach(d => {
                const val = s.values[d.id] || 0;
                rowHTML += `
                    <td>
                        <input type="number" value="${val}" 
                               onchange="updateDSeriesValue(${s.id}, ${d.id}, this.value)" 
                               style="width:80px;" step="0.01">
                    </td>
                `;
            });
            
            row.innerHTML = rowHTML;
            seriesBody.appendChild(row);
        });
    }

    function sortDDates() {
        const useDates = document.getElementById('useDates')?.checked !== false;
        
        if (useDates) {
            dDates.sort((a, b) => new Date(a.date) - new Date(b.date));
        }
        updateDTable();
        saveAllData();
    }

    function calculerD() {
        const ctx = document.getElementById('graphD')?.getContext('2d');
        if (!ctx) return;
        
        if (chartD) chartD.destroy();
        
        const useDates = document.getElementById('useDates')?.checked !== false;

        }
        
        const datasets = dSeries.map(s => ({

                responsive: true,
                maintainAspectRatio: false,

                    y: {

                        grid: { color: 'var(--border-light)' },
                        ticks: { color: 'var(--text-secondary)' }
                    },
                    x: {
                        grid: { color: 'var(--border-light)' },
                        ticks: { color: 'var(--text-secondary)' }
                    }
                }
            }
        });
        saveAllData();
    }

    }

    function calculerF() {
        const ctx = document.getElementById('graphF')?.getContext('2d');
        if (!ctx) return;
        
        if (chartF) chartF.destroy();
        
        historiqueF = [];
        const blocks = document.querySelectorAll('#multiFContainer .ref-block');
        
        const visibleBlocks = Array.from(blocks).filter(block => {
            return !block.classList.contains('hidden-legend');
        });
        
        visibleBlocks.forEach(block => {
            historiqueF.push({
                label: block.querySelector('.labelF').value || "Sans nom",
                val: parseFloat(block.querySelector('.valF').value) || 0,
                color: block.querySelector('.colorF').value || '#3498db',
                type: block.getAttribute('data-original-value') ? 'original' : 'merged'
            });
        });
        
        if (historiqueF.length === 0) {
            document.getElementById('tableauF').querySelector('tbody').innerHTML = '<tr><td colspan="5" style="text-align: center; color: var(--text-muted); padding: 20px;">Aucune catégorie visible</td></tr>';
            document.getElementById('tableauFTotal').innerHTML = '';
            return;
        }
        
        const total = historiqueF.reduce((sum, cat) => sum + cat.val, 0);
        
        const data = {
            labels: historiqueF.map(cat => cat.label),
            datasets: [{
                data: historiqueF.map(cat => cat.val),
                backgroundColor: historiqueF.map(cat => cat.color),
                borderWidth: 2,
                borderColor: 'var(--bg-secondary)'
            }]
        };
        
        chartF = new Chart(ctx, {
            type: 'pie',
            data: data,
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { 
                        position: 'right',
                        onClick: function(evt, legendItem, legend) {
                            evt.stopPropagation();
                            
                            const index = legendItem.index;
                            const label = this.chart.data.labels[index];
                            
                            const blocks = document.querySelectorAll('#multiFContainer .ref-block');
                            blocks.forEach(block => {
                                if (block.querySelector('.labelF').value === label) {
                                    block.classList.toggle('hidden-legend');
                                }
                            });
                            
                            recalculatePercentagesF();
                        }
                    },
                    title: { display: true, text: `Répartition - Total: ${total.toFixed(2)}` },
                    datalabels: {
                        formatter: (value, context) => {
                            if (total === 0) return '0%';
                            const percentage = (value / total * 100).toFixed(1);
                            return context.chart.data.labels[context.dataIndex] + '\n' + percentage + '%';
                        },
                        color: '#fff',
                        font: { weight: 'bold', size: 12 },
                        textAlign: 'center'
                    }
                }
            },
            plugins: [ChartDataLabels]
        });
        
        updateFTable(total);
        saveAllData();
    }

    function recalculatePercentagesF() {
        const blocks = document.querySelectorAll('#multiFContainer .ref-block');
        
        const visibleBlocks = Array.from(blocks).filter(block => {
            return !block.classList.contains('hidden-legend');
        });
        
        const totalVisible = visibleBlocks.reduce((sum, block) => {
            const value = parseFloat(block.querySelector('.valF').value) || 0;
            return sum + value;
        }, 0);
        
        updateFTable(totalVisible);
        updateVisibleCount();
        calculerF();
    }

    function updateFTable(totalVisible = null) {
        const tbody = document.querySelector('#tableauF tbody');
        const tfoot = document.getElementById('tableauFTotal');
        
        if (!tbody || !tfoot) return;
        
        const blocks = document.querySelectorAll('#multiFContainer .ref-block');
        
        let total;
        if (totalVisible !== null) {
            total = totalVisible;
        } else {
            total = Array.from(blocks).reduce((sum, block) => {
                const value = parseFloat(block.querySelector('.valF').value) || 0;
                return sum + value;
            }, 0);
        }
        
        tbody.innerHTML = '';
        
        blocks.forEach(block => {
            const label = block.querySelector('.labelF').value || 'Sans nom';
            const value = parseFloat(block.querySelector('.valF').value) || 0;
            const color = block.querySelector('.colorF').value || '#3498db';
            const type = block.getAttribute('data-original-value') ? 'original' : 'merged';
            const isHidden = block.classList.contains('hidden-legend');
            
            const percentage = (total === 0 || isHidden) ? 0 : (value / total * 100);
            
            const typeBadge = type === 'merged' ?
                '<span style="background: #9b59b6; color: white; padding: 2px 6px; border-radius: 3px; font-size: 11px;">Fusionnée</span>' :
                '<span style="background: var(--success-color); color: white; padding: 2px 6px; border-radius: 3px; font-size: 11px;">Originale</span>';
            
            const visibilityBadge = isHidden ? 
                '<span style="background: var(--warning-color); color: white; padding: 2px 6px; border-radius: 3px; font-size: 11px; margin-left: 5px;">Cachée</span>' :
                '<span style="background: var(--success-color); color: white; padding: 2px 6px; border-radius: 3px; font-size: 11px; margin-left: 5px;">Visible</span>';
            
            tbody.innerHTML += `<tr style="${isHidden ? 'opacity: 0.5; background: var(--bg-tertiary);' : ''}">
                <td>
                    <span style="display:inline-block;width:12px;height:12px;background:${color};margin-right:8px;border-radius:2px;"></span>
                    ${label}
                </td>
                <td style="text-align: center;">${value.toFixed(2)}</td>
                <td style="text-align: center;">
                    ${isHidden ? '-' : `${percentage.toFixed(1)}%`}
                </td>
                <td style="text-align: center;">${typeBadge}</td>
                <td style="text-align: center;">${visibilityBadge}</td>
            </tr>`;
        });
        
        tfoot.innerHTML = `
            <tr style="font-weight: bold; background: rgba(52, 152, 219, 0.1);">
                <td style="text-align: center;">TOTAL (${totalVisible !== null ? 'visibles' : 'tous'})</td>
                <td style="text-align: center;">${total.toFixed(2)}</td>
                <td style="text-align: center;">100%</td>
                <td></td>
                <td></td>
            </tr>
        `;
    }

            btn.style.background = 'var(--bg-tertiary)';
            btn.style.color = 'var(--text-primary)';

            btn.style.color = 'white';
            btn.setAttribute('data-selected', 'true');
        }
        updateSelectedCount();
        saveAllData();
    }

    function mergeSelectedCategories() {
        const selected = Array.from(document.querySelectorAll('.fusion-toggle-btn[data-selected="true"]'));
        if (selected.length < 2) {
            showNotification('Sélectionnez au moins 2 catégories à fusionner.', 'warning');
            return;
        }

    function calculerG() {
        const resultDiv = document.getElementById("resultG");
        const blocks = document.querySelectorAll('#multiGContainer .ref-block');
        
        if (blocks.length === 0) {
            resultDiv.innerHTML = '<p style="color: var(--text-muted); text-align: center; padding: 20px;">Ajoutez un bloc TVA pour commencer</p>';
            return;
        }
        
        let globalHTML = "";
        
        blocks.forEach((block, blockIndex) => {
            const blocNom = block.querySelector('.bloc-nom-input').value || `Bloc ${blockIndex + 1}`;
            const globalTVA = parseFloat(block.querySelector('.global-tva-input').value) || 20;
            
            const lignes = block.querySelectorAll('.ligne-tva');
            let totalHT = 0;
            let totalTVA = 0;
            let totalTTC = 0;
            
            let tableHTML = `
                <div style="margin: 20px 0; padding: 20px; background: var(--bg-tertiary); border-radius: 10px; border: 1px solid var(--border-color);">
                    <h4 style="color: var(--text-primary); margin: 0 0 15px 0;">${blocNom} (TVA par défaut: ${globalTVA}%)</h4>
                    <table style="width:100%; margin-top:10px; border-collapse: collapse;">
                        <thead>
                            <tr>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">Libellé</th>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">Montant</th>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">Type</th>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">TVA</th>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">HT</th>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">TVA (€)</th>
                                <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">TTC</th>
                            </tr>
                        </thead>
                        <tbody>
            `;
            
            lignes.forEach(ligne => {
                const libelle = ligne.querySelector('.libelleG').value || "";
                const valeur = parseFloat(ligne.querySelector('.valeurG').value) || 0;
                const type = ligne.querySelector('.typeG').value;
                const tva = parseFloat(ligne.querySelector('.tvaIndividuelleG').value) || globalTVA;
                
                let ht, ttc, tvaEur;
                
                if (type === "HT") {
                    ht = valeur;
                    tvaEur = ht * (tva / 100);
                    ttc = ht + tvaEur;
                } else {
                    ttc = valeur;
                    ht = ttc / (1 + (tva / 100));
                    tvaEur = ttc - ht;
                }
                
                totalHT += ht;
                totalTVA += tvaEur;
                totalTTC += ttc;
                
                tableHTML += `
                    <tr style="border-bottom: 1px solid var(--border-light);">
                        <td style="padding: 10px;">${libelle}</td>
                        <td style="padding: 10px; text-align: right;">${valeur.toFixed(2)} €</td>
                        <td style="padding: 10px; text-align: center;">${type}</td>
                        <td style="padding: 10px; text-align: right;">${tva.toFixed(1)}%</td>
                        <td style="padding: 10px; text-align: right;">${ht.toFixed(2)} €</td>
                        <td style="padding: 10px; text-align: right;">${tvaEur.toFixed(2)} €</td>
                        <td style="padding: 10px; text-align: right;">${ttc.toFixed(2)} €</td>
                    </tr>
                `;
            });
            
            tableHTML += `
                    </tbody>
                    <tfoot style="font-weight:bold; background:rgba(52, 152, 219, 0.1);">
                        <tr>
                            <td colspan="4" style="padding: 10px; border-top: 2px solid var(--border-color);">TOTAUX</td>
                            <td style="padding: 10px; text-align: right; border-top: 2px solid var(--border-color);">${totalHT.toFixed(2)} €</td>
                            <td style="padding: 10px; text-align: right; border-top: 2px solid var(--border-color);">${totalTVA.toFixed(2)} €</td>
                            <td style="padding: 10px; text-align: right; border-top: 2px solid var(--border-color);">${totalTTC.toFixed(2)} €</td>
                        </tr>
                    </tfoot>
                </table>
            </div>
            `;
            
            globalHTML += tableHTML;
        });
        
        resultDiv.innerHTML = globalHTML;
        saveAllData();
    }

    async function loadExchangeRates() {
        showNotification('Chargement des taux de change...', 'info');
        
        try {
            const cached = localStorage.getItem(EXCHANGE_RATES_KEY);
            const cacheDate = localStorage.getItem('exchange_rates_date');
            
            if (cached && cacheDate) {
                const cacheTime = new Date(cacheDate);
                const now = new Date();
                const hoursDiff = (now - cacheTime) / (1000 * 60 * 60);
                
                if (hoursDiff < 24) {
                    exchangeRates = JSON.parse(cached);
                    console.log('Taux de change chargés depuis le cache');
                    updateCurrencySelects();
                    convertCurrency();
                    updateAllRates();
                    showNotification('Taux de change chargés (cache)', 'success');
                    return;
                }
            }
            
            console.log('Récupération des taux de change depuis l\'API...');
            
            const response = await fetch(`https://api.exchangerate-api.com/v4/latest/EUR`);
            
            if (!response.ok) {
                throw new Error(`API error: ${response.status}`);
            }
            
            const data = await response.json();
            exchangeRates = data.rates;
            exchangeRates['EUR'] = 1;
            
            localStorage.setItem(EXCHANGE_RATES_KEY, JSON.stringify(exchangeRates));
            localStorage.setItem('exchange_rates_date', new Date().toISOString());
            
            console.log('Taux de change mis à jour depuis l\'API:', exchangeRates);
            updateCurrencySelects();
            convertCurrency();
            updateAllRates();
            showNotification('Taux de change actualisés', 'success');
            
        } catch (error) {
            console.error('Erreur chargement taux de change:', error);
            
            exchangeRates = {
                'EUR': 1, 'USD': 1.08, 'GBP': 0.85, 'JPY': 160, 'CHF': 0.95,
                'CAD': 1.46, 'AUD': 1.64, 'CNY': 7.78, 'INR': 89.5,
                'BRL': 5.45, 'KRW': 1445, 'MXN': 18.2, 'SGD': 1.45,
                'HKD': 8.45, 'NZD': 1.78, 'SEK': 11.2, 'NOK': 11.5,
                'DKK': 7.45, 'PLN': 4.32, 'TRY': 34.8, 'THB': 38.9,
                'IDR': 17200, 'MYR': 5.12, 'PHP': 60.8, 'VND': 27000,
                'AED': 3.97, 'SAR': 4.05, 'ZAR': 20.1, 'EGP': 50.8
            };
            
            updateCurrencySelects();
            convertCurrency();
            updateAllRates();
            showNotification('Taux de change par défaut (API indisponible)', 'warning');
        }
    }

    function updateCurrencySelects() {
        const fromSelect = document.getElementById('fromCurrency');
        const toSelect = document.getElementById('toCurrency');
        
        if (!fromSelect || !toSelect) return;
        
        fromSelect.innerHTML = '';
        toSelect.innerHTML = '';
        
        const sortedCurrencies = Object.keys(exchangeRates).sort();
        
        sortedCurrencies.forEach(currency => {
            const option1 = document.createElement('option');
            option1.value = currency;
            option1.textContent = `${currency}`;
            fromSelect.appendChild(option1);
            
            const option2 = document.createElement('option');
            option2.value = currency;
            option2.textContent = `${currency}`;
            toSelect.appendChild(option2);
        });
        
        const lastFromCurrency = localStorage.getItem('lastFromCurrency') || 'EUR';
        const lastToCurrency = localStorage.getItem('lastToCurrency') || 'USD';
        
        if (sortedCurrencies.includes(lastFromCurrency)) {
            fromSelect.value = lastFromCurrency;
        }
        if (sortedCurrencies.includes(lastToCurrency)) {
            toSelect.value = lastToCurrency;
        }
        
        updateFlags();
    }

    function updateFlags() {
        const fromCurrency = document.getElementById('fromCurrency')?.value;
        const toCurrency = document.getElementById('toCurrency')?.value;
        
        const fromFlag = document.getElementById('fromFlag');
        const toFlag = document.getElementById('toFlag');
        
        if (fromCurrency && currencyFlags[fromCurrency] && fromFlag) {
            fromFlag.src = `https://flagcdn.com/w20/${currencyFlags[fromCurrency]}.png`;
            fromFlag.style.display = 'block';
        } else if (fromFlag) {
            fromFlag.src = '';
            fromFlag.style.display = 'none';
        }
        
        if (toCurrency && currencyFlags[toCurrency] && toFlag) {
            toFlag.src = `https://flagcdn.com/w20/${currencyFlags[toCurrency]}.png`;
            toFlag.style.display = 'block';
        } else if (toFlag) {
            toFlag.src = '';
            toFlag.style.display = 'none';
        }
        
        if (fromCurrency) localStorage.setItem('lastFromCurrency', fromCurrency);
        if (toCurrency) localStorage.setItem('lastToCurrency', toCurrency);
        
        convertCurrency();
    }

    function convertCurrency() {
        const fromCurrency = document.getElementById('fromCurrency')?.value;
        const toCurrency = document.getElementById('toCurrency')?.value;
        const fromAmount = parseFloat(document.getElementById('fromAmount')?.value) || 0;
        const toAmount = document.getElementById('toAmount');
        
        if (!fromCurrency || !toCurrency || !toAmount || !exchangeRates[fromCurrency] || !exchangeRates[toCurrency]) {
            if (toAmount) toAmount.value = '0.00';
            return;
        }
        
        try {
            const amountInEur = fromAmount / exchangeRates[fromCurrency];
            const convertedAmount = amountInEur * exchangeRates[toCurrency];
            
            toAmount.value = convertedAmount.toFixed(2);
            
            const rate = (exchangeRates[toCurrency] / exchangeRates[fromCurrency]).toFixed(4);
            const rateInfo = document.getElementById('rateInfo');
            if (rateInfo) {
                rateInfo.innerHTML = `Taux de change: 1 ${fromCurrency} = ${rate} ${toCurrency}`;
            }
            saveAllData();
        } catch (error) {
            console.error('Erreur conversion:', error);
            if (toAmount) toAmount.value = '0.00';
        }
    }

    function switchCurrencies() {

    function updateAllRates() {
        const container = document.getElementById('allRatesGrid');
        if (!container) return;
        
        let html = '';
        const baseCurrency = 'EUR';
        
        Object.entries(exchangeRates)
            .sort(([a], [b]) => a.localeCompare(b))
            .forEach(([currency, rate]) => {
                if (currency === baseCurrency) return;
                
                const flagCode = currencyFlags[currency] || 'un';
                
                html += `
                    <div style="padding: 12px; background: var(--bg-secondary); border-radius: 8px; border: 1px solid var(--border-color); transition: all 0.3s; cursor: pointer;" onclick="setCurrencyPair('${currency}')">
                        <div style="display: flex; align-items: center; gap: 10px; margin-bottom: 8px;">
                            <img src="https://flagcdn.com/w20/${flagCode}.png" style="width: 24px; height: 18px; border-radius: 3px; object-fit: cover;">
                            <div style="font-weight: bold; color: var(--text-primary);">${currency}</div>
                        </div>
                        <div style="font-size: 18px; color: var(--accent-color); margin-top: 5px;">${rate.toFixed(4)}</div>
                        <div style="font-size: 12px; color: var(--text-muted);">1 ${baseCurrency} = ${rate.toFixed(4)} ${currency}</div>
                    </div>
                `;
        });
        
        container.innerHTML = html;
    }

    function setCurrencyPair(currency) {

        document.getElementById('toCurrency').value = currency;
        document.getElementById('fromAmount').value = '100';
        updateFlags();
        convertCurrency();
        showNotification(`Paire EUR/${currency} sélectionnée`, 'info');
    }

    function saveConversionToHistory() {
        const fromCurrency = document.getElementById('fromCurrency').value;
        const toCurrency = document.getElementById('toCurrency').value;
        const fromAmount = parseFloat(document.getElementById('fromAmount').value) || 0;
        const toAmount = parseFloat(document.getElementById('toAmount').value) || 0;
        
        if (fromAmount === 0) {
            showNotification('Entrez un montant à convertir', 'warning');
            return;
        }
        
        const conversion = {
            id: Date.now(),
            date: new Date().toLocaleString('fr-FR'),
            from: { currency: fromCurrency, amount: fromAmount },
            to: { currency: toCurrency, amount: toAmount },
            rate: (exchangeRates[toCurrency] / exchangeRates[fromCurrency]).toFixed(4)
        };
        
        conversionHistory.unshift(conversion);
        if (conversionHistory.length > 20) {
            conversionHistory = conversionHistory.slice(0, 20);
        }
        
        localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
        updateConversionHistoryDisplay();
        showNotification('Conversion sauvegardée', 'success');
        saveAllData();
    }

    function updateConversionHistoryDisplay() {
        const container = document.getElementById('conversionHistoryList');
        if (!container) return;
        
        if (conversionHistory.length === 0) {
            container.innerHTML = `
                <div style="padding: 30px; text-align: center; color: var(--text-muted);">
                    <i class="fas fa-exchange-alt" style="font-size: 2rem; margin-bottom: 10px; opacity: 0.5;"></i>
                    <p>Aucune conversion dans l'historique</p>
                </div>
            `;
            return;
        }
        
        let html = '';
        
        conversionHistory.forEach(conv => {
            html += `
                <div class="history-item" style="display: grid; grid-template-columns: 1fr 1fr auto auto; gap: 10px; padding: 12px; border-bottom: 1px solid var(--border-color); align-items: center; background: var(--bg-secondary);">
                    <div style="font-size: 12px; color: var(--text-muted);">${conv.date}</div>
                    <div style="font-weight: bold; color: var(--text-primary);">${conv.from.amount.toFixed(2)} ${conv.from.currency}</div>
                    <div style="text-align: center; color: var(--text-primary);">→</div>
                    <div style="font-weight: bold; color: var(--success-color);">${conv.to.amount.toFixed(2)} ${conv.to.currency}</div>
                    <div style="font-size: 12px; color: var(--text-muted); grid-column: 1 / -1;">Taux: ${conv.rate}</div>
                    <button onclick="deleteConversion(${conv.id})" style="padding: 5px 10px; background: var(--error-color); color: white; border: none; border-radius: 4px; cursor: pointer; font-size: 12px; grid-column: 4;">
                        <i class="fas fa-trash"></i>
                    </button>
                </div>
            `;
        });
        
        container.innerHTML = html;
    }

    function deleteConversion(id) {
        conversionHistory = conversionHistory.filter(conv => conv.id !== id);
        localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
        updateConversionHistoryDisplay();
        showNotification('Conversion supprimée', 'success');
        saveAllData();
    }

    function clearConversionHistory() {
        if (conversionHistory.length === 0) return;
        
        if (confirm('Effacer tout l\'historique des conversions ?')) {
            conversionHistory = [];
            localStorage.setItem('conversionHistory', JSON.stringify([]));
            updateConversionHistoryDisplay();
            showNotification('Historique effacé', 'success');
            saveAllData();
        }
    }

    function exportHistoryToCSV() {
        if (conversionHistory.length === 0) {
            showNotification('Historique vide', 'warning');
            return;
        }
        
        let csv = 'Date;Montant Source;Devise Source;Montant Cible;Devise Cible;Taux de change\n';
        
        conversionHistory.forEach(conv => {
            csv += `${conv.date};${conv.from.amount};${conv.from.currency};${conv.to.amount};${conv.to.currency};${conv.rate}\n`;
        });
        
        const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
        const link = document.createElement('a');
        const url = URL.createObjectURL(blob);
        
        link.setAttribute('href', url);
        link.setAttribute('download', `historique_conversions_${new Date().toISOString().split('T')[0]}.csv`);
        link.style.visibility = 'hidden';
        
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        
        showNotification('Historique exporté en CSV', 'success');
    }

    function saveAllData() {
        try {
            const appData = {
                timestamp: new Date().toISOString(),
                version: '12.0',
                modeA: collectModeAData(),
                modeC: cBlocks,
                modeB: {
                    categories: budgetCategories,
                    budget: document.getElementById('budget')?.value || '2000',
                    catCount: budgetCatCount
                },
                modeD: { dates: dDates, series: dSeries, useDates: document.getElementById('useDates')?.checked || true },
                modeF: {
                    categories: historiqueF,
                    blocks: Array.from(document.querySelectorAll('#multiFContainer .ref-block')).map(block => ({
                        label: block.querySelector('.labelF').value,
                        value: block.querySelector('.valF').value,
                        color: block.querySelector('.colorF').value,
                        originalValue: block.getAttribute('data-original-value'),
                        isHidden: block.classList.contains('hidden-legend')
                    }))
                },
                modeG: collectModeGData(),
                modeH: {
                    conversions: conversionHistory,
                    fromCurrency: document.getElementById('fromCurrency')?.value || 'EUR',
                    toCurrency: document.getElementById('toCurrency')?.value || 'USD',
                    fromAmount: document.getElementById('fromAmount').value || '100'
                },
                calculator: calculatorHistory,
                recentCalculations: recentCalculations,
                shortcuts: keyboardShortcuts,
                settings: settings,
                notes: notes,
                exchangeRates: exchangeRates
            };
            
            localStorage.setItem(APP_STORAGE_KEY, JSON.stringify(appData));
            localStorage.setItem(NOTES_STORAGE_KEY, JSON.stringify(notes));
            
            console.log('💾 Données sauvegardées (version 12.0)');
            return appData;
        } catch (error) {
            console.error('❌ Erreur sauvegarde:', error);
            return null;
        }
    }

    function loadAllData() {
        try {
            const saved = localStorage.getItem(APP_STORAGE_KEY);
            if (saved) {
                const data = JSON.parse(saved);
                
                if (data.modeA) restoreModeAData(data.modeA);
                if (data.modeC) {
                    cBlocks = data.modeC;
                    cBlockCounter = cBlocks.length + 1;
                }
                if (data.modeB) {
                    budgetCategories = data.modeB.categories || [];
                    budgetCatCount = data.modeB.catCount || 1;
                    if (document.getElementById('budget')) {
                        document.getElementById('budget').value = data.modeB.budget || '2000';
                    }
                }
                if (data.modeD) {
                    dDates = data.modeD.dates || [];
                    dSeries = data.modeD.series || [];
                    if (document.getElementById('useDates')) {
                        document.getElementById('useDates').checked = data.modeD.useDates !== false;
                    }
                }
                if (data.modeF) {
                    historiqueF = data.modeF.categories || [];
                    const blocks = data.modeF.blocks || [];
                    const container = document.getElementById("multiFContainer");
                    container.innerHTML = '';
                    
                    blocks.forEach(blockData => {
                        const id = Date.now() + Math.random();
                        const div = document.createElement("div");
                        div.className = "ref-block";
                        div.setAttribute('data-category-id', id);
                        
                        if (blockData.originalValue) {
                            div.setAttribute('data-original-value', blockData.originalValue);
                        }
                        
                        if (blockData.isHidden) {
                            div.classList.add('hidden-legend');
                        }
                        
                        div.innerHTML = `
                            <div style="display: flex; align-items: center; gap: 10px; flex-wrap: wrap;">
                                <input type="text" class="labelF" placeholder="Catégorie" value="${blockData.label}" style="width:150px;" oninput="autoCalculF()">
                                <input type="number" class="valF" placeholder="Valeur" value="${blockData.value}" style="width:100px;" oninput="autoCalculF()">
                                <input type="color" class="colorF" value="${blockData.color}" oninput="autoCalculF()">
                                <div class="fusion-toggle-btn" onclick="toggleFusionSelection(this)" data-selected="false" data-original-value="${blockData.originalValue || blockData.value}" style="width: 32px; height: 32px; border: 2px solid var(--border-color); border-radius: 4px; cursor: pointer; display: flex; align-items: center; justify-content: center; font-weight: bold; user-select: none;">✗</div>
                                <button onclick="removeF(this)" class="delete-btn"><i class="fas fa-trash"></i></button>
                            </div>
                        `;
                        
                        container.appendChild(div);
                    });
                    
                    if (blocks.length === 0) {
                        addFBlock();
                        addFBlock();
                    }
                }
                if (data.modeG) restoreModeGData(data.modeG);
                if (data.modeH) {
                    conversionHistory = data.modeH.conversions || [];
                    updateConversionHistoryDisplay();
                }
                if (data.calculator) {
                    calculatorHistory = data.calculator;
                }
                if (data.recentCalculations) {
                    recentCalculations = data.recentCalculations;
                }
                if (data.shortcuts) {
                    keyboardShortcuts = data.shortcuts;
                }
                if (data.settings) {
                    settings = { ...settings, ...data.settings };
                }
                if (data.notes) {
                    notes = data.notes;
                    console.log('📝 Notes chargées depuis sauvegarde principale:', notes.length);
                }
                if (data.exchangeRates) {
                    exchangeRates = data.exchangeRates;
                }
                
                console.log('✅ Données restaurées (version 12.0)');
                
                if (data.modeF) {
                    updateSelectedCount();
                    updateVisibleCount();
                    recalculatePercentagesF();
                }
                
                return true;
            }
        } catch (error) {
            console.error('❌ Erreur chargement:', error);
        }
        
        initDefaultData();
        return false;
    }

        const fBlocks = document.querySelectorAll('#multiFContainer .ref-block');

            addFBlock();
            addFBlock();
        }
    }

    function createBackup(name = null) {
        const backupData = saveAllData();
        const backupName = name || prompt('Nom de la sauvegarde :', `Sauvegarde_${new Date().toLocaleDateString('fr-FR')}`);
        
        if (!backupName) return;
        
        const backup = {
            id: Date.now(),
            name: backupName,
            timestamp: new Date().toISOString(),
            data: backupData,
            size: JSON.stringify(backupData).length
        };
        
        let backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
        backups.unshift(backup);
        
        if (backups.length > 10) {
            backups = backups.slice(0, 10);
        }
        
        localStorage.setItem(BACKUP_STORAGE_KEY, JSON.stringify(backups));
        displayBackupList();
        showNotification(`Sauvegarde "${backupName}" créée`, 'success');
    }

    function handleBackupFileSelect(event) {
        const file = event.target.files[0];
        if (!file) return;
        
        const reader = new FileReader();
        reader.onload = function(e) {
            try {
                const backupData = JSON.parse(e.target.result);
                
                if (confirm(`Importer la sauvegarde "${backupData.name || 'Sans nom'}" ?`)) {
                    const backup = {
                        id: Date.now(),
                        name: backupData.name || 'Importé',
                        timestamp: new Date().toISOString(),
                        data: backupData
                    };
                    
                    let backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
                    backups.unshift(backup);
                    localStorage.setItem(BACKUP_STORAGE_KEY, JSON.stringify(backups));
                    
                    displayBackupList();
                    showNotification('Sauvegarde importée avec succès', 'success');
                }
            } catch (error) {
                console.error('Erreur import:', error);
                showNotification('Format de fichier invalide', 'error');
            }
        };
        reader.readAsText(file);
    }
