  <script>
    function showMode(modeId) {
      ["modeA","modeB","modeC","modeD"].forEach(id => {
        document.getElementById(id).classList.add("hidden");
      });
      document.getElementById(modeId).classList.remove("hidden");
    }

    // MODE A : Fiches colorées 📈📉
    function calculerA() {
      const ref = parseFloat(document.getElementById("refA").value);
      const valeurs = document.getElementById("valeursA").value.trim().split(/\s+/).map(Number);
      const cardsContainer = document.getElementById("resultCardsA");
      cardsContainer.innerHTML = "";

      if (isNaN(ref) || ref === 0) {
        cardsContainer.innerHTML = `<div class="result-card perte"><div class="emoji">⚠️</div><div class="content"><div class="title">Erreur</div><div class="sub">Merci d'entrer une valeur de référence valide.</div></div></div>`;
        return;
      }

      valeurs.forEach(v => {
        if (isNaN(v)) return;
        const diffNum = v - ref;
        const diffPct = ((v - ref) / ref) * 100;
        const isGain = diffNum > 0;
        const absDiff = Math.abs(diffNum);
        const card = document.createElement("div");
        card.className = "result-card " + (isGain ? "gain" : diffNum < 0 ? "perte" : "");
        card.innerHTML = `
          <div class="emoji">${isGain ? "📈" : diffNum < 0 ? "📉" : "⚪"}</div>
          <div class="content">
            <div class="title">${isGain ? "Gain" : diffNum < 0 ? "Perte" : "Aucune variation"} : ${diffNum >= 0 ? "+" : ""}${diffNum.toFixed(2)}</div>
            <div class="sub">${diffPct.toFixed(2)}% par rapport à la valeur de référence (${ref})</div>
          </div>`;
        cardsContainer.appendChild(card);
      });
    }

    // MODE B
    function calculerB() {
      const base = parseFloat(document.getElementById("valeurB").value);
      const pourcents = document.getElementById("pourcentB").value.trim().split(/\s+/).map(Number);
      let res = "";
      if (isNaN(base)) {
        res = "⚠️ Merci d'entrer une valeur de départ valide.";
      } else {
        pourcents.forEach(p => {
          if (!isNaN(p)) {
            let nv = base * (1 + p / 100);
            res += `${p >= 0 ? "Gain" : "Perte"} de ${p}% → ${nv.toFixed(2)}\n`;
          }
        });
      }
      document.getElementById("resultatB").innerText = res;
    }

    // MODE C
    let chartC;
    function calculerC() {
      const titre = document.getElementById("titreC").value.trim() || "Graphique";
      const lignes = document.getElementById("seriesC").value.trim().split("\n").filter(l => l.trim().length);
      if (!lignes.length) {
        document.getElementById("resultatC").innerText = "⚠️ Merci d'entrer au moins une série de données.";
        return;
      }

      let res = "";
      let series = [];

      lignes.forEach(ligne => {
        let parts = ligne.trim().split(/\s+/);
        if (parts.length < 2) return;
        let nom = parts[0];
        let valeurs = parts.slice(1).map(Number).filter(v => !isNaN(v));
        if (valeurs.length < 2) return;
        series.push({ nom, valeurs });

        res += `${nom} :\n`;
        let croissanceTotale = 0;
        for (let i = 1; i < valeurs.length; i++) {
          let diff = ((valeurs[i] - valeurs[i - 1]) / valeurs[i - 1]) * 100;
          res += `  Étape ${i} : ${diff.toFixed(2)}%\n`;
          croissanceTotale += diff;
        }
        let moyenne = croissanceTotale / (valeurs.length - 1);
        res += `👉 Moyenne : ${moyenne.toFixed(2)}%\n\n`;
      });

      document.getElementById("resultatC").innerText = res || "Pas de séries valides.";

      const ctx = document.getElementById("graphC").getContext("2d");
      if (chartC) chartC.destroy();
      chartC = new Chart(ctx, {
        type: 'line',
        data: {
          labels: series[0]?.valeurs.map((_, i) => "Étape " + (i + 1)) || [],
          datasets: series.map((s, idx) => ({
            label: s.nom,
            data: s.valeurs,
            borderColor: `hsl(${idx * 60}, 70%, 50%)`,
            tension: 0.2,
            fill: false
          }))
        },
        options: {
          responsive: true,
          plugins: {
            title: {
              display: true,
              text: titre,
              font: { size: 18, weight: 'bold' },
              color: '#2c3e50'
            },
            legend: { position: 'bottom' }
          }
        }
      });
    }

    // MODE D
    let chartD;
    function calculerD() {
      const raw = document.getElementById("categoriesD").value.trim();
      if (!raw) return alert("Merci d'entrer un titre puis des données.");
      const lignes = raw.split("\n").map(l => l.trim()).filter(l => l.length);
      if (lignes.length < 2) return alert("Merci d'entrer un titre puis au moins une catégorie.");

      const titre = lignes[0];
      const dataLignes = lignes.slice(1);

      const labels = [];
      const valeurs = [];

      dataLignes.forEach(ligne => {
        const parts = ligne.split(/\s+/);
        if (parts.length >= 2) {
          const nom = parts.slice(0, -1).join(" ");
          const val = parseFloat(parts[parts.length - 1]);
          if (!isNaN(val)) {
            labels.push(nom);
            valeurs.push(val);
          }
        }
      });

      const ctx = document.getElementById("graphD").getContext("2d");
      if (chartD) chartD.destroy();

      chartD = new Chart(ctx, {
        type: 'pie',
        data: { labels, datasets: [{ data: valeurs, backgroundColor: labels.map((_, i) => `hsl(${i*60},70%,60%)`) }] },
        options: {
          responsive: true,
          plugins: {
            title: { display: true, text: titre, font: { size: 18, weight: 'bold' }, color: '#2c3e50' },
            legend: {
              position: 'bottom',
              labels: {
                generateLabels(chart) {
                  const ds = chart.data.datasets[0];
                  const meta = chart.getDatasetMeta(0);
                  const visibles = ds.data.filter((_, i) => !meta.data[i].hidden);
                  const total = visibles.reduce((a,b)=>a+b, 0) || 1;
                  return chart.data.labels.map((label, i) => {
                    const val = ds.data[i];
                    const hidden = meta.data[i].hidden;
                    const pct = hidden ? '0.0' : ((val / total) * 100).toFixed(1);
                    return {
                      text: `${label} (${val}${hidden ? "" : ` | ${pct}%`})`,
                      fillStyle: ds.backgroundColor[i],
                      hidden,
                      index: i
                    };
                  });
                }
              },
              onClick(e, legendItem, legend) {
                const index = legendItem.index;
                const ci = legend.chart;
                const meta = ci.getDatasetMeta(0);
                meta.data[index].hidden = !meta.data[index].hidden;
                ci.update();
              }
            },
            tooltip: {
              callbacks: {
                label(ctx) {
                  const ds = ctx.chart.data.datasets[0];
                  const meta = ctx.chart.getDatasetMeta(0);
                  const visibles = ds.data.filter((_, i) => !meta.data[i].hidden);
                  const total = visibles.reduce((a,b)=>a+b,0) || 1;
                  const val = ctx.raw;
                  const pct = (val / total * 100).toFixed(1);
                  return `${ctx.label} → ${val} (${pct}%)`;
                }
              }
            },
            datalabels: {
              color: '#fff',
              font: { weight: 'bold', size: 12 },
              formatter(value, ctx) {
                const ds = ctx.chart.data.datasets[0];
                const meta = ctx.chart.getDatasetMeta(0);
                const visibles = ds.data.filter((_, i) => !meta.data[i].hidden);
                const total = visibles.reduce((a,b)=>a+b,0) || 1;
                if (meta.data[ctx.dataIndex].hidden) return '';
                return (value / total * 100).toFixed(1) + '%';
              }
            }
          }
        },
        plugins: [ChartDataLabels]
      });
    }

    // TÉLÉCHARGEMENT MULTI-FORMATS
    function downloadChart(mode) {
      const format = document.getElementById(`format${mode}`).value;
      const canvas = document.getElementById(`graph${mode}`);
      if (!canvas) return alert("Aucun graphique à télécharger.");
      if (!format) return alert("Veuillez choisir un format.");

      const mime = (format === "pdf") ? "image/png" : `image/${format}`;
      const dataURL = canvas.toDataURL(mime);

      if (format === "pdf") {
        const { jsPDF } = window.jspdf;
        const pdf = new jsPDF();
        const imgProps = pdf.getImageProperties(dataURL);
        const pdfWidth = pdf.internal.pageSize.getWidth();
        const pdfHeight = (imgProps.height * pdfWidth) / imgProps.width;
        pdf.addImage(dataURL, "PNG", 0, 10, pdfWidth, pdfHeight);
        pdf.save(`graphique_${mode}.pdf`);
      } else {
        const a = document.createElement("a");
        a.href = dataURL;
        a.download = `graphique_${mode}.${format}`;
        a.click();
      }
    }
