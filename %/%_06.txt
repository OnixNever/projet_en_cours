function calculerA(){
  const cont=document.getElementById("multiAContainer"); const resDiv=document.getElementById("resultA"); resDiv.innerHTML="";
  Array.from(cont.getElementsByClassName("ref-block")).forEach(block=>{
    const ref=parseFloat(block.querySelector(".refA").value);
    const vals=block.querySelector(".valsA").value.trim().split(/\s+/).map(Number);
    if(isNaN(ref)||vals.length===0) return;
    let html=`<h4>Analyse pour ${ref.toFixed(2)}€</h4><table><tr><th>Valeur</th><th>Différence (€)</th><th>%</th></tr>`;
    vals.forEach(v=>{ 
      if(!isNaN(v)){ 
        const diff=v-ref; 
        const pct=(diff/ref*100); 
        const col=diff>0?"#2ecc71":diff<0?"#e74c3c":"#95a5a6"; 
        html+=`<tr style="background:${col};color:white;"><td>${v}</td><td>${diff>=0?'+':''}${diff.toFixed(2)}</td><td>${pct.toFixed(2)}%</td></tr>`;
      }
    });
    html+="</table>"; resDiv.innerHTML+=html;
  });
  historique.A=resDiv.innerHTML;
}

function calculerD(){
  const cont = document.getElementById("multiDContainer");
  const datasets = [];
  let labelsSet = new Set();

  Array.from(cont.getElementsByClassName("ref-block")).forEach((block,i)=>{
    const label = block.querySelector(".labelD")?.value || `Série ${i+1}`;
    const eye = block.querySelector(".eyeD");
    const visible = eye.dataset.visible === "true";
    if(!visible) return;

    const dataMap = {};
    Array.from(block.querySelectorAll(".eventsDContainer div")).forEach(row=>{
      const date = row.querySelector(".dateD").value;
      const val = parseFloat(row.querySelector(".valD").value);
      if(date && !isNaN(val)){
        dataMap[date] = val;
        labelsSet.add(date);
      }
    });

    datasets.push({
      label: label,
      data: dataMap,
      borderColor: ["#3498db","#2ecc71","#e74c3c","#f1c40f","#9b59b6","#1abc9c","#95a5a6"][i%7],
      fill: false,
      tension: 0.2
    });
  });

  const labels = Array.from(labelsSet).sort();
  datasets.forEach(ds=>{
    ds.data = labels.map(lab => ds.data[lab] ?? null);
  });

  if(chartD) chartD.destroy();
  const ctx = document.getElementById("graphD").getContext("2d");
  chartD = new Chart(ctx,{
    type:'line',
    data:{labels: labels, datasets: datasets},
    options:{plugins:{legend:{display:true}}}
  });

  historique.D = "Graphique multi-séries de l'évolution temporelle généré.";
}

function calculerF(){
  const lines=[];
  Array.from(document.getElementById("multiFContainer").getElementsByClassName("ref-block")).forEach(block=>{
    const label=block.querySelector(".labelF")?.value;
    const val=parseFloat(block.querySelector(".valF")?.value);
    if(label && !isNaN(val)) lines.push({label,val});
  });
  if(chartF) chartF.destroy();
  const ctx=document.getElementById("graphF").getContext("2d");
  chartF=new Chart(ctx,{
    type:'pie',
    data:{
      labels:lines.map(l=>l.label),
      datasets:[{
        data:lines.map(l=>l.val),
        backgroundColor:['#3498db','#2ecc71','#e74c3c','#f1c40f','#9b59b6','#1abc9c','#95a5a6']
      }]
    },
    options:{
      plugins:{
        legend:{
          display:true,
          onClick:(e, legendItem, legend)=>{
            const index=legendItem.index;
            const ci=legend.chart;
            const meta=ci.getDatasetMeta(0);
            meta.data[index].hidden=!meta.data[index].hidden;
            ci.update();
            // recalculer les pourcentages visibles
            const visibleTotal=ci.data.datasets[0].data.reduce((sum,val,i)=>meta.data[i].hidden?sum:sum+val,0);
            ci.data.labels=lines.map((l,i)=>meta.data[i].hidden?l.label:`${l.label} (${((l.val/visibleTotal)*100).toFixed(1)}%)`);
            ci.update();
          }
        },
        datalabels:{
          formatter:(value,ctx)=>{
            const meta=ctx.chart.getDatasetMeta(0);
            const visibleTotal=ctx.chart.data.datasets[0].data.reduce((sum,val,i)=>meta.data[i].hidden?sum:sum+val,0);
            return ((value/visibleTotal)*100).toFixed(1)+'%';
          },
          color:'#fff', font:{weight:'bold'}
        }
      }
    },
    plugins:[ChartDataLabels]
  });
  historique.F="Graphique circulaire généré avec pourcentages recalculés dynamiquement.";
}

async function genererRapport(){

  const doc = new jsPDF({orientation:'p', unit:'pt', format:'a4'});
  let y = 40;

  for (const m of ["A","B","C","D","E","F","G"]) {
    if (document.getElementById("include"+m)?.checked) {
      doc.setFontSize(14);
      doc.text(`Mode ${m}`, 40, y);
      y += 20;
      doc.setFontSize(10);
      const text = historique[m]?.replace(/<[^>]+>/g,' ').trim() || '';
      const lines = doc.splitTextToSize(text, 520);
      doc.text(lines, 40, y);
      y += lines.length * 12 + 10;

      // Ajouter les graphiques au PDF
      let chartImg = null;
      if (m === "D" && chartD) chartImg = chartD.toBase64Image();
      if (m === "E" && chartE) chartImg = chartE.toBase64Image();
      if (m === "F" && chartF) chartImg = chartF.toBase64Image();

      if (chartImg) {
        if (y > 700) { doc.addPage(); y = 40; }
        doc.addImage(chartImg, "PNG", 80, y, 400, 300);
        y += 320;
      }

      if (y > 760) { doc.addPage(); y = 40; }
    }
  }

  doc.save("rapport_global.pdf");
}

// --- Sélectionner / désélectionner tous les modes ---
function toggleAllModes(masterCheckbox) {
  const allModes = ["A","B","C","D","E","F","G"];
  allModes.forEach(m => {
    const checkbox = document.getElementById("include" + m);
    if (checkbox) checkbox.checked = masterCheckbox.checked;
  });
}
