  <script>
    function showMode(modeId) {
      document.getElementById("modeA").classList.add("hidden");
      document.getElementById("modeB").classList.add("hidden");
      document.getElementById("modeC").classList.add("hidden");
      document.getElementById(modeId).classList.remove("hidden");
    }

    // Mode A : comparaison à une référence
    function calculerA() {
      const ref = parseFloat(document.getElementById("refA").value);
      const valeurs = document.getElementById("valeursA").value.split(" ").map(Number);
      let res = "";
      if (isNaN(ref) || ref === 0) {
        res = "⚠️ Merci d'entrer une valeur de référence valide (≠ 0).";
      } else {
        valeurs.forEach(v => {
          if (!isNaN(v)) {
            let diff = ((v - ref) / ref) * 100;
            res += `${v} → ${diff >= 0 ? "Gain" : "Perte"} : ${diff.toFixed(2)}% sur la valeur de référence (${ref})\n`;
          }
        });
      }
      document.getElementById("resultatA").innerText = res;
    }

    // Mode B : évolution + moyenne + graphique
    let chartB;
    function calculerB() {
      const lignes = document.getElementById("seriesB").value.trim().split("\n");
      let res = "";
      let series = [];

      lignes.forEach((ligne, idx) => {
        let parts = ligne.trim().split(/\s+/);
        if (parts.length < 3) return; 
        let nom = parts[0];
        let valeurs = parts.slice(1).map(Number).filter(v => !isNaN(v));
        
        if (valeurs.length < 2) return;

        series.push({ nom, valeurs });

        res += `${nom} :\n`;
        let croissanceTotale = 0;
        let nbCroissances = 0;

        for (let i = 1; i < valeurs.length; i++) {
          let diff = ((valeurs[i] - valeurs[i-1]) / valeurs[i-1]) * 100;
          res += `  ${valeurs[i-1]} → ${valeurs[i]} : ${diff.toFixed(2)}%\n`;
          croissanceTotale += diff;
          nbCroissances++;
        }

        let moyenne = croissanceTotale / nbCroissances;
        res += `👉 Moyenne de croissance : ${moyenne.toFixed(2)}%\n\n`;
      });

      document.getElementById("resultatB").innerText = res;

      // Graphique
      const ctx = document.getElementById("graphB").getContext("2d");
      if (chartB) chartB.destroy();
      chartB = new Chart(ctx, {
        type: 'line',
        data: {
          labels: series[0].valeurs.map((_, i) => "Étape " + (i+1)),
          datasets: series.map((s, idx) => ({
            label: s.nom,
            data: s.valeurs,
            borderColor: `hsl(${idx*60},70%,50%)`,
            fill: false
          }))
        }
      });
    }

    // Mode C : appliquer gains et pertes
    function calculerC() {
      const base = parseFloat(document.getElementById("valeurC").value);
      const pourcents = document.getElementById("pourcentC").value.split(" ").map(Number);
      let res = "";
      if (isNaN(base)) {
        res = "⚠️ Merci d'entrer une valeur de départ valide.";
      } else {
        pourcents.forEach(p => {
          if (!isNaN(p)) {
            let nv = base * (1 + p/100);
            res += `${p >= 0 ? "Gain" : "Perte"} de ${p}% → ${nv.toFixed(2)}\n`;
          }
        });
      }
      document.getElementById("resultatC").innerText = res;
    }
  </script>
