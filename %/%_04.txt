<script>
let chartC, chartD;
let historique={A:'',B:'',C:'',D:''};

function showMode(id){
  ["modeA","modeB","modeC","modeD","modeE"].forEach(m=>document.getElementById(m).classList.add("hidden"));
  document.getElementById(id).classList.remove("hidden");
}

function addABlock(){
  const cont=document.getElementById("multiAContainer");
  const div=document.createElement("div"); div.className="ref-block";
  div.innerHTML=`<input type="number" class="refA" placeholder="Valeur de référence">
  <input type="text" class="valsA" placeholder="Valeurs à comparer (ex: 90 150)">
  <button type="button" onclick="removeRef(this)">🗑️</button>`;
  cont.appendChild(div);
}

function addBBlock(){
  const cont=document.getElementById("multiBContainer");
  const div=document.createElement("div"); div.className="ref-block";
  div.innerHTML=`<input type="number" class="valeurB" placeholder="Valeur de départ">
  <input type="text" class="pourcentB" placeholder="Pourcentages (ex: 10 -5 20)">
  <button type="button" onclick="removeRef(this)">🗑️</button>`;
  cont.appendChild(div);
}

function removeRef(btn){btn.parentNode.remove();}

function calculerA(){
  const cont=document.getElementById("multiAContainer"); const resDiv=document.getElementById("resultA"); resDiv.innerHTML="";
  Array.from(cont.getElementsByClassName("ref-block")).forEach(block=>{
    const ref=parseFloat(block.querySelector(".refA").value);
    const vals=block.querySelector(".valsA").value.trim().split(/\s+/).map(Number);
    if(isNaN(ref)||vals.length===0) return;
    let html=`<h4>Analyse pour ${ref.toFixed(2)}€</h4><table><tr><th>Valeur</th><th>Différence (€)</th><th>%</th></tr>`;
    vals.forEach(v=>{
      if(!isNaN(v)){
        const diff=v-ref; const pct=(diff/ref*100);
        const col=diff>0?"#2ecc71":diff<0?"#e74c3c":"#95a5a6";
        html+=`<tr style="background:${col};color:white;"><td>${v}</td><td>${diff>=0?'+':''}${diff.toFixed(2)}</td><td>${pct.toFixed(2)}%</td></tr>`;
      }
    });
    html+="</table>";
    resDiv.innerHTML+=html;
  });
  historique.A=resDiv.innerHTML;
}

function calculerB(){
  const cont=document.getElementById("multiBContainer"); const resDiv=document.getElementById("resultB"); resDiv.innerHTML="";
  Array.from(cont.getElementsByClassName("ref-block")).forEach(block=>{
    const val=parseFloat(block.querySelector(".valeurB").value);
    const taux=block.querySelector(".pourcentB").value.trim().split(/\s+/).map(Number);
    if(isNaN(val)||taux.length===0) return;
    let html=`<h4>Analyse pour ${val.toFixed(2)}€</h4><table><tr><th>Taux (%)</th><th>Résultat (€)</th></tr>`;
    taux.forEach(p=>{
      if(!isNaN(p)){
        const r=val*(1+p/100);
        const col=p>0?"#2ecc71":p<0?"#e74c3c":"#95a5a6";
        html+=`<tr style="background:${col};color:white;"><td>${p}%</td><td>${r.toFixed(2)}</td></tr>`;
      }
    });
    html+="</table>"; resDiv.innerHTML+=html;
  });
  historique.B=resDiv.innerHTML;
}

function calculerC(){
  const titre=document.getElementById("titreC").value.trim()||"Graphique";
  const lignes=document.getElementById("seriesC").value.trim().split("\n").filter(l=>l.trim().length);
  if(!lignes.length)return alert("Ajoute au moins une série");
  const series=lignes.map(l=>{const p=l.split(/\s+/); return {nom:p[0],valeurs:p.slice(1).map(Number)}}); 
  const maxLen=Math.max(...series.map(s=>s.valeurs.length));
  const labels=Array.from({length:maxLen},(_,i)=>"Étape "+(i+1));
  const ctx=document.getElementById("graphC").getContext("2d");
  if(chartC) chartC.destroy();
  chartC=new Chart(ctx,{type:"line",data:{labels,datasets:series.map((s,i)=>({label:s.nom,data:s.valeurs,borderColor:`hsl(${i*60},70%,50%)`,fill:false,tension:0.2}))},options:{plugins:{title:{display:true,text:titre},legend:{position:"bottom"}}}});
  historique.C=titre;
}

function calculerD(){
  const raw=document.getElementById("categoriesD").value.trim();
  if(!raw)return alert("Merci d’entrer des données");
  const lignes=raw.split("\n").filter(l=>l.trim().length);
  const titre=lignes[0]; 
  const labels=[]; const valeurs=[];
  lignes.slice(1).forEach(l=>{const p=l.split(/\s+/); labels.push(p.slice(0,-1).join(" ")); valeurs.push(parseFloat(p[p.length-1]));});
  const ctx=document.getElementById("graphD").getContext("2d");
  if(chartD) chartD.destroy();
  chartD=new Chart(ctx,{
    type:'pie',
    data:{labels,datasets:[{data:valeurs,backgroundColor:labels.map((_,i)=>`hsl(${i*60},70%,60%)`)}]},
    options:{plugins:{
      title:{display:true,text:titre},
      legend:{position:"bottom"},
      datalabels:{color:'#fff',formatter:(val,ctx)=>((val/ctx.chart.data.datasets[0].data.reduce((a,b)=>a+b,0))*100).toFixed(1)+'%'}
    }},
    plugins:[ChartDataLabels]
  });
  historique.D={titre,labels,valeurs};
}

/* Rapport global */
function previsualiserRapport(){
  const cont=document.getElementById("previewContainer"); cont.innerHTML="";
  if(document.getElementById("includeA").checked)
    cont.innerHTML+=`<div class='preview-section'><h3>Mode A</h3>${historique.A||"<i>Pas de données</i>"}</div>`;
  if(document.getElementById("includeB").checked)
    cont.innerHTML+=`<div class='preview-section'><h3>Mode B</h3>${historique.B||"<i>Pas de données</i>"}</div>`;
  if(document.getElementById("includeC").checked && chartC)
    cont.innerHTML+=`<div class='preview-section'><h3>Mode C : ${historique.C}</h3><img src='${chartC.toBase64Image()}' width='500'></div>`;
  if(document.getElementById("includeD").checked && chartD)
    cont.innerHTML+=`<div class='preview-section'><h3>Mode D : ${historique.D.titre}</h3><img src='${chartD.toBase64Image()}' width='400'></div>`;
}

async function genererRapport(){
  const cont=document.getElementById("previewContainer");
  if(!cont.innerHTML) return alert("Rien à exporter !");
  const { jsPDF } = window.jspdf;
  const pdf = new jsPDF('p','mm','a4');
  const canvas = await html2canvas(cont,{scale:2});
  const img = canvas.toDataURL("image/png");
  const pdfWidth = pdf.internal.pageSize.getWidth();
  const pdfHeight = (canvas.height * pdfWidth) / canvas.width;
  pdf.addImage(img, 'PNG', 0, 0, pdfWidth, pdfHeight);
  pdf.save("rapport_global.pdf");
}
