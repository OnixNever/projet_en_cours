function parseNumber(value) {
    if (typeof value === 'string') {
        value = value.replace(',', '.');
    }
    const num = parseFloat(value);
    return isNaN(num) ? null : num;
}

function formatDateForDisplay(dateString) {
    if (!dateString) return '';
    if (dateString.match(/^\d{4}-\d{2}-\d{2}$/)) {
        const [yyyy, mm, dd] = dateString.split('-');
        return `${dd}/${mm}/${yyyy}`;
    }
    return dateString;
}

function calculerA(){
    const cont = document.getElementById("multiAContainer");
    const resDiv = document.getElementById("resultA"); resDiv.innerHTML = "";
    Array.from(cont.getElementsByClassName("ref-block")).forEach(block => {
        const ref = parseNumber(block.querySelector(".refA").value);
        const raw = (block.querySelector(".valsA").value || "").trim();
        if(raw === "") return;
        const vals = raw.split(/\s+/).map(parseNumber).filter(v => v !== null); 
        if(ref === null && vals.length === 0) return;
        
        let html = `<h4>Analyse pour ${ref === null ? 'n/a': ref.toFixed(2)}€</h4><table class="result-table"><tr><th>Valeur</th><th>Différence (€)</th><th>%</th></tr>`;
        vals.forEach(v => {
            const diff = ref === null ? v : v - ref;
            const pct = (ref === null || ref === 0) ? null : (diff / ref * 100);
            const col = diff > 0 ? "#2ecc71" : diff < 0 ? "#e74c3c" : "#95a5a6";
            html += `<tr style="background:${col}; color:#fff;"><td>${v.toFixed(2)}</td><td>${diff>=0?'+':''}${diff.toFixed(2)}</td><td>${pct===null? '—' : pct.toFixed(2)+'%'}</td></tr>`;
        });
        html += "</table>";
        resDiv.innerHTML += html;
    });
    historique.A = resDiv.innerHTML;
}

function calculerB(){

        let html = `<h4>Analyse pour ${val.toFixed(2)}€</h4><table class="result-table"><tr><th>Taux (%)</th><th>Variation (€)</th><th>Résultat (€)</th></tr>`;

            const r = val * (1 + p/100);
            const variation = r - val; // VARIABLE CORRECTEMENT DÉCLARÉE
            const col = p > 0 ? "#2ecc71" : p < 0 ? "#e74c3c" : "#95a5a6";
            html += `<tr style="background:${col}; color:#fff;">
                <td>${p}%</td>
                <td>${variation >= 0 ? '+' : ''}${variation.toFixed(2)}</td>
                <td>${r.toFixed(2)}</td>
            </tr>`;
        });
        html += "</table>";
        resDiv.innerHTML += html;
    });
    historique.B = resDiv.innerHTML;
}

function updateCBlock(rangeInput){

    rangeInput.max = max;

    updateC();
}

function updateC(){

        if(vals.length){

function addDDate(defaultDate = null) {
    if (!defaultDate) {
        const today = new Date();
        const yyyy = today.getFullYear();
        const mm = String(today.getMonth() + 1).padStart(2, '0');
        const dd = String(today.getDate()).padStart(2, '0');
        defaultDate = `${yyyy}-${mm}-${dd}`;
    }
    const newDateId = Date.now() + Math.random(); 
    dDates.push({ id: newDateId, date: defaultDate });
    dSeries.forEach(s => { s.values[newDateId] = ''; });
    updateDTable();
}

function removeDDate(dateId) {
    dDates = dDates.filter(d => d.id !== dateId);
    dSeries.forEach(s => { delete s.values[dateId]; });
    updateDTable();
}

function updateDDateValue(dateId, newDateString) {
    const dateObj = dDates.find(d => d.id === dateId);
    if (dateObj) {
        dateObj.date = newDateString;
    }
    updateDTable(false);
}

function sortDDates() {

function addDSeries() {
    const count = dSeries.length + 1;
    const newSeriesId = Date.now() + Math.random();
    
    const newSeries = { 
        id: newSeriesId, 
        label: `Série ${count}`, 
        color: D_COLORS[dSeries.length % D_COLORS.length],

function removeDSeries(seriesId) {
    dSeries = dSeries.filter(s => s.id !== seriesId);

function toggleDSeriesVisibility(seriesId, eyeButton) {
    const series = dSeries.find(s => s.id === seriesId);
    if (series) {
        series.visible = !series.visible;
        eyeButton.textContent = series.visible ? "👁️" : "✖️"; 
        eyeButton.title = series.visible ? "Cacher cette série" : "Afficher cette série";

function updateDSeriesProp(seriesId, prop, value) {
    const series = dSeries.find(s => s.id === seriesId);
    if (series) {
        series[prop] = value;

        calculerD();
    }
}

function hexToRgbArr(hex){
    const c = (hex||'#000000').replace('#','');
    return [parseInt(c.substr(0,2),16), parseInt(c.substr(2,2),16), parseInt(c.substr(4,2),16)];
}

function rgbArrToHex(arr){
    return "#" + arr.map(n => n.toString(16).padStart(2,'0')).join('');
}

function mettreAJourTableauF(){
    const tbody = document.querySelector("#tableauF tbody");
    tbody.innerHTML = "";
    const total = historiqueF.reduce((s,c) => s + (parseFloat(c.val) || 0), 0);
    
    historiqueF.forEach(cat => {
        const pourcent = total > 0 ? (cat.val / total * 100).toFixed(1) : 0;
        const tr = document.createElement("tr");
        tr.style.backgroundColor = cat.color;
        
        const rgb = hexToRgbArr(cat.color);
        const luminance = (0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]) / 255;
        tr.style.color = luminance > 0.5 ? '#000' : '#fff';

        let sousHTML = "";
        if(cat.sub && cat.sub.length > 0){
            const isDarkMode = document.body.classList.contains('dark-mode');
            const subTextColor = isDarkMode ? '#ecf0f1' : '#333';
            const subBgColor = isDarkMode ? '#4a627a' : '#fff';
            sousHTML = `<div style="text-align:left; padding:4px; color:${subTextColor}; background:${subBgColor}; border-radius:4px; margin:2px; font-size:12px;">` + cat.sub.map(sc => {
                const pct = total > 0 ? ((sc.val / total) * 100).toFixed(1) : 0;
                return `<div>${sc.label}: ${sc.val.toFixed(2)} (${pct}%)</div>`;
            }).join("") + '</div>';
        } 
        tr.innerHTML = `<td>${cat.label}</td><td>${cat.val.toFixed(2)}</td><td>${pourcent}%</td><td>${sousHTML}</td>`;
        tbody.appendChild(tr);
    });
    calculerF();
}

function fusionnerFCategories(){
    const cont = document.getElementById("multiFContainer");
    const selectedBlocks = Array.from(cont.getElementsByClassName("ref-block")).filter(d => d.dataset.selectionnee === "true");
    
    if(selectedBlocks.length < 2){ alert("Sélectionnez au moins 2 catégories à fusionner !"); return; }

    let nomNouvelle = prompt("Entrez le nom de la nouvelle catégorie fusionnée :", "Catégorie Fusionnée");
    if(!nomNouvelle) nomNouvelle = "Catégorie Fusionnée";

    let valTotale = 0;
    const rgbs = [];
    const sous = [];
    const elementsToRemove = [];

    selectedBlocks.forEach(block => {
        const h = historiqueF.find(c => c.element === block);
        if(h){
            valTotale += h.val;
            rgbs.push(hexToRgbArr(h.color));
            sous.push({ label: h.label, val: h.val, color: h.color, isParent: true });
            if(h.sub && h.sub.length) sous.push(...h.sub);
            elementsToRemove.push(h);
        }
    });

    const moy = rgbs[0].map((_,i) => Math.round(rgbs.reduce((s,c) => s + c[i], 0) / rgbs.length));
    const couleurFusion = rgbArrToHex(moy);

    elementsToRemove.forEach(h => {
        h.element.remove();
        const idx = historiqueF.findIndex(c => c === h);
        if(idx !== -1) historiqueF.splice(idx,1);
    });

    addFBlock({ 
        label: nomNouvelle, 
        val: valTotale, 
        color: couleurFusion, 
        sub: sous 
    });

    mettreAJourTableauF();
    calculerF();
}

function propagateGlobalTvaToBlock(blockId) {

    const lines = block.querySelectorAll('.ligne-tva');

function calculerG(){
    const cont = document.getElementById("multiGContainer");
    const resDiv = document.getElementById("resultG"); 
    resDiv.innerHTML = "";

    const blocks = Array.from(cont.getElementsByClassName('g-block'));
    if (blocks.length === 0) {
        resDiv.innerHTML = "<p>Aucun bloc TVA défini.</p>";
        historique.G = resDiv.innerHTML;
        return;
    }

    let globalHTML = "";
    
    blocks.forEach((block, idx) => {
        const blockName = block.querySelector('.bloc-nom-input')?.value || `Bloc ${idx+1}`;
        const globalTva = parseFloat(block.querySelector('.global-tva-input')?.value) || 0;
        const lines = Array.from(block.querySelectorAll('.ligne-tva'));

        let html = `<h4>${blockName} (TVA globale ${globalTva}%)</h4>`;
        html += `<table class="result-table"><tr>
            <th>Libellé</th><th>Montant donné</th><th>Taux TVA</th><th>Montant TVA</th><th>Montant HT</th><th>Montant TTC</th>
        </tr>`;

        let totalTVA = 0, totalHT = 0, totalTTC = 0;
        lines.forEach(l => {
            const lib = l.querySelector('.libelleG')?.value || '';
            const raw = parseFloat(l.querySelector('.valeurG')?.value);
            if (isNaN(raw)) return;
            const type = l.querySelector('.typeG')?.value || 'HT';
            const tva = parseFloat(l.querySelector('.tvaIndividuelleG')?.value);
            const applicableTva = isNaN(tva) ? globalTva : tva;

            const factor = 1 + applicableTva/100;
            let montantHT, montantTTC, tvaVal;
            if (type === 'HT') {
                montantHT = raw;
                montantTTC = montantHT * factor;
                tvaVal = montantTTC - montantHT;
            } else {
                montantTTC = raw;
                montantHT = montantTTC / factor;
                tvaVal = montantTTC - montantHT;
            }

            totalTVA += tvaVal;
            totalHT += montantHT;
            totalTTC += montantTTC;

            html += `<tr>
                <td style="text-align:left;">${lib}</td>
                <td>${raw.toFixed(2)} € (${type})</td>
                <td>${applicableTva}%</td>
                <td>${tvaVal.toFixed(2)} €</td>
                <td>${montantHT.toFixed(2)} €</td>
                <td>${montantTTC.toFixed(2)} €</td>
            </tr>`;
        });

        html += `<tr style="font-weight:bold; background:#ecf0f1; color:#333;">
            <td style="text-align:left;">Totaux</td>
            <td>—</td><td>—</td>
            <td>${totalTVA.toFixed(2)} €</td>
            <td>${totalHT.toFixed(2)} €</td>
            <td>${totalTTC.toFixed(2)} €</td>
        </tr>`;

        html += `</table>`;
        resDiv.innerHTML += html;
        globalHTML += html; // Stocker aussi pour l'historique
    });

    historique.G = globalHTML; // CORRECTION : stocker le HTML complet
}

    const modeCheckboxes = document.querySelectorAll('#mode_rapport_global input[type="checkbox"]:not(#selectAllModes)');
    modeCheckboxes.forEach(cb => cb.checked = isChecked);
}

                        display: true,
                        position: 'bottom',
                        labels: {
                            font: { size: 12 },
                            padding: 20,
                            usePointStyle: true
                        }
                    };
                }
                
                // Créer le nouveau graphique
                new Chart(previewCanvas.getContext('2d'), {
                    type: chartType,
                    data: chartData,
                    options: chartOptions
                });
            }
        });
    }, 300);
}

/* =========================
    NOUVELLE FONCTION POUR PDF AVEC TEXTE SÉLECTIONNABLE
    ========================= */

async function genererRapport() {
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF('p', 'mm', 'a4');
    
    // Configuration de base
    const pageWidth = doc.internal.pageSize.width;
    const margin = 20;
    const contentWidth = pageWidth - 2 * margin;
    let y = margin;
    
    // Styles améliorés
    const styles = {
        title: { size: 24, bold: true, color: [40, 40, 40] },
        subtitle: { size: 18, bold: true, color: [60, 60, 60] },
        section: { size: 16, bold: true, color: [80, 80, 80] },
        normal: { size: 12, bold: false, color: [0, 0, 0] },
        small: { size: 10, bold: false, color: [100, 100, 100] },
        tableHeader: { size: 10, bold: true, color: [255, 255, 255] },
        tableRow: { size: 9, bold: false, color: [60, 60, 60] }
    };
    
    // Fonction pour ajouter une nouvelle page si nécessaire
    function checkPageHeight(additionalHeight) {
        if (y + additionalHeight > 280) {
            doc.addPage();
            y = margin;
            return true;
        }
        return false;
    }
    
    // Fonction pour appliquer un style
    function applyStyle(style) {
        doc.setFontSize(style.size);
        doc.setTextColor(...style.color);
        doc.setFont("helvetica", style.bold ? "bold" : "normal");
    }
    
    // Fonction pour ajouter du texte avec retour à la ligne automatique
    function addText(text, x = margin, style = styles.normal) {
        applyStyle(style);
        
        const lineHeight = style.size * 0.35;
        const maxWidth = contentWidth;
        
        // Découper le texte en lignes
        const lines = doc.splitTextToSize(text, maxWidth);
        
        // Vérifier si on a assez de place
        checkPageHeight(lines.length * lineHeight);
        
        // Ajouter chaque ligne
        lines.forEach(line => {
            doc.text(line, x, y);
            y += lineHeight;
        });
        
        return lines.length;
    }
    
    // Fonction utilitaire pour convertir hex en RGB
    function hexToRgb(hex) {
        if (!hex || hex.length !== 7) return [0, 0, 0];
        const r = parseInt(hex.slice(1, 3), 16);
        const g = parseInt(hex.slice(3, 5), 16);
        const b = parseInt(hex.slice(5, 7), 16);
        return [r, g, b];
    }
    
    // Fonction pour calculer les largeurs de colonnes intelligentes
    function calculateColumnWidths(headers, rows) {
        const minWidth = 25;
        const maxWidth = 70;
        
        // Largeurs par défaut selon le type de colonne
        const defaultWidths = {
            'Valeur': 40,
            'Différence': 45,
            '%': 25,
            'Taux': 35,
            'Variation': 45,
            'Résultat': 50,
            'Valeur initiale': 55,
            'Date': 50,
            'Catégorie': 65,
            'Valeur (€)': 50,
            'Pourcentage': 40,
            'Sous-catégories': 75,
            'Libellé': 65,
            'Montant': 45,
            'TVA %': 35,
            'Montant TVA': 50,
            'HT': 35,
            'TTC': 35
        };
        
        const colWidths = [];
        
        for (let i = 0; i < headers.length; i++) {
            let width = 45; // Largeur par défaut
            
            // Largeur prédéfinie selon le nom de la colonne
            for (const [key, presetWidth] of Object.entries(defaultWidths)) {
                if (headers[i].includes(key)) {
                    width = presetWidth;
                    break;
                }
            }
            
            // Ajuster selon le contenu réel
            let maxContentLength = headers[i].length;
            rows.forEach(row => {
                if (row[i] && row[i].toString) {
                    const cellText = row[i].toString();
                    // Ne pas compter les symboles € et % dans la longueur
                    const effectiveLength = cellText.replace(/[€%+\-]/g, '').trim().length;
                    maxContentLength = Math.max(maxContentLength, effectiveLength);
                }
            });
            
            // Calcul basé sur la longueur (3.5mm par caractère)
            const calculatedWidth = Math.min(maxWidth, Math.max(minWidth, maxContentLength * 3.5));
            width = Math.max(width, calculatedWidth);
            
            colWidths.push(width);
        }
        
        // Ajuster pour ne pas dépasser la largeur disponible
        const totalWidth = colWidths.reduce((a, b) => a + b, 0);
        if (totalWidth > contentWidth) {
            const scaleFactor = contentWidth / totalWidth;
            return colWidths.map(width => Math.floor(width * scaleFactor * 0.95));
        }
        
        return colWidths;
    }
    
    // Fonction pour ajouter un tableau avec alignement PRÉCIS
    function addTable(headers, rows, x = margin, colors = null, modeType = '') {
        // 1. Calculer les largeurs de colonnes
        const colWidths = calculateColumnWidths(headers, rows);
        const rowHeight = 8;
        const headerHeight = 10;
        const tableWidth = colWidths.reduce((a, b) => a + b, 0);
        
        // 2. Position X centrée si le tableau est plus petit que la largeur
        let tableX = x;
        if (tableWidth < contentWidth) {
            tableX = x + (contentWidth - tableWidth) / 2;
        }
        
        // 3. Vérifier la hauteur disponible
        checkPageHeight((rows.length * rowHeight) + headerHeight + 15);
        
        // 4. COULEURS D'EN-TÊTE PAR MODE
        const headerColors = {
            'A': [52, 152, 219],    // Bleu
            'B': [46, 204, 113],    // Vert
            'C': [155, 89, 182],    // Violet
            'D': [241, 196, 15],    // Jaune
            'F': [231, 76, 60],     // Rouge
            'G': [26, 188, 156]     // Turquoise
        };
        const headerBgColor = headerColors[modeType] || [52, 152, 219];
        
        // 5. EN-TÊTE DU TABLEAU - TEXTE BIEN CENTRÉ
        applyStyle({ size: 10, bold: true, color: [255, 255, 255] });
        let currentX = tableX;
        
        headers.forEach((header, i) => {
            // Fond de l'en-tête
            doc.setFillColor(...headerBgColor);
            doc.rect(currentX, y, colWidths[i], headerHeight, 'F');
            
            // TEXTE PARFAITEMENT CENTRÉ
            const text = header;
            const fontSize = 10;
            // Calculer la largeur exacte du texte
            const textWidth = (doc.getStringUnitWidth(text) * fontSize) / doc.internal.scaleFactor;
            const textX = currentX + (colWidths[i] - textWidth) / 2;
            
            doc.text(text, textX, y + 7);
            
            currentX += colWidths[i];
        });
        y += headerHeight;
        
        // 6. LIGNES DU TABLEAU - ALIGNEMENT PRÉCIS
        applyStyle({ size: 9, bold: false, color: [40, 40, 40] });
        
        rows.forEach((row, rowIndex) => {
            checkPageHeight(rowHeight);
            
            currentX = tableX;
            
            row.forEach((cell, colIndex) => {
                const cellText = cell.toString();
                
                // DÉTERMINER LA POSITION X EXACTE
                let textX;
                
                if (cellText.includes('€') || cellText.includes('%') || 
                    (!isNaN(parseFloat(cellText.replace(',', '.').replace('€', '').replace('%', '').trim())) && 
                     cellText.trim() !== '')) {
                    // ALIGNEMENT À DROITE pour nombres, monnaies, pourcentages
                    const textWidth = (doc.getStringUnitWidth(cellText) * 9) / doc.internal.scaleFactor;
                    textX = currentX + colWidths[colIndex] - textWidth - 3;
                } else if (colIndex === 0) {
                    // Première colonne : alignement à GAUCHE
                    textX = currentX + 3;
                } else {
                    // Autres colonnes : CENTRÉ
                    const textWidth = (doc.getStringUnitWidth(cellText) * 9) / doc.internal.scaleFactor;
                    textX = currentX + (colWidths[colIndex] - textWidth) / 2;
                }
                
                // Fond alterné
                if (rowIndex % 2 === 0) {
                    doc.setFillColor(250, 250, 250);
                } else {
                    doc.setFillColor(255, 255, 255);
                }
                doc.rect(currentX, y, colWidths[colIndex], rowHeight, 'F');
                
                // Bordures légères
                doc.setDrawColor(230, 230, 230);
                doc.setLineWidth(0.1);
                doc.rect(currentX, y, colWidths[colIndex], rowHeight);
                
                // Texte
                doc.text(cellText, textX, y + 6);
                
                currentX += colWidths[colIndex];
            });
            
            y += rowHeight;
        });
        
        y += 12; // Espace après le tableau
    }
    
    // Fonctions de traitement pour chaque mode avec design spécifique
    async function processModeA() {
        if (!historique.A) return;
        
        addText("Mode A : Comparaison de valeurs", margin, styles.subtitle);
        y += 8;
        
        const parser = new DOMParser();
        const docHTML = parser.parseFromString(historique.A, 'text/html');
        const tables = docHTML.querySelectorAll('table');
        
        tables.forEach((table, tableIndex) => {
            const headers = ['Valeur (€)', 'Différence (€)', 'Pourcentage'];
            const rows = [];
            const rowColors = [];
            
            table.querySelectorAll('tbody tr').forEach(tr => {
                const row = [];
                tr.querySelectorAll('td').forEach(td => {
                    row.push(td.textContent.trim());
                });
                if (row.length > 0) {
                    rows.push(row);
                    
                    // Déterminer la couleur selon la valeur
                    const diffText = row[1];
                    let color;
                    
                    if (diffText.includes('+')) {
                        color = [46, 204, 113]; // Vert pour gain
                    } else if (diffText.includes('-')) {
                        color = [231, 76, 60]; // Rouge pour perte
                    } else {
                        color = [149, 165, 166]; // Gris pour neutre
                    }
                    
                    rowColors.push(color);
                }
            });
            
            if (headers.length > 0 && rows.length > 0) {
                // Ajouter un titre pour chaque bloc
                if (tableIndex > 0) {
                    y += 5;
                }
                
                addTable(headers, rows, margin, rowColors, 'A');
            }
        });
    }

    }

                        color = [46, 204, 113]; // Vert pour gain
                    } else if (variationText.includes('-')) {
                        color = [231, 76, 60]; // Rouge pour perte

                        color = [149, 165, 166]; // Gris pour neutre
                    }
                    
                    rowColors.push(color);
                }
            });
            
            if (rows.length > 0) {

        });
    }
    
    async function processModeD() {
        if (!chartD || dSeries.length === 0) return;
        
        addText("Mode D : Évolution temporelle", margin, styles.subtitle);
        y += 12;
        
        try {
            // Créer un graphique haute résolution pour le PDF

            canvas.height = 700;
            canvas.style.display = 'none';
            document.body.appendChild(canvas);

    async function processModeF() {

    function processModeG() {
        if (!historique.G) return;
        
        addText("Mode G : Calculateur TVA", margin, styles.subtitle);
        y += 8;
        
        const parser = new DOMParser();
        const docHTML = parser.parseFromString(historique.G, 'text/html');
        const tables = docHTML.querySelectorAll('table');
        
        tables.forEach((table, tableIndex) => {
            // Titre du bloc
            const previousSibling = table.previousElementSibling;
            if (previousSibling && previousSibling.tagName === 'H4') {
                addText(previousSibling.textContent, margin, { size: 14, bold: true, color: [80, 80, 80] });
                y += 3;
            }
            
            const headers = ['Libellé', 'Montant', 'TVA %', 'Montant TVA', 'HT', 'TTC'];
            const rows = [];
            
            // Extraire toutes les lignes
            const allRows = [];
            table.querySelectorAll('tbody tr').forEach(tr => {
                const row = [];
                tr.querySelectorAll('td').forEach(td => {
                    row.push(td.textContent.trim());
                });
                if (row.length > 0) allRows.push(row);
            });
            
            // Séparer les lignes détaillées des totaux
            const detailRows = allRows.slice(0, -1);
            const totalRow = allRows[allRows.length - 1];
            
            // Ajouter les lignes détaillées
            detailRows.forEach(row => rows.push(row));
            
            // Ajouter une ligne vide de séparation visuelle
            if (detailRows.length > 0) {
                rows.push(['', '', '', '', '', '']);
            }
            
            // Ajouter la ligne de totaux
            rows.push(totalRow);
            
            if (rows.length > 0) {
                addTable(headers, rows, margin, null, 'G');
            }
            
            y += 5;
        });
    }

        { id: 'B', title: 'Mode B : Calcul de gains et pertes', process: processModeB },
        { id: 'C', title: 'Mode C : Application de pourcentages par bloc', process: processModeC },
        { id: 'D', title: 'Mode D : Évolution temporelle', process: processModeD },
        { id: 'E', title: 'Mode E : Graphique linéaire multi-séries', process: processModeE },
        { id: 'F', title: 'Mode F : Répartition circulaire', process: processModeF },
        { id: 'G', title: 'Mode G : Calculateur TVA', process: processModeG }
    ];
    
    let modeCount = 0;
    
    // Exécuter les modes sélectionnés
    for (const mode of modes) {
        if (document.getElementById("include" + mode.id).checked) {
            modeCount++;
            
            // Ajouter un saut de page entre les modes (sauf le premier)
            if (mode.id !== 'A' && y > 100) {
                // Ligne de séparation fine
                doc.setDrawColor(220, 220, 220);
                doc.setLineWidth(0.5);
                doc.line(margin, y - 10, pageWidth - margin, y - 10);

                if (y > 220) {
                    doc.addPage();
                    y = margin;
                    
                    // En-tête de page
                    doc.setFillColor(250, 250, 250);
                    doc.rect(0, 0, pageWidth, 30, 'F');
                    addText("Rapport Global Calculateur (suite)", margin, { size: 14, bold: true, color: [80, 80, 80] });
                    y += 20;
                }
            }
            
            await mode.process();
        }
    }
    
    // Message si aucun mode sélectionné
    if (modeCount === 0) {
        addText("Aucun mode sélectionné", pageWidth / 2, { size: 16, bold: true, color: [150, 150, 150] }, true);
        addText("Veuillez sélectionner au moins un mode dans l'interface", margin, { size: 12, bold: false, color: [180, 180, 180] });
    }

    const pageCount = doc.internal.getNumberOfPages();
    for (let i = 1; i <= pageCount; i++) {

            `Page ${i} / ${pageCount}`,

            285,
            { align: 'right' }
        );
        
        doc.text(
            `Calculateur Global • ${now.getFullYear()}`,
            margin,
            285
        );
    }
    
    // Sauvegarder le PDF avec un nom significatif
    const fileName = `Rapport_Calculateur_${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, '0')}${String(now.getDate()).padStart(2, '0')}_${String(now.getHours()).padStart(2, '0')}${String(now.getMinutes()).padStart(2, '0')}.pdf`;
    doc.save(fileName);
