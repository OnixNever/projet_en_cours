function parseNumber(value) {
    if (typeof value === 'string') {
        value = value.replace(',', '.');
    }
    const num = parseFloat(value);
    return isNaN(num) ? null : num;
}

function formatDateForDisplay(dateString) {
    if (!dateString) return '';
    if (dateString.match(/^\d{4}-\d{2}-\d{2}$/)) {
        const [yyyy, mm, dd] = dateString.split('-');
        return `${dd}/${mm}/${yyyy}`;
    }
    return dateString;
}

function calculerA(){
    const cont = document.getElementById("multiAContainer");
    const resDiv = document.getElementById("resultA"); resDiv.innerHTML = "";
    Array.from(cont.getElementsByClassName("ref-block")).forEach(block => {
        const ref = parseNumber(block.querySelector(".refA").value);
        const raw = (block.querySelector(".valsA").value || "").trim();
        if(raw === "") return;
        const vals = raw.split(/\s+/).map(parseNumber).filter(v => v !== null); 
        if(ref === null && vals.length === 0) return;
        
        let html = `<h4>Analyse pour ${ref === null ? 'n/a': ref.toFixed(2)}€</h4><table class="result-table"><tr><th>Valeur</th><th>Différence (€)</th><th>%</th></tr>`;
        vals.forEach(v => {
            const diff = ref === null ? v : v - ref;
            const pct = (ref === null || ref === 0) ? null : (diff / ref * 100);
            const col = diff > 0 ? "#2ecc71" : diff < 0 ? "#e74c3c" : "#95a5a6";
            html += `<tr style="background:${col}; color:#fff;"><td>${v.toFixed(2)}</td><td>${diff>=0?'+':''}${diff.toFixed(2)}</td><td>${pct===null? '—' : pct.toFixed(2)+'%'}</td></tr>`;
        });
        html += "</table>";
        resDiv.innerHTML += html;
    });
    historique.A = resDiv.innerHTML;
}

function calculerB(){

            const r = val * (1 + p/100);
            const variation = r - val; // NOUVEAU : Calcul de la variation

                <td>${p}%</td>
                <td>${variation >= 0 ? '+' : ''}${variation.toFixed(2)}</td>
                <td>${r.toFixed(2)}</td>
            </tr>`;
        });
        html += "</table>";
        resDiv.innerHTML += html;
    });
    historique.B = resDiv.innerHTML;
}

function updateC(){

function addDDate(defaultDate = null) {
    if (!defaultDate) {
        const today = new Date();
        const yyyy = today.getFullYear();
        const mm = String(today.getMonth() + 1).padStart(2, '0');
        const dd = String(today.getDate()).padStart(2, '0');
        defaultDate = `${yyyy}-${mm}-${dd}`;
    }
    const newDateId = Date.now() + Math.random(); 
    dDates.push({ id: newDateId, date: defaultDate });
    dSeries.forEach(s => { s.values[newDateId] = ''; });
    updateDTable();
}

function removeDDate(dateId) {
    dDates = dDates.filter(d => d.id !== dateId);
    dSeries.forEach(s => { delete s.values[dateId]; });
    updateDTable();
}

function updateDDateValue(dateId, newDateString) {
    const dateObj = dDates.find(d => d.id === dateId);
    if (dateObj) {
        dateObj.date = newDateString;
    }
    updateDTable(false);
}

function addDSeries() {

        label: `Série ${count}`,

function hexToRgbArr(hex){
    const c = (hex||'#000000').replace('#','');
    return [parseInt(c.substr(0,2),16), parseInt(c.substr(2,2),16), parseInt(c.substr(4,2),16)];
}

function rgbArrToHex(arr){
    return "#" + arr.map(n => n.toString(16).padStart(2,'0')).join('');
}

function mettreAJourTableauF(){
    const tbody = document.querySelector("#tableauF tbody");
    tbody.innerHTML = "";
    const total = historiqueF.reduce((s,c) => s + (parseFloat(c.val) || 0), 0);
    
    historiqueF.forEach(cat => {
        const pourcent = total > 0 ? (cat.val / total * 100).toFixed(1) : 0;
        const tr = document.createElement("tr");
        tr.style.backgroundColor = cat.color;
        
        const rgb = hexToRgbArr(cat.color);
        const luminance = (0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]) / 255;
        tr.style.color = luminance > 0.5 ? '#000' : '#fff';

        let sousHTML = "";
        if(cat.sub && cat.sub.length > 0){
            const isDarkMode = document.body.classList.contains('dark-mode');
            const subTextColor = isDarkMode ? '#ecf0f1' : '#333';
            const subBgColor = isDarkMode ? '#4a627a' : '#fff';
            sousHTML = `<div style="text-align:left; padding:4px; color:${subTextColor}; background:${subBgColor}; border-radius:4px; margin:2px; font-size:12px;">` + cat.sub.map(sc => {
                const pct = total > 0 ? ((sc.val / total) * 100).toFixed(1) : 0;
                return `<div>${sc.label}: ${sc.val.toFixed(2)} (${pct}%)</div>`;
            }).join("") + '</div>';
        } 
        tr.innerHTML = `<td>${cat.label}</td><td>${cat.val.toFixed(2)}</td><td>${pourcent}%</td><td>${sousHTML}</td>`;
        tbody.appendChild(tr);
    });
    calculerF();
}

function fusionnerFCategories(){
    const cont = document.getElementById("multiFContainer");
    const selectedBlocks = Array.from(cont.getElementsByClassName("ref-block")).filter(d => d.dataset.selectionnee === "true");
    
    if(selectedBlocks.length < 2){ alert("Sélectionnez au moins 2 catégories à fusionner !"); return; }

    let nomNouvelle = prompt("Entrez le nom de la nouvelle catégorie fusionnée :", "Catégorie Fusionnée");
    if(!nomNouvelle) nomNouvelle = "Catégorie Fusionnée";

    let valTotale = 0;
    const rgbs = [];
    const sous = [];
    const elementsToRemove = [];

    selectedBlocks.forEach(block => {
        const h = historiqueF.find(c => c.element === block);
        if(h){
            valTotale += h.val;
            rgbs.push(hexToRgbArr(h.color));
            sous.push({ label: h.label, val: h.val, color: h.color, isParent: true });
            if(h.sub && h.sub.length) sous.push(...h.sub);
            elementsToRemove.push(h);
        }
    });

    const moy = rgbs[0].map((_,i) => Math.round(rgbs.reduce((s,c) => s + c[i], 0) / rgbs.length));
    const couleurFusion = rgbArrToHex(moy);

    elementsToRemove.forEach(h => {
        h.element.remove();
        const idx = historiqueF.findIndex(c => c === h);
        if(idx !== -1) historiqueF.splice(idx,1);
    });

    addFBlock({ 
        label: nomNouvelle, 
        val: valTotale, 
        color: couleurFusion, 
        sub: sous 
    });

    mettreAJourTableauF();
    calculerF();
}

function calculerG(){
    const cont = document.getElementById("multiGContainer");
    const resDiv = document.getElementById("resultG"); resDiv.innerHTML = "";

    const blocks = Array.from(cont.getElementsByClassName('g-block'));
    if (blocks.length === 0) {
        resDiv.innerHTML = "<p>Aucun bloc TVA défini.</p>";
        historique.G = resDiv.innerHTML;
        return;
    }

    blocks.forEach((block, idx) => {
        const blockName = block.querySelector('.bloc-nom-input')?.value || `Bloc ${idx+1}`;
        const globalTva = parseFloat(block.querySelector('.global-tva-input')?.value) || 0;
        const lines = Array.from(block.querySelectorAll('.ligne-tva'));

        let html = `<h4>${blockName} (TVA globale ${globalTva}%)</h4>`;
        html += `<table class="result-table"><tr>
            <th>Libellé</th><th>Montant donné</th><th>Taux TVA</th><th>Montant TVA</th><th>Montant HT</th><th>Montant TTC</th>
        </tr>`;

        let totalTVA = 0, totalHT = 0, totalTTC = 0;
        lines.forEach(l => {
            const lib = l.querySelector('.libelleG')?.value || '';
            const raw = parseFloat(l.querySelector('.valeurG')?.value);
            if (isNaN(raw)) return;
            const type = l.querySelector('.typeG')?.value || 'HT';
            const tva = parseFloat(l.querySelector('.tvaIndividuelleG')?.value);
            const applicableTva = isNaN(tva) ? globalTva : tva;

            const factor = 1 + applicableTva/100;
            let montantHT, montantTTC, tvaVal;
            if (type === 'HT') {
                montantHT = raw;
                montantTTC = montantHT * factor;
                tvaVal = montantTTC - montantHT;
            } else {
                montantTTC = raw;
                montantHT = montantTTC / factor;
                tvaVal = montantTTC - montantHT;
            }

            totalTVA += tvaVal;
            totalHT += montantHT;
            totalTTC += montantTTC;

            html += `<tr>
                <td style="text-align:left;">${lib}</td>
                <td>${raw.toFixed(2)} € (${type})</td>
                <td>${applicableTva}%</td>
                <td>${tvaVal.toFixed(2)} €</td>
                <td>${montantHT.toFixed(2)} €</td>
                <td>${montantTTC.toFixed(2)} €</td>
            </tr>`;
        });

        html += `<tr style="font-weight:bold; background:#ecf0f1; color:#333;">
            <td style="text-align:left;">Totaux</td>
            <td>—</td><td>—</td>
            <td>${totalTVA.toFixed(2)} €</td>
            <td>${totalHT.toFixed(2)} €</td>
            <td>${totalTTC.toFixed(2)} €</td>
        </tr>`;

        html += `</table>`;
        resDiv.innerHTML += html;
    });

    historique.G = resDiv.innerHTML;
}

async function genererRapport(){
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF('p', 'mm', 'a4');
    let y = 20;
    const pageWidth = doc.internal.pageSize.getWidth();
    const margin = 20;
    const contentWidth = pageWidth - 2 * margin;

    // Titre principal
    doc.setFontSize(24);
    doc.setTextColor(40);
    doc.text("Rapport Global Calculateur", pageWidth / 2, y, { align: "center" });
    y += 10;
    
    doc.setFontSize(12);
    doc.setTextColor(100);
    const dateStr = new Date().toLocaleDateString('fr-FR', {
        weekday: 'long', year: 'numeric', month: 'long', day: 'numeric'
    });
    doc.text(`Généré le ${dateStr}`, pageWidth / 2, y, { align: "center" });
    y += 15;

    const modes = [
        { id: 'A', title: 'Mode A : Comparaison de valeurs', isChart: false },
        { id: 'B', title: 'Mode B : Calcul de gains et pertes', isChart: false },
        { id: 'C', title: 'Mode C : Application de pourcentages par bloc', isChart: false },
        { id: 'D', title: 'Mode D : Évolution temporelle', isChart: true },
        { id: 'E', title: 'Mode E : Graphique linéaire multi-séries', isChart: true },
        { id: 'F', title: 'Mode F : Répartition circulaire', isChart: true },
        { id: 'G', title: 'Mode G : Calculateur TVA', isChart: false }
    ];

    for (const mode of modes) {
        if (!document.getElementById("include" + mode.id).checked || !historique[mode.id]) {
            continue;
        }

        // Nouvelle page pour chaque mode (sauf la première)
        if (y > 250) {
            doc.addPage();
            y = 20;
        } else if (mode.id !== 'A') {
            // Séparateur entre modes
            doc.setLineWidth(0.2);
            doc.line(margin, y, pageWidth - margin, y);
            y += 10;
        }

        // Titre du mode
        doc.setFontSize(18);
        doc.setTextColor(30);
        doc.setFont("helvetica", "bold");
        
        // Vérifier si le titre tiendra sur la page
        if (y > 270) {
            doc.addPage();
            y = 20;
        }
        
        doc.text(mode.title, margin, y);
        y += 12;

        if (mode.isChart) {
            // Pour les graphiques, on utilise une capture d'écran simplifiée
            let canvas;
            let chart;
            
            switch(mode.id) {
                case 'D': canvas = document.getElementById('graphD'); chart = chartD; break;
                case 'E': canvas = document.getElementById('graphE'); chart = chartE; break;
                case 'F': canvas = document.getElementById('graphF'); chart = chartF; break;
            }

            if (chart && canvas && canvas.style.display !== 'none') {
                try {
                    // Image du graphique
                    const imgData = chart.toBase64Image();
                    const imgWidth = contentWidth;
                    const imgHeight = (imgWidth * 0.75); // Ratio 4:3
                    
                    if (y + imgHeight > 280) {
                        doc.addPage();
                        y = 20;
                    }
                    
                    doc.addImage(imgData, 'PNG', margin, y, imgWidth, imgHeight);
                    y += imgHeight + 10;
                    
                    // Pour le mode F, ajouter aussi le tableau
                    if (mode.id === 'F') {
                        if (y > 270) {
                            doc.addPage();
                            y = 20;
                        }
                        
                        doc.setFontSize(14);
                        doc.text("Tableau récapitulatif", margin, y);
                        y += 8;
                        
                        // Créer une image du tableau
                        const tableauF = document.getElementById('tableauF');
                        if (tableauF) {
                            const tableauCanvas = await html2canvas(tableauF, {
                                scale: 2,
                                backgroundColor: '#ffffff',
                                logging: false
                            });
                            
                            const tableauData = tableauCanvas.toDataURL('image/png');
                            const tableauHeight = (contentWidth * tableauCanvas.height) / tableauCanvas.width;
                            
                            if (y + tableauHeight > 280) {
                                doc.addPage();
                                y = 20;
                            }
                            
                            doc.addImage(tableauData, 'PNG', margin, y, contentWidth, tableauHeight);
                            y += tableauHeight + 10;
                        }
                    }
                } catch (error) {
                    console.error(`Erreur avec le graphique ${mode.id}:`, error);
                    doc.setFontSize(12);
                    doc.setTextColor(150);
                    doc.text("Graphique non disponible", margin, y);
                    y += 10;
                }
            } else {
                doc.setFontSize(12);
                doc.setTextColor(150);
                doc.text("Aucune donnée graphique disponible", margin, y);
                y += 10;
            }
        } else {
            // Pour les modes sans graphique (A, B, C, G)
            // On crée un conteneur temporaire pour la capture
            const tempContainer = document.createElement('div');
            tempContainer.style.width = contentWidth + 'mm';
            tempContainer.style.padding = '10px';
            tempContainer.style.backgroundColor = '#ffffff';
            tempContainer.style.fontFamily = 'Arial, sans-serif';
            tempContainer.style.fontSize = '12px';
            
            // Copier le contenu historique dans le conteneur temporaire
            const tempContent = document.createElement('div');
            tempContent.innerHTML = historique[mode.id];
            
            // Nettoyer les styles inline problématiques
            const tables = tempContent.querySelectorAll('table');
            tables.forEach(table => {
                table.style.width = '100%';
                table.style.borderCollapse = 'collapse';
                table.style.margin = '0';
                table.style.fontSize = '10px';
            });
            
            tempContainer.appendChild(tempContent);
            document.body.appendChild(tempContainer);
            
            try {
                const canvas = await html2canvas(tempContainer, {
                    scale: 2,
                    backgroundColor: '#ffffff',
                    logging: false,
                    useCORS: true
                });
                
                const imgData = canvas.toDataURL('image/png');
                const imgHeight = (contentWidth * canvas.height) / canvas.width;
                
                // Gestion de la pagination pour les images longues
                let currentY = y;
                const maxHeightPerPage = 280 - currentY;
                
                if (imgHeight <= maxHeightPerPage) {
                    // L'image tient sur une page
                    doc.addImage(imgData, 'PNG', margin, currentY, contentWidth, imgHeight);
                    y += imgHeight + 10;
                } else {
                    // L'image doit être divisée sur plusieurs pages
                    let remainingHeight = imgHeight;
                    let sourceY = 0;
                    
                    while (remainingHeight > 0) {
                        const pageHeight = Math.min(remainingHeight, maxHeightPerPage);
                        
                        doc.addImage(
                            imgData, 
                            'PNG', 
                            margin, 
                            currentY, 
                            contentWidth, 
                            pageHeight,
                            undefined,
                            undefined,
                            0,
                            sourceY
                        );
                        
                        remainingHeight -= pageHeight;
                        sourceY += pageHeight;
                        
                        if (remainingHeight > 0) {
                            doc.addPage();
                            currentY = 20;
                            maxHeightPerPage = 280 - currentY;
                        } else {
                            y = currentY + pageHeight + 10;
                        }
                    }
                }
                
            } catch (error) {
                console.error(`Erreur avec le mode ${mode.id}:`, error);
                doc.setFontSize(12);
                doc.setTextColor(150);
                doc.text("Contenu non disponible", margin, y);
                y += 10;
            }
            
            document.body.removeChild(tempContainer);
        }
        
        // Réinitialiser les styles de texte
        doc.setFont("helvetica", "normal");
        doc.setTextColor(0);
    }

    // Numéroter les pages
    const totalPages = doc.internal.getNumberOfPages();
    for (let i = 1; i <= totalPages; i++) {
        doc.setPage(i);
        doc.setFontSize(10);
        doc.setTextColor(150);
        doc.text(
            `Page ${i} sur ${totalPages}`,
            pageWidth - margin,
            287,
            { align: "right" }
        );
    }

    // Sauvegarder le PDF
    doc.save(`Rapport_Calculateur_${new Date().toISOString().slice(0,10)}.pdf`);
}
