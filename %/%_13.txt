function quickCalculation(type) {
    switch(type) {
        case 'percentage':
            const value = prompt('Entrez la valeur :', '100');
            const percentage = prompt('Entrez le pourcentage :', '20');
            if (value && percentage) {
                const result = (parseFloat(value) * parseFloat(percentage)) / 100;
                showMode('modeCalculator');
                setTimeout(() => {
                    addToExpression(`${value}*${percentage}/100`);
                    calculateResult();
                    addRecentCalculation('pourcentage', 
                        `Calcul de ${percentage}% de ${value}`, 
                        result
                    );
                }, 100);
            }
            break;
            
        case 'vat':
            const ht = prompt('Montant HT :', '100');
            const tva = prompt('Taux TVA (%) :', '20');
            if (ht && tva) {
                const ttc = parseFloat(ht) * (1 + parseFloat(tva)/100);
                showMode('modeG');
                setTimeout(() => {
                    // Simuler l'ajout d'une ligne TVA
                    addRecentCalculation('tva', 
                        `TVA: ${ht} HT → ${ttc.toFixed(2)} TTC (${tva}%)`, 
                        ttc
                    );
                }, 100);
            }
            break;
            
        case 'currency':
            const amount = prompt('Montant à convertir :', '100');
            if (amount) {
                showMode('modeH');
                setTimeout(() => {
                    document.getElementById('fromAmount').value = amount;
                    convertCurrency();
                    addRecentCalculation('devise', 
                        `Conversion: ${amount} EUR → USD`, 
                        parseFloat(document.getElementById('toAmount').value || 0)
                    );
                }, 100);
            }
            break;
    }
}

function addToExpression(value) {
    const expressionDiv = document.getElementById('calcExpression');
    const resultDiv = document.getElementById('calcResult');
    
    if (value === 'pi') {
        calcExpression += Math.PI;
    } else {
        calcExpression += value;
    }
    
    expressionDiv.textContent = calcExpression;
    resultDiv.textContent = calcExpression || '0';
}

function clearCalculator() {
    calcExpression = '';
    calcResult = '0';
    document.getElementById('calcExpression').textContent = '';
    document.getElementById('calcResult').textContent = '0';
}

function deleteLast() {
    calcExpression = calcExpression.slice(0, -1);
    document.getElementById('calcExpression').textContent = calcExpression;
    document.getElementById('calcResult').textContent = calcExpression || '0';
}

function calculateResult() {
    try {
        let expression = calcExpression;
        
        // Remplacer les fonctions mathématiques
        expression = expression.replace(/sqrt\(/g, 'Math.sqrt(');
        expression = expression.replace(/\^/g, '**');
        expression = expression.replace(/sin\(/g, 'Math.sin(');
        expression = expression.replace(/cos\(/g, 'Math.cos(');
        expression = expression.replace(/tan\(/g, 'Math.tan(');
        expression = expression.replace(/log\(/g, 'Math.log10(');
        expression = expression.replace(/ln\(/g, 'Math.log(');
        
        // Évaluer l'expression
        const result = eval(expression);
        
        if (isNaN(result) || !isFinite(result)) {
            throw new Error('Résultat non valide');
        }
        
        calcResult = result.toString();
        document.getElementById('calcResult').textContent = formatNumber(result);
        
        // Sauvegarder dans l'historique
        saveToCalcHistory(calcExpression, result);
        
        // Ajouter aux calculs récents
        addRecentCalculation('calculatrice', `Calcul: ${calcExpression}`, result);
        
        // Réinitialiser l'expression
        calcExpression = '';
        document.getElementById('calcExpression').textContent = '';
        
        saveAllData();
        
    } catch (error) {
        showNotification('Erreur de calcul : ' + error.message, 'error');
        document.getElementById('calcResult').textContent = 'Erreur';
    }
}

function saveToCalcHistory(expression, result) {
    const historyEntry = {
        id: Date.now(),
        expression: expression,
        result: result,
        timestamp: new Date().toLocaleString('fr-FR')
    };
    
    calculatorHistory.unshift(historyEntry);
    if (calculatorHistory.length > 20) {
        calculatorHistory = calculatorHistory.slice(0, 20);
    }
    
    updateCalcHistoryDisplay();
}

function updateCalcHistoryDisplay() {
    const container = document.getElementById('calcHistoryList');
    if (!container) return;
    
    if (calculatorHistory.length === 0) {
        container.innerHTML = '<div style="color: var(--text-muted); text-align: center; padding: 20px;">Aucun calcul récent</div>';
        return;
    }
    
    let html = '';
    calculatorHistory.forEach(entry => {
        html += `
            <div style="padding: 8px; border-bottom: 1px solid var(--border-light);">
                <div style="font-size: 12px; color: var(--text-muted);">${entry.timestamp}</div>
                <div style="font-weight: bold;">${entry.expression} = ${formatNumber(entry.result)}</div>
            </div>
        `;
    });
    
    container.innerHTML = html;
}

function clearCalcHistory() {

function autoCalculA(){
    const cont = document.getElementById("multiAContainer");
    const resDiv = document.getElementById("resultA"); 
    resDiv.innerHTML = "";
    
    Array.from(cont.getElementsByClassName("ref-block")).forEach((block, index) => {
        const ref = parseNumber(block.querySelector(".refA").value);
        const raw = (block.querySelector(".valsA").value || "").trim();
        if(raw === "") return;
        const vals = raw.split(/\s+/).map(parseNumber).filter(v => v !== null); 
        
        let html = `<div style="margin:15px 0;"><h4>Bloc ${index + 1} - Référence : ${ref.toFixed(2)}€</h4>`;
        html += `<table style="width:100%;"><tr><th>Valeur</th><th>Différence</th><th>%</th></tr>`;
        
        vals.forEach(v => {
            const diff = v - ref;
            const pct = (ref === 0) ? null : (diff / ref * 100);
            const col = diff > 0 ? "var(--success-color)" : diff < 0 ? "var(--error-color)" : "var(--text-muted)";
            html += `<tr style="background:${col}; color:#fff;">
                <td>${v.toFixed(2)} €</td>
                <td>${diff>=0?'+':''}${diff.toFixed(2)} €</td>
                <td>${pct===null?'—':pct.toFixed(2)+'%'}</td>
            </tr>`;
        });
        html += "</table></div>";
        resDiv.innerHTML += html;
    });
    saveAllData();
}

function updateCBlockName(blockId, newName) {

function updateBlockPercent(blockId, newPercent) {
    const block = cBlocks.find(b => b.id === blockId);
    if (block) {
        let percent = parseInt(newPercent) || 0;
        if (percent > 100) percent = 100;
        if (percent < -100) percent = -100;
        
        block.blockPercent = percent;
        block.variables.forEach(variable => {
            variable.percent = percent;
        });
        renderCBlocks();
        updateCResults();
        saveAllData();
    }
}

function updateVariableValue(blockId, variableId, newValue) {
    const block = cBlocks.find(b => b.id === blockId);
    if (block) {
        const variable = block.variables.find(v => v.id === variableId);
        if (variable) {
            variable.value = parseFloat(newValue) || 0;
            updateCResults();
            saveAllData();
        }
    }
}

function updateVariablePercent(blockId, variableId, newPercent) {
    const block = cBlocks.find(b => b.id === blockId);
    if (block) {
        const variable = block.variables.find(v => v.id === variableId);
        if (variable) {
            let percent = parseInt(newPercent) || 0;
            if (percent > 100) percent = 100;
            if (percent < -100) percent = -100;
            
            variable.percent = percent;
            renderCBlocks();
            updateCResults();
            saveAllData();
        }
    }
}

    const block = cBlocks.find(b => b.id === blockId);
    if (block) {

}

function removeVariable(blockId, variableId) {
    const block = cBlocks.find(b => b.id === blockId);
    if (block && block.variables.length > 1) {
        block.variables = block.variables.filter(v => v.id !== variableId);
        renderCBlocks();
        updateCResults();
        saveAllData();
    } else {
        showNotification('Un bloc doit avoir au moins une variable', 'warning');
    }
}

function removeCBlock(blockId) {
    if (cBlocks.length <= 1) {
        showNotification('Vous devez garder au moins un bloc', 'warning');
        return;
    }
    
    if (confirm('Voulez-vous vraiment supprimer ce bloc ?')) {
        cBlocks = cBlocks.filter(b => b.id !== blockId);
        renderCBlocks();
        updateCResults();
        showNotification('Bloc supprimé avec succès', 'success');
        saveAllData();
    }
}

function updateCResults() {
    const tbody = document.getElementById('cResultsBody');
    const tfoot = document.getElementById('cResultsTotal');
    let html = '';
    
    let totalInitial = 0;
    let totalFinal = 0;
    let totalVariation = 0;
    
    cBlocks.forEach(block => {
        block.variables.forEach(variable => {
            const initial = variable.value;
            const final = initial * (1 + variable.percent / 100);
            const variation = final - initial;
            
            totalInitial += initial;
            totalFinal += final;
            totalVariation += variation;
            
            html += `<tr>
                <td>${block.name}</td>
                <td>${variable.name}</td>
                <td>${initial.toFixed(2)}</td>
                <td>${variable.percent >= 0 ? '+' : ''}${variable.percent}%</td>
                <td>${final.toFixed(2)}</td>
                <td class="${variation >= 0 ? 'positive' : 'negative'}">${variation >= 0 ? '+' : ''}${variation.toFixed(2)}</td>
            </tr>`;
        });
    });
    
    tbody.innerHTML = html;
    
    tfoot.innerHTML = `
        <tr class="total-row">
            <td colspan="2"><strong>TOTAL GLOBAL</strong></td>
            <td><strong>${totalInitial.toFixed(2)}</strong></td>
            <td></td>
            <td><strong>${totalFinal.toFixed(2)}</strong></td>
            <td class="${totalVariation >= 0 ? 'positive' : 'negative'}">
                <strong>${totalVariation >= 0 ? '+' : ''}${totalVariation.toFixed(2)}</strong>
            </td>
        </tr>
    `;
    saveAllData();
}

        name: `Série ${dSeries.length + 1}`,
        values: {},
        color: '#' + Math.floor(Math.random()*16777215).toString(16)
    };
    dDates.forEach(d => {
        newSeries.values[d.id] = 0;
    });
    dSeries.push(newSeries);
    updateDTable();
    saveAllData();
}

    dSeries = dSeries.filter(s => s.id !== id);
    updateDTable();
    saveAllData();
}

function updateDSeriesColor(id, newColor) {

function updateDTable() {
    const dateHeader = document.getElementById('dateHeaderRow');
    const seriesBody = document.getElementById('seriesBody');
    const useDates = document.getElementById('useDates')?.checked !== false;
    
    let headerHTML = '<th style="min-width: 160px;">Série / Date</th>';
    dDates.forEach(d => {
        if (useDates) {
            headerHTML += `
                <th>
                    <input type="date" value="${d.date || new Date().toISOString().split('T')[0]}" onchange="updateDDate(${d.id}, this.value)" style="width:120px; background: var(--bg-secondary); color: var(--text-primary); border: 1px solid var(--border-color);">
                    <button onclick="removeDDate(${d.id})" style="padding:2px 6px;font-size:11px; background: var(--error-color); color: white; border: none; border-radius: 3px; cursor: pointer;">✖</button>
                </th>
            `;
        } else {
            headerHTML += `
                <th>
                    <input type="text" value="${d.label || `Point ${dDates.indexOf(d) + 1}`}" onchange="updateDPointLabel(${d.id}, this.value)" style="width:100px; background: var(--bg-secondary); color: var(--text-primary); border: 1px solid var(--border-color);">
                    <button onclick="removeDDate(${d.id})" style="padding:2px 6px;font-size:11px; background: var(--error-color); color: white; border: none; border-radius: 3px; cursor: pointer;">✖</button>
                </th>
            `;
        }
    });
    dateHeader.innerHTML = headerHTML;
    
    seriesBody.innerHTML = '';
    dSeries.forEach(s => {
        let row = document.createElement('tr');
        let rowHTML = `
            <td>
                <input type="text" value="${s.name}" onchange="updateDSeriesName(${s.id}, this.value)" style="width:120px; background: var(--bg-secondary); color: var(--text-primary); border: 1px solid var(--border-color);">
                <input type="color" value="${s.color}" onchange="updateDSeriesColor(${s.id}, this.value)" title="Couleur" style="width: 30px; height: 30px;">
                <button onclick="removeDSeries(${s.id})" style="padding:2px 6px;font-size:11px; background: var(--error-color); color: white; border: none; border-radius: 3px; cursor: pointer;">✖</button>

        dDates.forEach(d => {

                <td>

                </td>
            `;

function calculerF() {
    const ctx = document.getElementById('graphF')?.getContext('2d');
    if (!ctx) return;
    
    if (chartF) chartF.destroy();
    
    historiqueF = [];
    const blocks = document.querySelectorAll('#multiFContainer .ref-block');
    blocks.forEach(block => {
        historiqueF.push({
            label: block.querySelector('.labelF').value || "Sans nom",
            val: parseFloat(block.querySelector('.valF').value) || 0,
            color: block.querySelector('.colorF').value || '#3498db',
            type: block.getAttribute('data-original-value') ? 'original' : 'merged'
        });
    });
    
    if (historiqueF.length === 0) {
        document.getElementById('tableauF').querySelector('tbody').innerHTML = '<tr><td colspan="4" style="text-align: center; color: var(--text-muted); padding: 20px;">Aucune catégorie</td></tr>';
        document.getElementById('tableauFTotal').innerHTML = '';
        return;
    }
    
    const total = historiqueF.reduce((sum, cat) => sum + cat.val, 0);
    
    const data = {
        labels: historiqueF.map(cat => cat.label),
        datasets: [{
            data: historiqueF.map(cat => cat.val),
            backgroundColor: historiqueF.map(cat => cat.color),
            borderWidth: 2,
            borderColor: 'var(--bg-secondary)'
        }]
    };
    
    chartF = new Chart(ctx, {
        type: 'pie',
        data: data,
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                legend: { position: 'right' },
                title: { display: true, text: `Répartition - Total: ${total.toFixed(2)}` },
                datalabels: {
                    formatter: (value, context) => {
                        if (total === 0) return '0%';
                        const percentage = (value / total * 100).toFixed(1);
                        return context.chart.data.labels[context.dataIndex] + '\n' + percentage + '%';
                    },
                    color: '#fff',
                    font: { weight: 'bold', size: 12 },
                    textAlign: 'center'
                }
            }
        },
        plugins: [ChartDataLabels]
    });
    
    const tbody = document.querySelector('#tableauF tbody');
    const tfoot = document.getElementById('tableauFTotal');
    
    tbody.innerHTML = '';
    historiqueF.forEach(cat => {
        const percentage = total === 0 ? 0 : (cat.val / total * 100);
        const typeBadge = cat.type === 'merged' ? 
            '<span style="background: #9b59b6; color: white; padding: 2px 6px; border-radius: 3px; font-size: 11px;">Fusionnée</span>' :
            '<span style="background: var(--success-color); color: white; padding: 2px 6px; border-radius: 3px; font-size: 11px;">Originale</span>';
        
        tbody.innerHTML += `<tr>
            <td><span style="display:inline-block;width:12px;height:12px;background:${cat.color};margin-right:8px;border-radius:2px;"></span>${cat.label}</td>
            <td>${cat.val.toFixed(2)}</td>
            <td>${percentage.toFixed(1)}%</td>
            <td>${typeBadge}</td>
        </tr>`;
    });
    
    tfoot.innerHTML = `
        <tr style="font-weight: bold; background: rgba(52, 152, 219, 0.1);">
            <td colspan="2">TOTAL</td>
            <td>${total.toFixed(2)}</td>
            <td>100%</td>
            <td></td>
        </tr>
    `;
    saveAllData();
}

function toggleFusionSelection(btn) {
    const isSelected = btn.getAttribute('data-selected') === 'true';
    if (isSelected) {
        btn.textContent = '✗';
        btn.style.background = 'var(--bg-tertiary)';
        btn.style.color = 'var(--text-primary)';
        btn.setAttribute('data-selected', 'false');
    } else {
        btn.textContent = '✓';
        btn.style.background = 'var(--success-color)';
        btn.style.color = 'white';
        btn.setAttribute('data-selected', 'true');
    }
    updateSelectedCount();
    saveAllData();
}

function mergeSelectedCategories() {
    const selected = Array.from(document.querySelectorAll('.fusion-toggle-btn[data-selected="true"]'));
    if (selected.length < 2) {
        showNotification('Sélectionnez au moins 2 catégories à fusionner.', 'warning');
        return;
    }
    
    const fusionName = document.getElementById('fusionName').value || 'Fusionnée';
    let totalVal = 0;
    let color = '#3498db';
    
    // Stocker les informations des catégories originales pour une éventuelle séparation
    const originalCategories = [];
    
    selected.forEach((btn, index) => {
        const block = btn.closest('.ref-block');
        const label = block.querySelector('.labelF').value;
        const val = parseFloat(block.querySelector('.valF').value) || 0;
        totalVal += val;
        
        // Sauvegarder les informations originales
        originalCategories.push({
            label: label,
            value: val,
            color: block.querySelector('.colorF').value,
            originalValue: block.getAttribute('data-original-value')
        });
        
        if (index === 0) color = block.querySelector('.colorF').value;
    });
    
    // Supprimer les catégories sélectionnées
    selected.forEach(btn => {
        const block = btn.closest('.ref-block');
        block.remove();
    });
    
    // Ajouter la catégorie fusionnée avec les informations des originales
    const container = document.getElementById("multiFContainer");
    const id = Date.now();
    
    const div = document.createElement("div");
    div.className = "ref-block";
    div.setAttribute('data-category-id', id);
    div.setAttribute('data-original-categories', JSON.stringify(originalCategories));
    div.setAttribute('data-is-merged', 'true');
    
    div.innerHTML = `
        <div style="display: flex; align-items: center; gap: 10px; flex-wrap: wrap;">
            <input type="text" class="labelF" placeholder="Catégorie" value="${fusionName}" style="width:150px;" oninput="autoCalculF()">
            <input type="number" class="valF" placeholder="Valeur" value="${totalVal}" style="width:100px;" oninput="autoCalculF()">
            <input type="color" class="colorF" value="${color}" oninput="autoCalculF()">
            <div class="fusion-toggle-btn" onclick="toggleFusionSelection(this)" data-selected="false" data-original-categories='${JSON.stringify(originalCategories)}' style="width: 32px; height: 32px; border: 2px solid var(--border-color); border-radius: 4px; cursor: pointer; display: flex; align-items: center; justify-content: center; font-weight: bold; user-select: none;">✗</div>
            <button onclick="removeF(this)" class="delete-btn"><i class="fas fa-trash"></i></button>
        </div>
        <div style="margin-top: 10px; font-size: 12px; color: var(--text-muted);">
            <i class="fas fa-info-circle"></i> Fusion de ${selected.length} catégories
        </div>
    `;
    
    container.appendChild(div);
    
    document.getElementById('fusionName').value = '';
    selectAllForFusion(false);
    autoCalculF();
    showNotification(`${selected.length} catégories fusionnées en "${fusionName}"`, 'success');
    saveAllData();
}

function splitSelectedCategory() {
    const selected = Array.from(document.querySelectorAll('.fusion-toggle-btn[data-selected="true"]'));
    if (selected.length !== 1) {
        showNotification('Sélectionnez une seule catégorie fusionnée à séparer.', 'warning');
        return;
    }
    
    const btn = selected[0];
    const block = btn.closest('.ref-block');
    const originalCategoriesJSON = block.getAttribute('data-original-categories');
    
    if (!originalCategoriesJSON) {
        showNotification('Cette catégorie n\'est pas fusionnée ou ne peut pas être séparée.', 'warning');
        return;
    }
    
    const originalCategories = JSON.parse(originalCategoriesJSON);
    
    if (confirm(`Séparer cette catégorie en ${originalCategories.length} catégories originales ?`)) {
        // Supprimer la catégorie fusionnée
        block.remove();
        
        // Recréer les catégories originales
        const container = document.getElementById("multiFContainer");
        
        originalCategories.forEach(cat => {
            const id = Date.now() + Math.random();
            const div = document.createElement("div");
            div.className = "ref-block";
            div.setAttribute('data-category-id', id);
            div.setAttribute('data-original-value', cat.originalValue || cat.value);
            
            div.innerHTML = `
                <div style="display: flex; align-items: center; gap: 10px; flex-wrap: wrap;">
                    <input type="text" class="labelF" placeholder="Catégorie" value="${cat.label}" style="width:150px;" oninput="autoCalculF()">
                    <input type="number" class="valF" placeholder="Valeur" value="${cat.value}" style="width:100px;" oninput="autoCalculF()">
                    <input type="color" class="colorF" value="${cat.color}" oninput="autoCalculF()">
                    <div class="fusion-toggle-btn" onclick="toggleFusionSelection(this)" data-selected="false" data-original-value="${cat.originalValue || cat.value}" style="width: 32px; height: 32px; border: 2px solid var(--border-color); border-radius: 4px; cursor: pointer; display: flex; align-items: center; justify-content: center; font-weight: bold; user-select: none;">✗</div>
                    <button onclick="removeF(this)" class="delete-btn"><i class="fas fa-trash"></i></button>
                </div>
            `;
            
            container.appendChild(div);
        });
        
        selectAllForFusion(false);
        autoCalculF();
        showNotification(`Catégorie séparée en ${originalCategories.length} catégories originales`, 'success');
        saveAllData();
    }
}

}

// Système de sauvegarde des états de fusion
function saveFusionState() {
    const stateName = prompt('Nom de l\'état de fusion :', `État_${new Date().toLocaleDateString('fr-FR')}`);
    
    if (!stateName) return;
    
    const blocks = Array.from(document.querySelectorAll('#multiFContainer .ref-block')).map(block => ({
        label: block.querySelector('.labelF').value,
        value: block.querySelector('.valF').value,
        color: block.querySelector('.colorF').value,
        isSelected: block.querySelector('.fusion-toggle-btn').getAttribute('data-selected') === 'true',
        originalValue: block.getAttribute('data-original-value'),
        originalCategories: block.getAttribute('data-original-categories'),
        isMerged: block.getAttribute('data-is-merged') === 'true'
    }));
    
    const state = {
        id: Date.now(),
        name: stateName,
        timestamp: new Date().toISOString(),
        blocks: blocks,
        total: historiqueF.reduce((sum, cat) => sum + cat.val, 0)
    };
    
    fusionStates.unshift(state);
    if (fusionStates.length > 10) {
        fusionStates = fusionStates.slice(0, 10);
    }
    
    localStorage.setItem(FUSION_STATES_KEY, JSON.stringify(fusionStates));
    loadFusionStates();
    showNotification(`État "${stateName}" sauvegardé`, 'success');
}

function loadFusionState() {
    const container = document.getElementById('fusionStatesList');
    if (container.style.display === 'none') {
        container.style.display = 'block';
        return;
    }
    
    loadFusionStates();
}

function loadFusionStates() {
    const container = document.getElementById('fusionStatesList');
    fusionStates = JSON.parse(localStorage.getItem(FUSION_STATES_KEY) || '[]');
    
    if (fusionStates.length === 0) {
        container.innerHTML = '<div style="color: var(--text-muted); text-align: center; padding: 20px;">Aucun état sauvegardé</div>';
        container.style.display = 'block';
        return;
    }
    
    let html = '<h5 style="margin: 0 0 10px 0;">États sauvegardés :</h5>';
    
    fusionStates.forEach(state => {
        const date = new Date(state.timestamp).toLocaleString('fr-FR');
        html += `
            <div style="background: var(--bg-secondary); border: 1px solid var(--border-color); border-radius: 6px; padding: 10px; margin-bottom: 8px;">
                <div style="display: flex; justify-content: space-between; align-items: center;">
                    <div>
                        <div style="font-weight: bold;">${state.name}</div>
                        <div style="font-size: 11px; color: var(--text-muted);">${date} • ${state.blocks.length} catégories</div>
                    </div>
                    <div style="display: flex; gap: 5px;">
                        <button onclick="restoreFusionState(${state.id})" style="padding: 5px 10px; font-size: 12px;" class="secondary">
                            <i class="fas fa-undo"></i>
                        </button>
                        <button onclick="deleteFusionState(${state.id})" style="padding: 5px 10px; font-size: 12px;" class="delete-btn">
                            <i class="fas fa-trash"></i>
                        </button>
                    </div>
                </div>
            </div>
        `;
    });
    
    container.innerHTML = html;
    container.style.display = 'block';
}

function restoreFusionState(stateId) {
    const state = fusionStates.find(s => s.id === stateId);
    if (!state) return;
    
    if (confirm(`Restaurer l'état "${state.name}" ?\nLes catégories actuelles seront remplacées.`)) {
        const container = document.getElementById("multiFContainer");
        container.innerHTML = '';
        
        state.blocks.forEach(blockData => {
            const id = Date.now() + Math.random();
            const div = document.createElement("div");
            div.className = "ref-block";
            div.setAttribute('data-category-id', id);
            
            if (blockData.originalValue) {
                div.setAttribute('data-original-value', blockData.originalValue);
            }
            
            if (blockData.originalCategories) {
                div.setAttribute('data-original-categories', blockData.originalCategories);
            }
            
            if (blockData.isMerged) {
                div.setAttribute('data-is-merged', 'true');
            }
            
            div.innerHTML = `
                <div style="display: flex; align-items: center; gap: 10px; flex-wrap: wrap;">
                    <input type="text" class="labelF" placeholder="Catégorie" value="${blockData.label}" style="width:150px;" oninput="autoCalculF()">
                    <input type="number" class="valF" placeholder="Valeur" value="${blockData.value}" style="width:100px;" oninput="autoCalculF()">
                    <input type="color" class="colorF" value="${blockData.color}" oninput="autoCalculF()">
                    <div class="fusion-toggle-btn" onclick="toggleFusionSelection(this)" data-selected="${blockData.isSelected}" style="width: 32px; height: 32px; border: 2px solid var(--border-color); border-radius: 4px; cursor: pointer; display: flex; align-items: center; justify-content: center; font-weight: bold; user-select: none; background: ${blockData.isSelected ? 'var(--success-color)' : 'var(--bg-tertiary)'}; color: ${blockData.isSelected ? 'white' : 'var(--text-primary)'};">${blockData.isSelected ? '✓' : '✗'}</div>
                    <button onclick="removeF(this)" class="delete-btn"><i class="fas fa-trash"></i></button>
                </div>
            `;
            
            container.appendChild(div);
        });
        
        updateSelectedCount();
        autoCalculF();
        document.getElementById('fusionStatesList').style.display = 'none';
        showNotification(`État "${state.name}" restauré`, 'success');
        saveAllData();
    }
}

function deleteFusionState(stateId) {
    fusionStates = fusionStates.filter(s => s.id !== stateId);
    localStorage.setItem(FUSION_STATES_KEY, JSON.stringify(fusionStates));
    loadFusionStates();
    showNotification('État supprimé', 'success');
}

function calculerG() {
    const resultDiv = document.getElementById("resultG");
    const blocks = document.querySelectorAll('#multiGContainer .ref-block');
    
    if (blocks.length === 0) {
        resultDiv.innerHTML = '<p style="color: var(--text-muted); text-align: center; padding: 20px;">Ajoutez un bloc TVA pour commencer</p>';
        return;
    }
    
    let globalHTML = "";
    
    blocks.forEach((block, blockIndex) => {
        const blocNom = block.querySelector('.bloc-nom-input').value || `Bloc ${blockIndex + 1}`;
        const globalTVA = parseFloat(block.querySelector('.global-tva-input').value) || 20;
        
        const lignes = block.querySelectorAll('.ligne-tva');
        let totalHT = 0;
        let totalTVA = 0;
        let totalTTC = 0;
        
        let tableHTML = `
            <div style="margin: 20px 0; padding: 20px; background: var(--bg-tertiary); border-radius: 10px; border: 1px solid var(--border-color);">
                <h4 style="color: var(--text-primary); margin: 0 0 15px 0;">${blocNom} (TVA par défaut: ${globalTVA}%)</h4>
                <table style="width:100%; margin-top:10px; border-collapse: collapse;">
                    <thead>
                        <tr>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">Libellé</th>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">Montant</th>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">Type</th>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">TVA</th>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">HT</th>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">TVA (€)</th>
                            <th style="padding: 10px; border-bottom: 2px solid var(--border-color);">TTC</th>
                        </tr>
                    </thead>
                    <tbody>
        `;
        
        lignes.forEach(ligne => {
            const libelle = ligne.querySelector('.libelleG').value || "";
            const valeur = parseFloat(ligne.querySelector('.valeurG').value) || 0;
            const type = ligne.querySelector('.typeG').value;
            const tva = parseFloat(ligne.querySelector('.tvaIndividuelleG').value) || globalTVA;
            
            let ht, ttc, tvaEur;
            
            if (type === "HT") {
                ht = valeur;
                tvaEur = ht * (tva / 100);
                ttc = ht + tvaEur;
            } else {
                ttc = valeur;
                ht = ttc / (1 + (tva / 100));
                tvaEur = ttc - ht;
            }
            
            totalHT += ht;
            totalTVA += tvaEur;
            totalTTC += ttc;
            
            tableHTML += `
                <tr style="border-bottom: 1px solid var(--border-light);">
                    <td style="padding: 10px;">${libelle}</td>
                    <td style="padding: 10px; text-align: right;">${valeur.toFixed(2)} €</td>
                    <td style="padding: 10px; text-align: center;">${type}</td>
                    <td style="padding: 10px; text-align: right;">${tva.toFixed(1)}%</td>
                    <td style="padding: 10px; text-align: right;">${ht.toFixed(2)} €</td>
                    <td style="padding: 10px; text-align: right;">${tvaEur.toFixed(2)} €</td>
                    <td style="padding: 10px; text-align: right;">${ttc.toFixed(2)} €</td>
                </tr>
            `;
        });
        
        tableHTML += `
                    </tbody>
                    <tfoot style="font-weight:bold; background:rgba(52, 152, 219, 0.1);">
                        <tr>
                            <td colspan="4" style="padding: 10px; border-top: 2px solid var(--border-color);">TOTAUX</td>
                            <td style="padding: 10px; text-align: right; border-top: 2px solid var(--border-color);">${totalHT.toFixed(2)} €</td>
                            <td style="padding: 10px; text-align: right; border-top: 2px solid var(--border-color);">${totalTVA.toFixed(2)} €</td>
                            <td style="padding: 10px; text-align: right; border-top: 2px solid var(--border-color);">${totalTTC.toFixed(2)} €</td>
                        </tr>
                    </tfoot>
                </table>
            </div>
        `;
        
        globalHTML += tableHTML;
    });
    
    resultDiv.innerHTML = globalHTML;
    saveAllData();
}

function updateFlags() {
    const fromCurrency = document.getElementById('fromCurrency')?.value;
    const toCurrency = document.getElementById('toCurrency')?.value;
    
    const fromFlag = document.getElementById('fromFlag');
    const toFlag = document.getElementById('toFlag');
    
    if (fromCurrency && currencyFlags[fromCurrency] && fromFlag) {
        fromFlag.src = `https://flagcdn.com/w20/${currencyFlags[fromCurrency]}.png`;
        fromFlag.style.display = 'block';
    }
    
    if (toCurrency && currencyFlags[toCurrency] && toFlag) {
        toFlag.src = `https://flagcdn.com/w20/${currencyFlags[toCurrency]}.png`;
        toFlag.style.display = 'block';
    }
    
    if (fromCurrency) localStorage.setItem('lastFromCurrency', fromCurrency);
    if (toCurrency) localStorage.setItem('lastToCurrency', toCurrency);
    
    convertCurrency();
}

function convertCurrency() {
    const fromCurrency = document.getElementById('fromCurrency')?.value;
    const toCurrency = document.getElementById('toCurrency')?.value;
    const fromAmount = parseFloat(document.getElementById('fromAmount')?.value) || 0;
    const toAmount = document.getElementById('toAmount');
    
    if (!fromCurrency || !toCurrency || !toAmount || !exchangeRates[fromCurrency] || !exchangeRates[toCurrency]) {
        if (toAmount) toAmount.value = '0.00';
        return;
    }
    
    const amountInEur = fromAmount / exchangeRates[fromCurrency];
    const convertedAmount = amountInEur * exchangeRates[toCurrency];
    
    toAmount.value = convertedAmount.toFixed(2);
    
    const rate = (exchangeRates[toCurrency] / exchangeRates[fromCurrency]).toFixed(4);
    const rateInfo = document.getElementById('rateInfo');
    if (rateInfo) {
        rateInfo.innerHTML = `Taux de change: 1 ${fromCurrency} = ${rate} ${toCurrency}`;
    }
    saveAllData();
}

    toCurrency.value = tempCurrency;
    
    const tempAmount = fromAmount.value;
    fromAmount.value = toAmount.value;

}

function updateAllRates() {
    const container = document.getElementById('allRatesGrid');
    if (!container) return;
    
    let html = '';
    const baseCurrency = 'EUR';
    
    Object.entries(exchangeRates)
        .sort(([a], [b]) => a.localeCompare(b))

            if (currency === baseCurrency) return;
            
            const flagCode = currencyFlags[currency] || 'un';
            
            html += `
                <div style="padding: 12px; background: var(--bg-secondary); border-radius: 8px; border: 1px solid var(--border-color); transition: all 0.3s; cursor: pointer;" onclick="setCurrencyPair('${currency}')">

                        <div style="font-weight: bold; color: var(--text-primary);">${currency}</div>
                    </div>
                    <div style="font-size: 18px; color: var(--accent-color); margin-top: 5px;">${rate.toFixed(4)}</div>
                    <div style="font-size: 12px; color: var(--text-muted);">1 ${baseCurrency} = ${rate.toFixed(2)} ${currency}</div>

}

function saveConversionToHistory() {
    const fromCurrency = document.getElementById('fromCurrency').value;
    const toCurrency = document.getElementById('toCurrency').value;
    const fromAmount = parseFloat(document.getElementById('fromAmount').value) || 0;
    const toAmount = parseFloat(document.getElementById('toAmount').value) || 0;
    
    if (fromAmount === 0) {
        showNotification('Entrez un montant à convertir', 'warning');
        return;
    }
    
    const conversion = {
        id: Date.now(),
        date: new Date().toLocaleString('fr-FR'),
        from: { currency: fromCurrency, amount: fromAmount },
        to: { currency: toCurrency, amount: toAmount },
        rate: (toAmount / fromAmount).toFixed(4)
    };
    
    conversionHistory.unshift(conversion);
    if (conversionHistory.length > 20) {
        conversionHistory = conversionHistory.slice(0, 20);
    }
    
    localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
    updateConversionHistoryDisplay();
    showNotification('Conversion sauvegardée', 'success');
    saveAllData();
}

function updateConversionHistoryDisplay() {
    const container = document.getElementById('conversionHistoryList');
    if (!container) return;
    
    if (conversionHistory.length === 0) {
        container.innerHTML = `
            <div style="padding: 30px; text-align: center; color: var(--text-muted);">
                <i class="fas fa-exchange-alt" style="font-size: 2rem; margin-bottom: 10px; opacity: 0.5;"></i>
                <p>Aucune conversion dans l'historique</p>
            </div>
        `;
        return;
    }
    
    let html = '';
    
    conversionHistory.forEach(conv => {
        html += `
            <div class="history-item" style="display: grid; grid-template-columns: 1fr 1fr auto auto; gap: 10px; padding: 12px; border-bottom: 1px solid var(--border-color); align-items: center; background: var(--bg-secondary);">
                <div style="font-size: 12px; color: var(--text-muted);">${conv.date}</div>
                <div style="font-weight: bold; color: var(--text-primary);">${conv.from.amount.toFixed(2)} ${conv.from.currency}</div>
                <div style="text-align: center; color: var(--text-primary);">→</div>
                <div style="font-weight: bold; color: var(--success-color);">${conv.to.amount.toFixed(2)} ${conv.to.currency}</div>
                <div style="font-size: 12px; color: var(--text-muted); grid-column: 1 / -1;">Taux: ${conv.rate}</div>
                <button onclick="deleteConversion(${conv.id})" style="padding: 5px 10px; background: var(--error-color); color: white; border: none; border-radius: 4px; cursor: pointer; font-size: 12px; grid-column: 4;">
                    <i class="fas fa-trash"></i>
                </button>
            </div>
        `;
    });
    
    container.innerHTML = html;
}

function deleteConversion(id) {
    conversionHistory = conversionHistory.filter(conv => conv.id !== id);
    localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
    updateConversionHistoryDisplay();
    showNotification('Conversion supprimée', 'success');
    saveAllData();
}

function clearConversionHistory() {
    if (conversionHistory.length === 0) return;
    
    if (confirm('Effacer tout l\'historique des conversions ?')) {
        conversionHistory = [];
        localStorage.setItem('conversionHistory', JSON.stringify([]));
        updateConversionHistoryDisplay();
        showNotification('Historique effacé', 'success');
        saveAllData();
    }
}

function exportHistoryToCSV() {
    if (conversionHistory.length === 0) {
        showNotification('Historique vide', 'warning');
        return;
    }
    
    let csv = 'Date;Montant Source;Devise Source;Montant Cible;Devise Cible;Taux de change\n';
    
    conversionHistory.forEach(conv => {
        csv += `${conv.date};${conv.from.amount};${conv.from.currency};${conv.to.amount};${conv.to.currency};${conv.rate}\n`;
    });
    
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    const url = URL.createObjectURL(blob);
    
    link.setAttribute('href', url);
    link.setAttribute('download', `historique_conversions_${new Date().toISOString().split('T')[0]}.csv`);
    link.style.visibility = 'hidden';
    
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    
    showNotification('Historique exporté en CSV', 'success');
}

function saveAllData() {
    try {
        const appData = {
            timestamp: new Date().toISOString(),
            version: '6.0',
            modeA: collectModeAData(),
            modeC: cBlocks,
            modeD: { dates: dDates, series: dSeries, useDates: document.getElementById('useDates')?.checked || true },
            modeF: {
                categories: historiqueF,
                blocks: Array.from(document.querySelectorAll('#multiFContainer .ref-block')).map(block => ({
                    label: block.querySelector('.labelF').value,
                    value: block.querySelector('.valF').value,
                    color: block.querySelector('.colorF').value,
                    originalValue: block.getAttribute('data-original-value')
                }))
            },
            modeG: collectModeGData(),
            modeH: {
                conversions: conversionHistory,
                fromCurrency: document.getElementById('fromCurrency')?.value || 'EUR',
                toCurrency: document.getElementById('toCurrency')?.value || 'USD',
                fromAmount: document.getElementById('fromAmount')?.value || '100'
            },
            calculator: calculatorHistory,
            recentCalculations: recentCalculations,
            settings: settings
        };
        
        localStorage.setItem(APP_STORAGE_KEY, JSON.stringify(appData));
        
        // Sauvegarde automatique si activée
        if (settings.autoSave) {
            showNotificationAutoSave('Données sauvegardées');
        }
        
        return appData;
    } catch (error) {
        console.error('Erreur sauvegarde:', error);
        return null;
    }
}

function loadAllData() {
    try {
        const saved = localStorage.getItem(APP_STORAGE_KEY);
        if (saved) {
            const data = JSON.parse(saved);
            
            // Restaurer chaque mode
            if (data.modeA) restoreModeAData(data.modeA);
            if (data.modeC) {
                cBlocks = data.modeC;
                cBlockCounter = cBlocks.length + 1;
            }
            if (data.modeD) {
                dDates = data.modeD.dates || [];
                dSeries = data.modeD.series || [];
                if (document.getElementById('useDates')) {
                    document.getElementById('useDates').checked = data.modeD.useDates !== false;
                }
            }
            if (data.modeF) {
                historiqueF = data.modeF.categories || [];
                restoreModeFData(data.modeF.blocks || []);
            }
            if (data.modeG) restoreModeGData(data.modeG);
            if (data.modeH) {
                conversionHistory = data.modeH.conversions || [];
                updateConversionHistoryDisplay();
            }
            if (data.calculator) {
                calculatorHistory = data.calculator;
            }
            if (data.recentCalculations) {
                recentCalculations = data.recentCalculations;
            }
            if (data.settings) {
                settings = { ...settings, ...data.settings };
            }
            
            console.log('✅ Données restaurées');
            return true;
        }
    } catch (error) {
        console.error('❌ Erreur chargement:', error);
    }
    
    // Données par défaut
    initDefaultData();
    return false;
}

function initDefaultData() {
    if (cBlocks.length === 0) addCBlock();
    if (dDates.length === 0) {
        dDates.push({ id: Date.now(), date: new Date().toISOString().split('T')[0] });
        dDates.push({ id: Date.now() + 1, date: new Date(Date.now() + 86400000).toISOString().split('T')[0] });
        dSeries.push({
            id: Date.now() + 2,
            name: "Série 1",
            values: {},
            color: "#" + Math.floor(Math.random()*16777215).toString(16)
        });
    }
    conversionHistory = [];
    calculatorHistory = [];
    recentCalculations = [];
}

function showBackupModal() {
    document.getElementById('backupModal').classList.remove('hidden');
    displayBackupList();

function closeBackupModal() {
    document.getElementById('backupModal').classList.add('hidden');

function createBackup(name = null) {
    const backupData = saveAllData();
    const backupName = name || prompt('Nom de la sauvegarde :', `Sauvegarde_${new Date().toLocaleDateString('fr-FR')}`);
    
    if (!backupName) return;
    
    const backup = {
        id: Date.now(),
        name: backupName,
        timestamp: new Date().toISOString(),
        data: backupData,
        size: JSON.stringify(backupData).length
    };
    
    let backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
    backups.unshift(backup);
    
    if (backups.length > 10) {
        backups = backups.slice(0, 10);
    }
    
    localStorage.setItem(BACKUP_STORAGE_KEY, JSON.stringify(backups));
    displayBackupList();
    showNotification(`Sauvegarde "${backupName}" créée`, 'success');
}

function displayBackupList() {
    const container = document.getElementById('backupList');
    if (!container) return;
    
    const backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
    
    if (backups.length === 0) {
        container.innerHTML = `
            <div style="text-align: center; padding: 40px 20px; color: var(--text-muted);">
                <i class="fas fa-inbox fa-2x" style="margin-bottom: 15px;"></i>
                <p>Aucune sauvegarde disponible</p>
            </div>
        `;
        return;
    }
    
    let html = '';
    backups.forEach(backup => {
        const date = new Date(backup.timestamp).toLocaleString('fr-FR');
        const size = (backup.size / 1024).toFixed(2);
        
        html += `
            <div style="background: var(--bg-secondary); border: 1px solid var(--border-color); border-radius: 6px; padding: 15px; margin-bottom: 10px;">
                <div style="display: flex; justify-content: space-between; align-items: center;">
                    <div>
                        <div style="font-weight: 600; margin-bottom: 5px;">${backup.name}</div>
                        <div style="font-size: 12px; color: var(--text-muted);">${date} • ${size} Ko</div>
                    </div>
                    <div style="display: flex; gap: 5px;">
                        <button onclick="restoreBackup(${backup.id})" style="padding: 8px 12px;" class="secondary">
                            <i class="fas fa-undo"></i>
                        </button>
                        <button onclick="deleteBackup(${backup.id})" style="padding: 8px 12px;" class="delete-btn">
                            <i class="fas fa-trash"></i>
                        </button>
                    </div>
                </div>
            </div>
        `;
    });
    
    container.innerHTML = html;
}

function restoreBackup(backupId) {
    const backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
    const backup = backups.find(b => b.id === backupId);
    
    if (!backup) {
        showNotification('Sauvegarde introuvable', 'error');
        return;
    }
    
    if (confirm(`Restaurer la sauvegarde "${backup.name}" ?\nLes données actuelles seront écrasées.`)) {
        try {
            // Restaurer les données principales
            if (backup.data.modeA) restoreModeAData(backup.data.modeA);
            if (backup.data.modeC) {
                cBlocks = backup.data.modeC;
                cBlockCounter = cBlocks.length + 1;
            }
            if (backup.data.modeD) {
                dDates = backup.data.modeD.dates || [];
                dSeries = backup.data.modeD.series || [];
            }
            if (backup.data.modeF) {
                historiqueF = backup.data.modeF.categories || [];
                restoreModeFData(backup.data.modeF.blocks || []);
            }
            if (backup.data.modeG) restoreModeGData(backup.data.modeG);
            if (backup.data.modeH) {
                conversionHistory = backup.data.modeH.conversions || [];
            }
            if (backup.data.calculator) {
                calculatorHistory = backup.data.calculator;
            }
            if (backup.data.recentCalculations) {
                recentCalculations = backup.data.recentCalculations;
            }
            
            // Mettre à jour l'affichage
            updateAllModes();
            updateDashboard();
            showNotification(`Sauvegarde "${backup.name}" restaurée`, 'success');
            closeBackupModal();
        } catch (error) {
            showNotification('Erreur restauration', 'error');
        }
    }
}

function deleteBackup(backupId) {
    let backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
    const backup = backups.find(b => b.id === backupId);
    
    if (!backup) return;
    
    if (confirm(`Supprimer la sauvegarde "${backup.name}" ?`)) {
        backups = backups.filter(b => b.id !== backupId);
        localStorage.setItem(BACKUP_STORAGE_KEY, JSON.stringify(backups));
        displayBackupList();
        showNotification('Sauvegarde supprimée', 'success');
    }
}

function clearAllBackups() {
    const backups = JSON.parse(localStorage.getItem(BACKUP_STORAGE_KEY) || '[]');
    if (backups.length === 0) return;
    
    if (confirm(`Supprimer toutes les ${backups.length} sauvegardes ?`)) {
        localStorage.removeItem(BACKUP_STORAGE_KEY);
        displayBackupList();
        showNotification('Toutes les sauvegardes supprimées', 'success');
    }
}

function exportAllData() {
    const data = saveAllData();
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    
    const dateStr = new Date().toISOString().split('T')[0];
    a.href = url;
    a.download = `calculateur_sauvegarde_${dateStr}.json`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
    
    showNotification('Données exportées', 'success');
}

function handleBackupFileSelect(event) {
    const file = event.target.files[0];
    if (!file) return;
    
    const reader = new FileReader();
    reader.onload = function(e) {
        try {
            const data = JSON.parse(e.target.result);
            
            if (confirm('Importer ces données ?')) {
                // Sauvegarder d'abord l'état actuel
                createBackup('Avant import');
                
                // Restaurer les nouvelles données
                if (data.modeA) restoreModeAData(data.modeA);
                if (data.modeC) {
                    cBlocks = data.modeC;
                    cBlockCounter = cBlocks.length + 1;
                }
                if (data.modeD) {
                    dDates = data.modeD.dates || [];
                    dSeries = data.modeD.series || [];
                }
                if (data.modeF) {
                    historiqueF = data.modeF.categories || [];
                    restoreModeFData(data.modeF.blocks || []);
                }
                if (data.modeG) restoreModeGData(data.modeG);
                if (data.modeH) {
                    conversionHistory = data.modeH.conversions || [];
                }
                if (data.calculator) {
                    calculatorHistory = data.calculator;
                }
                if (data.recentCalculations) {
                    recentCalculations = data.recentCalculations;
                }
                
                updateAllModes();
                updateDashboard();
                showNotification('Données importées', 'success');
                closeBackupModal();
            }
        } catch (error) {
            showNotification('Fichier invalide', 'error');
        }
    };
    reader.readAsText(file);
    event.target.value = '';
}
