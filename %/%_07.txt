// conversion hex -> rgb array (utile pour moyenne couleur)
function hexToRgbArr(hex){
  const c = (hex||'#000000').replace('#','');
  return [parseInt(c.substr(0,2),16), parseInt(c.substr(2,2),16), parseInt(c.substr(4,2),16)];
}
function rgbArrToHex(arr){
  return "#" + arr.map(n => n.toString(16).padStart(2,'0')).join('');
}

function calculerA(){
  const cont = document.getElementById("multiAContainer");
  const resDiv = document.getElementById("resultA"); resDiv.innerHTML = "";
  Array.from(cont.getElementsByClassName("ref-block")).forEach(block => {
    const refRaw = block.querySelector(".refA").value;
    const ref = parseFloat(refRaw);
    const raw = (block.querySelector(".valsA").value || "").trim();
    if(raw === "") return;
    const vals = raw.split(/\s+/).map(Number).filter(v => !isNaN(v));
    if(isNaN(ref) && vals.length===0) return;
    let html = `<h4>Analyse pour ${isNaN(ref)? 'n/a': ref.toFixed(2)}€</h4><table><tr><th>Valeur</th><th>Différence (€)</th><th>%</th></tr>`;
    vals.forEach(v => {
      const diff = isNaN(ref) ? v : v - ref;
      const pct = (isNaN(ref) || ref === 0) ? null : (diff / ref * 100);
      const col = diff > 0 ? "#2ecc71" : diff < 0 ? "#e74c3c" : "#95a5a6";
      html += `<tr style="background:${col}; color:#fff;"><td>${v}</td><td>${diff>=0?'+':''}${diff.toFixed(2)}</td><td>${pct===null? '—' : pct.toFixed(2)+'%'}</td></tr>`;
    });
    html += "</table>";
    resDiv.innerHTML += html;
  });
  historique.A = resDiv.innerHTML;
}

function updateCBlock(rangeInput){
  const block = rangeInput.closest(".ref-block");

  rangeInput.min = -max;
  rangeInput.max = max;

function updateC(){
  const cont = document.getElementById("multiCContainer");
  let html = "";
  Array.from(cont.getElementsByClassName("ref-block")).forEach((block,i) => {
    const valsStr = (block.querySelector(".valsC")?.value || "").trim();
    const vals = valsStr ? valsStr.split(/\s+/).map(Number).filter(v => !isNaN(v)) : [];
    const pct = parseFloat(block.querySelector(".rangeC")?.value) || 0;
    if(vals.length){
      html += `<h4>Bloc ${i+1} (${pct}%)</h4><table><tr><th>Valeur</th><th>Résultat</th></tr>`;
      vals.forEach(v => {
        const r = v * (1 + pct/100);
        const col = pct > 0 ? "#2ecc71" : pct < 0 ? "#e74c3c" : "#95a5a6";
        html += `<tr style="background:${col}; color:#fff;"><td>${v}</td><td>${r.toFixed(2)}</td></tr>`;
      });
      html += "</table>";
    }
  });
  document.getElementById("resultC").innerHTML = html;
  historique.C = html;
}

function calculerD(){
  const cont = document.getElementById("multiDContainer");
  const datasets = [];
  const labelsSet = new Set();

  Array.from(cont.getElementsByClassName("ref-block")).forEach((block,i) => {
    const label = block.querySelector(".labelD")?.value || `Série ${i+1}`;
    const eye = block.querySelector(".eyeD");
    const visible = !eye || eye.dataset.visible === "true";
    if(!visible) return;

    const dataMap = {};
    Array.from(block.querySelectorAll(".eventsDContainer div")).forEach(row => {
      const date = (row.querySelector(".dateD").value || "").trim();
      const val = parseFloat(row.querySelector(".valD").value);
      if(date && !isNaN(val)){
        dataMap[date] = val;
        labelsSet.add(date);
      }
    });

    datasets.push({
      label: label,
      data: dataMap,
      borderColor: ["#3498db","#2ecc71","#e74c3c","#f1c40f","#9b59b6","#1abc9c","#95a5a6"][i%7],
      fill: false,
      tension: 0.2
    });
  });

  // tri simple des labels (texte). On garde tri lexicographique ici.
  const labels = Array.from(labelsSet).sort();

  // transformer map -> array alignée avec labels
  datasets.forEach(ds => {
    ds.data = labels.map(lab => ds.data[lab] ?? null);
  });

  if(chartD) chartD.destroy();
  const ctx = document.getElementById("graphD").getContext("2d");
  chartD = new Chart(ctx, {
    type: 'line',
    data: { labels: labels, datasets: datasets },
    options: { plugins: { legend: { display: true } } }
  });

  historique.D = "Graphique multi-séries de l'évolution temporelle généré.";
}

/* =========================
   MODE F : Graphique circulaire
   - Ajout / sélection / fusion / tableau récap
   ========================= */
const paletteF = ["#3498db","#2ecc71","#e74c3c","#f1c40f","#9b59b6","#1abc9c","#95a5a6"];
let compteurF = 0;

function addFBlock(){
  const cont = document.getElementById("multiFContainer");
  const div = document.createElement("div"); div.className = "ref-block";
  const couleurParDefaut = paletteF[compteurF % paletteF.length];
  compteurF++;
  div.dataset.selectionnee = "false";
  div.innerHTML = `
    <input type="text" class="labelF" placeholder="Catégorie">
    <input type="number" class="valF" placeholder="Valeur">
    <input type="color" class="colorF" value="${couleurParDefaut}" title="Choisir la couleur">
    <button type="button" class="selectF">❌</button>
    <button type="button" onclick="removeF(this)">🗑️</button>
  `;

  // gestion sélection/désélection
  const btnSelect = div.querySelector(".selectF");
  btnSelect.addEventListener("click", () => {
    const isSelected = div.dataset.selectionnee === "true";
    div.dataset.selectionnee = isSelected ? "false" : "true";
    btnSelect.textContent = isSelected ? "❌" : "✅";
  });

  // mettre à jour historiqueF à chaque modification
  const inputVal = div.querySelector(".valF");
  const inputLabel = div.querySelector(".labelF");
  const inputColor = div.querySelector(".colorF");

  function majHistorique(){
    let cat = historiqueF.find(c => c.element === div);
    if(!cat){
      cat = { element: div, label: inputLabel.value || "Catégorie", val: parseFloat(inputVal.value) || 0, color: inputColor.value, sub: [] };
      historiqueF.push(cat);
    } else {
      cat.label = inputLabel.value || "Catégorie";
      cat.val = parseFloat(inputVal.value) || 0;
      cat.color = inputColor.value;
    }
    mettreAJourTableauF();
    calculerF();
  }

  inputVal.addEventListener("input", majHistorique);
  inputLabel.addEventListener("input", majHistorique);
  inputColor.addEventListener("input", majHistorique);

  cont.appendChild(div);
  majHistorique();
}

function removeF(btn){ 
  const block = btn.parentNode;
  // retirer du tableau historique si présent
  const idx = historiqueF.findIndex(c => c.element === block);
  if(idx !== -1) historiqueF.splice(idx,1);
  block.remove(); 
  mettreAJourTableauF();
  calculerF();
}

function calculerF(){
  // générer le graphique à partir des blocs valides
  const cont = document.getElementById("multiFContainer");
  const lines = [];
  Array.from(cont.getElementsByClassName("ref-block")).forEach(block => {
    const label = (block.querySelector(".labelF")?.value || "").trim();
    const val = parseFloat(block.querySelector(".valF")?.value);
    const color = block.querySelector(".colorF")?.value || paletteF[0];
    if(label && !isNaN(val)) lines.push({ label, val, color });
  });

  if(chartF) chartF.destroy();
  const ctx = document.getElementById("graphF").getContext("2d");
  chartF = new Chart(ctx, {
    type: 'pie',
    data: {
      labels: lines.map(l => l.label),
      datasets: [{ data: lines.map(l => l.val), backgroundColor: lines.map(l => l.color) }]
    },
    options: {
      plugins: {
        legend: { display: true },
        datalabels: {
          formatter: (value, ctx) => {
            const meta = ctx.chart.getDatasetMeta(0);
            const visibleTotal = ctx.chart.data.datasets[0].data.reduce((sum, v, idx) => (meta.data[idx].hidden ? sum : sum + v), 0);
            return visibleTotal === 0 ? '' : ((value / visibleTotal) * 100).toFixed(1) + '%';
          },
          color: '#fff',
          font: { weight: 'bold' }
        }
      }
    },
    plugins: [ChartDataLabels]
  });

  historique.F = "Graphique circulaire généré.";
  // mettre à jour tableau si besoin
  mettreAJourTableauF();
}

function mettreAJourTableauF(){
  // reconstruit le tableau d'après historiqueF (définit lors des ajouts / modifs)
  const tbody = document.querySelector("#tableauF tbody");
  tbody.innerHTML = "";
  // recalculer total en utilisant historiqueF
  const total = historiqueF.reduce((s,c) => s + (parseFloat(c.val) || 0), 0);
  historiqueF.forEach(cat => {
    const pourcent = total > 0 ? (cat.val / total * 100).toFixed(1) : 0;
    const tr = document.createElement("tr");
    tr.style.backgroundColor = cat.color;
    tr.style.color = "#fff";
    let sousHTML = "";
    if(cat.sub && cat.sub.length > 0){
      sousHTML = cat.sub.map(sc => {
        const pct = total > 0 ? ((sc.val / total) * 100).toFixed(1) : 0;
        return `<div style="margin-left:10px;">${sc.label}: ${sc.val} (${pct}%)</div>`;
      }).join("");
    }
    tr.innerHTML = `<td>${cat.label}</td><td>${cat.val}</td><td>${pourcent}%</td><td>${sousHTML}</td>`;
    tbody.appendChild(tr);
  });
}

/* Fusionner catégories sélectionnées (crée un nouveau bloc et conserve les sous-cat) */
function fusionnerFCategories(){
  const cont = document.getElementById("multiFContainer");
  const selected = Array.from(cont.getElementsByClassName("ref-block")).filter(d => d.dataset.selectionnee === "true");
  if(selected.length < 2){ alert("Sélectionnez au moins 2 catégories à fusionner !"); return; }

  let nomNouvelle = prompt("Entrez le nom de la nouvelle catégorie fusionnée :", "Nouvelle catégorie");
  if(!nomNouvelle) nomNouvelle = "Nouvelle catégorie";

  // total valeurs et moyenne couleurs
  let valTotale = 0;
  const rgbs = [];
  const sous = [];

  selected.forEach(d => {
    const val = parseFloat(d.querySelector(".valF")?.value) || 0;
    valTotale += val;
    rgbs.push(hexToRgbArr(d.querySelector(".colorF")?.value || '#000000'));
    // récupération des sous-cat déjà enregistrées
    const h = historiqueF.find(c => c.element === d);
    if(h){
      if(h.sub && h.sub.length) sous.push(...h.sub);
      sous.push({ label: h.label, val: h.val, color: h.color });
    }
  });

  // moyenne RGB
  const moy = rgbs[0].map((_,i) => Math.round(rgbs.reduce((s,c) => s + c[i], 0) / rgbs.length));
  const couleurFusion = rgbArrToHex(moy);

  // suppression des anciennes catégories (DOM + historique)
  selected.forEach(d => {
    const idx = historiqueF.findIndex(c => c.element === d);
    if(idx !== -1) historiqueF.splice(idx,1);
    d.remove();
  });

  // création du nouveau bloc
  const div = document.createElement("div"); div.className = "ref-block";
  div.dataset.selectionnee = "false";
  div.innerHTML = `
    <input type="text" class="labelF" placeholder="Catégorie" value="${nomNouvelle}">
    <input type="number" class="valF" placeholder="Valeur" value="${valTotale}">
    <input type="color" class="colorF" value="${couleurFusion}">
    <button type="button" class="selectF">❌</button>
    <button type="button" onclick="removeF(this)">🗑️</button>
  `;

  // bouton sélection
  const btnSelect = div.querySelector(".selectF");
  btnSelect.addEventListener("click", () => {
    const isSelected = div.dataset.selectionnee === "true";
    div.dataset.selectionnee = isSelected ? "false" : "true";
    btnSelect.textContent = isSelected ? "❌" : "✅";
  });

  // ajouter au DOM + historiqueF
  document.getElementById("multiFContainer").appendChild(div);
  const cat = { element: div, label: nomNouvelle, val: valTotale, color: couleurFusion, sub: sous };
  historiqueF.push(cat);

  // lier modifications au nouveau bloc (comme addFBlock le ferait)
  const inputVal = div.querySelector(".valF");
  const inputLabel = div.querySelector(".labelF");
  const inputColor = div.querySelector(".colorF");
  function majHistoriqueNouvel(){
    const c = historiqueF.find(x => x.element === div);
    if(c){
      c.label = inputLabel.value || "Catégorie";
      c.val = parseFloat(inputVal.value) || 0;
      c.color = inputColor.value;
    }
    mettreAJourTableauF();
    calculerF();
  }
  inputVal.addEventListener("input", majHistoriqueNouvel);
  inputLabel.addEventListener("input", majHistoriqueNouvel);
  inputColor.addEventListener("input", majHistoriqueNouvel);

  // mise à jour visuelle
  mettreAJourTableauF();
  calculerF();
}

function calculerG(){
  const cont = document.getElementById("multiGContainer");
  let html = `<table><tr><th>#</th><th>Montant</th><th>Type</th><th>% TVA</th><th>Valeur TVA (€)</th><th>Résultat</th></tr>`;
  Array.from(cont.getElementsByClassName("ref-block")).forEach((block,i) => {
    const mode = block.querySelector(".tvaMode").value;
    const val = parseFloat(block.querySelector(".valeurG").value);
    const tva = parseFloat(block.querySelector(".tauxTVA").value);
    if(isNaN(val) || isNaN(tva)) return;
    let res, tvaValue;
    if(mode === "htTtc"){
      res = val * (1 + tva/100);
      tvaValue = res - val;
    } else {
      res = val / (1 + tva/100);
      tvaValue = val - res;
    }
    html += `<tr><td>${i+1}</td><td>${val.toFixed(2)}</td><td>${mode === "htTtc" ? "HT → TTC" : "TTC → HT"}</td><td>${tva.toFixed(2)}%</td><td>${tvaValue.toFixed(2)}</td><td>${res.toFixed(2)}</td></tr>`;
  });
  html += "</table>";
  document.getElementById("resultG").innerHTML = html;
  historique.G = html;
}

async function genererRapport(){
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF({orientation:'p', unit:'pt', format:'a4'});
  let y = 40;
  for(const m of ["A","B","C","D","E","F","G"]){
    if(document.getElementById("include"+m)?.checked){
      doc.setFontSize(14);
      doc.text(`Mode ${m} : ${modeDescriptions[m]}`, 40, y);
      y += 20;
      doc.setFontSize(10);
      const text = (historique[m] || "").replace(/<[^>]+>/g,' ').trim();
      const lines = doc.splitTextToSize(text, 520);
      doc.text(lines, 40, y);
      y += lines.length * 12 + 10;

      // Ajouter images de graphiques si existantes
      let chartImg = null;
      if(m === "D" && chartD) chartImg = chartD.toBase64Image();
      if(m === "E" && chartE) chartImg = chartE.toBase64Image();
      if(m === "F" && chartF) chartImg = chartF.toBase64Image();

      if(chartImg){
        if(y > 700){ doc.addPage(); y = 40; }
        doc.addImage(chartImg, "PNG", 80, y, 400, 300);
        y += 320;
      }
      if(y > 760){ doc.addPage(); y = 40; }
    }
  }
  doc.save("rapport_global.pdf");
}
