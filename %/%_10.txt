function autoCalculA(){
    const cont = document.getElementById("multiAContainer");
    const resDiv = document.getElementById("resultA"); 
    resDiv.innerHTML = "";
    
    Array.from(cont.getElementsByClassName("ref-block")).forEach((block, index) => {
        const ref = parseNumber(block.querySelector(".refA").value);
        const raw = (block.querySelector(".valsA").value || "").trim();
        if(raw === "") return;
        const vals = raw.split(/\s+/).map(parseNumber).filter(v => v !== null); 
        
        let html = `<div style="margin: 15px 0;"><h4>Bloc ${index + 1} - Référence : ${ref === null ? 'n/a': ref.toFixed(2)}€</h4>`;
        html += `<table style="width:100%;"><tr><th>Valeur</th><th>Différence</th><th>%</th></tr>`;
        
        vals.forEach(v => {
            const diff = ref === null ? v : v - ref;
            const pct = (ref === null || ref === 0) ? null : (diff / ref * 100);
            const col = diff > 0 ? "#2ecc71" : diff < 0 ? "#e74c3c" : "#95a5a6";
            html += `<tr style="background:${col}; color:#fff;">
                <td>${v.toFixed(2)} €</td>
                <td>${diff>=0?'+':''}${diff.toFixed(2)} €</td>
                <td>${pct===null? '—' : pct.toFixed(2)+'%'}</td>
            </tr>`;
        });
        html += "</table></div>";
        resDiv.innerHTML += html;
    });
}

function addDDate(defaultDate = null) {
    if (!defaultDate) {
        const today = new Date();
        const yyyy = today.getFullYear();
        const mm = String(today.getMonth() + 1).padStart(2, '0');
        const dd = String(today.getDate()).padStart(2, '0');
        defaultDate = `${yyyy}-${mm}-${dd}`;
    }
    const newDateId = Date.now() + Math.random();
    dDates.push({ id: newDateId, date: defaultDate });
    dSeries.forEach(s => { s.values[newDateId] = ''; });
    updateDTable();
}

function sortDDates() {
    dDates.sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
    updateDTable();
}

function addDSeries() {
    const count = dSeries.length + 1;
    const newSeriesId = Date.now() + Math.random();
    const colors = ["#3498db","#2ecc71","#e74c3c","#f1c40f","#9b59b6","#1abc9c"];
    
    const newSeries = { 
        id: newSeriesId, 
        label: `Série ${count}`, 
        color: colors[dSeries.length % colors.length], 
        values: {},
        visible: true
    };

    dDates.forEach(d => {
        newSeries.values[d.id] = '';
    });

    dSeries.push(newSeries);
    updateDTable();
}

function removeDDate(dateId) {
    dDates = dDates.filter(d => d.id !== dateId);
    dSeries.forEach(s => { delete s.values[dateId]; });
    updateDTable();
}

function updateDDateValue(dateId, newDateString) {
    const dateObj = dDates.find(d => d.id === dateId);
    if (dateObj) {
        dateObj.date = newDateString;
    }
    updateDTable(false);

function updateDValue(seriesId, dateId, value) {
    const series = dSeries.find(s => s.id === seriesId);
    if (series) {
        series.values[dateId] = parseNumber(value) || 0;
    }

function updateDTable(recreate = true) {
    const dateHeaderRow = document.getElementById("dateHeaderRow");
    const seriesBody = document.getElementById("seriesBody");

    if(recreate) {
        dDates.sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
    }

    dateHeaderRow.innerHTML = '<th style="min-width: 160px;">Série / Date</th>';
    dDates.forEach(d => {
        const th = document.createElement('th');
        th.innerHTML = `
            <div style="display: flex; flex-direction: column; align-items: center; gap: 4px;">
                <input type="date" class="date-input-d" value="${d.date}" 
                       onchange="updateDDateValue(${d.id}, this.value)">
                <button type="button" onclick="removeDDate(${d.id})" style="padding: 3px 8px; font-size: 12px;">🗑️</button>
            </div>
        `;
        dateHeaderRow.appendChild(th);
    });
    
    seriesBody.innerHTML = '';
    dSeries.forEach((s, index) => {
        const tr = document.createElement('tr');
        tr.dataset.seriesId = s.id;

        const thSeries = document.createElement('th');
        thSeries.innerHTML = `
            <div class="series-header-d">
                <input type="text" class="series-name-input" value="${s.label}" 
                       oninput="updateDSeriesProp(${s.id}, 'label', this.value)" style="width:110px;">
                <input type="color" value="${s.color}" 
                       onchange="updateDSeriesProp(${s.id}, 'color', this.value)">
                <button type="button" onclick="removeDSeries(${s.id})" style="padding: 3px 8px; font-size: 12px;">🗑️</button>
            </div>
        `;
        thSeries.style.textAlign = 'left';
        tr.appendChild(thSeries);

        dDates.forEach(d => {
            const rawVal = s.values[d.id];
            const currentValue = rawVal !== null && rawVal !== undefined ? rawVal : '';
            
            const td = document.createElement('td');
            td.innerHTML = `<input type="number" class="value-input-d" placeholder="0" value="${currentValue}"
                                   oninput="updateDValue(${s.id}, ${d.id}, this.value)">`;
            tr.appendChild(td);
        });
        
        seriesBody.appendChild(tr);
    });
}

function removeDSeries(seriesId) {

    updateDTable();
}

function updateDSeriesProp(seriesId, prop, value) {
    const series = dSeries.find(s => s.id === seriesId);
    if (series) {
        series[prop] = value;
    }
    calculerD();
}

function calculerD(){
    const canvasD = document.getElementById("graphD");
    const ctx = canvasD.getContext('2d');
    
    if (chartD) chartD.destroy();
    
    if (dDates.length === 0 || dSeries.length === 0) {
        canvasD.style.display = 'none';
        return;
    }
    
    const labels = dDates.map(d => {
        const date = new Date(d.date);
        return date.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric' });
    });
    
    const datasets = [];
    dSeries.forEach((s) => {
        if (!s.visible) return;
        
        const dataPoints = [];
        dDates.forEach(d => {
            const val = parseNumber(s.values[d.id]);
            dataPoints.push(val !== null ? val : 0);
        });
        
        datasets.push({
            label: s.label,
            data: dataPoints,
            borderColor: s.color,
            backgroundColor: s.color + '40',
            borderWidth: 3,
            fill: true,
            tension: 0.2,
            pointRadius: 6
        });
    });
    
    if (datasets.length === 0) {
        canvasD.style.display = 'none';
        return;
    }
    
    chartD = new Chart(ctx, {
        type: 'line',
        data: { labels, datasets },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: { legend: { display: true, position: 'top' } },
            scales: {
                y: { beginAtZero: true, grid: { color: 'rgba(0,0,0,0.1)' } },
                x: { grid: { color: 'rgba(0,0,0,0.1)' } }
            }
        }
    });
    
    canvasD.style.display = 'block';
}

    mettreAJourTableauF();

function calculerF(){
    const canvasF = document.getElementById("graphF");
    const ctx = canvasF.getContext('2d');
    
    if (chartF) chartF.destroy();
    
    const validCategories = historiqueF.filter(cat => cat.label && cat.val > 0);
    
    if (validCategories.length === 0) {
        canvasF.style.display = 'none';
        return;
    }
    
    const labels = validCategories.map(cat => cat.label);
    const data = validCategories.map(cat => cat.val);
    const colors = validCategories.map(cat => cat.color);
    
    chartF = new Chart(ctx, {
        type: 'pie',
        data: {
            labels: labels,
            datasets: [{
                data: data,
                backgroundColor: colors,
                borderWidth: 2
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                legend: { display: true, position: 'right' },
                tooltip: {
                    callbacks: {
                        label: function(context) {
                            const total = data.reduce((a, b) => a + b, 0);
                            const percentage = ((context.parsed / total) * 100).toFixed(1);
                            return `${context.label}: ${context.parsed} € (${percentage}%)`;
                        }
                    }
                }
            }
        }
    });
    
    canvasF.style.display = 'block';
}

function mettreAJourTableauF(){
    const tbody = document.querySelector("#tableauF tbody");
    tbody.innerHTML = "";
    
    const total = historiqueF.reduce((sum, cat) => sum + (cat.val || 0), 0);
    
    historiqueF.forEach(cat => {
        const pourcent = total > 0 ? ((cat.val / total) * 100).toFixed(1) : 0;
        const tr = document.createElement("tr");
        tr.innerHTML = `<td>${cat.label}</td><td>${cat.val.toFixed(2)} €</td><td>${pourcent}%</td><td>${cat.id}</td>`;
        tbody.appendChild(tr);
    });
    
    if (historiqueF.length > 0) {
        const trTotal = document.createElement("tr");
        trTotal.style.fontWeight = "bold";
        trTotal.innerHTML = `<td>Total</td><td>${total.toFixed(2)} €</td><td>100%</td><td>-</td>`;
        tbody.appendChild(trTotal);
    }
}

function mergeSelectedCategories() {
    const selectedCategories = Array.from(document.querySelectorAll('.select-for-fusion:checked'))
        .map(cb => cb.closest('.ref-block'))
        .filter(block => block)
        .map(block => {
            const catId = parseInt(block.getAttribute('data-category-id'));
            return historiqueF.find(cat => cat.id === catId);
        })
        .filter(cat => cat);
    
    if (selectedCategories.length < 2) {
        showNotification('Veuillez sélectionner au moins 2 catégories à fusionner', 'error');
        return;
    }
    
    const fusionName = document.getElementById('fusionName').value.trim();
    if (!fusionName) {
        showNotification('Veuillez donner un nom à la nouvelle catégorie', 'error');
        return;
    }
    
    // Calculer la nouvelle valeur (somme)
    const newValue = selectedCategories.reduce((sum, cat) => sum + cat.val, 0);
    
    // Calculer la nouvelle couleur (moyenne des couleurs)
    const colorSum = selectedCategories.reduce((sum, cat) => {
        const hex = cat.color.replace('#', '');
        const r = parseInt(hex.substr(0, 2), 16);
        const g = parseInt(hex.substr(2, 2), 16);
        const b = parseInt(hex.substr(4, 2), 16);
        return {
            r: sum.r + r,
            g: sum.g + g,
            b: sum.b + b
        };
    }, {r: 0, g: 0, b: 0});
    
    const avgColor = {
        r: Math.round(colorSum.r / selectedCategories.length),
        g: Math.round(colorSum.g / selectedCategories.length),
        b: Math.round(colorSum.b / selectedCategories.length)
    };
    
    const newColor = `#${avgColor.r.toString(16).padStart(2, '0')}${avgColor.g.toString(16).padStart(2, '0')}${avgColor.b.toString(16).padStart(2, '0')}`;
    
    // Supprimer les anciennes catégories
    selectedCategories.forEach(cat => {
        const idx = historiqueF.findIndex(c => c.id === cat.id);
        if (idx !== -1) {
            historiqueF.splice(idx, 1);
            cat.element.remove();
        }
    });
    
    // Ajouter la nouvelle catégorie fusionnée
    addFBlock({
        label: fusionName,
        val: newValue,
        color: newColor,
        isFusion: true
    });
    
    // Réinitialiser le champ de nom
    document.getElementById('fusionName').value = '';
    
    // Désélectionner toutes les cases
    selectAllForFusion(false);
    
    showNotification(`Catégories fusionnées : ${fusionName} = ${newValue.toFixed(2)} €`, 'success');
}

function autoCalculG(){
    const cont = document.getElementById("multiGContainer");
    const resDiv = document.getElementById("resultG"); 
    resDiv.innerHTML = "";

    const blocks = Array.from(cont.getElementsByClassName('g-block'));
    if (blocks.length === 0) {
        resDiv.innerHTML = "<p>Aucun bloc TVA défini.</p>";
        return;
    }

    blocks.forEach((block, idx) => {
        const blockName = block.querySelector('.bloc-nom-input')?.value || `Bloc ${idx+1}`;
        const globalTva = parseFloat(block.querySelector('.global-tva-input')?.value) || 20;
        const lines = Array.from(block.querySelectorAll('.ligne-tva'));

        let html = `<h4>${blockName} (TVA globale ${globalTva}%)</h4>`;
        html += `<table style="width:100%; margin:15px 0; border-collapse:collapse;">
            <tr>
                <th>Libellé</th><th>Montant donné</th><th>Taux TVA</th><th>Montant TVA</th><th>Montant HT</th><th>Montant TTC</th>
            </tr>`;

        let totalTVA = 0, totalHT = 0, totalTTC = 0;
        
        lines.forEach(l => {
            const lib = l.querySelector('.libelleG')?.value || '';
            const raw = parseFloat(l.querySelector('.valeurG')?.value);
            if (isNaN(raw)) return;
            
            const type = l.querySelector('.typeG')?.value || 'HT';
            const tva = parseFloat(l.querySelector('.tvaIndividuelleG')?.value);
            const applicableTva = isNaN(tva) ? globalTva : tva;

            const factor = 1 + applicableTva/100;
            let montantHT, montantTTC, tvaVal;
            
            if (type === 'HT') {
                montantHT = raw;
                montantTTC = montantHT * factor;
                tvaVal = montantTTC - montantHT;
            } else {
                montantTTC = raw;
                montantHT = montantTTC / factor;
                tvaVal = montantTTC - montantHT;
            }

            totalTVA += tvaVal;
            totalHT += montantHT;
            totalTTC += montantTTC;

            html += `<tr>
                <td style="text-align:left;">${lib}</td>
                <td>${raw.toFixed(2)} € (${type})</td>
                <td>${applicableTva}%</td>
                <td>${tvaVal.toFixed(2)} €</td>
                <td>${montantHT.toFixed(2)} €</td>
                <td>${montantTTC.toFixed(2)} €</td>
            </tr>`;
        });

        html += `<tr style="font-weight:bold; background:#2c3e50; color:white;">
            <td style="text-align:left;">Totaux</td>
            <td>—</td><td>—</td>
            <td>${totalTVA.toFixed(2)} €</td>
            <td>${totalHT.toFixed(2)} €</td>
            <td>${totalTTC.toFixed(2)} €</td>
        </tr>`;
        html += `</table>`;
        
        resDiv.innerHTML += html;
    });
}

async function loadExchangeRates() {
    try {
        document.getElementById('lastUpdate').textContent = 'Chargement des taux...';
        
        const response = await axios.get('https://api.exchangerate-api.com/v4/latest/EUR');
        exchangeRates = response.data.rates;
        
        const defaultRates = {
            'EUR': 1, 'USD': 1.08, 'GBP': 0.86, 'CHF': 0.95, 
            'JPY': 161.5, 'CAD': 1.46, 'AUD': 1.65, 'CNY': 7.85,
            'INR': 89.2, 'RUB': 99.5, 'BRL': 5.32, 'KRW': 1440
        };
        
        exchangeRates = { ...defaultRates, ...exchangeRates };
        
        const now = new Date();
        document.getElementById('lastUpdate').textContent = 
            `Dernière actualisation: ${now.toLocaleTimeString('fr-FR', {hour: '2-digit', minute:'2-digit'})}`;
        
        updateCurrencySelectors();
        updateRateInfo();
        convertCurrency();
        updateAllRatesDisplay();
        
        setTimeout(loadExchangeRates, 5 * 60 * 1000);
        
    } catch (error) {
        console.log('Utilisation des taux par défaut');
        document.getElementById('lastUpdate').textContent = 'Taux par défaut (hors ligne)';
        
        exchangeRates = {
            'EUR': 1, 'USD': 1.08, 'GBP': 0.86, 'CHF': 0.95, 
            'JPY': 161.5, 'CAD': 1.46, 'AUD': 1.65, 'CNY': 7.85,
            'INR': 89.2, 'RUB': 99.5, 'BRL': 5.32, 'KRW': 1440,
            'MXN': 18.2, 'SGD': 1.45, 'HKD': 8.45, 'NZD': 1.78
        };
        
        updateCurrencySelectors();
        updateRateInfo();
        convertCurrency();
        updateAllRatesDisplay();
        
        setTimeout(loadExchangeRates, 60000);
    }
}

function updateCurrencySelectors() {
    const fromSelect = document.getElementById('fromCurrency');
    const toSelect = document.getElementById('toCurrency');
    
    fromSelect.innerHTML = '';
    toSelect.innerHTML = '';
    
    const mainCurrencies = ['EUR', 'USD', 'GBP', 'CHF', 'JPY', 'CAD', 'AUD', 'CNY'];
    
    mainCurrencies.forEach(currency => {
        if (currencyNames[currency]) {
            const option1 = document.createElement('option');
            option1.value = currency;
            option1.textContent = `${currency} - ${currencyNames[currency]}`;
            fromSelect.appendChild(option1);
            
            const option2 = document.createElement('option');
            option2.value = currency;
            option2.textContent = `${currency} - ${currencyNames[currency]}`;
            toSelect.appendChild(option2);
        }
    });
    
    fromSelect.value = 'EUR';
    toSelect.value = 'USD';
    updateFlags();
}

function updateFlags() {
    const from = document.getElementById('fromCurrency').value;
    const to = document.getElementById('toCurrency').value;
    
    if (currencyFlags[from]) {
        document.getElementById('fromFlag').src = `https://flagcdn.com/w20/${currencyFlags[from]}.png`;
    }
    if (currencyFlags[to]) {
        document.getElementById('toFlag').src = `https://flagcdn.com/w20/${currencyFlags[to]}.png`;
    }
    
    updateRateInfo();
    convertCurrency();
}

function switchCurrencies() {
    const fromSelect = document.getElementById('fromCurrency');
    const toSelect = document.getElementById('toCurrency');
    const fromAmount = document.getElementById('fromAmount');
    const toAmount = document.getElementById('toAmount');
    
    // Échanger les devises
    const tempCurrency = fromSelect.value;
    fromSelect.value = toSelect.value;
    toSelect.value = tempCurrency;
    
    // Échanger les montants (en gardant la conversion)
    const tempAmount = fromAmount.value;
    fromAmount.value = toAmount.value.replace(' €', '');
    convertCurrency();
    
    updateFlags();
    showNotification('Devises inversées', 'info');
}

function updateRateInfo() {
    const from = document.getElementById('fromCurrency').value;
    const to = document.getElementById('toCurrency').value;
    
    if (exchangeRates[from] && exchangeRates[to]) {
        const rate = exchangeRates[to] / exchangeRates[from];
        const rateReverse = exchangeRates[from] / exchangeRates[to];
        
        document.getElementById('rateInfo').innerHTML = 
            `💰 Taux: 1 ${from} = <strong>${rate.toFixed(4)}</strong> ${to} | 1 ${to} = <strong>${rateReverse.toFixed(4)}</strong> ${from}`;
    }
}

function convertCurrency() {
    const from = document.getElementById('fromCurrency').value;
    const to = document.getElementById('toCurrency').value;
    const amount = parseFloat(document.getElementById('fromAmount').value) || 0;
    
    if (exchangeRates[from] && exchangeRates[to]) {
        const rate = exchangeRates[to] / exchangeRates[from];
        const result = amount * rate;
        
        document.getElementById('toAmount').value = result.toFixed(2) + ' €';
    }
}

function updateAllRatesDisplay() {
    const container = document.getElementById('allRatesGrid');

function saveConversionToHistory() {
    const fromAmount = document.getElementById('fromAmount').value;
    const fromCurrency = document.getElementById('fromCurrency').value;
    const toAmount = document.getElementById('toAmount').value;
    const toCurrency = document.getElementById('toCurrency').value;
    
    if (!fromAmount || fromAmount <= 0) {
        showNotification('Veuillez entrer un montant valide', 'error');
        return;
    }
    
    const conversion = {
        id: Date.now(),
        date: new Date().toLocaleString('fr-FR'),
        fromAmount: parseFloat(fromAmount),
        fromCurrency: fromCurrency,
        toAmount: parseFloat(toAmount.replace(' €', '')),
        toCurrency: toCurrency,
        rate: (parseFloat(toAmount.replace(' €', '')) / parseFloat(fromAmount)).toFixed(4)
    };
    
    conversionHistory.unshift(conversion);
    if (conversionHistory.length > 50) {
        conversionHistory = conversionHistory.slice(0, 50);
    }
    
    localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
    updateConversionHistoryDisplay();
    showNotification('Conversion sauvegardée dans l\'historique !', 'success');
}

function updateConversionHistoryDisplay() {
    const container = document.getElementById('conversionHistoryList');
    if (!container) return;
    
    container.innerHTML = '';
    
    if (conversionHistory.length === 0) {
        container.innerHTML = '<p style="text-align: center; color: #666; padding: 20px;">Aucune conversion dans l\'historique</p>';
        return;
    }
    
    conversionHistory.forEach(conversion => {
        const item = document.createElement('div');
        item.className = 'history-item';
        item.innerHTML = `
            <div class="history-date">${conversion.date}</div>
            <div class="history-amount">${conversion.fromAmount.toFixed(2)} ${conversion.fromCurrency}</div>
            <div class="history-result">${conversion.toAmount.toFixed(2)} ${conversion.toCurrency}</div>
            <div class="history-actions">
                <button onclick="loadConversion(${conversion.id})" style="padding: 5px 10px; background: #3498db; color: white; border: none; border-radius: 4px; cursor: pointer;">↻</button>
                <button onclick="deleteConversion(${conversion.id})" style="padding: 5px 10px; background: #e74c3c; color: white; border: none; border-radius: 4px; cursor: pointer;">🗑️</button>
            </div>
        `;
        container.appendChild(item);
    });
}

function loadConversion(id) {
    const conversion = conversionHistory.find(c => c.id === id);
    if (conversion) {
        document.getElementById('fromAmount').value = conversion.fromAmount;
        document.getElementById('fromCurrency').value = conversion.fromCurrency;
        document.getElementById('toCurrency').value = conversion.toCurrency;
        updateFlags();
        convertCurrency();
        showNotification('Conversion chargée !', 'success');
    }
}

function deleteConversion(id) {
    conversionHistory = conversionHistory.filter(c => c.id !== id);
    localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
    updateConversionHistoryDisplay();
    showNotification('Conversion supprimée !', 'success');
}

function clearConversionHistory() {
    if (confirm('Voulez-vous vraiment effacer tout l\'historique des conversions ?')) {
        conversionHistory = [];
        localStorage.setItem('conversionHistory', JSON.stringify(conversionHistory));
        updateConversionHistoryDisplay();
        showNotification('Historique effacé !', 'success');
    }
}

function exportHistoryToCSV() {
    if (conversionHistory.length === 0) {
        showNotification('Aucune conversion à exporter', 'warning');
        return;
    }
    
    let csv = 'Date;Devise Source;Montant Source;Devise Cible;Montant Cible;Taux\n';
    conversionHistory.forEach(conv => {
        csv += `${conv.date};${conv.fromCurrency};${conv.fromAmount.toFixed(2)};${conv.toCurrency};${conv.toAmount.toFixed(2)};${conv.rate}\n`;
    });
    
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    const url = URL.createObjectURL(blob);
    link.setAttribute('href', url);
    link.setAttribute('download', `historique_conversions_${new Date().toISOString().split('T')[0]}.csv`);
    link.style.visibility = 'hidden';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    showNotification('CSV exporté avec succès', 'success');
}

async function downloadPDFReport() {
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF('p', 'mm', 'a4');
    const pageWidth = doc.internal.pageSize.width;
    const pageHeight = doc.internal.pageSize.height;
    let yPos = 20;
    
    // En-tête
    doc.setFontSize(20);
    doc.setTextColor(44, 62, 80);
    doc.text('Rapport Global', pageWidth / 2, yPos, { align: 'center' });
    
    doc.setFontSize(10);
    doc.setTextColor(102, 102, 102);
    yPos += 10;
    doc.text(`Généré le ${new Date().toLocaleString('fr-FR')}`, pageWidth / 2, yPos, { align: 'center' });
    yPos += 15;
    
    // Fonction pour ajouter une page si nécessaire
    function checkPageBreak(neededSpace) {
        if (yPos + neededSpace > pageHeight - 20) {
            doc.addPage();
            yPos = 20;
            return true;
        }
        return false;
    }
    
    // Fonction pour ajouter une section
    function addSection(title, content, color) {
        checkPageBreak(20);
        doc.setFontSize(14);
        doc.setTextColor(color.r, color.g, color.b);
        doc.text(title, 14, yPos);
        yPos += 8;
        
        doc.setDrawColor(color.r, color.g, color.b);
        doc.setLineWidth(0.5);
        doc.line(14, yPos, pageWidth - 14, yPos);
        yPos += 10;
        
        if (content && content.trim() !== '') {
            doc.setFontSize(10);
            doc.setTextColor(0, 0, 0);
            const lines = doc.splitTextToSize(content.substring(0, 500), pageWidth - 28);
            doc.text(lines, 14, yPos);
            yPos += (lines.length * 6) + 15;
        } else {
            doc.setFontSize(10);
            doc.setTextColor(153, 153, 153);
            doc.text('Aucune donnée', 14, yPos);
            yPos += 10;
        }
        
        // Ligne de séparation
        doc.setDrawColor(221, 221, 221);
        doc.setLineWidth(0.3);
        doc.line(14, yPos, pageWidth - 14, yPos);
        yPos += 15;
    }
    
    // Mode A
    if (document.getElementById('reportA').checked) {
        const resultA = document.getElementById('resultA').textContent || '';
        addSection('Mode A : Comparer à des valeurs', resultA, {r: 52, g: 152, b: 219});
    }
    
    // Mode B
    if (document.getElementById('reportB').checked) {
        const resultB = document.getElementById('resultB').textContent || '';
        addSection('Mode B : Gains et pertes', resultB, {r: 46, g: 204, b: 113});
    }
    
    // Mode C
    if (document.getElementById('reportC').checked) {
        const resultC = document.getElementById('resultC').textContent || '';
        addSection('Mode C : Pourcentage par bloc', resultC, {r: 231, g: 76, b: 60});
    }
    
    // Mode G
    if (document.getElementById('reportG').checked) {
        const resultG = document.getElementById('resultG').textContent || '';
        addSection('Mode G : Calculateur TVA', resultG, {r: 230, g: 126, b: 34});
    }
    
    // Mode H
    if (document.getElementById('reportH').checked) {
        checkPageBreak(20);
        doc.setFontSize(14);
        doc.setTextColor(52, 73, 94);
        doc.text('Mode H : Convertisseur de Devises', 14, yPos);
        yPos += 15;
        
        // Dernière conversion
        const fromAmount = document.getElementById('fromAmount').value;
        const fromCurrency = document.getElementById('fromCurrency').value;
        const toAmount = document.getElementById('toAmount').value;
        const toCurrency = document.getElementById('toCurrency').value;

function toggleVoiceAssistant() {
    if (!('webkitSpeechRecognition' in window) && !('SpeechRecognition' in window)) {
        alert('La reconnaissance vocale n\'est pas supportée par votre navigateur. Essayez Chrome ou Edge.');
        return;
    }
    
    if (!isListening) {
        startVoiceRecognition();
    } else {
        stopVoiceRecognition();
    }
}

function startVoiceRecognition() {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    voiceRecognition = new SpeechRecognition();
    voiceRecognition.continuous = true;
    voiceRecognition.interimResults = true;
    voiceRecognition.lang = 'fr-FR';
    
    voiceRecognition.onstart = function() {
        isListening = true;
        document.getElementById('voiceStatus').classList.add('recording');
        document.getElementById('voiceText').textContent = 'Écoute en cours...';
        document.getElementById('voiceStatus').style.display = 'flex';
        document.getElementById('voiceIcon').textContent = '🔴';
        showVoiceFeedback('Assistant vocal activé', 'Dites "mode A" ou "rapport"');
    };
    
    voiceRecognition.onresult = function(event) {
        let interimTranscript = '';
        let finalTranscript = '';
        
        for (let i = event.resultIndex; i < event.results.length; i++) {
            const transcript = event.results[i][0].transcript;
            if (event.results[i].isFinal) {
                finalTranscript += transcript;
            } else {
                interimTranscript += transcript;
            }
        }
        
        if (finalTranscript) {
            processVoiceCommand(finalTranscript);
        }
        
        document.getElementById('voiceText').textContent = interimTranscript || finalTranscript || 'Écoute en cours...';
    };
    
    voiceRecognition.onerror = function(event) {
        console.error('Erreur de reconnaissance vocale:', event.error);
    };
    
    voiceRecognition.onend = function() {
        if (isListening) {
            // Redémarrer l'écoute
            setTimeout(() => {
                if (isListening && voiceRecognition) {
                    voiceRecognition.start();
                }
            }, 100);
        } else {
            document.getElementById('voiceStatus').classList.remove('recording');
            document.getElementById('voiceText').textContent = 'Assistant vocal prêt';
            document.getElementById('voiceIcon').textContent = '🎤';
        }
    };
    
    voiceRecognition.start();
}

function stopVoiceRecognition() {
    isListening = false;
    if (voiceRecognition) {
        voiceRecognition.stop();
    }
    document.getElementById('voiceStatus').classList.remove('recording');
    document.getElementById('voiceText').textContent = 'Assistant vocal prêt';
    document.getElementById('voiceIcon').textContent = '🎤';
    showVoiceFeedback('Assistant vocal désactivé', '');
}

function processVoiceCommand(command) {
    command = command.toLowerCase().trim();
    console.log('Commande vocale:', command);
    
    // Navigation simple entre modes
    if (command.includes('mode a') || command.includes('comparer') || command.includes('référence')) {
        showMode('modeA');
        showVoiceFeedback(command, '📊 Mode A ouvert');
        return;
    } 
    else if (command.includes('mode b') || command.includes('gains') || command.includes('pertes')) {
        showMode('modeB');
        showVoiceFeedback(command, '📈 Mode B ouvert');
        return;
    } 
    else if (command.includes('mode c') || command.includes('pourcentage')) {
        showMode('modeC');
        showVoiceFeedback(command, '🎚️ Mode C ouvert');
        return;
    } 
    else if (command.includes('mode d') || command.includes('évolution') || command.includes('temporel')) {
        showMode('modeD');
        showVoiceFeedback(command, '📈 Mode D ouvert');
        return;
    } 
    else if (command.includes('mode e') || command.includes('linéaire') || command.includes('séries')) {
        showMode('modeE');
        showVoiceFeedback(command, '📈 Mode E ouvert');
        return;
    } 
    else if (command.includes('mode f') || command.includes('circulaire') || command.includes('camembert')) {
        showMode('modeF');
        showVoiceFeedback(command, '🟣 Mode F ouvert');
        return;
    } 
    else if (command.includes('mode g') || command.includes('tva') || command.includes('facture')) {
        showMode('modeG');
        showVoiceFeedback(command, '💰 Mode G ouvert');
        return;
    } 
    else if (command.includes('mode h') || command.includes('devises') || command.includes('conversion')) {
        showMode('modeH');
        showVoiceFeedback(command, '💱 Mode H ouvert');
        return;
    } 
    else if (command.includes('rapport') || command.includes('pdf') || command.includes('rapport global')) {
        showMode('mode_rapport_global');
        showVoiceFeedback(command, '📄 Rapport ouvert');
        return;
    } 
    else if (command.includes('menu') || command.includes('accueil') || command.includes('retour')) {
        showMode('none');
        showVoiceFeedback(command, '🏠 Menu principal');
        return;
    }
    
    // Si on ne comprend pas, donner une indication simple
    showVoiceFeedback(command, 'Dites "mode A" ou "rapport" pour naviguer');
}

    const display = document.getElementById('voiceCommandDisplay');

    actionText.textContent = action;
    
    display.classList.add('active');
    
    setTimeout(() => {
        display.classList.remove('active');
    }, 2000);
}
