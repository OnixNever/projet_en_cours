function hexToRgbArr(hex){

    return [parseInt(c.substr(0,2),16), parseInt(c.substr(2,2),16), parseInt(c.substr(4,2),16)];

function formatDateForDisplay(dateString) {
    if (!dateString) return '';
    if (dateString.match(/^\d{4}-\d{2}-\d{2}$/)) {
        const [yyyy, mm, dd] = dateString.split('-');
        return `${dd}/${mm}/${yyyy}`;
    }
    return dateString;
}
function parseNumber(value) {
    if (typeof value === 'string') {
        // Remplace la virgule par le point pour la compatibilité
        value = value.replace(',', '.');
    }
    const num = parseFloat(value);
    return isNaN(num) ? null : num;
}

function calculerA(){
    const cont = document.getElementById("multiAContainer");
    const resDiv = document.getElementById("resultA"); resDiv.innerHTML = "";
    Array.from(cont.getElementsByClassName("ref-block")).forEach(block => {
        const ref = parseNumber(block.querySelector(".refA").value);
        const raw = (block.querySelector(".valsA").value || "").trim();
        if(raw === "") return;
        // V19 : Utilisation de parseNumber pour les valeurs
        const vals = raw.split(/\s+/).map(parseNumber).filter(v => v !== null); 
        if(ref === null && vals.length === 0) return;
        
        let html = `<h4>Analyse pour ${ref === null ? 'n/a': ref.toFixed(2)}€</h4><table class="result-table"><tr><th>Valeur</th><th>Différence (€)</th><th>%</th></tr>`;
        vals.forEach(v => {
            const diff = ref === null ? v : v - ref;
            const pct = (ref === null || ref === 0) ? null : (diff / ref * 100);
            const col = diff > 0 ? "#2ecc71" : diff < 0 ? "#e74c3c" : "#95a5a6";
            html += `<tr style="background:${col}; color:#fff;"><td>${v.toFixed(2)}</td><td>${diff>=0?'+':''}${diff.toFixed(2)}</td><td>${pct===null? '—' : pct.toFixed(2)+'%'}</td></tr>`;
        });
        html += "</table>";
        resDiv.innerHTML += html;
    });
    historique.A = resDiv.innerHTML;
}

function addDDate(defaultDate = null) {
    if (!defaultDate) {
        const today = new Date();
        const yyyy = today.getFullYear();
        const mm = String(today.getMonth() + 1).padStart(2, '0');
        const dd = String(today.getDate()).padStart(2, '0');
        defaultDate = `${yyyy}-${mm}-${dd}`;
    }
    const newDateId = Date.now() + Math.random(); 
    dDates.push({ id: newDateId, date: defaultDate });
    // Ajouter une valeur vide pour chaque nouvelle série
    dSeries.forEach(s => { s.values[newDateId] = ''; });
    
    updateDTable();
    
    // V19 : Focus automatique sur la nouvelle date
    // On doit le faire APRES le re-rendu de la table
    const newDateInput = document.querySelector(`#dateHeaderRow th:last-child input[type="date"]`);
    if(newDateInput) newDateInput.focus();
}

function removeDDate(dateId) {
    dDates = dDates.filter(d => d.id !== dateId);
    dSeries.forEach(s => { delete s.values[dateId]; });
    updateDTable();
}

function updateDDateValue(dateId, newDateString) {
    const dateObj = dDates.find(d => d.id === dateId);
    if (dateObj) {
        dateObj.date = newDateString;
    }
    updateDTable(false); 
}

function sortDDates() {
    dDates.sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
    updateDTable();
}

function addDSeries() {
    const count = dSeries.length + 1;
    const newSeriesId = Date.now() + Math.random();
    
    const newSeries = { 
        id: newSeriesId, 
        label: `Série ${count}`, 
        color: D_COLORS[dSeries.length % D_COLORS.length], 
        values: {},
        visible: true
    };

    dDates.forEach(d => {
        newSeries.values[d.id] = '';
    });

    dSeries.push(newSeries);
    updateDTable();
}

function removeDSeries(seriesId) {
    dSeries = dSeries.filter(s => s.id !== seriesId);
    updateDTable();
}

function toggleDSeriesVisibility(seriesId, eyeButton) {
    const series = dSeries.find(s => s.id === seriesId);
    if (series) {
        series.visible = !series.visible;
        // V19 : Utilisation d'icônes plus claires pour la visibilité
        eyeButton.textContent = series.visible ? "👁️" : "✖️"; 
        eyeButton.title = series.visible ? "Cacher cette série" : "Afficher cette série";
    }
    calculerD();
}

function updateDSeriesProp(seriesId, prop, value) {
    const series = dSeries.find(s => s.id === seriesId);
    if (series) {
        series[prop] = value;
    }
    if(prop === 'label' || prop === 'color') {
        calculerD();
    }
}

function updateDValue(seriesId, dateId, value) {
    const series = dSeries.find(s => s.id === seriesId);
    const parsedValue = parseNumber(value);
    if (series) {
        series.values[dateId] = parsedValue;
    }
    calculerD();
}

function updateDTable(recreate = true) {
    const dateHeaderRow = document.getElementById("dateHeaderRow");
    const seriesBody = document.getElementById("seriesBody");

    if(recreate) {
        dDates.sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
    }

    // 1. Reconstruction des en-têtes de dates (colonnes)
    dateHeaderRow.innerHTML = '<th style="min-width: 150px;">Série / Date</th>'; 

    dDates.forEach(d => {
        const th = document.createElement('th');
        th.innerHTML = `
            <div class="date-header-actions">
                <input type="date" class="date-input-d" value="${d.date}" 
                       onchange="updateDDateValue(${d.id}, this.value)">
                <button type="button" onclick="removeDDate(${d.id})">🗑️</button>
            </div>
        `;
        dateHeaderRow.appendChild(th);
    });
    
    // 2. Reconstruction des corps de séries (lignes)
    seriesBody.innerHTML = '';
    dSeries.forEach(s => {
        const tr = document.createElement('tr');
        tr.dataset.seriesId = s.id;

        // Cellule de la série (première colonne)
        const thSeries = document.createElement('th');
        thSeries.innerHTML = `
            <div class="series-header-d">
                <input type="text" class="series-name-input" value="${s.label}" title="Nom de la série"
                       oninput="updateDSeriesProp(${s.id}, 'label', this.value)">
                <input type="color" value="${s.color}" title="Couleur"
                       onchange="updateDSeriesProp(${s.id}, 'color', this.value)">
                <span class="eyeD" style="cursor:pointer;" data-visible="${s.visible}" 
                      onclick="toggleDSeriesVisibility(${s.id}, this)"
                      title="${s.visible ? "Cacher cette série" : "Afficher cette série"}">
                    ${s.visible ? "👁️" : "✖️"}
                </span>
                <button type="button" onclick="removeDSeries(${s.id})">🗑️</button>
            </div>
        `;
        thSeries.style.textAlign = 'left';
        tr.appendChild(thSeries);

        // Cellules de valeurs (colonnes de dates)
        dDates.forEach(d => {
            const rawVal = s.values[d.id];
            const currentValue = rawVal !== null && rawVal !== undefined ? rawVal : '';
            
            const td = document.createElement('td');
            td.innerHTML = `<input type="number" class="value-input-d" placeholder="0" value="${currentValue}"
                                   oninput="updateDValue(${s.id}, ${d.id}, this.value)">`;
            tr.appendChild(td);
        });
        
        seriesBody.appendChild(tr);
    });

    calculerD();
}

function calculerD(){

                    beginAtZero: true,

                    ticks: { color: axisColor }
                }, 
                x: {
                     grid: { color: gridColor },
                     ticks: { color: axisColor }
                }
            },
            plugins: { 
                legend: { display: true, labels: { color: axisColor } },
                tooltip: { mode: 'index', intersect: false }

    };

function mettreAJourTableauF(){
    const tbody = document.querySelector("#tableauF tbody");
    tbody.innerHTML = "";
    const total = historiqueF.reduce((s,c) => s + (parseFloat(c.val) || 0), 0);
    
    historiqueF.forEach(cat => {
        const pourcent = total > 0 ? (cat.val / total * 100).toFixed(1) : 0;
        const tr = document.createElement("tr");
        tr.style.backgroundColor = cat.color;
        
        const rgb = hexToRgbArr(cat.color);
        const luminance = (0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]) / 255;
        tr.style.color = luminance > 0.5 ? '#000' : '#fff'; 

        let sousHTML = "";
        if(cat.sub && cat.sub.length > 0){
            const isDarkMode = document.body.classList.contains('dark-mode');
            const subTextColor = isDarkMode ? '#ecf0f1' : '#333';
            const subBgColor = isDarkMode ? '#4a627a' : '#fff';

            sousHTML = `<div style="text-align:left; padding:4px; color:${subTextColor}; background:${subBgColor}; border-radius:4px; margin:2px; font-size:12px;">` + cat.sub.map(sc => {
                const pct = total > 0 ? ((sc.val / total) * 100).toFixed(1) : 0;
                return `<div>${sc.label}: ${sc.val.toFixed(2)} (${pct}%)</div>`;
            }).join("") + '</div>';
        } 
        tr.innerHTML = `<td>${cat.label}</td><td>${cat.val.toFixed(2)}</td><td>${pourcent}%</td><td>${sousHTML}</td>`;
        tbody.appendChild(tr);
    });
    calculerF(); 
}

function fusionnerFCategories(){
    const cont = document.getElementById("multiFContainer");
    const selectedBlocks = Array.from(cont.getElementsByClassName("ref-block")).filter(d => d.dataset.selectionnee === "true");
    
    if(selectedBlocks.length < 2){ alert("Sélectionnez au moins 2 catégories à fusionner !"); return; }

    let nomNouvelle = prompt("Entrez le nom de la nouvelle catégorie fusionnée :", "Catégorie Fusionnée");
    if(!nomNouvelle) nomNouvelle = "Catégorie Fusionnée";

    let valTotale = 0;
    const rgbs = [];
    const sous = [];
    const elementsToRemove = [];

    selectedBlocks.forEach(block => {
        // Récupère l'objet historique correspondant à ce block
        const h = historiqueF.find(c => c.element === block);
        if(h){
            valTotale += h.val;
            rgbs.push(hexToRgbArr(h.color));
            
            // Ajoute la catégorie elle-même et ses sous-catégories à la liste des sous-catégories de la fusion
            sous.push({ label: h.label, val: h.val, color: h.color, isParent: true });
            if(h.sub && h.sub.length) sous.push(...h.sub);
            
            // Marque l'élément pour suppression du DOM et de l'historiqueF
            elementsToRemove.push(h);
        }
    });

    // Calcul de la couleur moyenne
    const moy = rgbs[0].map((_,i) => Math.round(rgbs.reduce((s,c) => s + c[i], 0) / rgbs.length));
    const couleurFusion = rgbArrToHex(moy);

    // Supprime les blocs du DOM et de l'historiqueF
    elementsToRemove.forEach(h => {
        h.element.remove();
        const idx = historiqueF.findIndex(c => c === h);
        if(idx !== -1) historiqueF.splice(idx,1);
    });

    // On recrée un bloc pour la catégorie fusionnée
    addFBlock({ 
        label: nomNouvelle, 
        val: valTotale, 
        color: couleurFusion, 
        sub: sous 
    });

    mettreAJourTableauF();
    calculerF();
}

    addGLigneToBlock(blockId, "Ligne 1", '', 'HT', globalTva);

function addGLigneToBlock(blockId, labelDefault = "", valDefault = "", typeDefault = 'HT', tvaDefault = 21) {
    const block = document.getElementById(blockId);
    if (!block) return;
    const container = block.querySelector('.ligneGContainer');
    const row = document.createElement('div');
    row.className = 'ligne-tva';
    row.style.display = 'flex';
    row.style.gap = '6px';
    row.style.alignItems = 'center';

    row.innerHTML = `

          <option value="HT"${typeDefault==='HT' ? ' selected':''}>HT</option>
          <option value="TTC"${typeDefault==='TTC' ? ' selected':''}>TTC</option>
        </select>
        <input type="number" class="tvaIndividuelleG" placeholder="TVA (%)" value="${tvaDefault}" style="width:70px;">
        <button type="button" onclick="removeNode(this)" title="Supprimer la ligne">✖️</button>

    // écouteurs pour recalculer automatiquement

function propagateGlobalTvaToBlock(blockId) {
    const block = document.getElementById(blockId);
    if (!block) return;
    const global = parseFloat(block.querySelector('.global-tva-input').value) || 0;
    const lines = block.querySelectorAll('.ligne-tva');
    lines.forEach(line => {
        // remplacer la valeur TVA individuelle par la globale
        const tvaInput = line.querySelector('.tvaIndividuelleG');
        if (tvaInput) {
            tvaInput.value = global;
            updateLineCalc(line, blockId);
        }
    });
}

function updateLineCalc(lineElem, blockId) {
    if (!lineElem) return;
    const montant = parseFloat(lineElem.querySelector('.valeurG').value) || 0;
    const type = lineElem.querySelector('.typeG').value;
    const tva = parseFloat(lineElem.querySelector('.tvaIndividuelleG').value);
    const resSpan = lineElem.querySelector('.resultatG');
    if (isNaN(tva)) { if (resSpan) resSpan.textContent = ''; return; }

    let ht, ttc;
    if (type === 'HT') {
        ht = montant;
        ttc = ht * (1 + tva / 100);
    } else {
        ttc = montant;
        ht = ttc / (1 + tva / 100);
    }

    if (resSpan) {
        resSpan.textContent = `HT ${ht.toFixed(2)} € — TTC ${ttc.toFixed(2)} €`;
    }
}

/* calculerG : utilise maintenant blocs, lignes et leurs inputs */
function calculerG(){
    const cont = document.getElementById("multiGContainer");
    const resDiv = document.getElementById("resultG"); resDiv.innerHTML = "";

    const blocks = Array.from(cont.getElementsByClassName('g-block'));
    if (blocks.length === 0) {
        resDiv.innerHTML = "<p>Aucun bloc TVA défini.</p>";
        historique.G = resDiv.innerHTML;
        return;
    }

    blocks.forEach((block, idx) => {
        const blockName = block.querySelector('.bloc-nom-input')?.value || `Bloc ${idx+1}`;
        const globalTva = parseFloat(block.querySelector('.global-tva-input')?.value) || 0;
        const lines = Array.from(block.querySelectorAll('.ligne-tva'));

        // préparer tableau html
        let html = `<h4>${blockName} (TVA globale ${globalTva}%)</h4>`;
        html += `<table class="result-table"><tr>
            <th>Libellé</th><th>Montant donné</th><th>Taux TVA</th><th>Montant TVA</th><th>Montant HT</th><th>Montant TTC</th>
        </tr>`;

        let totalTVA = 0, totalHT = 0, totalTTC = 0;
        lines.forEach(l => {
            const lib = l.querySelector('.libelleG')?.value || '';
            const raw = parseFloat(l.querySelector('.valeurG')?.value);
            if (isNaN(raw)) return; // ignorer lignes vides
            const type = l.querySelector('.typeG')?.value || 'HT';
            const tva = parseFloat(l.querySelector('.tvaIndividuelleG')?.value);
            const applicableTva = isNaN(tva) ? globalTva : tva;

            const factor = 1 + applicableTva/100;
            let montantHT, montantTTC, tvaVal;
            if (type === 'HT') {
                montantHT = raw;
                montantTTC = montantHT * factor;
                tvaVal = montantTTC - montantHT;
            } else {
                montantTTC = raw;
                montantHT = montantTTC / factor;
                tvaVal = montantTTC - montantHT;
            }

            totalTVA += tvaVal;
            totalHT += montantHT;
            totalTTC += montantTTC;

            html += `<tr>
                <td style="text-align:left;">${lib}</td>
                <td>${raw.toFixed(2)} € (${type})</td>
                <td>${applicableTva}%</td>
                <td>${tvaVal.toFixed(2)} €</td>
                <td>${montantHT.toFixed(2)} €</td>
                <td>${montantTTC.toFixed(2)} €</td>
            </tr>`;
        });

        html += `<tr style="font-weight:bold; background:#ecf0f1; color:#333;">
            <td style="text-align:left;">Totaux</td>
            <td>—</td><td>—</td>
            <td>${totalTVA.toFixed(2)} €</td>
            <td>${totalHT.toFixed(2)} €</td>
            <td>${totalTTC.toFixed(2)} €</td>
        </tr>`;

        html += `</table>`;
        resDiv.innerHTML += html;
    });

    historique.G = resDiv.innerHTML;
}

async function genererRapport(){
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF('p', 'mm', 'a4'); 
    let y = 10;
    const pageWidth = doc.internal.pageSize.getWidth();
    const margin = 15; 
    const contentWidth = pageWidth - 2 * margin;

    doc.setFontSize(22);
    doc.text("Rapport Global Calculateur", pageWidth / 2, y, null, null, "center");
    y += 10;
    doc.setFontSize(10);
    doc.text(`Généré le ${new Date().toLocaleDateString('fr-FR')}`, pageWidth / 2, y, null, null, "center");
    y += 10;
    doc.setLineWidth(0.5);
    doc.line(margin, y, pageWidth - margin, y); 
    y += 10;

    const modes = [
        { id: 'A', title: 'Mode A : Comparer à des valeurs' },
        { id: 'B', title: 'Mode B : Gains et pertes' },
        { id: 'C', title: 'Mode C : Pourcentage par bloc' },
        { id: 'D', title: 'Mode D : Évolution temporelle', isChart: true, chartData: chartDataD, ratio: 2/1 }, 
        { id: 'E', title: 'Mode E : Graphique linéaire multi-séries', isChart: true, chartData: chartDataE, ratio: 2/1 },
        { id: 'F', title: 'Mode F : Graphique circulaire', isChart: true, chartData: chartDataF, ratio: 1/1 }
        ,
        { id: 'G', title: 'Mode G : Calculateur TVA (par ligne nominative)' }
    ];

    for (const mode of modes) {
        if (document.getElementById("include" + mode.id).checked && historique[mode.id]) {
            doc.setFontSize(16);
            if (y > 270) { doc.addPage(); y = 20; }
            doc.text(mode.title, margin, y);
            y += 8;
            doc.setLineWidth(0.2);
            doc.line(margin, y, pageWidth - margin, y);
            y += 4;

            if (mode.isChart && mode.chartData) {
                
                const isPieChart = mode.id === 'F';
                
                const CHART_WIDTH = isPieChart ? 600 : 800;
                const CHART_HEIGHT = isPieChart ? CHART_WIDTH : (CHART_WIDTH / mode.ratio); 
                const EXPORT_SCALE = isPieChart ? 6 : 4; 

                const canvasId = `tempCanvas${mode.id}`;
                let chartCanvas = document.getElementById(canvasId);

                if(!chartCanvas){
                    chartCanvas = document.createElement('canvas');
                    chartCanvas.id = canvasId;
                    chartCanvas.width = CHART_WIDTH; 
                    chartCanvas.height = CHART_HEIGHT; 
                    chartCanvas.style.display = 'none';
                    document.body.appendChild(chartCanvas);
                } else {
                    chartCanvas.width = CHART_WIDTH; chartCanvas.height = CHART_HEIGHT;
                    chartCanvas.getContext('2d').clearRect(0, 0, chartCanvas.width, chartCanvas.height);
                }

                const chartConfig = JSON.parse(JSON.stringify(mode.chartData));
                
                // Créer le graphique temporaire
                const plugins = isPieChart ? [ChartDataLabels] : [];
                let tempChart = new Chart(chartCanvas.getContext('2d'), {
                    ...chartConfig,
                    plugins: plugins
                });
                
                // Attendre le rendu du graphique (300ms)
                await new Promise(resolve => setTimeout(resolve, 300));
                
                const finalCanvas = await html2canvas(chartCanvas, { 
                    scale: EXPORT_SCALE, 
                    logging: false, 
                    useCORS: true,
                    backgroundColor: 'rgba(255,255,255,0)' 
                });
                const finalImgData = finalCanvas.toDataURL('image/png');
                
                const finalImgProps = doc.getImageProperties(finalImgData);
                const pdfHeight = (finalImgProps.height * contentWidth) / finalImgProps.width;

                if (y + pdfHeight + 10 > 280) { doc.addPage(); y = 20; }
                doc.addImage(finalImgData, 'PNG', margin, y, contentWidth, pdfHeight); 
                y += pdfHeight + 5; 
                
                tempChart.destroy();
                chartCanvas.remove(); 
                
                if (isPieChart) {
                    // Ajout du tableau récapitulatif du Mode F 
                    const tempDivF = document.createElement('div');
                    tempDivF.innerHTML = `<h4>Tableau Récapitulatif des catégories</h4>${document.getElementById("tableauF").outerHTML}`;
                    
                    const wrapperF = document.createElement('div');
                    wrapperF.appendChild(tempDivF);
                    wrapperF.style.display = 'inline-block';
                    wrapperF.style.width = contentWidth + 'mm'; 
                    document.body.appendChild(wrapperF);

                    const canvasF = await html2canvas(wrapperF, { scale: 3, logging: false });
                    const imgDataF = canvasF.toDataURL('image/png');

                    const imgPropsF = doc.getImageProperties(imgDataF);
                    const pdfHeightF = (imgPropsF.height * contentWidth) / imgPropsF.width;

                    if (y + pdfHeightF + 10 > 280) { doc.addPage(); y = 20; }
                    doc.addImage(imgDataF, 'PNG', margin, y, contentWidth, pdfHeightF);
                    y += pdfHeightF + 5;
                    
                    wrapperF.remove();
                }

            } else {
                // Gestion des tableaux HTML (A, B, C, G)
                const tempDiv = document.createElement('div');
                tempDiv.innerHTML = historique[mode.id];
                
                const wrapper = document.createElement('div');
                wrapper.appendChild(tempDiv);
                wrapper.style.display = 'inline-block';
                wrapper.style.width = contentWidth + 'mm'; 
                wrapper.style.paddingLeft = '5px'; 
                document.body.appendChild(wrapper);

                const canvas = await html2canvas(wrapper, { scale: 3, logging: false }); 
                const imgData = canvas.toDataURL('image/png');

                const imgProps = doc.getImageProperties(imgData);
                const pdfHeight = (imgProps.height * contentWidth) / imgProps.width;

                if (y + pdfHeight + 10 > 280) { doc.addPage(); y = 20; }
                doc.addImage(imgData, 'PNG', margin, y, contentWidth, pdfHeight);
                y += pdfHeight + 5;
                
                wrapper.remove();
            }

            y += 5; 
        }
    }

    doc.save('Rapport_Calculateur_Global_V24.pdf');
}
