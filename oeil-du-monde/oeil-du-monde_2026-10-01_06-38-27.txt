function sharpen(ctx, w, h, a) {
  const src = ctx.getImageData(0, 0, w, h);
  const dst = ctx.createImageData(w, h);
  const s = src.data, d = dst.data, c = 1 + 4 * a;
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      const i = (y * w + x) * 4;
      if (x === 0 || y === 0 || x === w - 1 || y === h - 1) {
        d[i] = s[i]; d[i+1] = s[i+1]; d[i+2] = s[i+2]; d[i+3] = s[i+3]; continue;
      }
      const u = i - w * 4, b = i + w * 4;
      for (let k = 0; k < 3; k++) {

      d[i+3] = s[i+3];

}

function liftShadows(ctx, w, h, a) {
  const g = 1 + a * 1.4, lut = new Uint8ClampedArray(256);
  for (let i = 0; i < 256; i++) lut[i] = 255 * Math.pow(i / 255, 1 / g);
  const im = ctx.getImageData(0, 0, w, h), d = im.data;
  for (let i = 0; i < d.length; i += 4) { d[i] = lut[d[i]]; d[i+1] = lut[d[i+1]]; d[i+2] = lut[d[i+2]]; }
  ctx.putImageData(im, 0, 0);
}

const ProcTile = L.TileLayer.extend({
  initialize(url, opt) {
    L.TileLayer.prototype.initialize.call(this, url, opt);
    this.on('tileunload', e => { e.tile._dead = true; });
  },
  createTile(coords, done) {
    const size = this.getTileSize();
    const canvas = document.createElement('canvas');
    canvas.width = size.x; canvas.height = size.y;
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => this._process(img, canvas, size).then(() => done(null, canvas), () => done(null, canvas));
    img.onerror = e => done(e, canvas);
    let url = this.getTileUrl(coords);
    if (this._bust) url += (url.includes('?') ? '&' : '?') + '_=' + this._bust;
    img.src = url;
    return canvas;
  },
  async _process(img, canvas, size) {
    const ctx = canvas.getContext('2d', { willReadFrequently:true });
    ctx.drawImage(img, 0, 0, size.x, size.y);
    if (!this.options.proc) return;
    let w = size.x, h = size.y;
    try {
      if (aiOn && upscaler && !canvas._dead) {
        const out = await enqueue(() => canvas._dead ? null : upscaler.upscale(canvas, { patchSize:64, padding:4 }));
        if (out) {
          const im = new Image(); im.src = out; await im.decode();
          canvas.width = w = im.width; canvas.height = h = im.height;
          ctx.drawImage(im, 0, 0);
        }
      }
      if (shadowAmount > 0) liftShadows(ctx, w, h, shadowAmount);
      if (sharpAmount > 0) sharpen(ctx, w, h, sharpAmount);
    } catch (e) { /* canvas protégé par le serveur ou IA indisponible : l'image reste telle quelle */ }
  },
  refresh() { this._bust = Date.now(); this.redraw(); }
});

function liftShadows(ctx, w, h, a) {
  const g = 1 + a * 1.4, lut = new Uint8ClampedArray(256);
  for (let i = 0; i < 256; i++) lut[i] = 255 * Math.pow(i / 255, 1 / g);
  const im = ctx.getImageData(0, 0, w, h), d = im.data;
  for (let i = 0; i < d.length; i += 4) { d[i] = lut[d[i]]; d[i+1] = lut[d[i+1]]; d[i+2] = lut[d[i+2]]; }

/* Flou par boîte séparable sur un tableau de luminance */
function boxBlur(src, w, h, r) {
  const tmp = new Float32Array(w * h), out = new Float32Array(w * h), n = 2 * r + 1;
  for (let y = 0; y < h; y++) {
    let acc = 0; const o = y * w;
    for (let k = -r; k <= r; k++) acc += src[o + Math.min(w - 1, Math.max(0, k))];
    for (let x = 0; x < w; x++) {
      tmp[o + x] = acc / n;
      acc += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)];
    }
  }
  for (let x = 0; x < w; x++) {
    let acc = 0;
    for (let k = -r; k <= r; k++) acc += tmp[Math.min(h - 1, Math.max(0, k)) * w + x];
    for (let y = 0; y < h; y++) {
      out[y * w + x] = acc / n;
      acc += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x];
    }
  }
  return out;
}

   + un peu de contraste local (échelle large) pour faire ressortir toits, routes, haies.
   Le seuil évite d'amplifier le bruit de compression JPEG. */
function sharpen(ctx, w, h, a) {
  const im = ctx.getImageData(0, 0, w, h), d = im.data, N = w * h;
  const L = new Float32Array(N);
  for (let i = 0, p = 0; p < N; i += 4, p++) L[p] = 0.299 * d[i] + 0.587 * d[i+1] + 0.114 * d[i+2];
  const sc = Math.max(1, Math.round(w / 256));
  const fine = boxBlur(L, w, h, sc), wide = boxBlur(L, w, h, 4 * sc);
  const T = 1.5;
  for (let i = 0, p = 0; p < N; i += 4, p++) {
    let f = L[p] - fine[p];
    f = Math.abs(f) < T ? 0 : f - Math.sign(f) * T;
    const add = a * (2.2 * f + 0.7 * (L[p] - wide[p]));
    for (let k = 0; k < 3; k++) { const v = d[i+k] + add; d[i+k] = v < 0 ? 0 : v > 255 ? 255 : v; }

}

/* Éclaircit les zones sombres (courbe gamma) : révèle ce qui est noyé dans l'ombre */
function liftShadows(ctx, w, h, a) {
  const g = 1 + a * 1.4, lut = new Uint8ClampedArray(256);
  for (let i = 0; i < 256; i++) lut[i] = 255 * Math.pow(i / 255, 1 / g);
  const im = ctx.getImageData(0, 0, w, h), d = im.data;
  for (let i = 0; i < d.length; i += 4) { d[i] = lut[d[i]]; d[i+1] = lut[d[i+1]]; d[i+2] = lut[d[i+2]]; }
  ctx.putImageData(im, 0, 0);
}

/* Flou par boîte séparable sur un tableau de luminance */
function boxBlur(src, w, h, r) {
  const tmp = new Float32Array(w * h), out = new Float32Array(w * h), n = 2 * r + 1;
  for (let y = 0; y < h; y++) {
    let acc = 0; const o = y * w;
    for (let k = -r; k <= r; k++) acc += src[o + Math.min(w - 1, Math.max(0, k))];
    for (let x = 0; x < w; x++) {
      tmp[o + x] = acc / n;
      acc += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)];
    }
  }
  for (let x = 0; x < w; x++) {
    let acc = 0;
    for (let k = -r; k <= r; k++) acc += tmp[Math.min(h - 1, Math.max(0, k)) * w + x];
    for (let y = 0; y < h; y++) {
      out[y * w + x] = acc / n;
      acc += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x];
    }
  }
  return out;
}

/* Détails : masque flou sur la luminance uniquement (pas de franges colorées)
   + un peu de contraste local (échelle large) pour faire ressortir toits, routes, haies.
   Le seuil évite d'amplifier le bruit de compression JPEG. */
function sharpen(ctx, w, h, a) {
  const im = ctx.getImageData(0, 0, w, h), d = im.data, N = w * h;
  const L = new Float32Array(N);
  for (let i = 0, p = 0; p < N; i += 4, p++) L[p] = 0.299 * d[i] + 0.587 * d[i+1] + 0.114 * d[i+2];
  const sc = Math.max(1, Math.round(w / 256));
  const fine = boxBlur(L, w, h, sc), wide = boxBlur(L, w, h, 4 * sc);
  const T = 1.5;
  for (let i = 0, p = 0; p < N; i += 4, p++) {
    let f = L[p] - fine[p];
    f = Math.abs(f) < T ? 0 : f - Math.sign(f) * T;
    const add = a * (2.2 * f + 0.7 * (L[p] - wide[p]));
    for (let k = 0; k < 3; k++) { const v = d[i+k] + add; d[i+k] = v < 0 ? 0 : v > 255 ? 255 : v; }
  }
  ctx.putImageData(im, 0, 0);
}

async function pump() {
  if (aiBusy) return;
  aiBusy = true;
  while (aiQueue.length) {
    const t = aiQueue.pop();
    try {
      const out = await t.job();

      tmp[o + x] = acc / n;
      acc += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)];
    }
  }
  for (let x = 0; x < w; x++) {
    let acc = 0;
    for (let k = -r; k <= r; k++) acc += tmp[Math.min(h - 1, Math.max(0, k)) * w + x];
    for (let y = 0; y < h; y++) {
      out[y * w + x] = acc / n;
      acc += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x];
    }
  }
  return out;
}

/* Détails : masque flou sur la luminance + contraste local, avec seuil anti-bruit JPEG */
function sharpen(ctx, w, h, a) {
  const im = ctx.getImageData(0, 0, w, h), d = im.data, N = w * h;
  const L = new Float32Array(N);
  for (let i = 0, p = 0; p < N; i += 4, p++) L[p] = 0.299 * d[i] + 0.587 * d[i+1] + 0.114 * d[i+2];
  const sc = Math.max(1, Math.round(w / 256));
  const fine = boxBlur(L, w, h, sc), wide = boxBlur(L, w, h, 4 * sc);
  const T = 1.5;
  for (let i = 0, p = 0; p < N; i += 4, p++) {
    let f = L[p] - fine[p];
    f = Math.abs(f) < T ? 0 : f - Math.sign(f) * T;
    const add = a * (2.2 * f + 0.7 * (L[p] - wide[p]));
    for (let k = 0; k < 3; k++) { const v = d[i+k] + add; d[i+k] = v < 0 ? 0 : v > 255 ? 255 : v; }
  }
  ctx.putImageData(im, 0, 0);
}

/* Une tuile 256×256 → tuile 512×512. On ajoute 8 px de marge (bords répliqués) avant le réseau
   puis on les retire, pour éviter les coutures visibles entre tuiles. */
async function aiUpscale(src) {
  const P = 8, w = src.width, h = src.height, W = w + 2 * P, H = h + 2 * P;
  const pc = document.createElement('canvas'); pc.width = W; pc.height = H;
  const p = pc.getContext('2d', { willReadFrequently:true });
  p.drawImage(src, P, P);
  p.drawImage(src, 0, 0, w, 1, P, 0, w, P);
  p.drawImage(src, 0, h - 1, w, 1, P, P + h, w, P);
  p.drawImage(src, 0, 0, 1, h, 0, P, P, h);
  p.drawImage(src, w - 1, 0, 1, h, P + w, P, P, h);
  p.drawImage(src, 0, 0, 1, 1, 0, 0, P, P);
  p.drawImage(src, w - 1, 0, 1, 1, P + w, 0, P, P);
  p.drawImage(src, 0, h - 1, 1, 1, 0, P + h, P, P);
  p.drawImage(src, w - 1, h - 1, 1, 1, P + w, P + h, P, P);

  const px = p.getImageData(0, 0, W, H).data, n = W * H, f = new Float32Array(3 * n);
  for (let i = 0; i < n; i++) { f[i] = px[4*i] / 255; f[n + i] = px[4*i+1] / 255; f[2*n + i] = px[4*i+2] / 255; }

  const res = await session.run({ [session.inputNames[0]]: new ort.Tensor('float32', f, [1, 3, H, W]) });
  const o = res[session.outputNames[0]], OH = o.dims[2], OW = o.dims[3], od = o.data, m = OW * OH;

  const bc = document.createElement('canvas'); bc.width = OW; bc.height = OH;
  const bx = bc.getContext('2d'), bi = bx.createImageData(OW, OH), b = bi.data;
  for (let i = 0; i < m; i++) {
    const r = od[i], g = od[m + i], bl = od[2*m + i];
    b[4*i]   = r  < 0 ? 0 : r  > 1 ? 255 : r  * 255;
    b[4*i+1] = g  < 0 ? 0 : g  > 1 ? 255 : g  * 255;
    b[4*i+2] = bl < 0 ? 0 : bl > 1 ? 255 : bl * 255;
    b[4*i+3] = 255;
  }
  bx.putImageData(bi, 0, 0);
  if (o.dispose) o.dispose();

  const k = OW / W;
  const rc = document.createElement('canvas'); rc.width = 2 * w; rc.height = 2 * h;
  const rx = rc.getContext('2d'); rx.imageSmoothingQuality = 'high';
  rx.drawImage(bc, P * k, P * k, w * k, h * k, 0, 0, 2 * w, 2 * h);
  return rc;
}

/* File d'attente : une tuile à la fois, pause entre deux tuiles pour ménager la carte graphique.
   Dernier arrivé, premier servi : quand on déplace la carte, les tuiles visibles passent d'abord. */
const aiQueue = [];
let aiBusy = false;
const sleep = ms => new Promise(r => setTimeout(r, ms));
function enqueue(job) { return new Promise((res, rej) => { aiQueue.push({ job, res, rej }); pump(); }); }
function clearQueue() { aiQueue.splice(0).forEach(t => t.res(null)); }
async function pump() {
  if (aiBusy) return;
  aiBusy = true;
  while (aiQueue.length) {
    const t = aiQueue.pop();
    try {
      const out = await t.job();
      t.res(out);
      say('ai', '<b>IA</b> : ' + aiQueue.length + ' tuile(s) en attente');
      if (out && coolMs) await sleep(coolMs);
    } catch (e) { t.rej(e); say('ai', '<b>IA</b> : erreur sur une tuile (' + (e.message || e) + ')'); }
  }
  aiBusy = false;
  say('ai', '<b>IA</b> : terminé');
}

const ProcTile = L.TileLayer.extend({
  initialize(url, opt) {
    L.TileLayer.prototype.initialize.call(this, url, opt);
    this.on('tileunload', e => { e.tile._dead = true; });
  },
  createTile(coords, done) {
    const size = this.getTileSize();
    const canvas = document.createElement('canvas');
    canvas.width = size.x; canvas.height = size.y;
    const clean = this.getTileUrl(coords);
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => this._process(img, canvas, size, clean).then(() => done(null, canvas), () => done(null, canvas));
    img.onerror = e => done(e, canvas);
    img.src = this._bust ? clean + (clean.includes('?') ? '&' : '?') + '_=' + this._bust : clean;
    return canvas;
  },
  async _process(img, canvas, size, key) {
    const ctx = canvas.getContext('2d', { willReadFrequently:true });
    ctx.drawImage(img, 0, 0, size.x, size.y);
    if (!this.options.proc || rawView) return;
    let w = size.x, h = size.y;
    try {
      if (aiOn && session) {
        const ck = loadedModel + '|' + key;
        let big = cacheGet(ck);
        if (!big && !canvas._dead) {
          const snapshot = document.createElement('canvas');
          snapshot.width = size.x; snapshot.height = size.y;
          snapshot.getContext('2d').drawImage(canvas, 0, 0);
          big = await enqueue(async () => {
            if (canvas._dead) return null;
            const c = await aiUpscale(snapshot);
            cacheSet(ck, c);
            return c;
          });
        }
        if (big && !canvas._dead) {
          canvas.width = w = big.width; canvas.height = h = big.height;
          ctx.drawImage(big, 0, 0);
        }
      }
      if (shadowAmount > 0) liftShadows(ctx, w, h, shadowAmount);
      if (sharpAmount > 0) sharpen(ctx, w, h, sharpAmount);
    } catch (e) { /* canvas protégé par le serveur ou IA indisponible : l'image reste telle quelle */ }
  },
  /* Vrai rechargement (nouvelles images) : on vide le cache IA */
  refresh() { this._bust = Date.now(); aiCache.clear(); this.redraw(); },
  /* Simple ré-application des réglages : mêmes images, cache IA conservé */
  restyle() { this.redraw(); }
});

/* Flou par boîte séparable sur un tableau de luminance */
function boxBlur(src, w, h, r) {
  const tmp = new Float32Array(w * h), out = new Float32Array(w * h), n = 2 * r + 1;
  for (let y = 0; y < h; y++) {
    let acc = 0; const o = y * w;
    for (let k = -r; k <= r; k++) acc += src[o + Math.min(w - 1, Math.max(0, k))];
    for (let x = 0; x < w; x++) {
      tmp[o + x] = acc / n;
      acc += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)];
    }
  }
  for (let x = 0; x < w; x++) {
    let acc = 0;
    for (let k = -r; k <= r; k++) acc += tmp[Math.min(h - 1, Math.max(0, k)) * w + x];
    for (let y = 0; y < h; y++) {
      out[y * w + x] = acc / n;
      acc += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x];
    }
  }
  return out;
}

/* Détails : masque flou sur la luminance + contraste local, avec seuil anti-bruit JPEG */
function sharpen(ctx, w, h, a) {
  const im = ctx.getImageData(0, 0, w, h), d = im.data, N = w * h;
  const L = new Float32Array(N);
  for (let i = 0, p = 0; p < N; i += 4, p++) L[p] = 0.299 * d[i] + 0.587 * d[i+1] + 0.114 * d[i+2];
  const sc = Math.max(1, Math.round(w / 256));
  const fine = boxBlur(L, w, h, sc), wide = boxBlur(L, w, h, 4 * sc);
  const T = 1.5;
  for (let i = 0, p = 0; p < N; i += 4, p++) {
    let f = L[p] - fine[p];
    f = Math.abs(f) < T ? 0 : f - Math.sign(f) * T;
    const add = a * (2.2 * f + 0.7 * (L[p] - wide[p]));
    for (let k = 0; k < 3; k++) { const v = d[i+k] + add; d[i+k] = v < 0 ? 0 : v > 255 ? 255 : v; }
  }
  ctx.putImageData(im, 0, 0);
}

/* Une tuile 256×256 → tuile 512×512. On ajoute 8 px de marge (bords répliqués) avant le réseau
   puis on les retire, pour éviter les coutures visibles entre tuiles. */
async function aiUpscale(src) {
  const P = 8, w = src.width, h = src.height, W = w + 2 * P, H = h + 2 * P;
  const pc = document.createElement('canvas'); pc.width = W; pc.height = H;
  const p = pc.getContext('2d', { willReadFrequently:true });
  p.drawImage(src, P, P);
  p.drawImage(src, 0, 0, w, 1, P, 0, w, P);
  p.drawImage(src, 0, h - 1, w, 1, P, P + h, w, P);
  p.drawImage(src, 0, 0, 1, h, 0, P, P, h);
  p.drawImage(src, w - 1, 0, 1, h, P + w, P, P, h);
  p.drawImage(src, 0, 0, 1, 1, 0, 0, P, P);
  p.drawImage(src, w - 1, 0, 1, 1, P + w, 0, P, P);
  p.drawImage(src, 0, h - 1, 1, 1, 0, P + h, P, P);
  p.drawImage(src, w - 1, h - 1, 1, 1, P + w, P + h, P, P);

  const px = p.getImageData(0, 0, W, H).data, n = W * H, f = new Float32Array(3 * n);
  for (let i = 0; i < n; i++) { f[i] = px[4*i] / 255; f[n + i] = px[4*i+1] / 255; f[2*n + i] = px[4*i+2] / 255; }

  const res = await session.run({ [session.inputNames[0]]: new ort.Tensor('float32', f, [1, 3, H, W]) });
  const o = res[session.outputNames[0]], OH = o.dims[2], OW = o.dims[3], od = o.data, m = OW * OH;

  const bc = document.createElement('canvas'); bc.width = OW; bc.height = OH;
  const bx = bc.getContext('2d'), bi = bx.createImageData(OW, OH), b = bi.data;
  for (let i = 0; i < m; i++) {
    const r = od[i], g = od[m + i], bl = od[2*m + i];
    b[4*i]   = r  < 0 ? 0 : r  > 1 ? 255 : r  * 255;
    b[4*i+1] = g  < 0 ? 0 : g  > 1 ? 255 : g  * 255;
    b[4*i+2] = bl < 0 ? 0 : bl > 1 ? 255 : bl * 255;
    b[4*i+3] = 255;
  }
  bx.putImageData(bi, 0, 0);
  if (o.dispose) o.dispose();

  const k = OW / W;
  const rc = document.createElement('canvas'); rc.width = 2 * w; rc.height = 2 * h;
  const rx = rc.getContext('2d'); rx.imageSmoothingQuality = 'high';
  rx.drawImage(bc, P * k, P * k, w * k, h * k, 0, 0, 2 * w, 2 * h);
  return rc;
}

/* File d'attente : une tuile à la fois, pause entre deux tuiles pour ménager la carte graphique.
   Dernier arrivé, premier servi : quand on déplace la carte, les tuiles visibles passent d'abord. */
const aiQueue = [];
let aiBusy = false;
const sleep = ms => new Promise(r => setTimeout(r, ms));
function enqueue(job) { return new Promise((res, rej) => { aiQueue.push({ job, res, rej }); pump(); }); }
function clearQueue() { aiQueue.splice(0).forEach(t => t.res(null)); }
async function pump() {
  if (aiBusy) return;
  aiBusy = true;
  while (aiQueue.length) {
    const t = aiQueue.pop();
    try {
      const out = await t.job();
      t.res(out);
      say('ai', '<b>IA</b> : ' + aiQueue.length + ' tuile(s) en attente');
      if (out && coolMs) await sleep(coolMs);
    } catch (e) { t.rej(e); say('ai', '<b>IA</b> : erreur sur une tuile (' + (e.message || e) + ')'); }
  }
  aiBusy = false;
  say('ai', '<b>IA</b> : terminé');
}

const ProcTile = L.TileLayer.extend({
  initialize(url, opt) {
    L.TileLayer.prototype.initialize.call(this, url, opt);
    this.on('tileunload', e => { e.tile._dead = true; });
  },
  createTile(coords, done) {
    const size = this.getTileSize();

    canvas.width = size.x; canvas.height = size.y;
    const clean = this.getTileUrl(coords);
    const img = new Image();

    img.onload = () => this._process(img, canvas, size, clean).then(() => done(null, canvas), () => done(null, canvas));
    img.onerror = e => done(e, canvas);
    img.src = this._bust ? clean + (clean.includes('?') ? '&' : '?') + '_=' + this._bust : clean;
    return canvas;
  },
  async _process(img, canvas, size, key) {

            cacheSet(ck, c);
            return c;
          });

          ctx.drawImage(big, 0, 0);

      if (shadowAmount > 0) liftShadows(ctx, w, h, shadowAmount);
      if (sharpAmount > 0) sharpen(ctx, w, h, sharpAmount);
    } catch (e) { /* canvas protégé par le serveur ou IA indisponible : l'image reste telle quelle */ }
  },
  /* Vrai rechargement (nouvelles images) : on vide le cache IA */
  refresh() { this._bust = Date.now(); aiCache.clear(); this.redraw(); },
  /* Simple ré-application des réglages : mêmes images, cache IA conservé */
  restyle() { this.redraw(); }
});

/* Flou par boîte séparable sur un tableau de luminance */
function boxBlur(src, w, h, r) {
  const tmp = new Float32Array(w * h), out = new Float32Array(w * h), n = 2 * r + 1;
  for (let y = 0; y < h; y++) {
    let acc = 0; const o = y * w;
    for (let k = -r; k <= r; k++) acc += src[o + Math.min(w - 1, Math.max(0, k))];
    for (let x = 0; x < w; x++) {
      tmp[o + x] = acc / n;
      acc += src[o + Math.min(w - 1, x + r + 1)] - src[o + Math.max(0, x - r)];
    }
  }
  for (let x = 0; x < w; x++) {
    let acc = 0;
    for (let k = -r; k <= r; k++) acc += tmp[Math.min(h - 1, Math.max(0, k)) * w + x];
    for (let y = 0; y < h; y++) {
      out[y * w + x] = acc / n;
      acc += tmp[Math.min(h - 1, y + r + 1) * w + x] - tmp[Math.max(0, y - r) * w + x];
    }
  }
  return out;
}

/* Détails : masque flou sur la luminance + contraste local, avec seuil anti-bruit JPEG */
function sharpen(ctx, w, h, a) {
  const im = ctx.getImageData(0, 0, w, h), d = im.data, N = w * h;
  const L = new Float32Array(N);
  for (let i = 0, p = 0; p < N; i += 4, p++) L[p] = 0.299 * d[i] + 0.587 * d[i+1] + 0.114 * d[i+2];
  const sc = Math.max(1, Math.round(w / 256));
  const fine = boxBlur(L, w, h, sc), wide = boxBlur(L, w, h, 4 * sc);
  const T = 1.5;
  for (let i = 0, p = 0; p < N; i += 4, p++) {
    let f = L[p] - fine[p];
    f = Math.abs(f) < T ? 0 : f - Math.sign(f) * T;
    const add = a * (2.2 * f + 0.7 * (L[p] - wide[p]));
    for (let k = 0; k < 3; k++) { const v = d[i+k] + add; d[i+k] = v < 0 ? 0 : v > 255 ? 255 : v; }
  }
  ctx.putImageData(im, 0, 0);
}

/* Une tuile 256×256 → tuile 512×512. On ajoute 8 px de marge (bords répliqués) avant le réseau
   puis on les retire, pour éviter les coutures visibles entre tuiles. */
async function aiUpscale(src) {
  const P = 8, w = src.width, h = src.height, W = w + 2 * P, H = h + 2 * P;
  const pc = document.createElement('canvas'); pc.width = W; pc.height = H;
  const p = pc.getContext('2d', { willReadFrequently:true });
  p.drawImage(src, P, P);
  p.drawImage(src, 0, 0, w, 1, P, 0, w, P);
  p.drawImage(src, 0, h - 1, w, 1, P, P + h, w, P);
  p.drawImage(src, 0, 0, 1, h, 0, P, P, h);
  p.drawImage(src, w - 1, 0, 1, h, P + w, P, P, h);
  p.drawImage(src, 0, 0, 1, 1, 0, 0, P, P);
  p.drawImage(src, w - 1, 0, 1, 1, P + w, 0, P, P);
  p.drawImage(src, 0, h - 1, 1, 1, 0, P + h, P, P);
  p.drawImage(src, w - 1, h - 1, 1, 1, P + w, P + h, P, P);

  const px = p.getImageData(0, 0, W, H).data, n = W * H, f = new Float32Array(3 * n);
  for (let i = 0; i < n; i++) { f[i] = px[4*i] / 255; f[n + i] = px[4*i+1] / 255; f[2*n + i] = px[4*i+2] / 255; }

  const res = await session.run({ [session.inputNames[0]]: new ort.Tensor('float32', f, [1, 3, H, W]) });
  const o = res[session.outputNames[0]], OH = o.dims[2], OW = o.dims[3], od = o.data, m = OW * OH;

  const bc = document.createElement('canvas'); bc.width = OW; bc.height = OH;
  const bx = bc.getContext('2d'), bi = bx.createImageData(OW, OH), b = bi.data;
  for (let i = 0; i < m; i++) {
    const r = od[i], g = od[m + i], bl = od[2*m + i];
    b[4*i]   = r  < 0 ? 0 : r  > 1 ? 255 : r  * 255;
    b[4*i+1] = g  < 0 ? 0 : g  > 1 ? 255 : g  * 255;
    b[4*i+2] = bl < 0 ? 0 : bl > 1 ? 255 : bl * 255;
    b[4*i+3] = 255;
  }
  bx.putImageData(bi, 0, 0);
  if (o.dispose) o.dispose();

  const k = OW / W;
  const rc = document.createElement('canvas'); rc.width = 2 * w; rc.height = 2 * h;
  const rx = rc.getContext('2d'); rx.imageSmoothingQuality = 'high';
  rx.drawImage(bc, P * k, P * k, w * k, h * k, 0, 0, 2 * w, 2 * h);
  return rc;
}

/* File d'attente : une tuile à la fois, pause entre deux tuiles pour ménager la carte graphique.
   Dernier arrivé, premier servi : quand on déplace la carte, les tuiles visibles passent d'abord. */
const aiQueue = [];
let aiBusy = false;
const sleep = ms => new Promise(r => setTimeout(r, ms));
function enqueue(job) { return new Promise((res, rej) => { aiQueue.push({ job, res, rej }); pump(); }); }
function clearQueue() { aiQueue.splice(0).forEach(t => t.res(null)); }
async function pump() {
  if (aiBusy) return;
  aiBusy = true;
  while (aiQueue.length) {
    const t = aiQueue.pop();
    try {
      const out = await t.job();
      t.res(out);
      say('ai', '<b>IA</b> : ' + aiQueue.length + ' tuile(s) en attente');
      if (out && coolMs) await sleep(coolMs);
    } catch (e) { t.rej(e); say('ai', '<b>IA</b> : erreur sur une tuile (' + (e.message || e) + ')'); }
  }
  aiBusy = false;
  say('ai', '<b>IA</b> : terminé');
}

/* Cache : bouger un curseur ne relance pas le réseau de neurones, seul le traitement léger est refait */
const AI_CACHE_MAX = 80;
const aiCache = new Map();
function cacheGet(k) { const v = aiCache.get(k); if (v) { aiCache.delete(k); aiCache.set(k, v); } return v; }
function cacheSet(k, v) { aiCache.set(k, v); if (aiCache.size > AI_CACHE_MAX) aiCache.delete(aiCache.keys().next().value); }

/* =====================================================
   Couche de tuiles avec traitement
   ===================================================== */
const ProcTile = L.TileLayer.extend({
  initialize(url, opt) {
    L.TileLayer.prototype.initialize.call(this, url, opt);
    this.on('tileunload', e => { e.tile._dead = true; });
  },
  createTile(coords, done) {
    const size = this.getTileSize();
    const canvas = document.createElement('canvas');
    canvas.width = size.x; canvas.height = size.y;
    const clean = this.getTileUrl(coords);
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => this._process(img, canvas, size, clean).then(() => done(null, canvas), () => done(null, canvas));
    img.onerror = e => done(e, canvas);
    img.src = this._bust ? clean + (clean.includes('?') ? '&' : '?') + '_=' + this._bust : clean;
    return canvas;
  },
  async _process(img, canvas, size, key) {
    const ctx = canvas.getContext('2d', { willReadFrequently:true });
    ctx.drawImage(img, 0, 0, size.x, size.y);
    if (!this.options.proc || rawView) return;
    let w = size.x, h = size.y;
    try {
      if (aiOn && session) {
        const ck = loadedModel + '|' + key;
        let big = cacheGet(ck);
        if (!big && !canvas._dead) {
          const snapshot = document.createElement('canvas');
          snapshot.width = size.x; snapshot.height = size.y;
          snapshot.getContext('2d').drawImage(canvas, 0, 0);
          big = await enqueue(async () => {
            if (canvas._dead) return null;
            const c = await aiUpscale(snapshot);
            cacheSet(ck, c);
            return c;
          });
        }
        if (big && !canvas._dead) {
          canvas.width = w = big.width; canvas.height = h = big.height;
          ctx.drawImage(big, 0, 0);
        }
      }
      if (shadowAmount > 0) liftShadows(ctx, w, h, shadowAmount);
      if (sharpAmount > 0) sharpen(ctx, w, h, sharpAmount);
    } catch (e) { /* canvas protégé par le serveur ou IA indisponible : l'image reste telle quelle */ }
  },
  /* Vrai rechargement (nouvelles images) : on vide le cache IA */
  refresh() { this._bust = Date.now(); aiCache.clear(); this.redraw(); },
  /* Simple ré-application des réglages : mêmes images, cache IA conservé */
  restyle() { this.redraw(); }
});
const tiles = (url, opt) => new ProcTile(url, opt);
