"""

DEFAULT_SETTINGS = {
    "theme": "light",
    # Boost au maximum par defaut : plus de mode Standard/Poussee, on tape
    # toujours large (voir MAX_RESULTS_PER_SOURCE plus bas).
    "results_per_source": 100,

# Plafond dur, quoi qu'il arrive : au-dela, une colonne devient illisible.
MAX_RESULTS_PER_SOURCE = 100

# ---------------------------------------------------------------------------
# Elargissement thematique via Ollama (IA locale, pas de cle API).
# Si Ollama n'est pas lance, ou si le modele n'est pas installe, la
# recherche continue de fonctionner normalement : elle affiche juste les
# resultats du terme exact, sans elargissement.
# ---------------------------------------------------------------------------
OLLAMA_URL = "http://localhost:11434/api/generate"
# Adapte ce nom au modele que tu as reellement installe (voir `ollama list`
# dans un terminal). "llama3.1" est un choix courant par defaut.
OLLAMA_MODEL = "llama3.1"
OLLAMA_TIMEOUT_SECONDS = 12
# Nombre maximum de termes proches generes par recherche.
MAX_RELATED_TERMS = 4

# ---------------------------------------------------------------------------
# Recherche par scraping direct (yt-dlp pour YouTube, ddgs pour le reste) :
# aucune cle API, aucune limite de requetes mensuelle.
# ---------------------------------------------------------------------------

# Duree de mise en cache d'une recherche (en secondes) : evite de refaire la
# meme requete reseau quand on retape plusieurs fois le meme mot-cle.
CACHE_TTL_SECONDS = 6 * 60 * 60  # 6 heures


def _format_duration(seconds):
    """Convertit une duree en secondes en chaine mm:ss ou hh:mm:ss."""
    if not seconds:
        return ""
    seconds = int(seconds)
    h, rem = divmod(seconds, 3600)
    m, s = divmod(rem, 60)
    if h:
        return f"{h}:{m:02d}:{s:02d}"
    return f"{m}:{s:02d}"


def _load_cache():
    if not os.path.exists(CACHE_PATH):
        return {}
    try:
        with open(CACHE_PATH, "r", encoding="utf-8") as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError):
        return {}


def _save_cache(cache):
    try:
        with open(CACHE_PATH, "w", encoding="utf-8") as f:
            json.dump(cache, f, ensure_ascii=False, indent=2)
    except OSError:
        pass


def _cached(key, fetch_fn):
    """Renvoie le resultat en cache s'il est encore valide, sinon appelle
    fetch_fn(), stocke le resultat et le renvoie. Une erreur reseau ne fait
    jamais planter la page : elle renvoie juste une liste vide."""
    cache = _load_cache()
    entry = cache.get(key)
    now = time.time()
    if entry and (now - entry.get("timestamp", 0)) < CACHE_TTL_SECONDS:
        return entry["data"]
    try:
        data = fetch_fn()
    except Exception:
        data = []
    cache[key] = {"timestamp": now, "data": data}
    _save_cache(cache)
    return data


def _related_terms_via_ollama(query, max_terms=MAX_RELATED_TERMS):
    """Demande a Ollama (IA locale) une petite liste de termes de recherche
    proches du sujet (ex. "tactique militaire" -> "strategie militaire",
    "doctrine militaire", "manoeuvre militaire"...). Sert a completer une
    colonne quand le terme exact ne suffit pas a atteindre le plafond.

    Si Ollama n'est pas lance, ou si le modele n'est pas installe, on
    renvoie simplement une liste vide : la recherche continue de
    fonctionner normalement, juste sans elargissement."""
    prompt = (
        "Tu aides a elargir une recherche web. Pour le sujet suivant, donne "
        f"{max_terms} termes de recherche courts et etroitement lies au meme "
        "sujet (synonymes, sous-themes, vocabulaire technique proche), en "
        "francais. Reponds UNIQUEMENT avec un tableau JSON de chaines de "
        f"caracteres, sans aucun texte autour. Sujet : \"{query}\""
    )
    try:
        resp = requests.post(
            OLLAMA_URL,
            json={
                "model": OLLAMA_MODEL,
                "prompt": prompt,
                "stream": False,
                "format": "json",
            },
            timeout=OLLAMA_TIMEOUT_SECONDS,
        )
        resp.raise_for_status()
        raw_text = resp.json().get("response", "")
        parsed = json.loads(raw_text)
    except Exception:
        return []

    if not isinstance(parsed, list):
        return []

    query_lower = query.strip().lower()
    terms = []
    for item in parsed:
        term = str(item).strip()
        if not term or term.lower() == query_lower:
            continue
        if term not in terms:
            terms.append(term)
        if len(terms) >= max_terms:
            break
    return terms


def fetch_youtube(query, limit):
    """Recherche YouTube via yt-dlp (mode ytsearch), sans cle API ni limite."""
    ydl_opts = {
        "quiet": True,
        "no_warnings": True,
        "extract_flat": True,  # ne recupere que les metadonnees, pas le flux video
        "skip_download": True,
    }
    items = []
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        info = ydl.extract_info(f"ytsearch{limit}:{query}", download=False)
        entries = (info or {}).get("entries") or []
        for v in entries[:limit]:
            video_id = v.get("id", "")
            # On construit l'URL de vignette standard de YouTube plutot que
            # de se fier au champ "thumbnails" (souvent absent ou vide en
            # mode extract_flat).
            thumb_url = f"https://i.ytimg.com/vi/{video_id}/hqdefault.jpg" if video_id else ""
            items.append({
                "id": video_id,
                "title": v.get("title", "Sans titre"),
                "channel": v.get("uploader") or v.get("channel") or "",
                "thumbnail_url": thumb_url,
                "duration": _format_duration(v.get("duration")),
                "link": v.get("url") or f"https://www.youtube.com/watch?v={video_id}",
            })
    return items


# ddgs s'arrete parfois tot sur la premiere page meme si on demande plus de
# resultats : on force une marge (x3, avec un plancher a 20 et un plafond a
# 200 pour eviter une requete trop lourde) pour obtenir des colonnes bien
# remplies malgre ce comportement.
def _ddgs_text(query, limit, **kwargs):
    target = min(max(limit * 3, 20), 200)
    with DDGS() as ddgs:
        results = list(ddgs.text(query, max_results=target, **kwargs))
    return results[:limit] if len(results) > limit else results


def fetch_pdf(query, limit):
    """Recherche de PDF via DuckDuckGo (ddgs), sans cle API ni limite."""
    items = []
    for item in _ddgs_text(f"{query} filetype:pdf", limit):
        link = item.get("href", "")
        # ddgs ne fournit ni la langue ni le nombre de pages d'un PDF :
        # on ne les affiche que si on peut les deviner simplement, sinon
        # on laisse le champ vide plutot que d'afficher une fausse info.
        language = "FR" if (".fr/" in link or "/fr/" in link) else ""
        items.append({
            "id": link,
            "title": item.get("title", "Sans titre"),
            "language": language,
            "pages_count": None,
            "file_url": link,
        })
    return items


def fetch_documentaries(query, limit):
    """Recherche de documentaires via DuckDuckGo (ddgs)."""
    items = []
    for item in _ddgs_text(f"{query} documentaire", limit):
        link = item.get("href", "")
        items.append({
            "id": link,
            "title": item.get("title", "Sans titre"),
            "channel": urlparse(link).netloc.replace("www.", ""),
            "description": item.get("body", ""),
            "link": link,
        })
    return items


def fetch_articles(query, limit):
    """Recherche d'articles via DuckDuckGo (ddgs)."""
    items = []
    for item in _ddgs_text(query, limit):
        link = item.get("href", "")
        items.append({
            "id": link,
            "title": item.get("title", "Sans titre"),
            "source_name": urlparse(link).netloc.replace("www.", ""),
            "link": link,
        })
    return items


FETCHERS = {
    "youtube": fetch_youtube,
    "pdf": fetch_pdf,
    "documentaries": fetch_documentaries,
    "articles": fetch_articles,
}


def _fetch_with_related(key, fetch_fn, query, limit):
    """Remplit une colonne en deux temps :
    1. D'abord tous les resultats du terme exact (jusqu'a "limit").
    2. S'il n'y en a pas assez, on complete en dessous avec des termes
       proches generes par Ollama (elargissement thematique), jusqu'a
       atteindre "limit" ou a court de termes proches.
    Les resultats du terme exact ne portent pas de "related_term" ; ceux
    ajoutes par elargissement si (pour que le frontend puisse les
    distinguer visuellement).
    """
    cache_key_exact = f"{key}:{query.lower()}:{limit}"
    exact_items = _cached(cache_key_exact, lambda: fetch_fn(query, limit))
    combined = list(exact_items)
    seen_ids = {item.get("id") for item in combined}

    remaining = limit - len(combined)
    if remaining <= 0:
        return combined

    related_cache_key = f"related_terms:{query.lower()}:{MAX_RELATED_TERMS}"
    related_terms = _cached(related_cache_key, lambda: _related_terms_via_ollama(query))

    for term in related_terms:
        if remaining <= 0:
            break
        cache_key_rel = f"{key}:{term.lower()}:{remaining}"
        rel_items = _cached(cache_key_rel, lambda fn=fetch_fn, t=term, r=remaining: fn(t, r))
        for item in rel_items:
            item_id = item.get("id")
            if item_id in seen_ids:
                continue
            item = dict(item)
            item["related_term"] = term
            combined.append(item)
            seen_ids.add(item_id)
            remaining -= 1
            if remaining <= 0:
                break

    return combined

def load_settings():
    if not os.path.exists(SETTINGS_PATH):
        return dict(DEFAULT_SETTINGS)
    with open(SETTINGS_PATH, "r", encoding="utf-8") as f:
        return json.load(f)


def save_settings(payload):
    with open(SETTINGS_PATH, "w", encoding="utf-8") as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)


@app.route("/")
def index():
    """Route principale : rend la page unique du tableau de bord."""
    settings = load_settings()
    return render_template_string(HTML_TEMPLATE, settings=settings)


@app.route("/api/results")
def api_results():
    """Renvoie les resultats agreges, filtres selon results_per_source.

    Query params optionnels :
      - q : terme de recherche

    Comportement :
      - Sans terme de recherche : on affiche les donnees de demonstration
        (aucune requete reseau).
      - Avec un terme de recherche : on scrape les 4 colonnes (youtube,
        pdf, documentaires, articles) en direct, avec un cache de 6h par
        mot-cle pour eviter de refaire la meme recherche inutilement.
        Chaque colonne est remplie d'abord avec le terme exact, puis, si
        besoin, completee avec des termes proches (elargissement
        thematique via Ollama en local) jusqu'au plafond.
    """
    settings = load_settings()
    query = request.args.get("q", "").strip()

    # Toujours boosté au max (plus de reglage Standard/Poussée) : on tape
    # directement le plafond dur.
    effective_limit = MAX_RESULTS_PER_SOURCE

    if not query:
        return jsonify({"settings": settings, "results": DATA})

    filtered = {}
    for key, fetch_fn in FETCHERS.items():
        filtered[key] = _fetch_with_related(key, fetch_fn, query, effective_limit)

    return jsonify({"settings": settings, "results": filtered})


@app.route("/api/settings", methods=["GET", "POST"])
def api_settings():
    """GET : renvoie les preferences actuelles. POST : les met a jour et les sauvegarde."""
    if request.method == "GET":
        return jsonify(load_settings())

    incoming = request.get_json(force=True, silent=True) or {}
    current = load_settings()

    if "theme" in incoming and incoming["theme"] in ("light", "dark"):
        current["theme"] = incoming["theme"]
    save_settings(current)
    return jsonify(current)


@app.route("/api/quota")
def api_quota():
    """Plus de quota a suivre desormais (scraping direct, illimite) :
    on renvoie toujours "indisponible" pour que le badge reste cache."""
    return jsonify({"available": False})
