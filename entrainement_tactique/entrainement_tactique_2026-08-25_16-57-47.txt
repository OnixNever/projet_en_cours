/* =========================================================
   MOTEUR AUDIO (Web Audio API — sons synthétisés, 100% local)
   ========================================================= */
let actx = null;
function ensureAudio(){ if(!actx) actx = new (window.AudioContext||window.webkitAudioContext)(); return actx; }

function noiseBuffer(ctx, duration){
  const buf = ctx.createBuffer(1, Math.max(1, Math.floor(ctx.sampleRate*duration)), ctx.sampleRate);
  const d = buf.getChannelData(0);
  for(let i=0;i<d.length;i++) d[i] = (Math.random()*2-1);
  return buf;
}

// panner: -1 (gauche) .. 0 (centre) .. 1 (droite)
function makeOutputChain(ctx, gainVal, pan){
  const gain = ctx.createGain(); gain.gain.value = gainVal;
  const panner = ctx.createStereoPanner ? ctx.createStereoPanner() : null;
  if(panner){ panner.pan.value = pan; gain.connect(panner); panner.connect(ctx.destination); return gain; }
  gain.connect(ctx.destination); return gain;
}

function shotVolume(){ return (+document.getElementById('volShots').value)/100; }
function voiceVolume(){ return (+document.getElementById('volVoice').value)/100; }

/* ---- Impact sec et réaliste : bruit filtré dominant + très courte impulsion basse (pas de "note" de percussion) ----
   Corrige le bug signalé : les anciennes versions "courte portée" / "sniper" sonnaient comme un coup de
   basse/percussion cartoon. La composante grave est désormais du bruit filtré très bref, pas un oscillateur
   accordé, pour éviter tout effet "note de musique". */
function shotCrack(ctx, t0, out, hardness){
  // claquement dominant : bruit passe-bande, attaque quasi instantanée, decay très court
  const src = ctx.createBufferSource();
  src.buffer = noiseBuffer(ctx, 0.1);
  const hp = ctx.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 700;
  const bp = ctx.createBiquadFilter(); bp.type = 'bandpass'; bp.frequency.value = 1900*hardness; bp.Q.value = 1.1;
  const env = ctx.createGain();
  env.gain.setValueAtTime(0.0001, t0);
  env.gain.exponentialRampToValueAtTime(1, t0+0.0015);
  env.gain.exponentialRampToValueAtTime(0.0001, t0+0.075);
  src.connect(hp); hp.connect(bp); bp.connect(env); env.connect(out);
  src.start(t0); src.stop(t0+0.1);

  // impulsion grave très brève (bruit passe-bas, pas un ton) — donne du "corps" sans sonner comme un tambour
  const psrc = ctx.createBufferSource();
  psrc.buffer = noiseBuffer(ctx, 0.04);
  const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 200*hardness;
  const penv = ctx.createGain();
  penv.gain.setValueAtTime(0.0001, t0);
  penv.gain.exponentialRampToValueAtTime(0.45, t0+0.002);
  penv.gain.exponentialRampToValueAtTime(0.0001, t0+0.035);
  psrc.connect(lp); lp.connect(penv); penv.connect(out);
  psrc.start(t0); psrc.stop(t0+0.04);
}

function playSemiAuto(pan){
  const ctx = ensureAudio();
  const out = makeOutputChain(ctx, shotVolume()*0.95, pan);
  shotCrack(ctx, ctx.currentTime, out, 1.0);
}

function playSniper(pan){
  const ctx = ensureAudio(); const t0 = ctx.currentTime;
  const vol = shotVolume();
  const out = makeOutputChain(ctx, vol*0.95, pan);

  // claquement initial, plus dur que les autres tirs
  shotCrack(ctx, t0, out, 1.9);

  // impact grave et long — bruit passe-bas avec fréquence de coupure qui chute, sans oscillateur accordé
  const boomOut = makeOutputChain(ctx, vol*0.7, pan);
  const boomSrc = ctx.createBufferSource(); boomSrc.buffer = noiseBuffer(ctx, 0.5);
  const boomFilter = ctx.createBiquadFilter(); boomFilter.type='lowpass';
  boomFilter.frequency.setValueAtTime(220, t0);
  boomFilter.frequency.exponentialRampToValueAtTime(60, t0+0.4);
  const boomEnv = ctx.createGain();
  boomEnv.gain.setValueAtTime(0.0001, t0);
  boomEnv.gain.exponentialRampToValueAtTime(1, t0+0.012);
  boomEnv.gain.exponentialRampToValueAtTime(0.0001, t0+0.5);
  boomSrc.connect(boomFilter); boomFilter.connect(boomEnv); boomEnv.connect(boomOut);
  boomSrc.start(t0); boomSrc.stop(t0+0.5);

  // sifflement aigu qui traverse l'air
  const crackOut = makeOutputChain(ctx, vol*0.55, pan);
  const crackSrc = ctx.createBufferSource(); crackSrc.buffer = noiseBuffer(ctx, 0.6);
  const crackFilter = ctx.createBiquadFilter(); crackFilter.type='highpass'; crackFilter.frequency.value=2500;
  const crackEnv = ctx.createGain();
  crackEnv.gain.setValueAtTime(0.0001, t0+0.02);
  crackEnv.gain.exponentialRampToValueAtTime(0.45, t0+0.03);
  crackEnv.gain.exponentialRampToValueAtTime(0.0001, t0+0.5);
  crackSrc.connect(crackFilter); crackFilter.connect(crackEnv); crackEnv.connect(crackOut);
  crackSrc.start(t0+0.02); crackSrc.stop(t0+0.6);

  // écho distant — signature reconnaissable du sniper
  [0.16, 0.34].forEach((delay, i)=>{
    const echoOut = makeOutputChain(ctx, vol*(0.28 - i*0.11), pan);
    const echoSrc = ctx.createBufferSource(); echoSrc.buffer = noiseBuffer(ctx, 0.3);
    const echoFilter = ctx.createBiquadFilter(); echoFilter.type='lowpass'; echoFilter.frequency.value=600;
    const echoEnv = ctx.createGain();
    echoEnv.gain.setValueAtTime(0.0001, t0+delay);
    echoEnv.gain.exponentialRampToValueAtTime(1, t0+delay+0.01);
    echoEnv.gain.exponentialRampToValueAtTime(0.0001, t0+delay+0.26);
    echoSrc.connect(echoFilter); echoFilter.connect(echoEnv); echoEnv.connect(echoOut);
    echoSrc.start(t0+delay); echoSrc.stop(t0+delay+0.3);
  });
}

function playRafale(pan){
  const ctx = ensureAudio();
  const out = makeOutputChain(ctx, shotVolume()*0.85, pan);
  const n = 6 + Math.floor(Math.random()*6); // 6 à 11 impacts
  let t = 0;
  for(let i=0;i<n;i++){
    shotCrack(ctx, ctx.currentTime + t, out, 0.85 + Math.random()*0.15);
    t += 0.075 + Math.random()*0.035; // léger jitter pour une cadence naturelle
  }
}

function randomPan(){
  if(!state.flags.spatial) return 0;
  const r = Math.random();
  if(r < 0.34) return -0.85;
  if(r < 0.68) return 0.85;
  return 0; // tir "de face" reste centré
}

function fireShot(){
  const types = Array.from(state.shotsSelected);
  if(types.length===0) return;
  const w = {semi:40, sniper:30, rafale:30};
  w.rafale += fatigueLevel*5; w.semi -= fatigueLevel*2; w.sniper -= fatigueLevel*3;
  if(state.difficulty==='difficile'){ w.rafale += 18; w.semi -= 10; w.sniper -= 8; }
  else if(state.difficulty==='facile'){ w.rafale -= 12; w.semi += 12; }
  const pool = [];
  types.forEach(t=>{ const weight = Math.max(5, w[t]||30); for(let i=0;i<weight;i++) pool.push(t); });
  const pick = pool[Math.floor(Math.random()*pool.length)];
  const pan = randomPan();
  if(pick==='semi') playSemiAuto(pan);
  else if(pick==='sniper') playSniper(pan);
  else playRafale(pan);
}

  droneNode = null;
}

function ambianceVent(ctx, vol){
  const src = ctx.createBufferSource();
  src.buffer = noiseBuffer(ctx, 4); src.loop = true;
  const filter = ctx.createBiquadFilter(); filter.type='lowpass'; filter.frequency.value=500;
  const gain = ctx.createGain(); gain.gain.value = vol*0.35;
  const lfo = ctx.createOscillator(); lfo.frequency.value=0.08;
  const lfoGain = ctx.createGain(); lfoGain.gain.value=250;
  lfo.connect(lfoGain); lfoGain.connect(filter.frequency);
  src.connect(filter); filter.connect(gain); gain.connect(ctx.destination);
  src.start(); lfo.start();
  return { stop(){ try{src.stop(); lfo.stop();}catch(e){} } };
}
function ambianceMer(ctx, vol){
  const src = ctx.createBufferSource();
  src.buffer = noiseBuffer(ctx, 4); src.loop = true;
  const filter = ctx.createBiquadFilter(); filter.type='lowpass'; filter.frequency.value=700;
  const gain = ctx.createGain(); gain.gain.value = vol*0.08;
  const lfo = ctx.createOscillator(); lfo.type='sine'; lfo.frequency.value=0.12;
  const lfoGain = ctx.createGain(); lfoGain.gain.value = vol*0.26;
  lfo.connect(lfoGain); lfoGain.connect(gain.gain);
  src.connect(filter); filter.connect(gain); gain.connect(ctx.destination);
  src.start(); lfo.start();
  return { stop(){ try{src.stop(); lfo.stop();}catch(e){} } };
}
function ambianceCirculation(ctx, vol){
  const src = ctx.createBufferSource();
  src.buffer = noiseBuffer(ctx, 4); src.loop = true;
  const filter = ctx.createBiquadFilter(); filter.type='lowpass'; filter.frequency.value=220;
  const gain = ctx.createGain(); gain.gain.value = vol*0.3;
  src.connect(filter); filter.connect(gain); gain.connect(ctx.destination);
  src.start();
  let stopped = false;
  function scheduleSwoosh(){
    if(stopped) return;
    const delay = 4000+Math.random()*9000;
    setTimeout(()=>{
      if(stopped) return;
      const s = ctx.createBufferSource(); s.buffer = noiseBuffer(ctx, 1.2);
      const f2 = ctx.createBiquadFilter(); f2.type='bandpass'; f2.frequency.value=500; f2.Q.value=0.6;
      const g2 = ctx.createGain();
      g2.gain.setValueAtTime(0.0001, ctx.currentTime);
      g2.gain.exponentialRampToValueAtTime(vol*0.5, ctx.currentTime+0.5);
      g2.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime+1.2);
      s.connect(f2); f2.connect(g2); g2.connect(ctx.destination);
      s.start(); s.stop(ctx.currentTime+1.2);
      scheduleSwoosh();
    }, delay);
  }
  scheduleSwoosh();
  return { stop(){ stopped=true; try{src.stop();}catch(e){} } };
}
function ambianceAnimaux(ctx, vol){
  let stopped = false;
  function chirp(){
    if(stopped) return;
    const delay = 1500+Math.random()*5000;
    setTimeout(()=>{
      if(stopped) return;
      const o = ctx.createOscillator(); o.type='sine';
      const base = 1800+Math.random()*1200;
      o.frequency.setValueAtTime(base, ctx.currentTime);
      o.frequency.exponentialRampToValueAtTime(base*0.7, ctx.currentTime+0.09);
      const g = ctx.createGain();
      g.gain.setValueAtTime(0.0001, ctx.currentTime);
      g.gain.exponentialRampToValueAtTime(vol*0.25, ctx.currentTime+0.01);
      g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime+0.1);
      o.connect(g); g.connect(ctx.destination);
      o.start(); o.stop(ctx.currentTime+0.12);
      chirp();
    }, delay);
  }
  chirp();
  return { stop(){ stopped = true; } };
}

let sessionTimers = [];

function startSession(){
  if(state.modules.colors && state.colorsSelected.size===0){
    alert("Sélectionnez au moins une couleur, ou désactivez le module Cibles/Couleurs.");
    return;
  }
  if(state.modules.shots && state.shotsSelected.size===0){
    alert("Sélectionnez au moins un type de tir, ou désactivez le module Sons de tir.");
    return;
  }
  if(state.modules.ambiance && state.ambianceSelected.size===0){
    alert("Sélectionnez au moins une ambiance, ou désactivez le module Ambiance sonore.");
    return;
  }
  if(state.prepPhysique && state.prepExercisesSelected.size===0){
    alert("Sélectionnez au moins un exercice pour la préparation physique.");
    return;
  }
  ensureAudio();
  if(actx.state==='suspended') actx.resume();

  fatigueLevel = 0; paused = false; running = true;
  blockReasons.clear(); blockLabel = '';
  pauseAccum = 0; sessionStartTime = Date.now();

  chronoTotalMs = state.timeMode==='chrono' ? (parseInt(document.getElementById('chronoMin').value)||5)*60000 : 0;

  document.getElementById('session').classList.add('live');
  document.getElementById('sessWord').textContent = '';
  document.getElementById('sessHint').textContent = 'SESSION EN COURS';
  document.getElementById('sessExercise').classList.toggle('show', state.prepPhysique);
  updateStatusLine();

  clockTicker = setInterval(updateClock, 250);

  if(state.modules.fatigue && !state.prepPhysique){
    fatigueTicker = setInterval(()=>{
      if(paused) return;
      fatigueLevel = Math.min(9, fatigueLevel+1);
      updateStatusLine();
    }, 30000);
  }

  if(state.modules.ambiance){
    Array.from(state.ambianceSelected).forEach(k=> startAmbiance(k));
  }

  if(state.modules.shots && !state.prepPhysique) scheduleNextShot();
  if(state.modules.colors) scheduleNextColor();
  if(state.modules.orders && !state.prepPhysique) scheduleNextOrder();
  if(state.modules.decoys && !state.prepPhysique) scheduleNextDecoy();
  if(state.modules.reload && !state.prepPhysique) scheduleNextReload();
  if(state.prepPhysique) startPrepSeries();

  if('mediaSession' in navigator){
    try{
      navigator.mediaSession.setActionHandler('play', ()=>{ if(paused) togglePause(); });
      navigator.mediaSession.setActionHandler('pause', ()=>{ if(!paused) togglePause(); });
      navigator.mediaSession.playbackState = 'playing';
    }catch(e){}
  }
}

function updateStatusLine(){
  const el = document.getElementById('sessStatus');
  const f = state.modules.fatigue ? `fatigue: +${fatigueLevel*10}%` : 'fatigue: off';
  el.textContent = f + (blockLabel ? '\n'+blockLabel : '');
}

function updateClock(){
  const el = document.getElementById('sessTimer');
  if(state.prepPhysique){
    const remain = Math.max(0, Math.ceil((prepSeriesEndAt - Date.now())/1000));
    const mm = String(Math.floor(remain/60)).padStart(2,'0');
    const ss = String(remain%60).padStart(2,'0');
    document.getElementById('sessExercise').textContent = `${PREP_LABELS[prepCurrentExercise]||''} — ${mm}:${ss}`;
  }
  if(state.timeMode==='continu'){ el.textContent = '∞'; return; }
  const elapsed = (Date.now() - sessionStartTime - pauseAccum);
  const remain = Math.max(0, chronoTotalMs - elapsed);
  const s = Math.ceil(remain/1000);
  const mm = String(Math.floor(s/60)).padStart(2,'0');
  const ss = String(s%60).padStart(2,'0');
  el.textContent = `${mm}:${ss}`;
  if(remain<=0) quitSession();
}

function colorIntervalRange(){
  if(state.prepPhysique) return [+document.getElementById('prepColorMin').value, +document.getElementById('prepColorMax').value];
  return [+document.getElementById('colorMin').value, +document.getElementById('colorMax').value];

function scheduleNextColor(){
  if(!running) return;
  const [min,max] = colorIntervalRange();
  const mult = state.prepPhysique ? 1 : (fatigueMult() * difficultyMult());
  const delay = randRange(min,max) * 1000 * mult;
  schedule(()=>{
    if(!paused && running && blockReasons.size===0){
      const multiFreq = state.modules.multi ? (+document.getElementById('multiFreq').value) : 0;
      if(state.modules.multi && !state.prepPhysique && Math.random()*100 < multiFreq){
        fireMultiTarget(()=> scheduleNextColor());

      } else {
        fireSingleTarget();
      }
    }
    scheduleNextColor();
  }, delay);

function fireSingleTarget(){
  const arr = Array.from(state.colorsSelected);
  if(arr.length===0) return;
  const c = arr[Math.floor(Math.random()*arr.length)];
  announceColor(c);
}
function fireMultiTarget(afterBattement){
  const arr = Array.from(state.colorsSelected);
  if(arr.length===0){ afterBattement(); return; }
  const count = Math.min(arr.length, 2 + Math.floor(Math.random()*2)); // 2 ou 3
  const shuffled = [...arr].sort(()=>Math.random()-0.5).slice(0,count);
  shuffled.forEach((c,i)=>{
    schedule(()=> announceColor(c), i*1300);
  });
  const battMin = +document.getElementById('multiBattMin').value, battMax = +document.getElementById('multiBattMax').value;
  const totalDelay = shuffled.length*1300 + randRange(battMin,battMax)*1000;
  schedule(afterBattement, totalDelay);
}

}

function scheduleNextOrder(){
  if(!running) return;
  const min = +document.getElementById('orderMin').value, max = +document.getElementById('orderMax').value;
  const delay = randRange(min,max) * 1000 * difficultyMult();
  schedule(()=>{
    if(!paused && running){
      const pool = Array.from(state.ordersSelected);
      if(pool.length>0){
        triggerOrder(pool[Math.floor(Math.random()*pool.length)]);
        return; // triggerOrder se charge de reprogrammer après sa propre durée
      }

    scheduleNextOrder();
  }, delay);

function triggerOrder(o){
  const isMove = MOVE_ORDERS.includes(o);
  const isBlocking = BLOCKING_ORDERS.includes(o);
  let duration;
  if(isMove){
    const mn = +document.getElementById('moveMin').value, mx = +document.getElementById('moveMax').value;
    duration = randRange(mn,mx);
  } else {
    const hn = +document.getElementById('holdMin').value, hx = +document.getElementById('holdMax').value;
    duration = state.flags.randomHold ? randRange(hn,hx) : hn;
  }
  speak(ORDER_LABELS[o]);
  flashWord(ORDER_LABELS[o], true);
  document.getElementById('session').style.backgroundColor = '#111';

  if(isBlocking) addBlock('order', ORDER_LABELS[o].toUpperCase());
  if(o==='fige') startDrone();

  schedule(()=>{
    if(o==='fige') stopDrone();
    if(isBlocking) removeBlock('order');
    scheduleNextOrder();
  }, duration*1000);
}
