            <option value="en-US">English 🇺🇸</option>
            <option value="ar-SA">العربية 🇸🇦</option>

<script>
    const mainBtn = document.getElementById('mainBtn');
    const textDisplay = document.getElementById('textDisplay');
    const status = document.getElementById('status');
    const langSelect = document.getElementById('langSelect');

    let recognition;
    let isListening = false;
    let finalTranscript = '';

    // Initialisation
    if ('webkitSpeechRecognition' in window) {
        recognition = new webkitSpeechRecognition();
        recognition.continuous = true;
        recognition.interimResults = true; // LA CLÉ POUR LA FLUIDITÉ

        recognition.onstart = () => {
            isListening = true;
            mainBtn.textContent = "Arrêter la dictée";
            mainBtn.classList.add('active');
            status.textContent = "Je vous écoute...";
        };

        recognition.onresult = (event) => {
            let interimTranscript = '';
            
            for (let i = event.resultIndex; i < event.results.length; ++i) {
                if (event.results[i].isFinal) {
                    insertAtCursor(event.results[i][0].transcript + ' ');
                } else {
                    interimTranscript += event.results[i][0].transcript;
                }
            }
            
            // Affichage temporaire du texte "en cours"
            updateInterimDisplay(interimTranscript);
        };

        recognition.onerror = (event) => {
            console.error("Erreur:", event.error);
            stopDictation();
        };

        recognition.onend = () => {
            if (isListening) recognition.start(); // Auto-restart pour ne jamais couper
        };
    }

    function insertAtCursor(text) {
        // Supprime l'affichage intérimaire avant d'insérer le texte final
        removeInterim();
        
        const sel = window.getSelection();
        if (sel.rangeCount) {
            const range = sel.getRangeAt(0);
            const node = document.createTextNode(text);
            range.insertNode(node);
            range.collapse(false);
            sel.removeAllRanges();
            sel.addRange(range);
        }
        updateStats();
    }

    function updateInterimDisplay(text) {
        removeInterim();
        if (text) {
            const span = document.createElement('span');
            span.id = 'interim-span';
            span.className = 'interim';
            span.textContent = text;
            
            const sel = window.getSelection();
            if (sel.rangeCount) {
                const range = sel.getRangeAt(0);
                range.insertNode(span);
            }
        }
    }

    function removeInterim() {
        const existing = document.getElementById('interim-span');
        if (existing) existing.remove();
    }

    function toggleDictation() {
        if (isListening) {
            stopDictation();
        } else {
            startDictation();
        }
    }

    function startDictation() {
        recognition.lang = langSelect.value;
        recognition.start();
        textDisplay.focus();
    }

    function stopDictation() {
        isListening = false;
        recognition.stop();
        mainBtn.textContent = "Démarrer la dictée";
        mainBtn.classList.remove('active');
        status.textContent = "Arrêté";
        removeInterim();
    }

    function updateStats() {
        const words = textDisplay.innerText.trim().split(/\s+/).length;
        document.getElementById('stats').textContent = `${textDisplay.innerText.trim() ? words : 0} mots`;
    }

    mainBtn.addEventListener('click', toggleDictation);
    
    // Sauvegarde auto de la langue
    langSelect.value = localStorage.getItem('pref_lang') || 'fr-FR';
    langSelect.onchange = () => localStorage.setItem('pref_lang', langSelect.value);

</script>
