<script>
    const mainBtn = document.getElementById('mainBtn');
    const textDisplay = document.getElementById('textDisplay');
    const status = document.getElementById('status');
    const langSelect = document.getElementById('langSelect');
    const clearTextBtn = document.getElementById('clearTextBtn');
    const copyContentBtn = document.getElementById('copyContentBtn');

    let recognition;
    let isListening = false;
    let finalTranscript = '';

    // Initialisation
    if ('webkitSpeechRecognition' in window) {
        recognition = new webkitSpeechRecognition();
        recognition.continuous = true;
        recognition.interimResults = true; // LA CLÉ POUR LA FLUIDITÉ

        recognition.onstart = () => {
            isListening = true;
            mainBtn.textContent = "Arrêter la dictée";
            mainBtn.classList.add('active');
            status.textContent = "Je vous écoute...";
        };

        recognition.onresult = (event) => {
            let interimTranscript = '';
            
            for (let i = event.resultIndex; i < event.results.length; ++i) {
                if (event.results[i].isFinal) {
                    insertAtCursor(event.results[i][0].transcript + ' ');
                } else {
                    interimTranscript += event.results[i][0].transcript;
                }
            }
            
            // Affichage temporaire du texte "en cours"
            updateInterimDisplay(interimTranscript);
        };

        recognition.onerror = (event) => {
            console.error("Erreur:", event.error);
            stopDictation();
        };

        recognition.onend = () => {
            if (isListening) recognition.start(); // Auto-restart pour ne jamais couper
        };
    }

    function insertAtCursor(text) {
        // Supprime l'affichage intérimaire avant d'insérer le texte final
        removeInterim();
        
        const sel = window.getSelection();
        if (sel.rangeCount) {
            const range = sel.getRangeAt(0);
            const node = document.createTextNode(text);
            range.insertNode(node);
            range.collapse(false);
            sel.removeAllRanges();
            sel.addRange(range);
        }
        updateStats();
    }

    function updateInterimDisplay(text) {
        removeInterim();
        if (text) {
            const span = document.createElement('span');
            span.id = 'interim-span';
            span.className = 'interim';
            span.textContent = text;
            
            const sel = window.getSelection();
            if (sel.rangeCount) {
                const range = sel.getRangeAt(0);
                range.insertNode(span);
            }
        }
    }

    function removeInterim() {
        const existing = document.getElementById('interim-span');
        if (existing) existing.remove();
    }

    function toggleDictation() {
        if (isListening) {
            stopDictation();
        } else {
            startDictation();
        }
    }

    function startDictation() {
        recognition.lang = langSelect.value;
        recognition.start();
        textDisplay.focus();
    }

    function stopDictation() {
        isListening = false;
        recognition.stop();
        mainBtn.textContent = "Démarrer la dictée";
        mainBtn.classList.remove('active');
        status.textContent = "Arrêté";
        removeInterim();
    }

    function updateStats() {
        const words = textDisplay.innerText.trim().split(/\s+/).length;
        document.getElementById('stats').textContent = `${textDisplay.innerText.trim() ? words : 0} mots`;
    }

    // Fonction pour vider le texte - SIMPLIFIÉE
    function clearText() {
        // Effacer le contenu directement
        textDisplay.innerHTML = '';
        
        // Mettre à jour les statistiques
        updateStats();
        
        // Donner le focus à la zone de texte
        textDisplay.focus();
        
        // Afficher un message de confirmation
        status.textContent = "Texte effacé";
        clearTextBtn.textContent = "Texte effacé !";
        clearTextBtn.style.background = "#dcfce7";
        clearTextBtn.style.color = "#166534";
        
        setTimeout(() => {
            clearTextBtn.textContent = "Vider le texte";
            clearTextBtn.style.background = "";
            clearTextBtn.style.color = "";
            if (!isListening) status.textContent = "Prêt";
        }, 1500);
    }

    // Fonction pour copier le contenu
    function copyContent() {
        const textToCopy = textDisplay.innerText.trim();
        
        if (!textToCopy) {
            status.textContent = "Aucun texte à copier";
            setTimeout(() => {
                if (!isListening) status.textContent = "Prêt";
            }, 2000);
            return;
        }
        
        // Utiliser l'API Clipboard moderne
        navigator.clipboard.writeText(textToCopy)
            .then(() => {
                status.textContent = "Texte copié dans le presse-papiers !";
                copyContentBtn.textContent = "Copié !";
                copyContentBtn.style.background = "#d1fae5";
                copyContentBtn.style.color = "#065f46";
                
                setTimeout(() => {
                    copyContentBtn.textContent = "Copier le contenu";
                    copyContentBtn.style.background = "";
                    copyContentBtn.style.color = "";
                    if (!isListening) status.textContent = "Prêt";
                }, 2000);
            })
            .catch(err => {
                console.error('Erreur lors de la copie: ', err);
                status.textContent = "Erreur lors de la copie";
                
                // Méthode de secours pour les anciens navigateurs
                const textArea = document.createElement('textarea');
                textArea.value = textToCopy;
                document.body.appendChild(textArea);
                textArea.select();
                document.execCommand('copy');
                document.body.removeChild(textArea);
                
                status.textContent = "Texte copié (méthode de secours)";
                setTimeout(() => {
                    if (!isListening) status.textContent = "Prêt";
                }, 2000);
            });
    }

    // Événements des boutons
    mainBtn.addEventListener('click', toggleDictation);
    clearTextBtn.addEventListener('click', clearText);
    copyContentBtn.addEventListener('click', copyContent);
    
    // Sauvegarde auto de la langue
    langSelect.value = localStorage.getItem('pref_lang') || 'fr-FR';
    langSelect.onchange = () => localStorage.setItem('pref_lang', langSelect.value);

</script>

        function checkMicrophonePermission() {
            // Vérifier si la permission a déjà été accordée (stockée en localStorage)

            if (savedPermission === 'true') {
                state.micPermissionGranted = true;

        function testMicrophoneSilently() {
            if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                // L'API n'est pas disponible
                showPermissionBanner();
                return;
            }
            
            // Tester l'accès au microphone sans demander de permission
            navigator.mediaDevices.getUserMedia({ audio: true })
                .then(stream => {
                    // Permission déjà accordée
                    state.micPermissionGranted = true;
                    localStorage.setItem('micPermissionGranted', 'true');
                    elements.permissionBanner.style.display = 'none';
                    
                    // Arrêter le stream immédiatement
                    stream.getTracks().forEach(track => track.stop());
                    
                    // Initialiser la reconnaissance vocale
                    initSpeechRecognition();
                })
                .catch(error => {
                    if (error.name === 'NotAllowedError' || error.name === 'PermissionDeniedError') {
                        // Permission refusée précédemment
                        showPermissionBanner();
                    } else if (error.name === 'NotFoundError' || error.name === 'DevicesNotFoundError') {
                        // Pas de microphone disponible
                        showNotification("❌ Aucun microphone détecté", "#ef4444", "❌");
                        elements.startBtn.disabled = true;
                    } else {
                        // Autre erreur
                        showPermissionBanner();
                    }
                });
        }

                bar.style.height = `${height}px`;
                bar.classList.toggle('active', index < activeBars);
            });
        }

            // Mettre à jour les paramètres de reconnaissance
            updateRecognitionSettings();

            const level = state.speechPatterns.adaptationLevel || 'high';

            // Mettre à jour les affichages

        function insertPonctuation(char) {
            const isArabic = elements.languageSelect.value.startsWith('ar-');
            
            // Convertir le caractère spécial
            let actualChar = char;
            if (char === '⏎') actualChar = '\n';
            if (char === ' ') actualChar = ' ';
            
            // Ajouter au contenu
            if (isArabic) {
                state.textContent = actualChar + state.textContent;
            } else {
                state.textContent += actualChar;
            }
            
            // Mettre à jour l'affichage
            updateTextDisplay();
            
            // Notifier
            const labels = {
                '.': 'Point ajouté',
                ',': 'Virgule ajoutée',
                ';': 'Point-virgule ajouté',
                ':': 'Deux-points ajoutés',
                '?': 'Point d\'interrogation ajouté',
                '!': 'Point d\'exclamation ajouté',
                '\n': 'Retour à la ligne ajouté',
                ' ': 'Espace ajouté'
            };
            
            if (labels[actualChar]) {

        function initSpeechRecognition() {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            
            if (!SpeechRecognition) {
                showNotification("❌ Reconnaissance vocale non supportée", "#ef4444", "❌");
                elements.startBtn.disabled = true;
                return;
            }
            
            state.recognition = new SpeechRecognition();
            configureRecognition();
            
            state.recognition.onstart = () => {
                state.isListening = true;
                state.speechStartTime = Date.now();
                updateUI();
                startMicMonitoring();
                showNotification("🎤 Dictée en cours...", "#10b981", "🎤");
            };
            
            state.recognition.onresult = (event) => {
                handleSpeechResult(event);
            };
            
            state.recognition.onerror = (event) => {
                handleRecognitionError(event);
            };
            
            state.recognition.onend = () => {
                if (state.isListening) {
                    setTimeout(() => {
                        try {
                            state.recognition.start();
                        } catch (e) {
                            console.error('Redémarrage échoué:', e);
                        }
                    }, 100);
                }
            };

        function configureRecognition() {
            if (!state.recognition) return;
            
            state.recognition.continuous = true;
            state.recognition.interimResults = true;
            state.recognition.maxAlternatives = 3;
            state.recognition.lang = elements.languageSelect.value;
        }

        function updateRecognitionSettings() {
            if (!state.recognition) return;
            
            // En mode auto, ajuster dynamiquement
            if (state.isAutoMode) {
                const speedFactor = Math.max(0.5, Math.min(2, 120 / Math.max(state.speechPatterns.wordsPerMinute, 60)));
                const confidenceFactor = Math.max(0.3, Math.min(0.95, state.speechPatterns.avgConfidence));
                
                state.speechPatterns.pauseDuration = 2000 * speedFactor;
                state.speechPatterns.sensitivity = 0.5 + (confidenceFactor * 0.5);
                
                updateSliderDisplays();
                
                if (confidenceFactor > 0.8 && speedFactor > 0.8) {
                    state.speechPatterns.adaptationLevel = 'high';
                } else if (confidenceFactor > 0.6 && speedFactor > 0.6) {
                    state.speechPatterns.adaptationLevel = 'medium';
                } else {
                    state.speechPatterns.adaptationLevel = 'low';
                }
                updateAdaptationLevel();
            }
        }

        function handleSpeechResult(event) {
            const results = event.results;
            const resultIndex = event.resultIndex;
            
            let finalText = '';
            let maxConfidence = 0;
            
            for (let i = resultIndex; i < results.length; i++) {
                const result = results[i];
                
                if (result.isFinal) {
                    let bestAlternative = result[0];
                    for (let j = 1; j < result.length; j++) {
                        if (result[j].confidence > bestAlternative.confidence) {
                            bestAlternative = result[j];
                        }
                    }
                    
                    const confidence = bestAlternative.confidence;
                    const transcript = bestAlternative.transcript;
                    
                    const confidenceThreshold = state.isAutoMode ? 
                        state.speechPatterns.avgConfidence * 0.9 : 
                        state.speechPatterns.avgConfidence;
                    
                    if (confidence >= confidenceThreshold) {
                        finalText += transcript + ' ';
                        maxConfidence = Math.max(maxConfidence, confidence);
                        
                        updateSpeechMetrics(transcript, confidence);
                    }
                    
                } else {
                    showInterimText(result[0].transcript);
                }
            }
            
            if (finalText.trim()) {
                state.textContent += finalText;
                updateTextDisplay();
                
                if (state.isAutoMode) {
                    learnFromResult(finalText, maxConfidence);
                }
            }
            
            updateMicAnimation(maxConfidence);
            updateStats();
        }

        function updateSpeechMetrics(transcript, confidence) {
            const words = transcript.split(/\s+/).length;
            state.totalWords += words;
            
            const elapsedMinutes = (Date.now() - state.speechStartTime) / 60000;
            state.speechPatterns.wordsPerMinute = elapsedMinutes > 0 ? 
                Math.round(state.totalWords / elapsedMinutes) : 0;
            
            state.confidenceScores.push(confidence);
            if (state.confidenceScores.length > 50) {
                state.confidenceScores.shift();
            }
            
            const avgConfidence = state.confidenceScores.length > 0 ?
                state.confidenceScores.reduce((a, b) => a + b, 0) / state.confidenceScores.length : 0.85;
            
            state.speechPatterns.avgConfidence = avgConfidence;
            
            elements.wpmValue.textContent = state.speechPatterns.wordsPerMinute;
            elements.accuracyValue.textContent = Math.round(avgConfidence * 100) + '%';
        }

        function learnFromResult(text, confidence) {
            const words = text.split(/\s+/).length;
            const speed = words / 3;
            
            if (confidence > 0.8 && speed > 2) {
                state.speechPatterns.sensitivity = Math.min(0.9, state.speechPatterns.sensitivity + 0.05);
            } else if (confidence < 0.6 || speed < 1) {
                state.speechPatterns.sensitivity = Math.max(0.3, state.speechPatterns.sensitivity - 0.05);
            }
            
            updateRecognitionSettings();
        }

        function updateTextDisplay() {
            const isArabic = elements.languageSelect.value.startsWith('ar-');
            
            if (isArabic) {
                elements.textDisplay.classList.add('arabic');
                elements.textDisplay.dir = 'rtl';
            } else {
                elements.textDisplay.classList.remove('arabic');
                elements.textDisplay.dir = 'ltr';
            }
            
            elements.textDisplay.textContent = state.textContent;
            updatePlaceholder();
            elements.textDisplay.scrollTop = elements.textDisplay.scrollHeight;
        }

        function showInterimText(text) {
            const displayText = state.textContent + text;
            elements.textDisplay.textContent = displayText;
            updatePlaceholder();
        }

        function updatePlaceholder() {
            if (state.textContent.trim() || elements.textDisplay.textContent.trim()) {
                elements.textPlaceholder.style.display = 'none';
            } else {
                elements.textPlaceholder.style.display = 'block';
            }
        }

        function updateStats() {
            const text = state.textContent || elements.textDisplay.textContent;
            const words = text.trim() ? text.trim().split(/\s+/).length : 0;
            const chars = text.length;
            
            elements.wordCount.textContent = words;
            elements.charCount.textContent = chars;
        }

        function startListening() {
            if (!state.micPermissionGranted) {
                showPermissionBanner();
                return;
            }
            
            if (!state.recognition) {
                initSpeechRecognition();
            }
            
            // Réinitialiser les statistiques
            state.speechStartTime = Date.now();
            state.totalWords = 0;
            state.confidenceScores = [];
            
            try {
                state.recognition.start();
            } catch (error) {
                showNotification("❌ Erreur de démarrage", "#ef4444", "❌");
                console.error(error);
            }
        }

        function stopListening() {
            if (state.recognition && state.isListening) {
                state.recognition.stop();
            }
            
            state.isListening = false;
            updateUI();
            stopMicMonitoring();
            showNotification("⏹️ Dictée arrêtée", "#64748b", "⏹️");
        }

        function clearText() {
            state.textContent = '';
            updateTextDisplay();
            updateStats();
            showNotification("🗑️ Texte effacé", "#64748b", "🗑️");
        }

        function copyToClipboard() {
            const text = state.textContent || elements.textDisplay.textContent;
            
            if (!text.trim()) {
                showNotification("⚠️ Aucun texte à copier", "#f59e0b", "⚠️");
                return;
            }

            navigator.clipboard.writeText(text)
                .then(() => {
                    showNotification("📋 Texte copié !", "#3b82f6", "📋");
                })
                .catch(err => {
                    const textArea = document.createElement('textarea');
                    textArea.value = text;
                    document.body.appendChild(textArea);
                    textArea.select();
                    document.execCommand('copy');
                    document.body.removeChild(textArea);
                    showNotification("📋 Texte copié !", "#3b82f6", "📋");
                });
        }

        function updateLanguage() {
            if (state.recognition) {
                state.recognition.lang = elements.languageSelect.value;
                if (state.isListening) {
                    state.recognition.stop();
                    setTimeout(() => state.recognition.start(), 100);
                }
            }
            updateTextDisplay();
        }

        function updateUI() {
            if (state.isListening) {
                elements.startBtn.disabled = true;
                elements.stopBtn.disabled = false;
            } else {
                elements.startBtn.disabled = false;
                elements.stopBtn.disabled = true;
            }
        }

        function handleRecognitionError(event) {
            console.error('Erreur reconnaissance:', event.error);
            
            if (event.error === 'no-speech') {
                // Pause normale
            } else if (event.error === 'audio-capture') {
                showNotification("❌ Microphone non disponible", "#ef4444", "❌");
                stopListening();
            } else if (event.error === 'not-allowed') {
                showNotification("🔒 Permission microphone refusée", "#ef4444", "🔒");
                stopListening();
                state.micPermissionGranted = false;
                localStorage.setItem('micPermissionGranted', 'false');
                showPermissionBanner();
            }
        }

        function startMicMonitoring() {
            stopMicMonitoring();
            
            // Animation aléatoire pour simuler l'activité
            state.micInterval = setInterval(() => {
                if (state.isListening) {
                    const level = 0.3 + Math.random() * 0.7;
                    updateMicAnimation(level);
                }
            }, 200);
        }

        function stopMicMonitoring() {
            if (state.micInterval) {
                clearInterval(state.micInterval);
                state.micInterval = null;
            }
            
            // Réinitialiser les barres
            state.micBars.forEach(bar => {
                bar.style.height = '10px';
                bar.classList.remove('active');
            });
        }

        function handleKeyboardShortcuts(e) {
            if (e.target === elements.textDisplay) return;
            
            // Espace pour démarrer/arrêter
            if (e.code === 'Space') {
                e.preventDefault();
                if (state.isListening) {
                    stopListening();
                } else {
                    startListening();
                }
            }
            
            // Ctrl/Cmd + M pour changer de mode
            if ((e.ctrlKey || e.metaKey) && e.key === 'm') {
                e.preventDefault();
                setMode(state.isAutoMode ? 'manual' : 'auto');
            }
            
            // Ctrl/Cmd + C pour copier
            if ((e.ctrlKey || e.metaKey) && e.key === 'c') {
                e.preventDefault();
                copyToClipboard();
            }
            
            // Ctrl/Cmd + K pour effacer
            if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
                e.preventDefault();
                clearText();
            }
        }

        function showNotification(message, color, icon) {
            elements.notificationText.textContent = message;
            elements.notificationIcon.textContent = icon;
            elements.notification.style.background = color;
            elements.notification.classList.add('show');
            
            setTimeout(() => {
                elements.notification.classList.remove('show');
            }, 2000);
        }
