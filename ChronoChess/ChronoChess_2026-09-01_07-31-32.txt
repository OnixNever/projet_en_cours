<script>
    let t1, t2, activePlayer = 1, isRunning = false, timer;

    // --- SYSTÈME DE SAUVEGARDE ---
    function saveAll() {
        localStorage.setItem('chess_base', document.getElementById('base-time').value);
        localStorage.setItem('chess_inc', document.getElementById('inc-time').value);
    }

    window.onload = () => {
        // Restaurer temps
        if(localStorage.getItem('chess_base')) document.getElementById('base-time').value = localStorage.getItem('chess_base');
        if(localStorage.getItem('chess_inc')) document.getElementById('inc-time').value = localStorage.getItem('chess_inc');
        
        // Restaurer thème
        const savedTheme = localStorage.getItem('chess_theme');
        if(savedTheme) {
            const opt = document.querySelector(`[data-theme="${savedTheme}"]`);
            if(opt) selectTheme(opt);
        }
    };

    function selectTheme(el) {
        document.querySelectorAll('.theme-option').forEach(o => o.classList.remove('active'));
        el.classList.add('active');
        const theme = el.getAttribute('data-theme');
        document.body.className = theme;
        localStorage.setItem('chess_theme', theme);
    }

    // --- JEU ---
    function launchGame() {
        saveAll();
        const mins = parseInt(document.getElementById('base-time').value) || 10;
        t1 = t2 = mins * 60;
        activePlayer = 1; isRunning = false;
        updateDisplay();
        document.getElementById('chess-overlay').style.display = 'flex';
        updateZones();
    }

    function togglePause() {
        isRunning = !isRunning;
        document.getElementById('pause-btn').textContent = isRunning ? "⏸" : "▶️";
        if(isRunning) {
            timer = setInterval(() => {
                if(activePlayer === 1) t1--; else t2--;
                if(t1 <= 0 || t2 <= 0) { clearInterval(timer); alert("Temps écoulé !"); }
                updateDisplay();
            }, 1000);
        } else clearInterval(timer);
    }

    function tapZone(p) {
        if(!isRunning || activePlayer !== p) return;
        const inc = parseInt(document.getElementById('inc-time').value) || 0;
        if(p === 1) { t1 += inc; activePlayer = 2; }
        else { t2 += inc; activePlayer = 1; }
        updateZones();
        updateDisplay();
    }

    function updateZones() {
        document.getElementById('p1-zone').className = `player-zone ${activePlayer === 1 ? 'active-player' : 'inactive-player'}`;
        document.getElementById('p2-zone').className = `player-zone ${activePlayer === 2 ? 'active-player' : 'inactive-player'}`;
    }

    function updateDisplay() {
        const fmt = (s) => Math.floor(s/60) + ":" + (s%60).toString().padStart(2, '0');
        document.getElementById('t1-val').textContent = fmt(t1);
        document.getElementById('t2-val').textContent = fmt(t2);
    }

    function exitGame() { clearInterval(timer); document.getElementById('chess-overlay').style.display = 'none'; }
    function resetGame() { clearInterval(timer); launchGame(); }
</script>
